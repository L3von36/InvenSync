    47→interface Expense {
    48→  id: string
    49→  organizationId: string
    50→  shopId?: string | null
    51→  category: string
    52→  amount: number
    53→  description?: string | null
    54→  expenseDate: string
    55→  isRecurring: boolean
    56→  recurringPeriod?: string | null
    57→  createdAt: string
    58→  updatedAt: string
    59→  shop?: { id: string; name: string } | null
    60→}
    61→
    62→const EXPENSE_CATEGORIES = [
    63→  { value: 'rent', label: 'Rent' },
    64→  { value: 'salary', label: 'Salary' },
    65→  { value: 'utilities', label: 'Utilities' },
    66→  { value: 'marketing', label: 'Marketing' },
    67→  { value: 'supplies', label: 'Supplies' },
    68→  { value: 'transport', label: 'Transport' },
    69→  { value: 'other', label: 'Other' },
    70→]
    71→
    72→const expenseFormSchema = z.object({
    73→  category: z.string().min(1, 'Category is required'),
    74→  amount: z.number({ message: 'Amount is required' }).positive('Amount must be greater than 0'),
    75→  description: z.string().optional(),
    76→  expenseDate: z.string().min(1, 'Date is required'),
    77→  shopId: z.string().optional(),
    78→  isRecurring: z.boolean().optional(),
    79→  recurringPeriod: z.string().optional(),
    80→})
    81→
    82→type ExpenseFormData = z.infer<typeof expenseFormSchema>

   103→function formatMonthLabel(monthKey: string): string {
   104→  const MONTH_LABELS: Record<string, string> = {
   105→    '01': 'Jan', '02': 'Feb', '03': 'Mar', '04': 'Apr',
   106→    '05': 'May', '06': 'Jun', '07': 'Jul', '08': 'Aug',
   107→    '09': 'Sep', '10': 'Oct', '11': 'Nov', '12': 'Dec',
   108→  }
   109→  const [year, month] = monthKey.split('-')
   110→  return `${MONTH_LABELS[month] || month} ${year.slice(2)}`
   111→}

   117→export function ExpensesPage() {
   118→  const { currentOrg, shops } = useAuthStore()
   119→  const [expenses, setExpenses] = useState<Expense[]>([])

   121→  const [error, setError] = useState<string | null>(null)

   123→  const [searchQuery, setSearchQuery] = useState('')

   126→
   127→  // Dialog states
   128→  const [dialogOpen, setDialogOpen] = useState(false)

   132→  const [isDeleting, setIsDeleting] = useState(false)
   133→

   135→    resolver: zodResolver(expenseFormSchema),
   136→    defaultValues: {
   137→      category: '',
   138→      amount: 0,
   139→      description: '',

   143→      recurringPeriod: '',

   145→  })

   147→  const isRecurring = form.watch('isRecurring')

   150→    if (!currentOrg) return

   152→    setError(null)

   154→      const params = new URLSearchParams({ organizationId: currentOrg.id })
   155→      if (categoryFilter && categoryFilter !== 'all') params.set('category', categoryFilter)
   156→
   157→      const response = await fetch(`/api/expenses?${params.toString()}`, {
   158→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   159→      })
   160→      if (!response.ok) {
   161→        const errData = await response.json().catch(() => ({}))
   162→        throw new Error(errData.error || 'Failed to fetch expenses')
   163→      }

   166→      setTotalExpenses(result.summary?.totalExpenses || 0)
   167→      setMonthlySummary(result.monthlySummary || [])

   171→      setIsLoading(false)
   172→    }

   183→        CATEGORY_LABELS[e.category]?.toLowerCase().includes(searchQuery.toLowerCase())
   184→      )
   185→    : expenses
   186→

   189→    form.reset({
   190→      category: '',
   191→      amount: 0,
   192→      description: '',
   193→      expenseDate: new Date().toISOString().split('T')[0],
   194→      shopId: '',
   195→      isRecurring: false,
   196→      recurringPeriod: '',

   198→    setDialogOpen(true)
   199→  }

   201→  function openEditDialog(expense: Expense) {
   202→    setEditingExpense(expense)

   204→      category: expense.category,
   205→      amount: expense.amount,
   206→      description: expense.description || '',
   207→      expenseDate: new Date(expense.expenseDate).toISOString().split('T')[0],
   208→      shopId: expense.shopId || '',
   209→      isRecurring: expense.isRecurring,

   213→  }
   214→
   215→  async function onSubmit(formData: ExpenseFormData) {
   216→    if (!currentOrg) return
   217→    setIsSaving(true)
   218→    try {
   219→      const payload = {
   220→        organizationId: currentOrg.id,
   221→        ...formData,
   222→        shopId: formData.shopId || null,
   223→        description: formData.description || null,
   224→        recurringPeriod: formData.isRecurring ? formData.recurringPeriod : null,
   225→      }
   226→
   227→      if (editingExpense) {
   228→        const res = await fetch(`/api/expenses/${editingExpense.id}?organizationId=${currentOrg.id}`, {
   229→          method: 'PUT',
   230→          headers: {
   231→            'Content-Type': 'application/json',
   232→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   233→          },
   234→          body: JSON.stringify(payload),
   235→        })
   236→        if (!res.ok) {
   237→          const errData = await res.json().catch(() => ({}))
   238→          throw new Error(errData.error || 'Failed to update expense')
   239→        }
   240→        toast.success('Expense updated successfully')
   241→      } else {
   242→        const res = await fetch('/api/expenses', {
   243→          method: 'POST',
   244→          headers: {
   245→            'Content-Type': 'application/json',
   246→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   247→          },
   248→          body: JSON.stringify(payload),
   249→        })
   250→        if (!res.ok) {
   251→          const errData = await res.json().catch(() => ({}))
   252→          throw new Error(errData.error || 'Failed to create expense')
   253→        }
   254→        toast.success('Expense created successfully')
   255→      }
   256→
   257→      setDialogOpen(false)
   258→      fetchExpenses()
   259→    } catch (err) {
   260→      toast.error(getNetworkErrorMessage(err))
   261→    } finally {
   262→      setIsSaving(false)
   263→    }
   264→  }
   265→
   266→  async function handleDelete() {

   270→      const res = await fetch(`/api/expenses/${deleteTarget.id}?organizationId=${currentOrg.id}`, {
   271→        method: 'DELETE',
   272→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   273→      })
   274→      if (!res.ok) {
   275→        const errData = await res.json().catch(() => ({}))
   276→        throw new Error(errData.error || 'Failed to delete expense')
   277→      }
   278→      toast.success('Expense deleted successfully')
   279→      setDeleteTarget(null)
   280→      fetchExpenses()
   281→    } catch (err) {
   282→      toast.error(getNetworkErrorMessage(err))
   283→    } finally {
   284→      setIsDeleting(false)
   285→    }
   286→  }

   289→    return <ErrorState title="Failed to load Expenses" message={error} onRetry={fetchExpenses} />

   325→      {/* Monthly Chart */}

   330→        </CardHeader>
   331→        <CardContent>
   332→          {isLoading ? (
   333→            <Skeleton className="h-[250px] w-full" />
   334→          ) : monthlySummary.length === 0 ? (
   335→            <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   336→              No expense data available yet
   337→            </div>
   338→          ) : (
   339→            <ResponsiveContainer width="100%" height={250}>
   340→              <BarChart data={monthlySummary.map(d => ({ ...d, month: formatMonthLabel(d.month) }))}>
   341→                <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />

   343→                <YAxis className="text-xs" tickFormatter={(v) => `${(v / 1000).toFixed(0)}k`} />
   344→                <Tooltip formatter={(value: number) => formatETB(value)} contentStyle={{ borderRadius: '8px', fontSize: '12px' }} />
   345→                <Bar dataKey="total" name="Expenses" fill="#ef4444" radius={[4, 4, 0, 0]} />
   346→              </BarChart>

   351→
   352→      {/* Filters */}
   353→      <div className="flex flex-col sm:flex-row gap-3">
   354→        <div className="relative flex-1">
   355→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   356→          <Input

   361→          />

   374→        <Button variant="outline" size="icon" onClick={fetchExpenses} disabled={isLoading}>
   375→          <RefreshCw className={`size-4 ${isLoading ? 'animate-spin' : ''}`} />
   376→        </Button>
   377→      </div>
   378→
   379→      {/* Expense List */}
   380→      <Card>
   381→        <CardContent className="p-0">
   382→          {isLoading ? (
   383→            <div className="p-6 space-y-3">
   384→              {Array.from({ length: 5 }).map((_, i) => (
   385→                <Skeleton key={i} className="h-12 w-full" />
   386→              ))}
   387→            </div>
   388→          ) : filteredExpenses.length === 0 ? (
   389→            <EmptyState
   390→              title="No expenses found"
   391→              message={categoryFilter !== 'all' ? 'Try changing the filter or add a new expense' : 'Start by adding your first expense'}
   392→              action={{ label: 'Add Expense', onClick: openCreateDialog }}

   395→            <>
   396→              {/* Desktop Table */}
   397→              <div className="hidden md:block">
   398→                <Table>

   400→                    <TableRow>
   401→                      <TableHead>Date</TableHead>
   402→                      <TableHead>Category</TableHead>
   403→                      <TableHead>Description</TableHead>
   404→                      <TableHead>Shop</TableHead>
   405→                      <TableHead className="text-right">Amount</TableHead>
   406→                      <TableHead>Recurring</TableHead>

   409→                  </TableHeader>

   466→              {/* Mobile Cards */}
   467→              <div className="md:hidden divide-y">

   519→
   520→      {/* Add/Edit Dialog */}
   521→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   522→        <DialogContent className="sm:max-w-[480px]">
   523→          <DialogHeader>

   525→            <DialogDescription>
   526→              {editingExpense ? 'Update the expense details below' : 'Fill in the details for the new expense'}

   528→          </DialogHeader>
   529→          <Form {...form}>
   530→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   531→              {/* Category */}

   537→                  value={form.watch('category')}
   538→                  onValueChange={(val) => form.setValue('category', val)}
   539→                >
   540→                  <SelectTrigger>
   541→                    <SelectValue placeholder="Select category" />
   542→                  </SelectTrigger>
   543→                  <SelectContent>
   544→                    {EXPENSE_CATEGORIES.map(cat => (
   545→                      <SelectItem key={cat.value} value={cat.value}>{cat.label}</SelectItem>

   554→              {/* Amount */}
   555→              <div className="space-y-2">
   556→                <label className="text-sm font-medium">
   557→                  Amount (ETB) <span className="text-destructive">*</span>
   558→                </label>
   559→                <Input
   560→                  type="number"
   561→                  step="0.01"
   562→                  min="0"
   563→                  placeholder="0.00"
   564→                  {...form.register('amount', { valueAsNumber: true })}
   565→                />
   566→                {form.formState.errors.amount && (
   567→                  <p className="text-xs text-destructive">{form.formState.errors.amount.message}</p>
   568→                )}
   569→              </div>
   570→
   571→              {/* Description */}
   572→              <div className="space-y-2">
   573→                <label className="text-sm font-medium">Description</label>
   574→                <Input
   575→                  placeholder="Optional description"
   576→                  {...form.register('description')}
   577→                />
   578→              </div>
   579→
   580→              {/* Date */}
   581→              <div className="space-y-2">
   582→                <label className="text-sm font-medium">
   583→                  Date <span className="text-destructive">*</span>
   584→                </label>
   585→                <Input
   586→                  type="date"
   587→                  {...form.register('expenseDate')}
   588→                />
   589→                {form.formState.errors.expenseDate && (
   590→                  <p className="text-xs text-destructive">{form.formState.errors.expenseDate.message}</p>
   591→                )}
   592→              </div>
   593→
   594→              {/* Shop */}
   595→              {shops.length > 1 && (
   596→                <div className="space-y-2">
   597→                  <label className="text-sm font-medium">Shop</label>
   598→                  <Select
   599→                    value={form.watch('shopId') || 'none'}
   600→                    onValueChange={(val) => form.setValue('shopId', val === 'none' ? '' : val)}

   604→                    </SelectTrigger>
   605→                    <SelectContent>
   606→                      <SelectItem value="none">All shops</SelectItem>
   607→                      {shops.map(shop => (
   608→                        <SelectItem key={shop.id} value={shop.id}>{shop.name}</SelectItem>
   609→                      ))}
   610→                    </SelectContent>
   611→                  </Select>
   612→                </div>
   613→              )}
   614→
   615→              {/* Recurring Toggle */}
   616→              <div className="flex items-center justify-between gap-3 rounded-lg border p-3">
   617→                <div className="space-y-0.5">
   618→                  <label className="text-sm font-medium">Recurring</label>
   619→                  <p className="text-xs text-muted-foreground">Set up a recurring expense</p>
   620→                </div>
   621→                <Switch
   622→                  checked={form.watch('isRecurring')}
   623→                  onCheckedChange={(val) => form.setValue('isRecurring', val)}
   624→                />
   625→              </div>
   626→
   627→              {/* Recurring Period */}
   628→              {isRecurring && (
   629→                <div className="space-y-2">
   630→                  <label className="text-sm font-medium">Recurring Period</label>
   631→                  <Select
   632→                    value={form.watch('recurringPeriod') || ''}
   633→                    onValueChange={(val) => form.setValue('recurringPeriod', val)}
   634→                  >
   635→                    <SelectTrigger>
   636→                      <SelectValue placeholder="Select frequency" />
   637→                    </SelectTrigger>
   638→                    <SelectContent>
   639→                      <SelectItem value="weekly">Weekly</SelectItem>
   640→                      <SelectItem value="monthly">Monthly</SelectItem>
   641→                      <SelectItem value="yearly">Yearly</SelectItem>
   642→                    </SelectContent>
   643→                  </Select>
   644→                </div>
   645→              )}
   646→
   647→              <DialogFooter>
   648→                <Button type="button" variant="outline" onClick={() => setDialogOpen(false)}>
   649→                  Cancel
   650→                </Button>
   651→                <Button type="submit" disabled={isSaving}>
   652→                  {isSaving ? (
   653→                    <>
   654→                      <Loader2 className="size-4 animate-spin" />
   655→                      <span>Saving...</span>
   656→                    </>
   657→                  ) : editingExpense ? 'Update' : 'Create'}
   658→                </Button>
   659→              </DialogFooter>
   660→            </form>
   661→          </Form>
   662→        </DialogContent>
   663→      </Dialog>
   664→
   665→      {/* Delete Confirmation */}
   666→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   667→        <AlertDialogContent>
   668→          <AlertDialogHeader>
   669→            <AlertDialogTitle>Delete Expense</AlertDialogTitle>
   670→            <AlertDialogDescription>
   671→              Are you sure you want to delete this expense of {deleteTarget ? formatETB(deleteTarget.amount) : ''}?
   672→              This action cannot be undone.
   673→            </AlertDialogDescription>
   674→          </AlertDialogHeader>
   675→          <AlertDialogFooter>
   676→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   677→            <AlertDialogAction
   678→              onClick={handleDelete}
   679→              disabled={isDeleting}
   680→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   681→            >
   682→              {isDeleting ? (
   683→                <>
   684→                  <Loader2 className="size-4 animate-spin" />
   685→                  Deleting...
   686→                </>
   687→              ) : 'Delete'}
   688→            </AlertDialogAction>
   689→          </AlertDialogFooter>
   690→        </AlertDialogContent>
   691→      </AlertDialog>
   692→    </div>
   693→  )
