    47→interface Expense {
    48→  id: string
    49→  organizationId: string
    50→  shopId?: string | null
    51→  category: string
    52→  amount: number
    53→  description?: string | null
    54→  expenseDate: string
    55→  isRecurring: boolean
    56→  recurringPeriod?: string | null
    57→  createdAt: string
    58→  updatedAt: string
    59→  shop?: { id: string; name: string } | null
    60→}
    61→
    62→const EXPENSE_CATEGORIES = [
    63→  { value: 'rent', label: 'Rent' },
    64→  { value: 'salary', label: 'Salary' },
    65→  { value: 'utilities', label: 'Utilities' },
    66→  { value: 'marketing', label: 'Marketing' },
    67→  { value: 'supplies', label: 'Supplies' },
    68→  { value: 'transport', label: 'Transport' },
    69→  { value: 'other', label: 'Other' },
    70→]
    71→
    72→const expenseFormSchema = z.object({
    73→  category: z.string().min(1, 'Category is required'),
    74→  amount: z.number({ message: 'Amount is required' }).positive('Amount must be greater than 0'),
    75→  description: z.string().optional(),
    76→  expenseDate: z.string().min(1, 'Date is required'),
    77→  shopId: z.string().optional(),
    78→  isRecurring: z.boolean().optional(),
    79→  recurringPeriod: z.string().optional(),
    80→})
    81→
    82→type ExpenseFormData = z.infer<typeof expenseFormSchema>

   103→function formatMonthLabel(monthKey: string): string {
   104→  const MONTH_LABELS: Record<string, string> = {
   105→    '01': 'Jan', '02': 'Feb', '03': 'Mar', '04': 'Apr',
   106→    '05': 'May', '06': 'Jun', '07': 'Jul', '08': 'Aug',
   107→    '09': 'Sep', '10': 'Oct', '11': 'Nov', '12': 'Dec',
   108→  }
   109→  const [year, month] = monthKey.split('-')
   110→  return `${MONTH_LABELS[month] || month} ${year.slice(2)}`
   111→}

   117→export function ExpensesPage() {

   133→

   137→      category: '',

   143→      recurringPeriod: '',

   150→    if (!currentOrg) return

   152→    setError(null)

   154→      const params = new URLSearchParams({ organizationId: currentOrg.id })

   156→
   157→      const response = await fetch(`/api/expenses?${params.toString()}`, {
   158→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   159→      })
   160→      if (!response.ok) {
   161→        const errData = await response.json().catch(() => ({}))
   162→        throw new Error(errData.error || 'Failed to fetch expenses')

   171→      setIsLoading(false)

   190→      category: '',
   191→      amount: 0,
   192→      description: '',
   193→      expenseDate: new Date().toISOString().split('T')[0],

   195→      isRecurring: false,
   196→      recurringPeriod: '',

   201→  function openEditDialog(expense: Expense) {

   205→      amount: expense.amount,
   206→      description: expense.description || '',

   213→  }
   214→
   215→  async function onSubmit(formData: ExpenseFormData) {
   216→    if (!currentOrg) return
   217→    setIsSaving(true)
   218→    try {
   219→      const payload = {
   220→        organizationId: currentOrg.id,
   221→        ...formData,
   222→        shopId: formData.shopId || null,
   223→        description: formData.description || null,
   224→        recurringPeriod: formData.isRecurring ? formData.recurringPeriod : null,
   225→      }
   226→
   227→      if (editingExpense) {
   228→        const res = await fetch(`/api/expenses/${editingExpense.id}?organizationId=${currentOrg.id}`, {
   229→          method: 'PUT',
   230→          headers: {
   231→            'Content-Type': 'application/json',
   232→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   233→          },
   234→          body: JSON.stringify(payload),
   235→        })
   236→        if (!res.ok) {
   237→          const errData = await res.json().catch(() => ({}))
   238→          throw new Error(errData.error || 'Failed to update expense')
   239→        }
   240→        toast.success('Expense updated successfully')
   241→      } else {
   242→        const res = await fetch('/api/expenses', {
   243→          method: 'POST',
   244→          headers: {
   245→            'Content-Type': 'application/json',
   246→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   247→          },
   248→          body: JSON.stringify(payload),
   249→        })
   250→        if (!res.ok) {
   251→          const errData = await res.json().catch(() => ({}))
   252→          throw new Error(errData.error || 'Failed to create expense')
   253→        }
   254→        toast.success('Expense created successfully')
   255→      }
   256→
   257→      setDialogOpen(false)
   258→      fetchExpenses()
   259→    } catch (err) {
   260→      toast.error(getNetworkErrorMessage(err))
   261→    } finally {
   262→      setIsSaving(false)
   263→    }
   264→  }

   272→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   273→      })

   275→        const errData = await res.json().catch(() => ({}))
   276→        throw new Error(errData.error || 'Failed to delete expense')
   277→      }
   278→      toast.success('Expense deleted successfully')
   279→      setDeleteTarget(null)
   280→      fetchExpenses()
   281→    } catch (err) {
   282→      toast.error(getNetworkErrorMessage(err))
   283→    } finally {
   284→      setIsDeleting(false)
   285→    }

   330→        </CardHeader>

   333→            <Skeleton className="h-[250px] w-full" />
   334→          ) : monthlySummary.length === 0 ? (
   335→            <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   336→              No expense data available yet
   337→            </div>
   338→          ) : (
   339→            <ResponsiveContainer width="100%" height={250}>
   340→              <BarChart data={monthlySummary.map(d => ({ ...d, month: formatMonthLabel(d.month) }))}>

   345→                <Bar dataKey="total" name="Expenses" fill="#ef4444" radius={[4, 4, 0, 0]} />

   375→        <Button variant="outline" size="icon" onClick={fetchExpenses} disabled={isLoading} aria-label="Refresh expenses">
   376→          <RefreshCw className={`size-4 ${isLoading ? 'animate-spin' : ''}`} />
   377→        </Button>
   378→      </div>
   379→
   380→      {/* Expense List */}
   381→      <Card>
   382→        <CardContent className="p-0">
   383→          {isLoading ? (

   385→              {Array.from({ length: 5 }).map((_, i) => (
   386→                <Skeleton key={i} className="h-12 w-full" />

   388→            </div>
   389→          ) : filteredExpenses.length === 0 ? (
   390→            <EmptyState
   391→              title="No expenses found"
   392→              message={categoryFilter !== 'all' ? 'Try changing the filter or add a new expense' : 'Start by adding your first expense'}
   393→              action={{ label: 'Add Expense', onClick: openCreateDialog }}

   396→            <>
   397→              {/* Desktop Table */}

   401→                    <TableRow>
   402→                      <TableHead>Date</TableHead>
   403→                      <TableHead>Category</TableHead>
   404→                      <TableHead>Description</TableHead>

   469→              {/* Mobile Cards */}

   517→              </div>
   518→            </>

   520→        </CardContent>
   521→      </Card>
   522→
   523→      {/* Add/Edit Dialog */}
   524→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   525→        <DialogContent className="sm:max-w-[480px]">
   526→          <DialogHeader>

   531→          </DialogHeader>
   532→          <Form {...form}>
   533→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   534→              {/* Category */}
   535→              <div className="space-y-2">
   536→                <label className="text-sm font-medium">
   537→                  Category <span className="text-destructive">*</span>
   538→                </label>
   539→                <Select
   540→                  value={form.watch('category')}
   541→                  onValueChange={(val) => form.setValue('category', val)}
   542→                >
   543→                  <SelectTrigger>
   544→                    <SelectValue placeholder="Select category" />

   546→                  <SelectContent>
   547→                    {EXPENSE_CATEGORIES.map(cat => (

   550→                  </SelectContent>
   551→                </Select>
   552→                {form.formState.errors.category && (
   553→                  <p className="text-xs text-destructive">{form.formState.errors.category.message}</p>
   554→                )}
   555→              </div>
   556→
   557→              {/* Amount */}
   558→              <div className="space-y-2">
   559→                <label className="text-sm font-medium">
   560→                  Amount (ETB) <span className="text-destructive">*</span>
   561→                </label>
   562→                <Input
   563→                  type="number"
   564→                  step="0.01"
   565→                  min="0"
   566→                  placeholder="0.00"
   567→                  {...form.register('amount', { valueAsNumber: true })}
   568→                />
   569→                {form.formState.errors.amount && (
   570→                  <p className="text-xs text-destructive">{form.formState.errors.amount.message}</p>
   571→                )}
   572→              </div>
   573→
   574→              {/* Description */}
   575→              <div className="space-y-2">
   576→                <label className="text-sm font-medium">Description</label>
   577→                <Input
   578→                  placeholder="Optional description"
   579→                  {...form.register('description')}
   580→                />
   581→              </div>
   582→
   583→              {/* Date */}
   584→              <div className="space-y-2">
   585→                <label className="text-sm font-medium">
   586→                  Date <span className="text-destructive">*</span>
   587→                </label>
   588→                <Input
   589→                  type="date"
   590→                  {...form.register('expenseDate')}
   591→                />
   592→                {form.formState.errors.expenseDate && (
   593→                  <p className="text-xs text-destructive">{form.formState.errors.expenseDate.message}</p>
   594→                )}
   595→              </div>
   596→
   597→              {/* Shop */}
   598→              {shops.length > 1 && (
   599→                <div className="space-y-2">
   600→                  <label className="text-sm font-medium">Shop</label>
   601→                  <Select
   602→                    value={form.watch('shopId') || 'none'}
   603→                    onValueChange={(val) => form.setValue('shopId', val === 'none' ? '' : val)}
   604→                  >
   605→                    <SelectTrigger>
   606→                      <SelectValue placeholder="All shops" />
   607→                    </SelectTrigger>
   608→                    <SelectContent>
   609→                      <SelectItem value="none">All shops</SelectItem>
   610→                      {shops.map(shop => (
   611→                        <SelectItem key={shop.id} value={shop.id}>{shop.name}</SelectItem>
   612→                      ))}
   613→                    </SelectContent>
   614→                  </Select>
   615→                </div>
   616→              )}
   617→
   618→              {/* Recurring Toggle */}
   619→              <div className="flex items-center justify-between gap-3 rounded-lg border p-3">
   620→                <div className="space-y-0.5">
   621→                  <label className="text-sm font-medium">Recurring</label>
   622→                  <p className="text-xs text-muted-foreground">Set up a recurring expense</p>
   623→                </div>
   624→                <Switch
   625→                  checked={form.watch('isRecurring')}
   626→                  onCheckedChange={(val) => form.setValue('isRecurring', val)}
   627→                />
   628→              </div>
   629→
   630→              {/* Recurring Period */}
   631→              {isRecurring && (
   632→                <div className="space-y-2">
   633→                  <label className="text-sm font-medium">Recurring Period</label>
   634→                  <Select
   635→                    value={form.watch('recurringPeriod') || ''}
   636→                    onValueChange={(val) => form.setValue('recurringPeriod', val)}
   637→                  >
   638→                    <SelectTrigger>
   639→                      <SelectValue placeholder="Select frequency" />
   640→                    </SelectTrigger>
   641→                    <SelectContent>
   642→                      <SelectItem value="weekly">Weekly</SelectItem>
   643→                      <SelectItem value="monthly">Monthly</SelectItem>
   644→                      <SelectItem value="yearly">Yearly</SelectItem>
   645→                    </SelectContent>
   646→                  </Select>
   647→                </div>
   648→              )}
   649→
   650→              <DialogFooter>
   651→                <Button type="button" variant="outline" onClick={() => setDialogOpen(false)}>
   652→                  Cancel
   653→                </Button>
   654→                <Button type="submit" disabled={isSaving}>
   655→                  {isSaving ? (
   656→                    <>
   657→                      <Loader2 className="size-4 animate-spin" />
   658→                      <span>Saving...</span>
   659→                    </>
   660→                  ) : editingExpense ? 'Update' : 'Create'}
   661→                </Button>
   662→              </DialogFooter>
   663→            </form>
   664→          </Form>
   665→        </DialogContent>
   666→      </Dialog>
   667→
   668→      {/* Delete Confirmation */}
   669→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   670→        <AlertDialogContent>
   671→          <AlertDialogHeader>
   672→            <AlertDialogTitle>Delete Expense</AlertDialogTitle>
   673→            <AlertDialogDescription>
   674→              Are you sure you want to delete this expense of {deleteTarget ? formatETB(deleteTarget.amount) : ''}?
   675→              This action cannot be undone.
   676→            </AlertDialogDescription>
   677→          </AlertDialogHeader>
   678→          <AlertDialogFooter>
   679→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   680→            <AlertDialogAction
   681→              onClick={handleDelete}
   682→              disabled={isDeleting}
   683→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   684→            >
   685→              {isDeleting ? (
   686→                <>
   687→                  <Loader2 className="size-4 animate-spin" />
   688→                  Deleting...
   689→                </>
   690→              ) : 'Delete'}
   691→            </AlertDialogAction>
   692→          </AlertDialogFooter>
   693→        </AlertDialogContent>
   694→      </AlertDialog>
   695→    </div>
   696→  )
