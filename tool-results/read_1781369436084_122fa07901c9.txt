   103→   103→function formatMonthLabel(monthKey: string): string {
   104→   104→  const MONTH_LABELS: Record<string, string> = {
   105→   105→    '01': 'Jan', '02': 'Feb', '03': 'Mar', '04': 'Apr',
   106→   106→    '05': 'May', '06': 'Jun', '07': 'Jul', '08': 'Aug',
   107→   107→    '09': 'Sep', '10': 'Oct', '11': 'Nov', '12': 'Dec',
   108→   108→  }
   109→   109→  const [year, month] = monthKey.split('-')
   110→   110→  return `${MONTH_LABELS[month] || month} ${year.slice(2)}`
   111→   111→}
