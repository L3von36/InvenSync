    35→function getStockStatus(quantity: number, threshold: number): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    36→  if (quantity === 0) return { label: 'Out of Stock', variant: 'destructive' }
    37→  if (quantity <= threshold) return { label: 'Low Stock', variant: 'secondary' }
    38→  return { label: 'In Stock', variant: 'default' }
    39→}
    40→
    41→function formatDate(dateStr: string): string {
    42→  return new Date(dateStr).toLocaleDateString('en-US', {
    43→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit'
    44→  })
    45→}

    80→// ============================================
    81→// Stock Adjustment Dialog
    82→// ============================================
    83→function StockAdjustmentDialog({
    84→  open,
    85→  onClose,
    86→  product,
    87→  orgId,
    88→  shopId,
    89→  onAdjusted,
    90→}: {
    91→  open: boolean
    92→  onClose: () => void
    93→  product: Product | null
    94→  orgId: string
    95→  shopId?: string
    96→  onAdjusted: () => void
    97→}) {
    98→  const [loading, setLoading] = useState(false)
    99→
   100→  const form = useForm<StockAdjustmentFormData>({
   101→    resolver: zodResolver(stockAdjustmentSchema),
   102→    defaultValues: { type: 'in', quantity: '' as unknown as number, reason: '' },
   103→  })
   104→
   105→  const watchedType = form.watch('type')
   106→  const watchedQty = form.watch('quantity')
   107→  const qty = typeof watchedQty === 'number' ? watchedQty : (parseInt(String(watchedQty)) || 0)
   108→  const previewStock = watchedType === 'in'
   109→    ? (product?.quantity ?? 0) + qty
   110→    : watchedType === 'out'
   111→    ? Math.max(0, (product?.quantity ?? 0) - qty)
   112→    : qty
   113→
   114→  const handleSubmit = async (data: StockAdjustmentFormData) => {
   115→    if (!product) return
   116→
   117→    setLoading(true)
   118→    try {
   119→      await api.addStockMovement(orgId, {
   120→        productId: product.id,
   121→        type: data.type,
   122→        quantity: data.quantity,
   123→        reason: data.reason.trim(),
   124→        shopId,
   125→      })
   126→      toast.success('Stock adjusted successfully')
   127→      onAdjusted()
   128→      onClose()
   129→    } catch (err: unknown) {
   130→      toast.error(getNetworkErrorMessage(err))
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }
   135→
   136→  useEffect(() => {
   137→    if (!open) {
   138→      form.reset({ type: 'in', quantity: '' as unknown as number, reason: '' })
   139→    }
   140→  }, [open, form])
   141→
   142→  if (!product) return null
   143→
   144→  return (
   145→    <Dialog open={open} onOpenChange={onClose}>
   146→      <DialogContent className="sm:max-w-md">
   147→        <DialogHeader>
   148→          <DialogTitle>Adjust Stock</DialogTitle>
   149→          <DialogDescription>
   150→            Adjust stock for <span className="font-semibold">{product.name}</span> (Current: {product.quantity})
   151→          </DialogDescription>
   152→        </DialogHeader>
   153→        <Form {...form}>
   154→          <form onSubmit={form.handleSubmit(handleSubmit)} className="space-y-4 py-2">
   155→            <FormSelectField
   156→              name="type"
   157→              label="Adjustment Type"
   158→              options={[
   159→                { value: 'in', label: 'Add Stock', icon: <Plus className="size-3" /> },
   160→                { value: 'out', label: 'Remove Stock', icon: <Minus className="size-3" /> },
   161→                { value: 'adjustment', label: 'Adjustment (Set)', icon: <RefreshCw className="size-3" /> },
   162→              ]}
   163→            />
   164→            <FormInputField
   165→              name="quantity"
   166→              label="Quantity"
   167→              placeholder="Enter quantity"
   168→              type="number"
   169→              required
   170→              min={1}
   171→            />
   172→            <FormInputField
   173→              name="reason"
   174→              label="Reason"
   175→              placeholder="Enter reason for adjustment"
   176→              required
   177→            />
   178→            {qty > 0 && (
   179→              <div className="rounded-lg border bg-muted/50 p-3">
   180→                <p className="text-sm text-muted-foreground">Preview</p>
   181→                <p className="text-sm font-bold">
   182→                  {product.quantity} → {previewStock}
   183→                </p>
   184→              </div>
   185→            )}
   186→            <DialogFooter>
   187→              <Button type="button" variant="outline" onClick={onClose}>Cancel</Button>
   188→              <FormSubmitButton isLoading={loading} loadingText="Adjusting...">
   189→                Confirm Adjustment
   190→              </FormSubmitButton>
   191→            </DialogFooter>
   192→          </form>
   193→        </Form>
   194→      </DialogContent>
   195→    </Dialog>
   196→  )
   197→}

   200→// All Products Tab
   201→// ============================================
   202→function AllProductsTab({
   203→  orgId,
   204→  shopId,
   205→  onAdjust,
   206→}: {
   207→  orgId: string
   208→  shopId?: string
   209→  onAdjust: (product: Product) => void
   210→}) {
   211→  const [products, setProducts] = useState<Product[]>([])
   212→  const [productTypes, setProductTypes] = useState<ProductType[]>([])
   213→  const [loading, setLoading] = useState(true)
   214→  const [search, setSearch] = useState('')
   215→  const [filterType, setFilterType] = useState<string>('all')
   216→  const [sortCol, setSortCol] = useState<string>('name')
   217→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   218→  const [page, setPage] = useState(1)
   219→  const [totalPages, setTotalPages] = useState(1)
   220→
   221→  const fetchData = useCallback(async () => {
   222→    try {
   223→      const [productsRes, typesRes] = await Promise.all([
   224→        api.getProducts(orgId, {
   225→          search: search || undefined,
   226→          productTypeId: filterType !== 'all' ? filterType : undefined,
   227→          page,
   228→          limit: 10,
   229→          shopId,
   230→        }),
   231→        api.getProductTypes(orgId, shopId),
   232→      ])
   233→      setProducts(productsRes.products)
   234→      setTotalPages(productsRes.pagination.totalPages)
   235→      setProductTypes(typesRes.productTypes)
   236→    } catch (error) {
   237→      toast.error(getNetworkErrorMessage(error))
   238→    } finally {
   239→      setLoading(false)
   240→    }
   241→  }, [orgId, shopId, search, filterType, page])
   242→
   243→  useEffect(() => {
   244→    fetchData()
   245→  }, [fetchData])
   246→
   247→  const handleSort = (col: string) => {
   248→    if (sortCol === col) {
   249→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   250→    } else {
   251→      setSortCol(col)
   252→      setSortDir('asc')
   253→    }
   254→  }
   255→
   256→  const sortedProducts = [...products].sort((a, b) => {
   257→    const dir = sortDir === 'asc' ? 1 : -1
   258→    switch (sortCol) {
   259→      case 'name': return a.name.localeCompare(b.name) * dir
   260→      case 'sku': return (a.sku || '').localeCompare(b.sku || '') * dir
   261→      case 'quantity': return (a.quantity - b.quantity) * dir
   262→      case 'costPrice': return (a.costPrice - b.costPrice) * dir
   263→      case 'sellingPrice': return (a.sellingPrice - b.sellingPrice) * dir
   264→      default: return 0
   265→    }
   266→  })
   267→
   268→  const SortHeader = ({ col, children }: { col: string; children: React.ReactNode }) => (
   269→    <TableHead className="cursor-pointer select-none hover:bg-muted/50" onClick={() => handleSort(col)}>
   270→      <span className="flex items-center gap-1">
   271→        {children}
   272→        <ArrowUpDown className={`size-3 ${sortCol === col ? 'opacity-100' : 'opacity-30'}`} />
   273→      </span>
   274→    </TableHead>
   275→  )
   276→
   277→  if (loading) {
   278→    return (
   279→      <div className="space-y-4">
   280→        <div className="flex flex-col sm:flex-row gap-3">
   281→          <Skeleton className="h-10 flex-1" />
   282→          <Skeleton className="h-10 w-40" />
   283→        </div>
   284→        {Array.from({ length: 5 }).map((_, i) => (
   285→          <Skeleton key={i} className="h-14 w-full" />
   286→        ))}
   287→      </div>
   288→    )
   289→  }
   290→
   291→  return (
   292→    <div className="space-y-4">
   293→      {/* Filters */}
   294→      <div className="flex flex-col sm:flex-row gap-3">
   295→        <div className="relative flex-1">
   296→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   297→          <Input
   298→            placeholder="Search products..."
   299→            className="pl-9"
   300→            aria-label="Search inventory"
   301→            value={search}
   302→            onChange={(e) => { setSearch(e.target.value); setPage(1) }}
   303→          />
   304→        </div>
   305→        <Select value={filterType} onValueChange={(v) => { setFilterType(v); setPage(1) }}>
   306→          <SelectTrigger className="w-full sm:w-48">
   307→            <Filter className="size-4 mr-2" />
   308→            <SelectValue placeholder="Filter by type" />
   309→          </SelectTrigger>
   310→          <SelectContent>
   311→            <SelectItem value="all">All Types</SelectItem>
   312→            {productTypes.map((pt) => (
   313→              <SelectItem key={pt.id} value={pt.id}>{pt.name}</SelectItem>
   314→            ))}
   315→          </SelectContent>
   316→        </Select>
   317→      </div>
   318→
   319→      {/* Table */}
   320→      <div className="rounded-lg border overflow-hidden">
   321→        {/* Desktop table */}
   322→        <div className="hidden md:block overflow-x-auto -mx-4 md:mx-0">
   323→        <Table>
   324→          <TableHeader>
   325→            <TableRow>
   326→              <SortHeader col="name">Product Name</SortHeader>
   327→              <SortHeader col="sku">SKU</SortHeader>
   328→              <TableHead>Type</TableHead>
   329→              <SortHeader col="quantity">Quantity</SortHeader>
   330→              <SortHeader col="costPrice">Cost Price</SortHeader>
   331→              <SortHeader col="sellingPrice">Selling Price</SortHeader>
   332→              <TableHead>Value</TableHead>
   333→              <TableHead>Status</TableHead>
   334→              <TableHead className="text-right">Actions</TableHead>
   335→            </TableRow>
   336→          </TableHeader>
   337→          <TableBody>
   338→            {sortedProducts.length === 0 ? (
   339→              <TableRow>
   340→                <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   341→                  No products found
   342→                </TableCell>
   343→              </TableRow>
   344→            ) : (
   345→              sortedProducts.map((product) => {
   346→                const status = getStockStatus(product.quantity, product.lowStockThreshold)
   347→                return (
   348→                  <TableRow key={product.id}>
   349→                    <TableCell className="font-medium max-w-[180px] truncate">{product.name}</TableCell>
   350→                    <TableCell className="text-muted-foreground text-sm">{product.sku || '—'}</TableCell>
   351→                    <TableCell>
   352→                      <Badge variant="outline" className="text-xs">{product.productType?.name || '—'}</Badge>
   353→                    </TableCell>
   354→                    <TableCell className="font-mono">{product.quantity}</TableCell>
   355→                    <TableCell className="text-sm">{formatETB(product.costPrice)}</TableCell>
   356→                    <TableCell className="text-sm">{formatETB(product.sellingPrice)}</TableCell>
   357→                    <TableCell className="text-sm font-medium">{formatETB(product.quantity * product.sellingPrice)}</TableCell>
   358→                    <TableCell>
   359→                      <Badge variant={status.variant} className="text-xs">{status.label}</Badge>
   360→                    </TableCell>
   361→                    <TableCell className="text-right">
   362→                      <Button size="sm" variant="outline" onClick={() => onAdjust(product)}>
   363→                        Adjust Stock
   364→                      </Button>
   365→                    </TableCell>
   366→                  </TableRow>
   367→                )
   368→              })
   369→            )}
   370→          </TableBody>
   371→        </Table>
   372→        </div>
   373→        {/* Mobile card view */}
   374→        <div className="md:hidden space-y-3 p-3">
   375→          {sortedProducts.length === 0 ? (
   376→            <EmptyState
   377→              title="No products found"
   378→              message="No products match your current filters."
   379→            />
   380→          ) : (
   381→            sortedProducts.map((product) => {
   382→              const status = getStockStatus(product.quantity, product.lowStockThreshold)
   383→              return (
   384→                <Card key={product.id} className="p-4">
   385→                  <div className="flex items-start justify-between mb-2">
   386→                    <div className="min-w-0 flex-1">
   387→                      <p className="font-semibold text-sm truncate">{product.name}</p>
   388→                      <p className="text-xs text-muted-foreground">{product.productType?.name || '—'} {product.sku ? `· ${product.sku}` : ''}</p>
   389→                    </div>
   390→                    <Badge variant={status.variant} className="text-xs shrink-0 ml-2">{status.label}</Badge>
   391→                  </div>
   392→                  <div className="flex items-center justify-between text-xs mt-2">
   393→                    <div className="flex items-center gap-3 text-muted-foreground">
   394→                      <span>Qty: <strong className="text-foreground">{product.quantity}</strong></span>
   395→                      <span>{formatETB(product.sellingPrice)}</span>
   396→                    </div>
   397→                    <Button size="sm" variant="outline" className="h-7 text-xs" onClick={() => onAdjust(product)}>Adjust</Button>
   398→                  </div>
   399→                </Card>
   400→              )
   401→            })
   402→          )}
   403→        </div>
   404→      </div>
   405→
   406→      {/* Pagination */}
   407→      {totalPages > 1 && (
   408→        <div className="flex items-center justify-between">
   409→          <p className="text-sm text-muted-foreground">Page {page} of {totalPages}</p>
   410→          <div className="flex gap-2">
   411→            <Button size="sm" variant="outline" disabled={page <= 1} onClick={() => setPage(page - 1)}>
   412→              <ChevronLeft className="size-4" />
   413→            </Button>
   414→            <Button size="sm" variant="outline" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>
   415→              <ChevronRight className="size-4" />
   416→            </Button>
   417→          </div>
   418→        </div>
   419→      )}
   420→    </div>
   421→  )
   422→}

   424→// ============================================
   425→// Low Stock Tab
   426→// ============================================
   427→function LowStockTab({
   428→  orgId,
   429→  shopId,
   430→  lowStockProducts,
   431→  onRestock,
   432→}: {
   433→  orgId: string
   434→  shopId?: string
   435→  lowStockProducts: InventoryStats['lowStockProducts']
   436→  onRestock: (product: Product) => void
   437→}) {
   438→  const [products, setProducts] = useState<InventoryStats['lowStockProducts']>([])
   439→  const [loading, setLoading] = useState(true)
   440→
   441→  useEffect(() => {
   442→    const fetchLowStock = async () => {
   443→      try {
   444→        const data = await api.getInventory(orgId, shopId)
   445→        setProducts(data.lowStockProducts)
   446→      } catch (error) {
   447→      toast.error(getNetworkErrorMessage(error))
   448→      } finally {
   449→        setLoading(false)
   450→      }
   451→    }
   452→    fetchLowStock()
   453→  }, [orgId, shopId])
   454→
   455→  useEffect(() => {
   456→    if (lowStockProducts && lowStockProducts.length > 0) {
   457→      setProducts(lowStockProducts)
   458→      setLoading(false)
   459→    }
   460→  }, [lowStockProducts])
   461→
   462→  if (loading) {
   463→    return (
   464→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   465→        {Array.from({ length: 6 }).map((_, i) => (
   466→          <Skeleton key={i} className="h-32" />
   467→        ))}
   468→      </div>
   469→    )
   470→  }
   471→
   472→  if (products.length === 0) {
   473→    return (
   474→      <EmptyState
   475→        title="All Stocked Up!"
   476→        message="No products are currently low on stock."
   477→        icon={<AlertTriangle className="size-7 text-emerald-500" />}
   478→      />
   479→    )
   480→  }
   481→
   482→  return (
   483→    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   484→      {products.map((product) => (
   485→        <Card key={product.id} className="border-amber-200 dark:border-amber-800">
   486→          <CardContent className="p-4">
   487→            <div className="flex items-start justify-between mb-2">
   488→              <div>
   489→                <h4 className="font-semibold text-sm truncate max-w-[200px]">{product.name}</h4>
   490→                <p className="text-xs text-muted-foreground">{product.productType.name}</p>
   491→              </div>
   492→              <Badge variant="secondary" className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300 shrink-0">
   493→                Low Stock
   494→              </Badge>
   495→            </div>
   496→            <Separator className="my-3" />
   497→            <div className="flex items-center justify-between text-sm">
   498→              <div>
   499→                <span className="text-muted-foreground">Current: </span>
   500→                <span className="font-bold text-amber-600 dark:text-amber-400">{product.quantity}</span>
   501→              </div>
   502→              <div>
   503→                <span className="text-muted-foreground">Threshold: </span>
   504→                <span className="font-medium">{product.lowStockThreshold}</span>
   505→              </div>
   506→            </div>
   507→            <Button
   508→              size="sm"
   509→              className="w-full mt-3"
   510→              onClick={() => onRestock({
   511→                id: product.id,
   512→                name: product.name,
   513→                quantity: product.quantity,
   514→                lowStockThreshold: product.lowStockThreshold,
   515→              } as Product)}
   516→            >
   517→              <Plus className="size-3 mr-1" /> Quick Restock
   518→            </Button>
   519→          </CardContent>
   520→        </Card>
   521→      ))}
   522→    </div>
   523→  )
   524→}

   529→function OutOfStockTab({
   530→  orgId,
   531→  shopId,
   532→  onRestock,
   533→}: {
   534→  orgId: string
   535→  shopId?: string
   536→  onRestock: (product: Product) => void
   537→}) {
   538→  const [products, setProducts] = useState<Product[]>([])
   539→  const [loading, setLoading] = useState(true)
   540→
   541→  useEffect(() => {
   542→    const fetchOutOfStock = async () => {
   543→      try {
   544→        const res = await api.getProducts(orgId, { limit: 100, shopId })
   545→        setProducts(res.products.filter(p => p.quantity === 0))
   546→      } catch (error) {
   547→      toast.error(getNetworkErrorMessage(error))
   548→      } finally {
   549→        setLoading(false)
   550→      }
   551→    }
   552→    fetchOutOfStock()
   553→  }, [orgId, shopId])
   554→
   555→  if (loading) {
   556→    return (
   557→      <div className="space-y-3">

   589→              </div>
   590→            </div>
   591→            <Button size="sm" onClick={() => onRestock(product)}>

   594→          </CardContent>
   595→        </Card>
   596→      ))}
   597→    </div>

   604→function StockHistoryTab({ orgId, shopId }: { orgId: string; shopId?: string }) {
   605→  const [movements, setMovements] = useState<StockMovement[]>([])
   606→  const [products, setProducts] = useState<Product[]>([])
   607→  const [loading, setLoading] = useState(true)
   608→  const [filterProduct, setFilterProduct] = useState<string>('all')
   609→  const [filterType, setFilterType] = useState<string>('all')
   610→
   611→  useEffect(() => {
   612→    const fetchHistory = async () => {
   613→      try {
   614→        const [invData, prodRes] = await Promise.all([
   615→          api.getInventory(orgId, shopId),
   616→          api.getProducts(orgId, { limit: 100, shopId }),
   617→        ])
   618→        setMovements(invData.recentMovements)
   619→        setProducts(prodRes.products)
   620→      } catch (error) {
   621→        toast.error(getNetworkErrorMessage(error))
   622→      } finally {
   623→        setLoading(false)
   624→      }
   625→    }
   626→    fetchHistory()
   627→  }, [orgId, shopId])
   628→
   629→  const filtered = movements.filter((m) => {
   630→    if (filterProduct !== 'all' && m.productId !== filterProduct) return false
   631→    if (filterType !== 'all' && m.type !== filterType) return false
   632→    return true
   633→  })
   634→
   635→  const typeLabel = (type: string) => {
   636→    switch (type) {
   637→      case 'in': return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300">Stock In</Badge>
   638→      case 'out': return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300">Stock Out</Badge>
   639→      case 'adjustment': return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300">Adjustment</Badge>
   640→      default: return <Badge variant="outline">{type}</Badge>
   641→    }
   642→  }
   643→
   644→  if (loading) {
   645→    return (
   646→      <div className="space-y-3">
   647→        <div className="flex flex-col sm:flex-row gap-3">
   648→          <Skeleton className="h-10 w-48" />
   649→          <Skeleton className="h-10 w-36" />
   650→        </div>
   651→        {Array.from({ length: 6 }).map((_, i) => (
   652→          <Skeleton key={i} className="h-14" />
   653→        ))}
   654→      </div>
   655→    )
   656→  }
   657→
   658→  return (
   659→    <div className="space-y-4">
   660→      {/* Filters */}
   661→      <div className="flex flex-col sm:flex-row gap-3">
   662→        <Select value={filterProduct} onValueChange={setFilterProduct}>
   663→          <SelectTrigger className="w-full sm:w-52">
   664→            <SelectValue placeholder="Filter by product" />
   665→          </SelectTrigger>
   666→          <SelectContent>
   667→            <SelectItem value="all">All Products</SelectItem>
   668→            {products.map((p) => (
   669→              <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   670→            ))}
   671→          </SelectContent>
   672→        </Select>
   673→        <Select value={filterType} onValueChange={setFilterType}>
   674→          <SelectTrigger className="w-full sm:w-40">
   675→            <SelectValue placeholder="Filter by type" />
   676→          </SelectTrigger>
   677→          <SelectContent>
   678→            <SelectItem value="all">All Types</SelectItem>
   679→            <SelectItem value="in">Stock In</SelectItem>
   680→            <SelectItem value="out">Stock Out</SelectItem>
   681→            <SelectItem value="adjustment">Adjustment</SelectItem>
   682→          </SelectContent>
   683→        </Select>
   684→      </div>
   685→
   686→      {/* Empty state or Table */}
   687→      {!loading && filtered.length === 0 ? (
   688→        <Card>
   689→          <CardContent className="py-12 text-center">
   690→            <History className="size-12 text-muted-foreground mx-auto mb-4" />
   691→            <h3 className="text-base font-semibold mb-1">
   692→              {filterProduct !== 'all' || filterType !== 'all' ? 'No matching stock movements' : 'No stock movements yet'}
   693→            </h3>
   694→            <p className="text-muted-foreground text-sm mb-4">
   695→              {filterProduct !== 'all' || filterType !== 'all'
   696→                ? 'Try adjusting your filters'
   697→                : 'Stock movements will appear here when you adjust product quantities'}
   698→            </p>
   699→          </CardContent>
   700→        </Card>
   701→      ) : (
   702→      <div className="rounded-lg border overflow-hidden">
   703→        {/* Desktop table */}
   704→        <div className="hidden md:block overflow-x-auto -mx-4 md:mx-0">
   705→          <Table>
   706→            <TableHeader>
   707→              <TableRow>
   708→                <TableHead>Date</TableHead>
   709→                <TableHead>Product</TableHead>
   710→                <TableHead>Type</TableHead>
   711→                <TableHead className="text-right">Quantity</TableHead>
   712→                <TableHead className="text-right">Previous Stock</TableHead>
   713→                <TableHead className="text-right">New Stock</TableHead>
   714→                <TableHead>Reason</TableHead>
   715→              </TableRow>
   716→            </TableHeader>
   717→            <TableBody>
   718→              {filtered.map((m) => (
   719→                <TableRow key={m.id}>
   720→                  <TableCell className="text-sm whitespace-nowrap">{formatDate(m.createdAt)}</TableCell>
   721→                  <TableCell className="font-medium text-sm">{m.product?.name || '—'}</TableCell>
   722→                  <TableCell>{typeLabel(m.type)}</TableCell>
   723→                  <TableCell className="text-right font-mono text-sm">
   724→                    {m.type === 'in' ? '+' : m.type === 'out' ? '-' : '±'}{m.quantity}
   725→                  </TableCell>
   726→                  <TableCell className="text-right font-mono text-sm text-muted-foreground">{m.previousStock}</TableCell>
   727→                  <TableCell className="text-right font-mono text-sm font-medium">{m.newStock}</TableCell>
   728→                  <TableCell className="text-sm text-muted-foreground max-w-[200px] truncate">{m.reason || '—'}</TableCell>
   729→                </TableRow>
   730→              ))}
   731→            </TableBody>
   732→          </Table>
   733→        </div>
   734→        {/* Mobile card view */}
   735→        <div className="md:hidden space-y-3 p-3">
   736→          {filtered.map((m) => (
   737→            <Card key={m.id} className="p-4">
   738→              <div className="flex items-start justify-between mb-2">
   739→                <div className="min-w-0 flex-1">
   740→                  <p className="font-semibold text-sm truncate">{m.product?.name || '—'}</p>
   741→                  <p className="text-xs text-muted-foreground">{formatDate(m.createdAt)}</p>
   742→                </div>
   743→                {typeLabel(m.type)}
   744→              </div>
   745→              <div className="flex items-center justify-between text-xs mt-2">
   746→                <div className="flex items-center gap-3 text-muted-foreground">
   747→                  <span>Qty: <strong className="text-foreground">{m.type === 'in' ? '+' : m.type === 'out' ? '-' : '±'}{m.quantity}</strong></span>
   748→                  <span>{m.previousStock} → {m.newStock}</span>
   749→                </div>
   750→              </div>
   751→              {m.reason && (
   752→                <p className="text-xs text-muted-foreground mt-1 truncate">{m.reason}</p>
   753→              )}
   754→            </Card>
   755→          ))}
   756→        </div>
   757→      </div>
   758→      )}
   759→    </div>
