    33→function formatETB(amount: number): string {
    34→  return new Intl.NumberFormat('en-ET', {
    35→    style: 'decimal',
    36→    minimumFractionDigits: 2,
    37→    maximumFractionDigits: 2,
    38→  }).format(amount) + ' ETB'
    39→}
    40→
    41→function getStockStatus(quantity: number, threshold: number): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    42→  if (quantity === 0) return { label: 'Out of Stock', variant: 'destructive' }
    43→  if (quantity <= threshold) return { label: 'Low Stock', variant: 'secondary' }
    44→  return { label: 'In Stock', variant: 'default' }
    45→}
    46→
    47→function formatDate(dateStr: string): string {
    48→  return new Date(dateStr).toLocaleDateString('en-US', {
    49→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit'
    50→  })
    51→}

    86→// ============================================
    87→// Stock Adjustment Dialog
    88→// ============================================
    89→function StockAdjustmentDialog({
    90→  open,
    91→  onClose,
    92→  product,
    93→  orgId,
    94→  shopId,
    95→  onAdjusted,
    96→}: {
    97→  open: boolean
    98→  onClose: () => void
    99→  product: Product | null
   100→  orgId: string
   101→  shopId?: string
   102→  onAdjusted: () => void
   103→}) {
   104→  const [loading, setLoading] = useState(false)
   105→
   106→  const form = useForm<StockAdjustmentFormData>({
   107→    resolver: zodResolver(stockAdjustmentSchema),
   108→    defaultValues: { type: 'in', quantity: '' as unknown as number, reason: '' },
   109→  })
   110→
   111→  const watchedType = form.watch('type')
   112→  const watchedQty = form.watch('quantity')
   113→  const qty = typeof watchedQty === 'number' ? watchedQty : (parseInt(String(watchedQty)) || 0)
   114→  const previewStock = watchedType === 'in'
   115→    ? (product?.quantity ?? 0) + qty
   116→    : watchedType === 'out'
   117→    ? Math.max(0, (product?.quantity ?? 0) - qty)
   118→    : qty
   119→
   120→  const handleSubmit = async (data: StockAdjustmentFormData) => {
   121→    if (!product) return
   122→
   123→    setLoading(true)
   124→    try {
   125→      await api.addStockMovement(orgId, {
   126→        productId: product.id,
   127→        type: data.type,
   128→        quantity: data.quantity,
   129→        reason: data.reason.trim(),
   130→        shopId,
   131→      })
   132→      toast.success('Stock adjusted successfully')
   133→      onAdjusted()
   134→      onClose()
   135→    } catch (err: unknown) {
   136→      toast.error(getNetworkErrorMessage(err))
   137→    } finally {
   138→      setLoading(false)
   139→    }
   140→  }
   141→
   142→  useEffect(() => {
   143→    if (!open) {
   144→      form.reset({ type: 'in', quantity: '' as unknown as number, reason: '' })
   145→    }
   146→  }, [open, form])
   147→
   148→  if (!product) return null
   149→
   150→  return (
   151→    <Dialog open={open} onOpenChange={onClose}>
   152→      <DialogContent className="sm:max-w-md">
   153→        <DialogHeader>
   154→          <DialogTitle>Adjust Stock</DialogTitle>
   155→          <DialogDescription>
   156→            Adjust stock for <span className="font-semibold">{product.name}</span> (Current: {product.quantity})
   157→          </DialogDescription>
   158→        </DialogHeader>
   159→        <Form {...form}>
   160→          <form onSubmit={form.handleSubmit(handleSubmit)} className="space-y-4 py-2">
   161→            <FormSelectField
   162→              name="type"
   163→              label="Adjustment Type"
   164→              options={[
   165→                { value: 'in', label: 'Add Stock', icon: <Plus className="size-3" /> },
   166→                { value: 'out', label: 'Remove Stock', icon: <Minus className="size-3" /> },
   167→                { value: 'adjustment', label: 'Adjustment (Set)', icon: <RefreshCw className="size-3" /> },
   168→              ]}
   169→            />
   170→            <FormInputField
   171→              name="quantity"
   172→              label="Quantity"
   173→              placeholder="Enter quantity"
   174→              type="number"
   175→              required
   176→              min={1}
   177→            />
   178→            <FormInputField
   179→              name="reason"
   180→              label="Reason"
   181→              placeholder="Enter reason for adjustment"
   182→              required
   183→            />
   184→            {qty > 0 && (
   185→              <div className="rounded-lg border bg-muted/50 p-3">
   186→                <p className="text-sm text-muted-foreground">Preview</p>
   187→                <p className="text-lg font-bold">
   188→                  {product.quantity} → {previewStock}
   189→                </p>
   190→              </div>
   191→            )}
   192→            <DialogFooter>
   193→              <Button type="button" variant="outline" onClick={onClose}>Cancel</Button>
   194→              <FormSubmitButton isLoading={loading} loadingText="Adjusting...">
   195→                Confirm Adjustment
   196→              </FormSubmitButton>
   197→            </DialogFooter>
   198→          </form>
   199→        </Form>
   200→      </DialogContent>
   201→    </Dialog>
   202→  )
   203→}

   206→// All Products Tab
   207→// ============================================
   208→function AllProductsTab({
   209→  orgId,
   210→  shopId,
   211→  onAdjust,
   212→}: {
   213→  orgId: string
   214→  shopId?: string
   215→  onAdjust: (product: Product) => void
   216→}) {
   217→  const [products, setProducts] = useState<Product[]>([])
   218→  const [productTypes, setProductTypes] = useState<ProductType[]>([])
   219→  const [loading, setLoading] = useState(true)
   220→  const [search, setSearch] = useState('')
   221→  const [filterType, setFilterType] = useState<string>('all')
   222→  const [sortCol, setSortCol] = useState<string>('name')
   223→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   224→  const [page, setPage] = useState(1)
   225→  const [totalPages, setTotalPages] = useState(1)
   226→
   227→  const fetchData = useCallback(async () => {
   228→    try {
   229→      const [productsRes, typesRes] = await Promise.all([
   230→        api.getProducts(orgId, {
   231→          search: search || undefined,
   232→          productTypeId: filterType !== 'all' ? filterType : undefined,
   233→          page,
   234→          limit: 10,
   235→          shopId,
   236→        }),
   237→        api.getProductTypes(orgId, shopId),
   238→      ])
   239→      setProducts(productsRes.products)
   240→      setTotalPages(productsRes.pagination.totalPages)
   241→      setProductTypes(typesRes.productTypes)
   242→    } catch (error) {
   243→      toast.error(getNetworkErrorMessage(error))
   244→    } finally {
   245→      setLoading(false)
   246→    }
   247→  }, [orgId, shopId, search, filterType, page])
   248→
   249→  useEffect(() => {
   250→    fetchData()
   251→  }, [fetchData])
   252→
   253→  const handleSort = (col: string) => {
   254→    if (sortCol === col) {
   255→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   256→    } else {
   257→      setSortCol(col)
   258→      setSortDir('asc')
   259→    }
   260→  }
   261→
   262→  const sortedProducts = [...products].sort((a, b) => {
   263→    const dir = sortDir === 'asc' ? 1 : -1
   264→    switch (sortCol) {
   265→      case 'name': return a.name.localeCompare(b.name) * dir
   266→      case 'sku': return (a.sku || '').localeCompare(b.sku || '') * dir
   267→      case 'quantity': return (a.quantity - b.quantity) * dir
   268→      case 'costPrice': return (a.costPrice - b.costPrice) * dir
   269→      case 'sellingPrice': return (a.sellingPrice - b.sellingPrice) * dir
   270→      default: return 0
   271→    }
   272→  })
   273→
   274→  const SortHeader = ({ col, children }: { col: string; children: React.ReactNode }) => (
   275→    <TableHead className="cursor-pointer select-none hover:bg-muted/50" onClick={() => handleSort(col)}>
   276→      <span className="flex items-center gap-1">
   277→        {children}
   278→        <ArrowUpDown className={`size-3 ${sortCol === col ? 'opacity-100' : 'opacity-30'}`} />
   279→      </span>
   280→    </TableHead>
   281→  )
   282→
   283→  if (loading) {
   284→    return (
   285→      <div className="space-y-4">
   286→        <div className="flex flex-col sm:flex-row gap-3">
   287→          <Skeleton className="h-10 flex-1" />
   288→          <Skeleton className="h-10 w-40" />
   289→        </div>
   290→        {Array.from({ length: 5 }).map((_, i) => (
   291→          <Skeleton key={i} className="h-14 w-full" />
   292→        ))}
   293→      </div>
   294→    )
   295→  }
   296→
   297→  return (
   298→    <div className="space-y-4">
   299→      {/* Filters */}
   300→      <div className="flex flex-col sm:flex-row gap-3">
   301→        <div className="relative flex-1">
   302→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   303→          <Input
   304→            placeholder="Search products..."
   305→            className="pl-9"
   306→            value={search}
   307→            onChange={(e) => { setSearch(e.target.value); setPage(1) }}
   308→          />
   309→        </div>
   310→        <Select value={filterType} onValueChange={(v) => { setFilterType(v); setPage(1) }}>
   311→          <SelectTrigger className="w-full sm:w-48">
   312→            <Filter className="size-4 mr-2" />
   313→            <SelectValue placeholder="Filter by type" />
   314→          </SelectTrigger>
   315→          <SelectContent>
   316→            <SelectItem value="all">All Types</SelectItem>
   317→            {productTypes.map((pt) => (
   318→              <SelectItem key={pt.id} value={pt.id}>{pt.name}</SelectItem>
   319→            ))}
   320→          </SelectContent>
   321→        </Select>
   322→      </div>
   323→
   324→      {/* Table */}
   325→      <div className="rounded-lg border overflow-hidden">
   326→        {/* Desktop table */}
   327→        <div className="hidden md:block overflow-x-auto">
   328→        <Table>
   329→          <TableHeader>
   330→            <TableRow>
   331→              <SortHeader col="name">Product Name</SortHeader>
   332→              <SortHeader col="sku">SKU</SortHeader>
   333→              <TableHead>Type</TableHead>
   334→              <SortHeader col="quantity">Quantity</SortHeader>
   335→              <SortHeader col="costPrice">Cost Price</SortHeader>
   336→              <SortHeader col="sellingPrice">Selling Price</SortHeader>
   337→              <TableHead>Value</TableHead>
   338→              <TableHead>Status</TableHead>
   339→              <TableHead className="text-right">Actions</TableHead>
   340→            </TableRow>
   341→          </TableHeader>
   342→          <TableBody>
   343→            {sortedProducts.length === 0 ? (
   344→              <TableRow>
   345→                <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   346→                  No products found
   347→                </TableCell>
   348→              </TableRow>
   349→            ) : (
   350→              sortedProducts.map((product) => {
   351→                const status = getStockStatus(product.quantity, product.lowStockThreshold)
   352→                return (
   353→                  <TableRow key={product.id}>
   354→                    <TableCell className="font-medium max-w-[180px] truncate">{product.name}</TableCell>
   355→                    <TableCell className="text-muted-foreground text-sm">{product.sku || '—'}</TableCell>
   356→                    <TableCell>
   357→                      <Badge variant="outline" className="text-xs">{product.productType?.name || '—'}</Badge>
   358→                    </TableCell>
   359→                    <TableCell className="font-mono">{product.quantity}</TableCell>
   360→                    <TableCell className="text-sm">{formatETB(product.costPrice)}</TableCell>
   361→                    <TableCell className="text-sm">{formatETB(product.sellingPrice)}</TableCell>
   362→                    <TableCell className="text-sm font-medium">{formatETB(product.quantity * product.sellingPrice)}</TableCell>
   363→                    <TableCell>
   364→                      <Badge variant={status.variant} className="text-xs">{status.label}</Badge>
   365→                    </TableCell>
   366→                    <TableCell className="text-right">
   367→                      <Button size="sm" variant="outline" onClick={() => onAdjust(product)}>
   368→                        Adjust Stock
   369→                      </Button>
   370→                    </TableCell>
   371→                  </TableRow>
   372→                )
   373→              })
   374→            )}
   375→          </TableBody>
   376→        </Table>
   377→        </div>
   378→        {/* Mobile card view */}
   379→        <div className="md:hidden space-y-3 p-3">
   380→          {sortedProducts.length === 0 ? (
   381→            <p className="text-center py-8 text-muted-foreground text-sm">No products found</p>
   382→          ) : (
   383→            sortedProducts.map((product) => {
   384→              const status = getStockStatus(product.quantity, product.lowStockThreshold)
   385→              return (
   386→                <Card key={product.id} className="p-4">
   387→                  <div className="flex items-start justify-between mb-2">
   388→                    <div className="min-w-0 flex-1">
   389→                      <p className="font-semibold text-sm truncate">{product.name}</p>
   390→                      <p className="text-xs text-muted-foreground">{product.productType?.name || '—'} {product.sku ? `· ${product.sku}` : ''}</p>
   391→                    </div>
   392→                    <Badge variant={status.variant} className="text-xs shrink-0 ml-2">{status.label}</Badge>
   393→                  </div>
   394→                  <div className="flex items-center justify-between text-xs mt-2">
   395→                    <div className="flex items-center gap-3 text-muted-foreground">
   396→                      <span>Qty: <strong className="text-foreground">{product.quantity}</strong></span>
   397→                      <span>{formatETB(product.sellingPrice)}</span>
   398→                    </div>
   399→                    <Button size="sm" variant="outline" className="h-7 text-xs" onClick={() => onAdjust(product)}>Adjust</Button>
   400→                  </div>
   401→                </Card>
   402→              )
   403→            })
   404→          )}
   405→        </div>
   406→      </div>
   407→
   408→      {/* Pagination */}
   409→      {totalPages > 1 && (
   410→        <div className="flex items-center justify-between">
   411→          <p className="text-sm text-muted-foreground">Page {page} of {totalPages}</p>
   412→          <div className="flex gap-2">
   413→            <Button size="sm" variant="outline" disabled={page <= 1} onClick={() => setPage(page - 1)}>
   414→              <ChevronLeft className="size-4" />
   415→            </Button>
   416→            <Button size="sm" variant="outline" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>
   417→              <ChevronRight className="size-4" />
   418→            </Button>
   419→          </div>
   420→        </div>
   421→      )}
   422→    </div>
   423→  )
   424→}

   429→function LowStockTab({
   430→  orgId,
   431→  shopId,
   432→  lowStockProducts,
   433→  onRestock,

   435→  orgId: string
   436→  shopId?: string

   438→  onRestock: (product: Product) => void
   439→}) {

   441→  const [loading, setLoading] = useState(true)
   442→
   443→  useEffect(() => {
   444→    const fetchLowStock = async () => {
   445→      try {
   446→        const data = await api.getInventory(orgId, shopId)
   447→        setProducts(data.lowStockProducts)
   448→      } catch (error) {
   449→      toast.error(getNetworkErrorMessage(error))
   450→      } finally {
   451→        setLoading(false)
   452→      }
   453→    }
   454→    fetchLowStock()
   455→  }, [orgId, shopId])
   456→
   457→  useEffect(() => {
   458→    if (lowStockProducts && lowStockProducts.length > 0) {
   459→      setProducts(lowStockProducts)
   460→      setLoading(false)
   461→    }
   462→  }, [lowStockProducts])
   463→
   464→  if (loading) {
   465→    return (
   466→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   467→        {Array.from({ length: 6 }).map((_, i) => (
   468→          <Skeleton key={i} className="h-32" />
   469→        ))}
   470→      </div>
   471→    )
   472→  }
   473→
   474→  if (products.length === 0) {
   475→    return (
   476→      <div className="text-center py-12">
   477→        <AlertTriangle className="size-12 text-emerald-500 mx-auto mb-3" />
   478→        <h3 className="text-base font-semibold">All Stocked Up!</h3>
   479→        <p className="text-muted-foreground">No products are currently low on stock.</p>
   480→      </div>
   481→    )
   482→  }
   483→
   484→  return (
   485→    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   486→      {products.map((product) => (
   487→        <Card key={product.id} className="border-amber-200 dark:border-amber-800">
   488→          <CardContent className="p-4">
   489→            <div className="flex items-start justify-between mb-2">
   490→              <div>
   491→                <h4 className="font-semibold text-sm truncate max-w-[200px]">{product.name}</h4>
   492→                <p className="text-xs text-muted-foreground">{product.productType.name}</p>
   493→              </div>
   494→              <Badge variant="secondary" className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300 shrink-0">
   495→                Low Stock
   496→              </Badge>
   497→            </div>
   498→            <Separator className="my-3" />
   499→            <div className="flex items-center justify-between text-sm">
   500→              <div>
   501→                <span className="text-muted-foreground">Current: </span>
   502→                <span className="font-bold text-amber-600 dark:text-amber-400">{product.quantity}</span>
   503→              </div>
   504→              <div>
   505→                <span className="text-muted-foreground">Threshold: </span>
   506→                <span className="font-medium">{product.lowStockThreshold}</span>
   507→              </div>
   508→            </div>
   509→            <Button
   510→              size="sm"
   511→              className="w-full mt-3"
   512→              onClick={() => onRestock({
   513→                id: product.id,
   514→                name: product.name,
   515→                quantity: product.quantity,
   516→                lowStockThreshold: product.lowStockThreshold,
   517→              } as Product)}
   518→            >
   519→              <Plus className="size-3 mr-1" /> Quick Restock
   520→            </Button>
   521→          </CardContent>
   522→        </Card>
   523→      ))}
   524→    </div>
   525→  )
   526→}

   531→function OutOfStockTab({
   532→  orgId,
   533→  shopId,
   534→  onRestock,
   535→}: {
   536→  orgId: string
   537→  shopId?: string
   538→  onRestock: (product: Product) => void
   539→}) {
   540→  const [products, setProducts] = useState<Product[]>([])
   541→  const [loading, setLoading] = useState(true)
   542→
   543→  useEffect(() => {
   544→    const fetchOutOfStock = async () => {
   545→      try {
   546→        const res = await api.getProducts(orgId, { limit: 100, shopId })
   547→        setProducts(res.products.filter(p => p.quantity === 0))
   548→      } catch (error) {
   549→      toast.error(getNetworkErrorMessage(error))
   550→      } finally {

   606→function StockHistoryTab({ orgId, shopId }: { orgId: string; shopId?: string }) {
   607→  const [movements, setMovements] = useState<StockMovement[]>([])
   608→  const [products, setProducts] = useState<Product[]>([])
   609→  const [loading, setLoading] = useState(true)
   610→  const [filterProduct, setFilterProduct] = useState<string>('all')
   611→  const [filterType, setFilterType] = useState<string>('all')
   612→
   613→  useEffect(() => {
   614→    const fetchHistory = async () => {
   615→      try {
   616→        const [invData, prodRes] = await Promise.all([
   617→          api.getInventory(orgId, shopId),
   618→          api.getProducts(orgId, { limit: 100, shopId }),
   619→        ])
   620→        setMovements(invData.recentMovements)
   621→        setProducts(prodRes.products)
   622→      } catch (error) {
   623→        toast.error(getNetworkErrorMessage(error))
   624→      } finally {
   625→        setLoading(false)
   626→      }
   627→    }
   628→    fetchHistory()
   629→  }, [orgId, shopId])
   630→
   631→  const filtered = movements.filter((m) => {
   632→    if (filterProduct !== 'all' && m.productId !== filterProduct) return false
   633→    if (filterType !== 'all' && m.type !== filterType) return false
   634→    return true
   635→  })
   636→
   637→  const typeLabel = (type: string) => {
   638→    switch (type) {
   639→      case 'in': return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300">Stock In</Badge>
   640→      case 'out': return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300">Stock Out</Badge>
   641→      case 'adjustment': return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300">Adjustment</Badge>
   642→      default: return <Badge variant="outline">{type}</Badge>
   643→    }
   644→  }
   645→
   646→  if (loading) {
   647→    return (
   648→      <div className="space-y-3">
   649→        <div className="flex flex-col sm:flex-row gap-3">
   650→          <Skeleton className="h-10 w-48" />
   651→          <Skeleton className="h-10 w-36" />
   652→        </div>
   653→        {Array.from({ length: 6 }).map((_, i) => (
   654→          <Skeleton key={i} className="h-14" />
   655→        ))}
   656→      </div>
   657→    )
   658→  }
   659→
   660→  return (
   661→    <div className="space-y-4">
   662→      {/* Filters */}
   663→      <div className="flex flex-col sm:flex-row gap-3">
   664→        <Select value={filterProduct} onValueChange={setFilterProduct}>
   665→          <SelectTrigger className="w-full sm:w-52">
   666→            <SelectValue placeholder="Filter by product" />
   667→          </SelectTrigger>
   668→          <SelectContent>
   669→            <SelectItem value="all">All Products</SelectItem>
   670→            {products.map((p) => (
   671→              <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   672→            ))}
   673→          </SelectContent>
   674→        </Select>
   675→        <Select value={filterType} onValueChange={setFilterType}>
   676→          <SelectTrigger className="w-full sm:w-40">
   677→            <SelectValue placeholder="Filter by type" />
   678→          </SelectTrigger>
   679→          <SelectContent>
   680→            <SelectItem value="all">All Types</SelectItem>
   681→            <SelectItem value="in">Stock In</SelectItem>
   682→            <SelectItem value="out">Stock Out</SelectItem>
   683→            <SelectItem value="adjustment">Adjustment</SelectItem>
   684→          </SelectContent>
   685→        </Select>
   686→      </div>
   687→
   688→      {/* Empty state or Table */}
   689→      {!loading && filtered.length === 0 ? (
   690→        <Card>
   691→          <CardContent className="py-12 text-center">
   692→            <History className="size-12 text-muted-foreground mx-auto mb-4" />
   693→            <h3 className="text-lg font-semibold mb-1">
   694→              {filterProduct !== 'all' || filterType !== 'all' ? 'No matching stock movements' : 'No stock movements yet'}
   695→            </h3>
   696→            <p className="text-muted-foreground text-sm mb-4">
   697→              {filterProduct !== 'all' || filterType !== 'all'
   698→                ? 'Try adjusting your filters'
   699→                : 'Stock movements will appear here when you adjust product quantities'}
   700→            </p>
   701→          </CardContent>
   702→        </Card>
   703→      ) : (
   704→      <div className="rounded-lg border overflow-hidden">
   705→        {/* Desktop table */}
   706→        <div className="hidden md:block overflow-x-auto">
   707→          <Table>
   708→            <TableHeader>
   709→              <TableRow>
   710→                <TableHead>Date</TableHead>
   711→                <TableHead>Product</TableHead>
   712→                <TableHead>Type</TableHead>
   713→                <TableHead className="text-right">Quantity</TableHead>
   714→                <TableHead className="text-right">Previous Stock</TableHead>

   716→                <TableHead>Reason</TableHead>
   717→              </TableRow>
   718→            </TableHeader>
   719→            <TableBody>
   720→              {filtered.map((m) => (

   722→                  <TableCell className="text-sm whitespace-nowrap">{formatDate(m.createdAt)}</TableCell>
   723→                  <TableCell className="font-medium text-sm">{m.product?.name || '—'}</TableCell>
   724→                  <TableCell>{typeLabel(m.type)}</TableCell>
   725→                  <TableCell className="text-right font-mono text-sm">

   729→                  <TableCell className="text-right font-mono text-sm font-medium">{m.newStock}</TableCell>

   732→              ))}
   733→            </TableBody>
   734→          </Table>
   735→        </div>
   736→        {/* Mobile card view */}
   737→        <div className="md:hidden space-y-3 p-3">
   738→          {filtered.map((m) => (
   739→            <Card key={m.id} className="p-4">
   740→              <div className="flex items-start justify-between mb-2">
   741→                <div className="min-w-0 flex-1">
   742→                  <p className="font-semibold text-sm truncate">{m.product?.name || '—'}</p>

   744→                </div>
   745→                {typeLabel(m.type)}
   746→              </div>
   747→              <div className="flex items-center justify-between text-xs mt-2">
   748→                <div className="flex items-center gap-3 text-muted-foreground">
   749→                  <span>Qty: <strong className="text-foreground">{m.type === 'in' ? '+' : m.type === 'out' ? '-' : '±'}{m.quantity}</strong></span>
   750→                  <span>{m.previousStock} → {m.newStock}</span>
   751→                </div>
   752→              </div>
