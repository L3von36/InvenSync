   104→function formatDate(dateStr: string | null | undefined): string {
   105→  if (!dateStr) return '—'
   106→  return new Date(dateStr).toLocaleDateString('en-US', {
   107→    year: 'numeric', month: 'short', day: 'numeric'
   108→  })
   109→}
   110→
   111→function formatDateTime(dateStr: string | null | undefined): string {
   112→  if (!dateStr) return '—'
   113→  return new Date(dateStr).toLocaleString('en-US', {
   114→    year: 'numeric', month: 'short', day: 'numeric',
   115→    hour: '2-digit', minute: '2-digit',
   116→  })
   117→}

   142→

   205→      setDetailLoading(false)

   207→  }
   208→

   211→    setAdjustAction(action)
   212→    setAdjustPoints('')
   213→    setAdjustReason('')

   235→          action: adjustAction,

   280→            Tier Thresholds
   281→          </CardTitle>
   282→        </CardHeader>
   283→        <CardContent>
   284→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   285→            {Object.entries(TIER_CONFIG).map(([key, config]) => (
   286→              <div key={key} className="flex items-center gap-2 rounded-lg border p-3">
   287→                <div className={`size-8 rounded-md flex items-center justify-center ${config.bgClass}`}>
   288→                  {config.icon}
   289→                </div>
   290→                <div>
   291→                  <p className="text-sm font-medium">{config.label}</p>
   292→                  <p className="text-xs text-muted-foreground">{config.minPoints}+ points</p>
   293→                </div>
   294→              </div>
   295→            ))}
   296→          </div>
   297→        </CardContent>
   298→      </Card>
   299→
   300→      {/* Stats Cards */}
   301→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   302→        <Card>
   303→          <CardContent className="p-4">
   304→            <div className="flex items-center gap-3">
   305→              <div className="size-10 rounded-lg bg-primary/10 flex items-center justify-center">
   306→                <Award className="size-5 text-primary" />
   307→              </div>

   312→            </div>
   313→          </CardContent>
   314→        </Card>

   316→          <CardContent className="p-4">

   319→                <Plus className="size-5 text-emerald-600" />
   320→              </div>
   321→              <div>
   322→                <p className="text-sm text-muted-foreground">Total Earned</p>

   324→              </div>
   325→            </div>
   326→          </CardContent>
   327→        </Card>

   330→            <div className="flex items-center gap-3">

   332→                <Minus className="size-5 text-red-600" />
   333→              </div>
   334→              <div>
   335→                <p className="text-sm text-muted-foreground">Total Redeemed</p>
   336→                <p className="text-base font-semibold">{totalRedeemed.toLocaleString()}</p>
   337→              </div>
   338→            </div>
   339→          </CardContent>
   340→        </Card>
   341→        <Card>
   342→          <CardContent className="p-4">
   343→            <div className="flex items-center gap-3">
   344→              <div className="size-10 rounded-lg bg-purple-100 dark:bg-purple-900/30 flex items-center justify-center">
   345→                <Gem className="size-5 text-purple-600" />
   346→              </div>
   347→              <div>
   348→                <p className="text-sm text-muted-foreground">Active Points</p>
   349→                <p className="text-base font-semibold">{totalPoints.toLocaleString()}</p>
   350→              </div>
   351→            </div>
   352→          </CardContent>

   362→          </div>
   363→        ))}
   364→      </div>
   365→
   366→      {/* Search */}
   367→      <div className="relative">
   368→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   369→        <Input
   370→          placeholder="Search by customer name..."
   371→          value={searchQuery}
   372→          onChange={(e) => setSearchQuery(e.target.value)}
   373→          className="pl-9"
   374→        />
   375→      </div>
   376→
   377→      {/* Content */}
   378→      {pageError ? (
   379→        <ErrorState title="Failed to load loyalty accounts" message={pageError} onRetry={fetchAccounts} />

   381→        <Card>
   382→          <CardContent className="p-6">
   383→            <div className="space-y-4">
   384→              {[...Array(5)].map((_, i) => (
   385→                <div key={i} className="flex items-center gap-4">
   386→                  <Skeleton className="size-10 rounded-full" />
   387→                  <div className="flex-1 space-y-2">
   388→                    <Skeleton className="h-4 w-40" />
   389→                    <Skeleton className="h-3 w-24" />
   390→                  </div>
   391→                  <Skeleton className="h-6 w-20" />
   392→                  <Skeleton className="h-4 w-16" />
   393→                </div>
   394→              ))}
   395→            </div>
   396→          </CardContent>
   397→        </Card>
   398→      ) : accounts.length === 0 ? (
   399→        <EmptyState
   400→          title="No loyalty accounts found"
   401→          message={searchQuery ? 'Try adjusting your search query' : 'Loyalty accounts will be created when customers earn points from purchases'}

   405→        <>
   406→          {/* Desktop Table */}

   409→              <Table>
   410→                <TableHeader>
   411→                  <TableRow>
   412→                    <TableHead>Customer</TableHead>
   413→                    <TableHead>Tier</TableHead>

   417→                    <TableHead className="text-right">Actions</TableHead>
   418→                  </TableRow>
   419→                </TableHeader>

   427→                            <p className="text-xs text-muted-foreground">{account.customer.phone}</p>

   453→                          <Button
   454→                            variant="ghost"
   455→                            size="icon"

   457→                            onClick={() => openAdjust(account, 'add')}

   459→                            <Plus className="size-4" />
   460→                          </Button>
   461→                          <Button
   462→                            variant="ghost"
   463→                            size="icon"
   464→                            className="size-8 text-red-600"
   465→                            onClick={() => openAdjust(account, 'deduct')}

   468→                          </Button>

   477→
   478→          {/* Mobile Cards */}
   479→          <div className="md:hidden space-y-3">
   480→            {accounts.map((account) => (

   482→                <CardContent className="p-4">

   504→                      onClick={() => openDetail(account)}

   506→                      <Eye className="size-3.5" />

   510→                      variant="outline"

   512→                      className="h-8 text-xs gap-1 text-emerald-600"

   514→                    >
   515→                      <Plus className="size-3.5" />
   516→                      Add
   517→                    </Button>
   518→                    <Button
   519→                      variant="outline"
   520→                      size="sm"
   521→                      className="h-8 text-xs gap-1 text-red-600"
   522→                      onClick={() => openAdjust(account, 'deduct')}
   523→                    >
   524→                      <Minus className="size-3.5" />

   527→                  </div>
   528→                </CardContent>

   531→          </div>
   532→
   533→          {/* Pagination */}

   561→
   562→      {/* ============================================ */}
   563→      {/* Detail Dialog */}

   565→      <Dialog open={showDetail} onOpenChange={setShowDetail}>
   566→        <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
