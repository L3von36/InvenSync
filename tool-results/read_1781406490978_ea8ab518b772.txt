   583→class ApiClient {
   584→  // Supabase Auth uses httpOnly cookies for session management.
   585→  // The token is no longer stored in localStorage for security.
   586→  // We keep a reference for backward compat but auth is primarily cookie-based.
   587→  private _token: string | null = null
   588→  // Guard flag: when true, 401 auto-logout is suppressed to prevent
   589→  // stale background requests from interfering with an active login attempt.
   590→  private _isLoggingIn: boolean = false
   591→
   592→  // Request deduplication: prevent concurrent identical GET requests
   593→  // If 3 components request the same endpoint simultaneously, only 1 fetch is made.
   594→  private _inflightRequests = new Map<string, Promise<unknown>>()
   595→
   596→  // Short-lived client-side response cache for GET requests (5s TTL)
   597→  // Prevents rapid re-fetches when multiple components mount simultaneously
   598→  private _responseCache = new Map<string, { data: unknown; expiresAt: number }>()
   599→  private static RESPONSE_CACHE_TTL = 5_000 // 5 seconds
   600→
   601→  /**
   602→   * Invalidate client-side response cache entries matching a prefix.
   603→   * Called after mutating operations to ensure stale data is not served.
   604→   */
   605→  invalidateCache(prefix: string): void {
   606→    // Invalidate our internal response cache
   607→    this._responseCache.forEach((_, key) => {
   608→      if (key.startsWith(prefix)) {
   609→        this._responseCache.delete(key)
   610→      }
   611→    })
   612→    // Also invalidate the shared client-cache (used by swrFetch)
   613→    try {
   614→      import('@/lib/client-cache').then(({ invalidateKeysByPrefix }) => {
   615→        invalidateKeysByPrefix(prefix)
   616→      }).catch(() => {
   617→        // client-cache may not be available during SSR
   618→      })
   619→    } catch {
   620→      // client-cache may not be available during SSR
   621→    }
   622→  }
   623→
   624→  private getToken(): string | null {
   625→    if (typeof window === 'undefined') return null
   626→    // Check memory first, then fallback to localStorage (for migration)
   627→    return this._token || localStorage.getItem('sb_token')
   628→  }
   629→
   630→  setToken(token: string): void {
   631→    if (typeof window === 'undefined') return
   632→    this._token = token
   633→    // Also keep in localStorage for migration period
   634→    localStorage.setItem('sb_token', token)
   635→  }
   636→
   637→  clearToken(): void {
   638→    if (typeof window === 'undefined') return
   639→    this._token = null
   640→    localStorage.removeItem('sb_token')
   641→  }
   642→
   643→  private async request<T>(
   644→    endpoint: string,
   645→    options: RequestInit = {}
   646→  ): Promise<T> {
   647→    const token = this.getToken()
   648→    const headers: Record<string, string> = {
   649→      'Content-Type': 'application/json',
   650→      ...(options.headers as Record<string, string>),
   651→    }
   652→
   653→    // Add JWT token to Authorization header if available
   654→    if (token) {
   655→      headers['Authorization'] = `Bearer ${token}`
   656→    }
   657→
   658→    // Offline interceptor: queue mutating requests, reject GET requests
   659→    const method = (options.method || 'GET').toUpperCase()
   660→    const isMutating = ['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)
   661→    const isGet = method === 'GET'
   662→
   663→    // ---- Request deduplication for GET requests ----
   664→    // If an identical GET request is already in-flight, return the same promise
   665→    // instead of firing a duplicate network request.
   666→    if (isGet) {
   667→      const cacheKey = `${method}:${endpoint}`
   668→
   669→      // Check short-lived response cache first
   670→      const cached = this._responseCache.get(cacheKey)
   671→      if (cached && Date.now() < cached.expiresAt) {
   672→        return cached.data as T
   673→      }
   674→
   675→      // Check for in-flight deduplication
   676→      const inflight = this._inflightRequests.get(cacheKey)
   677→      if (inflight) {
   678→        return inflight as Promise<T>
   679→      }
   680→    }
   681→
   682→    if (typeof window !== 'undefined' && !navigator.onLine) {
   683→      if (isMutating) {
   684→        // Queue the operation for later sync
   685→        try {
   686→          const { queueOperation } = await import('@/lib/offline-queue')
   687→          await queueOperation(
   688→            endpoint,
   689→            method,
   690→            headers,
   691→            options.body ? String(options.body) : null
   692→          )
   693→          // Return a mock "offline queued" response so callers don't break
   694→          return { offline: true, queued: true } as unknown as T
   695→        } catch {
   696→          // If queuing fails, fall through to normal error
   697→        }
   698→      }
   699→      throw new Error('You are offline. Please check your internet connection and try again.')
   700→    }
   701→
   702→    // Add timeout (30s) to prevent hanging requests
   703→    const controller = new AbortController()
   704→    const timeoutId = setTimeout(() => controller.abort(), 30000)
   705→    const signal = options.signal || controller.signal
   706→
   707→    // For GET requests, register the inflight promise for deduplication
   708→    const cacheKey = `${method}:${endpoint}`
   709→    let isInflightRegistered = false
   710→
   711→    if (isGet) {
   712→      isInflightRegistered = true
   713→    }
   714→
   715→    // Create the fetch promise (may be shared across callers)
   716→    const fetchPromise = (async () => {
   717→      let response: Response
   718→      try {
   719→        response = await fetch(endpoint, {
   720→          ...options,
   721→          headers,
   722→          signal,
   723→        })
   724→      } catch (err) {
   725→        clearTimeout(timeoutId)
   726→        // Clean up inflight registration
   727→        if (isInflightRegistered) {
   728→          this._inflightRequests.delete(cacheKey)
   729→        }
   730→        // If we got a network error during a mutating request, try to queue it
   731→        if (err instanceof TypeError && err.message === 'Failed to fetch' && isMutating) {
   732→          try {
   733→            const { queueOperation } = await import('@/lib/offline-queue')
   734→            await queueOperation(
   735→              endpoint,
   736→              method,
   737→              headers,
   738→              options.body ? String(options.body) : null
   739→            )
   740→            return { offline: true, queued: true } as unknown as T
   741→          } catch {
   742→            // Fall through to original error handling
   743→          }
   744→        }
   745→        if (err instanceof DOMException && err.name === 'AbortError') {
   746→          throw new Error('Request timed out — the server took too long to respond. Please try again.')
   747→        }
   748→        if (err instanceof TypeError && err.message === 'Failed to fetch') {
   749→          throw new Error('Network error — please check your internet connection and try again.')
   750→        }
   751→        throw new Error(err instanceof Error ? err.message : 'An unexpected network error occurred')
   752→      }
   753→
   754→      clearTimeout(timeoutId)
   755→
   756→      // Handle non-JSON responses gracefully
   757→      let data: Record<string, unknown>
   758→      try {
   759→        data = await response.json() as Record<string, unknown>
   760→      } catch {
   761→        // Clean up inflight registration on parse error
   762→        if (isInflightRegistered) {
   763→          this._inflightRequests.delete(cacheKey)
   764→        }
   765→        if (!response.ok) {
   766→          // Map common HTTP error codes to user-friendly messages — never expose raw status codes
   767→          if (response.status === 401) throw new Error('Your session has expired. Please log in again.')
   768→          if (response.status === 403) throw new Error('You do not have permission to perform this action')
   769→          if (response.status === 404) throw new Error('The requested resource was not found')
   770→          if (response.status === 429) throw new Error('Too many requests — please wait a moment and try again')
   771→          if (response.status >= 500) throw new Error('Server error — please try again later')
   772→          throw new Error('An unexpected error occurred. Please try again.')
   773→        }
   774→        throw new Error('Server returned an invalid response')
   775→      }
   776→
   777→      // Clean up inflight registration after response is parsed
   778→      if (isInflightRegistered) {
   779→        this._inflightRequests.delete(cacheKey)
   780→      }
   781→
   782→      if (!response.ok) {
   783→        const errorMessage = (data.error as string) || (data.message as string) || 'An unexpected error occurred. Please try again.'
   784→        const errorCode = data.code as string | undefined
   785→
   786→        // Handle database unreachable errors
   787→        if (response.status === 503 && errorCode === 'DB_UNREACHABLE') {
   788→          const dbError = new Error('Database is unreachable') as Error & { code: string }
   789→          dbError.code = 'DB_UNREACHABLE'
   790→          throw dbError
   791→        }
   792→
   793→        // Handle auth errors — clear token and trigger logout if 401 from a non-auth endpoint
   794→        // Auth endpoints (login, register, me) legitimately return 401 for wrong credentials
   795→        // Also suppress auto-logout during active login to prevent race conditions
   796→        const isAuthEndpoint = endpoint.startsWith('/api/auth/login') || endpoint.startsWith('/api/auth/register') || endpoint.startsWith('/api/auth/me')
   797→        if (response.status === 401 && token && !isAuthEndpoint && !this._isLoggingIn) {
   798→          // Only clear token and trigger logout if the token we sent is STILL the current token.
   799→          // This prevents a stale background request from clearing a freshly-set token
   800→          // after the user logged out and logged back in.
   801→          const currentToken = this.getToken()
   802→          if (token === currentToken) {
   803→            this.clearToken()
   804→            // Trigger logout via auth store to update UI state
   805→            try {
   806→              const { useAuthStore } = await import('@/lib/stores/auth-store')
   807→              const store = useAuthStore.getState()
   808→              if (store.isAuthenticated) {
   809→                // Use setTimeout to avoid race conditions with concurrent requests.
   810→                // If multiple 401s come in at the same time, only the first one will
   811→                // trigger logout (subsequent ones will see isAuthenticated = false).
   812→                setTimeout(() => {
   813→                  const currentStore = useAuthStore.getState()
   814→                  if (currentStore.isAuthenticated) {
   815→                    currentStore.logout()
   816→                  }
   817→                }, 100)
   818→              }
   819→            } catch {
   820→              // If import fails (e.g. during SSR), just clear the token
   821→            }
   822→          }
   823→          throw new Error('Your session has expired. Please log in again.')
   824→        }
   825→
   826→        throw new Error(errorMessage)
   827→      }
   828→
   829→      // Cache successful GET responses for short TTL
   830→      if (isGet) {
   831→        this._responseCache.set(cacheKey, {
   832→          data: data as T,
   833→          expiresAt: Date.now() + ApiClient.RESPONSE_CACHE_TTL,
   834→        })
   835→      }
   836→
   837→      return data as T
   838→    })()
   839→
   840→    // Register the inflight promise for GET deduplication
   841→    if (isGet) {
   842→      this._inflightRequests.set(cacheKey, fetchPromise)
   843→    }
   844→
   845→    return fetchPromise
   846→  }
