     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import { useForm } from 'react-hook-form'
     5→import { zodResolver } from '@hookform/resolvers/zod'
     6→import {
     7→  Users, UserPlus, Search, Eye, Pencil, Trash2,
     8→  TrendingUp, CreditCard, Calendar, Mail, Phone, MapPin,
     9→  ShoppingCart, DollarSign, UserCircle, ChevronLeft, ChevronRight, Loader2,
    10→} from 'lucide-react'
    11→import { api, type Customer, type Sale, type Debt } from '@/lib/api-client'
    12→import { useAuthStore } from '@/lib/stores/auth-store'
    13→import { customerSchema, type CustomerFormData } from '@/lib/validations'
    14→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Badge } from '@/components/ui/badge'
    18→import {
    19→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
    20→} from '@/components/ui/table'
    21→import {
    22→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription
    23→} from '@/components/ui/dialog'
    24→import {
    25→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    26→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    27→} from '@/components/ui/alert-dialog'
    28→import { Form } from '@/components/ui/form'
    29→import { FormInputField, FormTextareaField, FormSubmitButton } from '@/components/shared/form-fields'
    30→import { Separator } from '@/components/ui/separator'
    31→import { Skeleton } from '@/components/ui/skeleton'
    32→import { toast } from 'sonner'
    33→import { getNetworkErrorMessage } from '@/lib/validation'
    34→import { ErrorState, EmptyState } from '@/components/shared/error-states'
    35→
    36→// ============================================
    37→// Helpers
    38→// ============================================
    39→import { formatETB, formatDate } from '@/lib/format'
    40→
    41→// ============================================
    42→// Main Component
    43→// ============================================
    44→export function CustomersPage() {
    45→  const { currentOrg, currentShop } = useAuthStore()
    46→  const orgId = currentOrg?.id || ''
    47→  const shopId = currentShop?.id
    48→
    49→  const [customers, setCustomers] = useState<Customer[]>([])
    50→  const [loading, setLoading] = useState(true)
    51→  const [pageError, setPageError] = useState<string | null>(null)
    52→  const [searchQuery, setSearchQuery] = useState('')
    53→  const [page, setPage] = useState(1)
    54→  const [totalPages, setTotalPages] = useState(1)
    55→  const [total, setTotal] = useState(0)
    56→  const limit = 10
    57→
    58→  // Dialogs
    59→  const [showAddEdit, setShowAddEdit] = useState(false)
    60→  const [showDetail, setShowDetail] = useState(false)
    61→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false)
    62→  const [selectedCustomer, setSelectedCustomer] = useState<Customer | null>(null)
    63→  const [isEditing, setIsEditing] = useState(false)
    64→
    65→  // Form state
    66→  const [formSubmitting, setFormSubmitting] = useState(false)
    67→  const [deleting, setDeleting] = useState(false)
    68→
    69→  const form = useForm<CustomerFormData>({
    70→    resolver: zodResolver(customerSchema),
    71→    defaultValues: {
    72→      name: '',
    73→      email: '',
    74→      phone: '',
    75→      address: '',
    76→    },
    77→  })
    78→
    79→  // Customer detail data
    80→  const [customerSales, setCustomerSales] = useState<Sale[]>([])
    81→  const [customerDebts, setCustomerDebts] = useState<Debt[]>([])
    82→  const [detailLoading, setDetailLoading] = useState(false)
    83→  const [totalDebt, setTotalDebt] = useState(0)
    84→
    85→  // Stats
    86→  const [newThisMonth, setNewThisMonth] = useState(0)
    87→
    88→  const fetchCustomers = useCallback(async () => {
    89→    if (!orgId) return
    90→    setLoading(true)
    91→    setPageError(null)
    92→    try {
    93→      const data = await api.getCustomers(orgId, {
    94→        search: searchQuery || undefined,
    95→        page,
    96→        limit,
    97→        shopId,
    98→      })
    99→      setCustomers(data.customers)
   100→      setTotalPages(data.pagination.totalPages)
   101→      setTotal(data.pagination.total)
   102→
   103→      // Calculate new this month from all customers (fetch all)
   104→      if (page === 1 && !searchQuery) {
   105→        try {
   106→          const allData = await api.getCustomers(orgId, { limit: 100, shopId })
   107→          const now = new Date()
   108→          const newCount = allData.customers.filter((c) => {
   109→            const created = new Date(c.createdAt)
   110→            return created.getMonth() === now.getMonth() && created.getFullYear() === now.getFullYear()
   111→          }).length
   112→          setNewThisMonth(newCount)
   113→
   114→          // Total debt from debts API
   115→          const debtsData = await api.getDebts(orgId, { type: 'customer_debt', status: 'all', limit: '100', shopId })
   116→          const outstandingDebt = debtsData.debts
   117→            .filter((d) => d.status !== 'paid')
   118→            .reduce((sum, d) => sum + (d.amount - d.paidAmount), 0)
   119→          setTotalDebt(outstandingDebt)
   120→        } catch {
   121→          // Partial data load — stats may be incomplete but customers still show
   122→        }
   123→      }
   124→    } catch (err) {
   125→      const msg = getNetworkErrorMessage(err)
   126→      setPageError(msg)
   127→      toast.error(msg)
   128→    } finally {
   129→      setLoading(false)
   130→    }
   131→  }, [orgId, searchQuery, page, shopId])
   132→
   133→  useEffect(() => {
   134→    fetchCustomers()
   135→  }, [fetchCustomers])
   136→
   137→  // Reset page when search changes
   138→  useEffect(() => {
   139→    setPage(1)
   140→  }, [searchQuery])
   141→
   142→  // ============================================
   143→  // Handlers
   144→  // ============================================
   145→  const openAddDialog = () => {
   146→    setIsEditing(false)
   147→    setSelectedCustomer(null)
   148→    form.reset({ name: '', email: '', phone: '', address: '' })
   149→    setShowAddEdit(true)
   150→  }
   151→
   152→  const openEditDialog = (customer: Customer) => {
   153→    setIsEditing(true)
   154→    setSelectedCustomer(customer)
   155→    form.reset({
   156→      name: customer.name,
   157→      email: customer.email || '',
   158→      phone: customer.phone || '',
   159→      address: customer.address || '',
   160→    })
   161→    setShowAddEdit(true)
   162→  }
   163→
   164→  const handleSave = async (data: CustomerFormData) => {
   165→    if (!orgId) return
   166→
   167→    setFormSubmitting(true)
   168→    try {
   169→      if (isEditing && selectedCustomer) {
   170→        await api.updateCustomer(selectedCustomer.id, orgId, {
   171→          name: data.name.trim(),
   172→          email: data.email?.trim() || undefined,
   173→          phone: data.phone?.trim() || undefined,
   174→          address: data.address?.trim() || undefined,
   175→        })
   176→        toast.success('Customer updated successfully')
   177→      } else {
   178→        await api.createCustomer(orgId, {
   179→          name: data.name.trim(),
   180→          email: data.email?.trim() || undefined,
   181→          phone: data.phone?.trim() || undefined,
   182→          address: data.address?.trim() || undefined,
   183→          shopId,
   184→        })
   185→        toast.success('Customer created successfully')
   186→      }
   187→      setShowAddEdit(false)
   188→      fetchCustomers()
   189→    } catch (err) {
   190→      toast.error(getNetworkErrorMessage(err))
   191→    } finally {
   192→      setFormSubmitting(false)
   193→    }
   194→  }
   195→
   196→  const openDeleteConfirm = (customer: Customer) => {
   197→    setSelectedCustomer(customer)
   198→    setShowDeleteConfirm(true)
   199→  }
   200→
   201→  const handleDelete = async () => {
   202→    if (!selectedCustomer || !orgId) return
   203→    setDeleting(true)
   204→    try {
   205→      await api.deleteCustomer(selectedCustomer.id, orgId)
   206→      toast.success('Customer deleted successfully')
   207→      setShowDeleteConfirm(false)
   208→      setSelectedCustomer(null)
   209→      fetchCustomers()
   210→    } catch (err) {
   211→      toast.error(getNetworkErrorMessage(err))
   212→    } finally {
   213→      setDeleting(false)
   214→    }
   215→  }
   216→
   217→  const openDetail = async (customer: Customer) => {
   218→    setSelectedCustomer(customer)
   219→    setShowDetail(true)
   220→    setDetailLoading(true)
   221→    try {
   222→      // Fetch sales for this customer
   223→      const salesData = await api.getSales(orgId, { limit: 50, shopId })
   224→      const filteredSales = salesData.sales.filter((s) => s.customerId === customer.id)
   225→      setCustomerSales(filteredSales)
   226→
   227→      // Fetch debts for this customer
   228→      const debtsData = await api.getDebts(orgId, { type: 'customer_debt', limit: '100', shopId })
   229→      const filteredDebts = debtsData.debts.filter((d) => d.customerId === customer.id)
   230→      setCustomerDebts(filteredDebts)
   231→    } catch (err) {
   232→      toast.error(getNetworkErrorMessage(err))
   233→    } finally {
   234→      setDetailLoading(false)
   235→    }
   236→  }
   237→
   238→  // Calculate active this month (customers with sales this month)
   239→  const activeThisMonth = customers.filter((c) => {
   240→    return c._count && c._count.sales > 0
   241→  }).length
   242→
   243→  // ============================================
   244→  // Render
   245→  // ============================================
   246→  return (
   247→    <div className="space-y-6">
   248→      {/* Header */}
   249→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   250→        <div>
   251→          <h1 className="text-xl sm:text-2xl font-bold tracking-tight">Customers</h1>
   252→          <p className="text-muted-foreground text-sm">Manage your customer database and track purchase history</p>
   253→        </div>
   254→        <Button onClick={openAddDialog} className="gap-2">
   255→          <UserPlus className="size-4" />
   256→          Add Customer
   257→        </Button>
   258→      </div>
   259→
   260→      {/* Stats Cards */}
   261→      <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   262→        <Card>
   263→          <CardContent className="p-4">
   264→            <div className="flex items-center gap-3">
   265→              <div className="size-10 rounded-lg bg-primary/10 flex items-center justify-center">
   266→                <Users className="size-5 text-primary" />
   267→              </div>
   268→              <div>
   269→                <p className="text-sm text-muted-foreground">Total Customers</p>
   270→                <p className="text-xl sm:text-2xl font-bold">{total}</p>
   271→              </div>
   272→            </div>
   273→          </CardContent>
   274→        </Card>
   275→        <Card>
   276→          <CardContent className="p-4">
   277→            <div className="flex items-center gap-3">
   278→              <div className="size-10 rounded-lg bg-brand-50 dark:bg-brand-900/20 flex items-center justify-center">
   279→                <TrendingUp className="size-5 text-primary" />
   280→              </div>
   281→              <div>
   282→                <p className="text-sm text-muted-foreground">Active This Month</p>
   283→                <p className="text-xl sm:text-2xl font-bold">{activeThisMonth}</p>
   284→              </div>
   285→            </div>
   286→          </CardContent>
   287→        </Card>
   288→        <Card>
   289→          <CardContent className="p-4">
   290→            <div className="flex items-center gap-3">
   291→              <div className="size-10 rounded-lg bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   292→                <CreditCard className="size-5 text-red-600 dark:text-red-400" />
   293→              </div>
   294→              <div>
   295→                <p className="text-sm text-muted-foreground">Total Debt</p>
   296→                <p className="text-xl sm:text-2xl font-bold">{formatETB(totalDebt)}</p>
   297→              </div>
   298→            </div>
   299→          </CardContent>
   300→        </Card>
   301→        <Card>
   302→          <CardContent className="p-4">
   303→            <div className="flex items-center gap-3">
   304→              <div className="size-10 rounded-lg bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center">
   305→                <Calendar className="size-5 text-amber-600 dark:text-amber-400" />
   306→              </div>
   307→              <div>
   308→                <p className="text-sm text-muted-foreground">New This Month</p>
   309→                <p className="text-xl sm:text-2xl font-bold">{newThisMonth}</p>
   310→              </div>
   311→            </div>
   312→          </CardContent>
   313→        </Card>
   314→      </div>
   315→
   316→      {/* Search */}
   317→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   318→        <div className="relative w-full sm:w-72">
   319→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   320→          <Input
   321→            placeholder="Search customers..."
   322→            aria-label="Search customers"
   323→            value={searchQuery}
   324→            onChange={(e) => setSearchQuery(e.target.value)}
   325→            className="pl-9"
   326→          />
   327→        </div>
   328→      </div>
   329→
   330→      {/* Table */}
   331→      {loading ? (
   332→        <Card>
   333→          <CardContent className="p-6 space-y-4">
   334→            {Array.from({ length: 6 }).map((_, i) => (
   335→              <Skeleton key={i} className="h-12 w-full" />
   336→            ))}
   337→          </CardContent>
   338→        </Card>
   339→      ) : pageError && customers.length === 0 ? (
   340→        <ErrorState title="Failed to load customers" message={pageError} onRetry={fetchCustomers} />
   341→      ) : customers.length === 0 ? (
   342→        <EmptyState
   343→          title={searchQuery ? 'No customers match your search' : 'No customers yet'}
   344→          message={searchQuery ? 'Try a different search term' : 'Add your first customer to get started.'}
   345→          icon={<Users className="size-7 text-muted-foreground" />}
   346→          action={!searchQuery ? { label: 'Add Customer', onClick: openAddDialog } : undefined}
   347→        />
   348→      ) : (
   349→        <Card>
   350→          <CardContent className="p-0">
   351→            {/* Desktop table */}
   352→            <div className="hidden md:block overflow-x-auto -mx-4 md:mx-0">
   353→              <Table>
   354→                <TableHeader>
   355→                  <TableRow>
   356→                    <TableHead>Name</TableHead>
   357→                    <TableHead className="hidden lg:table-cell">Email</TableHead>
   358→                    <TableHead>Phone</TableHead>
   359→                    <TableHead className="hidden lg:table-cell text-right">Purchases</TableHead>
   360→                    <TableHead className="text-right">Outstanding Debt</TableHead>
   361→                    <TableHead className="hidden lg:table-cell">Created</TableHead>
   362→                    <TableHead className="text-right">Actions</TableHead>
   363→                  </TableRow>
   364→                </TableHeader>
   365→                <TableBody>
   366→                  {customers.map((customer) => (
   367→                    <TableRow
   368→                      key={customer.id}
   369→                      className="cursor-pointer hover:bg-muted/50"
   370→                      onClick={() => openDetail(customer)}
   371→                    >
   372→                      <TableCell>
   373→                        <div className="flex items-center gap-3">
   374→                          <div className="size-8 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   375→                            <UserCircle className="size-4 text-primary" />
   376→                          </div>
   377→                          <div>
   378→                            <p className="font-medium">{customer.name}</p>
   379→                            <p className="text-xs text-muted-foreground md:hidden">
   380→                              {customer.email || customer.phone || ''}
   381→                            </p>
   382→                          </div>
   383→                        </div>
   384→                      </TableCell>
   385→                      <TableCell className="hidden lg:table-cell text-muted-foreground">
   386→                        {customer.email || '—'}
   387→                      </TableCell>
   388→                      <TableCell className="hidden md:table-cell text-muted-foreground">
   389→                        {customer.phone || '—'}
   390→                      </TableCell>
   391→                      <TableCell className="hidden lg:table-cell text-right">
   392→                        <Badge variant="secondary">
   393→                          {customer._count?.sales || 0} sales
   394→                        </Badge>
   395→                      </TableCell>
   396→                      <TableCell className="text-right">
   397→                        <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800">
   398→                          {formatETB(0)}
   399→                        </Badge>
   400→                      </TableCell>
   401→                      <TableCell className="hidden lg:table-cell text-muted-foreground text-sm">
   402→                        {formatDate(customer.createdAt)}
   403→                      </TableCell>
   404→                      <TableCell className="text-right">
   405→                        <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   406→                          <Button
   407→                            variant="ghost"
   408→                            size="sm"
   409→                            className="h-9 w-9 sm:h-8 sm:w-8"
   410→                            onClick={() => openDetail(customer)}
   411→                            title="View Details"
   412→                            aria-label={`View ${customer.name}`}
   413→                          >
   414→                            <Eye className="size-4" />
   415→                          </Button>
   416→                          <Button
   417→                            variant="ghost"
   418→                            size="sm"
   419→                            className="h-9 w-9 sm:h-8 sm:w-8"
   420→                            onClick={() => openEditDialog(customer)}
   421→                            title="Edit"
   422→                            aria-label={`Edit ${customer.name}`}
   423→                          >
   424→                            <Pencil className="size-4" />
   425→                          </Button>
   426→                          <Button
   427→                            variant="ghost"
   428→                            size="sm"
   429→                            className="h-9 w-9 sm:h-8 sm:w-8 text-destructive hover:text-destructive"
   430→                            onClick={() => openDeleteConfirm(customer)}
   431→                            title="Delete"
   432→                            aria-label={`Delete ${customer.name}`}
   433→                          >
   434→                            <Trash2 className="size-4" />
   435→                          </Button>
   436→                        </div>
   437→                      </TableCell>
   438→                    </TableRow>
   439→                  ))}
   440→                </TableBody>
   441→              </Table>
   442→            </div>
   443→            {/* Mobile card view */}
   444→            <div className="md:hidden space-y-3 p-3">
   445→              {customers.map((customer) => (
   446→                <Card
   447→                  key={customer.id}
   448→                  className="p-4 cursor-pointer hover:bg-muted/50 transition-colors"
   449→                  onClick={() => openDetail(customer)}
   450→                >
   451→                  <div className="flex items-start justify-between mb-2">
   452→                    <div className="flex items-center gap-3">
   453→                      <div className="size-9 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   454→                        <UserCircle className="size-5 text-primary" />
   455→                      </div>
   456→                      <div className="min-w-0">
   457→                        <p className="font-semibold text-sm truncate">{customer.name}</p>
   458→                        <p className="text-xs text-muted-foreground truncate">{customer.email || customer.phone || 'No contact'}</p>
   459→                      </div>
   460→                    </div>
   461→                    <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800 shrink-0 text-xs">
   462→                      {formatETB(0)}
   463→                    </Badge>
   464→                  </div>
   465→                  <div className="flex items-center justify-between mt-2">
   466→                    <span className="text-xs text-muted-foreground">{customer._count?.sales || 0} sales</span>
   467→                    <div className="flex gap-1" onClick={(e) => e.stopPropagation()}>
   468→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0" onClick={() => openEditDialog(customer)}><Pencil className="size-3.5" /></Button>
   469→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0 text-destructive" onClick={() => openDeleteConfirm(customer)}><Trash2 className="size-3.5" /></Button>
   470→                    </div>
   471→                  </div>
   472→                </Card>
   473→              ))}
   474→            </div>
   475→          </CardContent>
   476→        </Card>
   477→      )}
   478→
   479→      {/* Pagination */}
   480→      {totalPages > 1 && (
   481→        <div className="flex items-center justify-between">
   482→          <p className="text-xs sm:text-sm text-muted-foreground">
   483→            Page {page} of {totalPages}
   484→          </p>
   485→          <div className="flex items-center gap-2">
   486→            <Button
   487→              variant="outline"
   488→              size="sm"
   489→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   490→              disabled={page === 1}
   491→              className="h-9 min-w-9 sm:min-w-auto"
   492→            >
   493→              <ChevronLeft className="size-4" />
   494→              <span className="hidden sm:inline ml-1">Previous</span>
   495→            </Button>
   496→            <Button
   497→              variant="outline"
   498→              size="sm"
   499→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   500→              disabled={page === totalPages}
   501→              className="h-9 min-w-9 sm:min-w-auto"
   502→            >
   503→              <span className="hidden sm:inline mr-1">Next</span>
   504→              <ChevronRight className="size-4" />
   505→            </Button>
   506→          </div>
   507→        </div>
   508→      )}
   509→
   510→      {/* Add/Edit Customer Dialog */}
   511→      <Dialog open={showAddEdit} onOpenChange={setShowAddEdit}>
   512→        <DialogContent className="sm:max-w-md">
   513→          <DialogHeader>
   514→            <DialogTitle>{isEditing ? 'Edit Customer' : 'Add New Customer'}</DialogTitle>
   515→            <DialogDescription>
   516→              {isEditing ? 'Update customer information' : 'Enter details for the new customer'}
   517→            </DialogDescription>
   518→          </DialogHeader>
   519→          <Form {...form}>
   520→            <form onSubmit={form.handleSubmit(handleSave)} className="space-y-4">
   521→              <FormInputField
   522→                name="name"
   523→                label="Name"
   524→                placeholder="Customer name"
   525→                required
   526→              />
   527→              <FormInputField
   528→                name="email"
   529→                label="Email"
   530→                placeholder="customer@email.com"
   531→                type="email"
   532→              />
   533→              <FormInputField
   534→                name="phone"
   535→                label="Phone"
   536→                placeholder="+251 9XX XXX XXXX"
   537→                type="tel"
   538→              />
   539→              <FormTextareaField
   540→                name="address"
   541→                label="Address"
   542→                placeholder="Customer address..."
   543→                rows={3}
   544→              />
   545→              <DialogFooter>
   546→                <Button variant="outline" type="button" onClick={() => setShowAddEdit(false)}>
   547→                  Cancel
   548→                </Button>
   549→                <FormSubmitButton isLoading={formSubmitting} loadingText={isEditing ? 'Updating...' : 'Saving...'}>
   550→                  {isEditing ? 'Update Customer' : 'Add Customer'}
