     1→'use client'
     2→

     4→import { useForm } from 'react-hook-form'
     5→import { zodResolver } from '@hookform/resolvers/zod'
     6→import {
     7→  Users, UserPlus, Search, Eye, Pencil, Trash2,
     8→  TrendingUp, CreditCard, Calendar, Mail, Phone, MapPin,
     9→  ShoppingCart, DollarSign, UserCircle, ChevronLeft, ChevronRight, Loader2,
    10→} from 'lucide-react'
    11→import { api, type Customer, type Sale, type Debt } from '@/lib/api-client'
    12→import { useAuthStore } from '@/lib/stores/auth-store'
    13→import { customerSchema, type CustomerFormData } from '@/lib/validations'
    14→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Badge } from '@/components/ui/badge'
    18→import {
    19→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
    20→} from '@/components/ui/table'
    21→import {
    22→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription
    23→} from '@/components/ui/dialog'
    24→import {
    25→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    26→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    27→} from '@/components/ui/alert-dialog'
    28→import { Form } from '@/components/ui/form'
    29→import { FormInputField, FormTextareaField, FormSubmitButton } from '@/components/shared/form-fields'
    30→import { Separator } from '@/components/ui/separator'
    31→import { Skeleton } from '@/components/ui/skeleton'
    32→import { toast } from 'sonner'
    33→import { getNetworkErrorMessage } from '@/lib/validation'
    34→import { ErrorState } from '@/components/shared/error-states'
    35→
    36→// ============================================
    37→// Helpers
    38→// ============================================
    39→function formatETB(amount: number): string {
    40→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    41→}
    42→
    43→function formatDate(dateStr: string | null | undefined): string {
    44→  if (!dateStr) return '—'
    45→  return new Date(dateStr).toLocaleDateString('en-US', {
    46→    year: 'numeric', month: 'short', day: 'numeric'
    47→  })
    48→}
    49→
    50→// ============================================
    51→// Main Component
    52→// ============================================
    53→export function CustomersPage() {
    54→  const { currentOrg, currentShop } = useAuthStore()
    55→  const orgId = currentOrg?.id || ''
    56→  const shopId = currentShop?.id
    57→
    58→  const [customers, setCustomers] = useState<Customer[]>([])
    59→  const [loading, setLoading] = useState(true)
    60→  const [pageError, setPageError] = useState<string | null>(null)
    61→  const [searchQuery, setSearchQuery] = useState('')
    62→  const [page, setPage] = useState(1)
    63→  const [totalPages, setTotalPages] = useState(1)
    64→  const [total, setTotal] = useState(0)
    65→  const limit = 10
    66→
    67→  // Dialogs
    68→  const [showAddEdit, setShowAddEdit] = useState(false)
    69→  const [showDetail, setShowDetail] = useState(false)
    70→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false)
    71→  const [selectedCustomer, setSelectedCustomer] = useState<Customer | null>(null)
    72→  const [isEditing, setIsEditing] = useState(false)
    73→
    74→  // Form state
    75→  const [formSubmitting, setFormSubmitting] = useState(false)
    76→  const [deleting, setDeleting] = useState(false)
    77→
    78→  const form = useForm<CustomerFormData>({
    79→    resolver: zodResolver(customerSchema),
    80→    defaultValues: {
    81→      name: '',
    82→      email: '',
    83→      phone: '',
    84→      address: '',
    85→    },
    86→  })
    87→
    88→  // Customer detail data
    89→  const [customerSales, setCustomerSales] = useState<Sale[]>([])
    90→  const [customerDebts, setCustomerDebts] = useState<Debt[]>([])
    91→  const [detailLoading, setDetailLoading] = useState(false)
    92→  const [totalDebt, setTotalDebt] = useState(0)
    93→
    94→  // Stats
    95→  const [newThisMonth, setNewThisMonth] = useState(0)
    96→
    97→  const fetchCustomers = useCallback(async () => {
    98→    if (!orgId) return
    99→    setLoading(true)
   100→    setPageError(null)
   101→    try {
   102→      const data = await api.getCustomers(orgId, {
   103→        search: searchQuery || undefined,
   104→        page,
   105→        limit,
   106→        shopId,
   107→      })
   108→      setCustomers(data.customers)
   109→      setTotalPages(data.pagination.totalPages)
   110→      setTotal(data.pagination.total)
   111→
   112→      // Calculate new this month from all customers (fetch all)
   113→      if (page === 1 && !searchQuery) {
   114→        try {
   115→          const allData = await api.getCustomers(orgId, { limit: 100, shopId })
   116→          const now = new Date()
   117→          const newCount = allData.customers.filter((c) => {
   118→            const created = new Date(c.createdAt)
   119→            return created.getMonth() === now.getMonth() && created.getFullYear() === now.getFullYear()
   120→          }).length
   121→          setNewThisMonth(newCount)
   122→
   123→          // Total debt from debts API
   124→          const debtsData = await api.getDebts(orgId, { type: 'customer_debt', status: 'all', limit: '100', shopId })
   125→          const outstandingDebt = debtsData.debts
   126→            .filter((d) => d.status !== 'paid')
   127→            .reduce((sum, d) => sum + (d.amount - d.paidAmount), 0)
   128→          setTotalDebt(outstandingDebt)
   129→        } catch {
   130→          // Partial data load — stats may be incomplete but customers still show
   131→        }
   132→      }
   133→    } catch (err) {
   134→      const msg = getNetworkErrorMessage(err)
   135→      setPageError(msg)
   136→      toast.error(msg)
   137→    } finally {
   138→      setLoading(false)
   139→    }
   140→  }, [orgId, searchQuery, page, shopId])
   141→
   142→  useEffect(() => {
   143→    fetchCustomers()
   144→  }, [fetchCustomers])
   145→
   146→  // Reset page when search changes
   147→  useEffect(() => {
   148→    setPage(1)
   149→  }, [searchQuery])
   150→
   151→  // ============================================
   152→  // Handlers
   153→  // ============================================
   154→  const openAddDialog = () => {
   155→    setIsEditing(false)
   156→    setSelectedCustomer(null)
   157→    form.reset({ name: '', email: '', phone: '', address: '' })
   158→    setShowAddEdit(true)
   159→  }
   160→
   161→  const openEditDialog = (customer: Customer) => {
   162→    setIsEditing(true)
   163→    setSelectedCustomer(customer)
   164→    form.reset({
   165→      name: customer.name,
   166→      email: customer.email || '',
   167→      phone: customer.phone || '',
   168→      address: customer.address || '',
   169→    })
   170→    setShowAddEdit(true)
   171→  }
   172→
   173→  const handleSave = async (data: CustomerFormData) => {
   174→    if (!orgId) return
   175→
   176→    setFormSubmitting(true)
   177→    try {
   178→      if (isEditing && selectedCustomer) {
   179→        await api.updateCustomer(selectedCustomer.id, orgId, {
   180→          name: data.name.trim(),
   181→          email: data.email?.trim() || undefined,
   182→          phone: data.phone?.trim() || undefined,
   183→          address: data.address?.trim() || undefined,
   184→        })
   185→        toast.success('Customer updated successfully')
   186→      } else {
   187→        await api.createCustomer(orgId, {
   188→          name: data.name.trim(),
   189→          email: data.email?.trim() || undefined,
   190→          phone: data.phone?.trim() || undefined,
   191→          address: data.address?.trim() || undefined,
   192→          shopId,
   193→        })
   194→        toast.success('Customer created successfully')
   195→      }
   196→      setShowAddEdit(false)
   197→      fetchCustomers()
   198→    } catch (err) {
   199→      toast.error(getNetworkErrorMessage(err))
   200→    } finally {
   201→      setFormSubmitting(false)
   202→    }
   203→  }
   204→
   205→  const openDeleteConfirm = (customer: Customer) => {
   206→    setSelectedCustomer(customer)
   207→    setShowDeleteConfirm(true)
   208→  }
   209→
   210→  const handleDelete = async () => {
   211→    if (!selectedCustomer || !orgId) return
   212→    setDeleting(true)
   213→    try {
   214→      await api.deleteCustomer(selectedCustomer.id, orgId)
   215→      toast.success('Customer deleted successfully')
   216→      setShowDeleteConfirm(false)
   217→      setSelectedCustomer(null)
   218→      fetchCustomers()
   219→    } catch (err) {
   220→      toast.error(getNetworkErrorMessage(err))
   221→    } finally {
   222→      setDeleting(false)
   223→    }
   224→  }
   225→
   226→  const openDetail = async (customer: Customer) => {
   227→    setSelectedCustomer(customer)
   228→    setShowDetail(true)
   229→    setDetailLoading(true)
   230→    try {
   231→      // Fetch sales for this customer
   232→      const salesData = await api.getSales(orgId, { limit: 50, shopId })
   233→      const filteredSales = salesData.sales.filter((s) => s.customerId === customer.id)
   234→      setCustomerSales(filteredSales)
   235→
   236→      // Fetch debts for this customer
   237→      const debtsData = await api.getDebts(orgId, { type: 'customer_debt', limit: '100', shopId })
   238→      const filteredDebts = debtsData.debts.filter((d) => d.customerId === customer.id)
   239→      setCustomerDebts(filteredDebts)
   240→    } catch (err) {
   241→      toast.error(getNetworkErrorMessage(err))
   242→    } finally {
   243→      setDetailLoading(false)
   244→    }
   245→  }
   246→
   247→  // Calculate active this month (customers with sales this month)
   248→  const activeThisMonth = customers.filter((c) => {
   249→    return c._count && c._count.sales > 0
   250→  }).length
   251→
   252→  // ============================================
   253→  // Render
   254→  // ============================================
   255→  return (
   256→    <div className="space-y-6">
   257→      {/* Header */}
   258→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   259→        <div>
   260→          <h1 className="text-xl sm:text-2xl font-bold tracking-tight">Customers</h1>
   261→          <p className="text-muted-foreground text-sm">Manage your customer database and track purchase history</p>
   262→        </div>
   263→        <Button onClick={openAddDialog} className="gap-2">
   264→          <UserPlus className="size-4" />
   265→          Add Customer
   266→        </Button>
   267→      </div>
   268→
   269→      {/* Stats Cards */}
   270→      <div className="grid grid-cols-2 sm:grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   271→        <Card>
   272→          <CardContent className="p-4">
   273→            <div className="flex items-center gap-3">
   274→              <div className="size-10 rounded-lg bg-primary/10 flex items-center justify-center">
   275→                <Users className="size-5 text-primary" />
   276→              </div>
   277→              <div>
   278→                <p className="text-sm text-muted-foreground">Total Customers</p>
   279→                <p className="text-xl sm:text-2xl font-bold">{total}</p>
   280→              </div>
   281→            </div>
   282→          </CardContent>
   283→        </Card>
   284→        <Card>
   285→          <CardContent className="p-4">
   286→            <div className="flex items-center gap-3">
   287→              <div className="size-10 rounded-lg bg-brand-50 dark:bg-brand-900/20 flex items-center justify-center">
   288→                <TrendingUp className="size-5 text-primary" />
   289→              </div>
   290→              <div>
   291→                <p className="text-sm text-muted-foreground">Active This Month</p>
   292→                <p className="text-xl sm:text-2xl font-bold">{activeThisMonth}</p>
   293→              </div>
   294→            </div>
   295→          </CardContent>
   296→        </Card>
   297→        <Card>
   298→          <CardContent className="p-4">
   299→            <div className="flex items-center gap-3">
   300→              <div className="size-10 rounded-lg bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   301→                <CreditCard className="size-5 text-red-600 dark:text-red-400" />
   302→              </div>
   303→              <div>
   304→                <p className="text-sm text-muted-foreground">Total Debt</p>
   305→                <p className="text-xl sm:text-2xl font-bold">{formatETB(totalDebt)}</p>
   306→              </div>
   307→            </div>
   308→          </CardContent>
   309→        </Card>
   310→        <Card>
   311→          <CardContent className="p-4">
   312→            <div className="flex items-center gap-3">
   313→              <div className="size-10 rounded-lg bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center">
   314→                <Calendar className="size-5 text-amber-600 dark:text-amber-400" />
   315→              </div>
   316→              <div>
   317→                <p className="text-sm text-muted-foreground">New This Month</p>
   318→                <p className="text-xl sm:text-2xl font-bold">{newThisMonth}</p>
   319→              </div>
   320→            </div>
   321→          </CardContent>
   322→        </Card>
   323→      </div>
   324→
   325→      {/* Search */}
   326→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   327→        <div className="relative w-full sm:w-72">
   328→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   329→          <Input
   330→            placeholder="Search customers..."
   331→            value={searchQuery}
   332→            onChange={(e) => setSearchQuery(e.target.value)}
   333→            className="pl-9"
   334→          />
   335→        </div>
   336→      </div>
   337→
   338→      {/* Table */}
   339→      {loading ? (
   340→        <Card>
   341→          <CardContent className="p-6 space-y-4">
   342→            {Array.from({ length: 6 }).map((_, i) => (
   343→              <Skeleton key={i} className="h-12 w-full" />
   344→            ))}
   345→          </CardContent>
   346→        </Card>
   347→      ) : pageError && customers.length === 0 ? (
   348→        <ErrorState title="Failed to load customers" message={pageError} onRetry={fetchCustomers} />
   349→      ) : customers.length === 0 ? (
   350→        <Card>
   351→          <CardContent className="py-12 text-center">
   352→            <Users className="size-12 text-muted-foreground mx-auto mb-4" />
   353→            <p className="text-muted-foreground font-medium">
   354→              {searchQuery ? 'No customers match your search' : 'No customers yet'}
   355→            </p>
   356→            <p className="text-xs text-muted-foreground mt-1">
   357→              {searchQuery ? 'Try a different search term' : 'Add your first customer to get started'}
   358→            </p>
   359→            {!searchQuery && (
   360→              <Button onClick={openAddDialog} className="mt-4 gap-2" variant="outline">
   361→                <UserPlus className="size-4" />
   362→                Add Customer
   363→              </Button>
   364→            )}
   365→          </CardContent>
   366→        </Card>
   367→      ) : (
   368→        <Card>
   369→          <CardContent className="p-0">
   370→            {/* Desktop table */}
   371→            <div className="hidden md:block overflow-x-auto">
   372→              <Table>
   373→                <TableHeader>
   374→                  <TableRow>
   375→                    <TableHead>Name</TableHead>
   376→                    <TableHead className="hidden lg:table-cell">Email</TableHead>
   377→                    <TableHead>Phone</TableHead>
   378→                    <TableHead className="hidden lg:table-cell text-right">Purchases</TableHead>
   379→                    <TableHead className="text-right">Outstanding Debt</TableHead>
   380→                    <TableHead className="hidden lg:table-cell">Created</TableHead>
   381→                    <TableHead className="text-right">Actions</TableHead>
   382→                  </TableRow>
   383→                </TableHeader>
   384→                <TableBody>
   385→                  {customers.map((customer) => (
   386→                    <TableRow
   387→                      key={customer.id}
   388→                      className="cursor-pointer hover:bg-muted/50"
   389→                      onClick={() => openDetail(customer)}
   390→                    >
   391→                      <TableCell>
   392→                        <div className="flex items-center gap-3">
   393→                          <div className="size-8 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   394→                            <UserCircle className="size-4 text-primary" />
   395→                          </div>
   396→                          <div>
   397→                            <p className="font-medium">{customer.name}</p>
   398→                            <p className="text-xs text-muted-foreground md:hidden">
   399→                              {customer.email || customer.phone || ''}
   400→                            </p>
   401→                          </div>
   402→                        </div>
   403→                      </TableCell>
   404→                      <TableCell className="hidden lg:table-cell text-muted-foreground">
   405→                        {customer.email || '—'}
   406→                      </TableCell>
   407→                      <TableCell className="hidden md:table-cell text-muted-foreground">
   408→                        {customer.phone || '—'}
   409→                      </TableCell>
   410→                      <TableCell className="hidden lg:table-cell text-right">
   411→                        <Badge variant="secondary">
   412→                          {customer._count?.sales || 0} sales
   413→                        </Badge>
   414→                      </TableCell>
   415→                      <TableCell className="text-right">
   416→                        <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800">
   417→                          {formatETB(0)}
   418→                        </Badge>
   419→                      </TableCell>
   420→                      <TableCell className="hidden lg:table-cell text-muted-foreground text-sm">
   421→                        {formatDate(customer.createdAt)}
   422→                      </TableCell>
   423→                      <TableCell className="text-right">
   424→                        <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   425→                          <Button
   426→                            variant="ghost"
   427→                            size="sm"
   428→                            onClick={() => openDetail(customer)}
   429→                            title="View Details"
   430→                          >
   431→                            <Eye className="size-4" />
   432→                          </Button>
   433→                          <Button
   434→                            variant="ghost"
   435→                            size="sm"
   436→                            onClick={() => openEditDialog(customer)}
   437→                            title="Edit"
   438→                          >
   439→                            <Pencil className="size-4" />
   440→                          </Button>
   441→                          <Button
   442→                            variant="ghost"
   443→                            size="sm"
   444→                            onClick={() => openDeleteConfirm(customer)}
   445→                            title="Delete"
   446→                            className="text-destructive hover:text-destructive"
   447→                          >
   448→                            <Trash2 className="size-4" />
   449→                          </Button>
   450→                        </div>
   451→                      </TableCell>
   452→                    </TableRow>
   453→                  ))}
   454→                </TableBody>
   455→              </Table>
   456→            </div>
   457→            {/* Mobile card view */}
   458→            <div className="md:hidden space-y-3 p-3">
   459→              {customers.map((customer) => (
   460→                <Card
   461→                  key={customer.id}
   462→                  className="p-4 cursor-pointer hover:bg-muted/50 transition-colors"
   463→                  onClick={() => openDetail(customer)}
   464→                >
   465→                  <div className="flex items-start justify-between mb-2">
   466→                    <div className="flex items-center gap-3">
   467→                      <div className="size-9 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   468→                        <UserCircle className="size-5 text-primary" />
   469→                      </div>
   470→                      <div className="min-w-0">
   471→                        <p className="font-semibold text-sm truncate">{customer.name}</p>
   472→                        <p className="text-xs text-muted-foreground truncate">{customer.email || customer.phone || 'No contact'}</p>
   473→                      </div>
   474→                    </div>
   475→                    <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800 shrink-0 text-xs">
   476→                      {formatETB(0)}
   477→                    </Badge>
   478→                  </div>
   479→                  <div className="flex items-center justify-between mt-2">
   480→                    <span className="text-xs text-muted-foreground">{customer._count?.sales || 0} sales</span>
   481→                    <div className="flex gap-1" onClick={(e) => e.stopPropagation()}>
   482→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0" onClick={() => openEditDialog(customer)}><Pencil className="size-3.5" /></Button>
   483→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0 text-destructive" onClick={() => openDeleteConfirm(customer)}><Trash2 className="size-3.5" /></Button>
   484→                    </div>
   485→                  </div>
   486→                </Card>
   487→              ))}
   488→            </div>
   489→          </CardContent>
   490→        </Card>
   491→      )}
   492→
   493→      {/* Pagination */}
   494→      {totalPages > 1 && (
   495→        <div className="flex items-center justify-between">
   496→          <p className="text-xs sm:text-sm text-muted-foreground">
   497→            Page {page} of {totalPages}
   498→          </p>
   499→          <div className="flex items-center gap-2">
   500→            <Button
   501→              variant="outline"
   502→              size="sm"
   503→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   504→              disabled={page === 1}
   505→              className="h-9 min-w-9 sm:min-w-auto"
   506→            >
   507→              <ChevronLeft className="size-4" />
   508→              <span className="hidden sm:inline ml-1">Previous</span>
   509→            </Button>
   510→            <Button
   511→              variant="outline"
   512→              size="sm"
   513→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   514→              disabled={page === totalPages}
   515→              className="h-9 min-w-9 sm:min-w-auto"
   516→            >
   517→              <span className="hidden sm:inline mr-1">Next</span>
   518→              <ChevronRight className="size-4" />
   519→            </Button>
   520→          </div>
   521→        </div>
   522→      )}
   523→
   524→      {/* Add/Edit Customer Dialog */}
   525→      <Dialog open={showAddEdit} onOpenChange={setShowAddEdit}>
   526→        <DialogContent className="sm:max-w-md">
   527→          <DialogHeader>
   528→            <DialogTitle>{isEditing ? 'Edit Customer' : 'Add New Customer'}</DialogTitle>
   529→            <DialogDescription>
   530→              {isEditing ? 'Update customer information' : 'Enter details for the new customer'}
   531→            </DialogDescription>
   532→          </DialogHeader>
   533→          <Form {...form}>
   534→            <form onSubmit={form.handleSubmit(handleSave)} className="space-y-4">
   535→              <FormInputField
   536→                name="name"
   537→                label="Name"
   538→                placeholder="Customer name"
   539→                required
   540→              />
   541→              <FormInputField
   542→                name="email"
   543→                label="Email"
   544→                placeholder="customer@email.com"
   545→                type="email"
   546→              />
   547→              <FormInputField
   548→                name="phone"
   549→                label="Phone"
   550→                placeholder="+251 9XX XXX XXXX"
   551→                type="tel"
   552→              />
   553→              <FormTextareaField
   554→                name="address"
