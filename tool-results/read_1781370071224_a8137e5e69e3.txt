    57→// Sales Stats Cards
    58→// ============================================
    59→function SalesStatsCards({ sales }: { sales: Sale[] }) {
    60→  const today = new Date()
    61→  today.setHours(0, 0, 0, 0)
    62→
    63→  const startOfWeek = new Date(today)
    64→  startOfWeek.setDate(today.getDate() - today.getDay())
    65→
    66→  const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
    67→
    68→  const todaySales = sales.filter(s => new Date(s.saleDate) >= today)
    69→  const weekSales = sales.filter(s => new Date(s.saleDate) >= startOfWeek)
    70→  const monthSales = sales.filter(s => new Date(s.saleDate) >= startOfMonth)
    71→
    72→  const todayTotal = todaySales.reduce((acc, s) => acc + s.total, 0)
    73→  const weekTotal = weekSales.reduce((acc, s) => acc + s.total, 0)
    74→  const monthTotal = monthSales.reduce((acc, s) => acc + s.total, 0)
    75→  const avgSale = sales.length > 0 ? sales.reduce((acc, s) => acc + s.total, 0) / sales.length : 0
    76→
    77→  const cards = [
    78→    { title: "Today's Sales", value: formatETB(todayTotal), icon: ShoppingCart, color: 'text-primary', bg: 'bg-primary/10' },
    79→    { title: 'This Week', value: formatETB(weekTotal), icon: Calendar, color: 'text-primary', bg: 'bg-brand-50 dark:bg-brand-900/20' },
    80→    { title: 'This Month', value: formatETB(monthTotal), icon: TrendingUp, color: 'text-amber-600', bg: 'bg-amber-100 dark:bg-amber-900/30' },
    81→    { title: 'Avg Sale Value', value: formatETB(avgSale), icon: DollarSign, color: 'text-primary', bg: 'bg-primary/10' },
    82→  ]
    83→
    84→  return (
    85→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
    86→      {cards.map((card) => (
    87→        <Card key={card.title}>
    88→          <CardContent className="p-4">
    89→            <div className="flex items-center gap-3">
    90→              <div className={`size-10 rounded-lg ${card.bg} flex items-center justify-center shrink-0`}>
    91→                <card.icon className={`size-5 ${card.color}`} />
    92→              </div>
    93→              <div className="min-w-0">
    94→                <p className="text-xs text-muted-foreground truncate">{card.title}</p>
    95→                <p className="text-xl sm:text-2xl font-bold truncate">{card.value}</p>
    96→              </div>
    97→            </div>
    98→          </CardContent>
    99→        </Card>
   100→      ))}
   101→    </div>
   102→  )
   103→}

   276→function AllSalesTab({
   277→  orgId,
   278→  shopId,
   279→  onViewSale,
   280→  onCreateSale,
   281→}: {
   282→  orgId: string
   283→  shopId?: string
   284→  onViewSale: (sale: Sale) => void
   285→  onCreateSale: () => void
   286→}) {
   287→  const [sales, setSales] = useState<Sale[]>([])
   288→  const [loading, setLoading] = useState(true)
   289→  const [error, setError] = useState<string | null>(null)
   290→  const [search, setSearch] = useState('')
   291→  const [filterStatus, setFilterStatus] = useState<string>('all')
   292→  const [filterPayment, setFilterPayment] = useState<string>('all')
   293→  const [page, setPage] = useState(1)
   294→  const [totalPages, setTotalPages] = useState(1)
   295→
   296→  const fetchSales = useCallback(async () => {
   297→    setLoading(true)
   298→    setError(null)
   299→    try {
   300→      const res = await api.getSales(orgId, {
   301→        status: filterStatus !== 'all' ? filterStatus : undefined,
   302→        page,
   303→        limit: 10,
   304→        shopId,
   305→      })
   306→      setSales(res.sales)
   307→      setTotalPages(res.pagination.totalPages)
   308→    } catch (err) {
   309→      const msg = getNetworkErrorMessage(err)
   310→      setError(msg)
   311→      toast.error(msg)
   312→    } finally {
   313→      setLoading(false)
   314→    }
   315→  }, [orgId, filterStatus, page, shopId])
   316→
   317→  useEffect(() => {
   318→    fetchSales()
   319→  }, [fetchSales])
   320→
   321→  const filteredSales = sales.filter((s) => {
   322→    if (filterPayment !== 'all' && s.paymentMethod !== filterPayment) return false
   323→    if (search) {
   324→      const q = search.toLowerCase()
   325→      const matchInvoice = s.invoiceNumber.toLowerCase().includes(q)
   326→      const matchCustomer = s.customer?.name.toLowerCase().includes(q)
   327→      if (!matchInvoice && !matchCustomer) return false
   328→    }
   329→    return true
   330→  })
   331→
   332→  if (loading) {
   333→    return (
   334→      <div className="space-y-4">
   335→        <div className="flex flex-col sm:flex-row gap-3">
   336→          <Skeleton className="h-10 flex-1" />
   337→          <Skeleton className="h-10 w-36" />
   338→          <Skeleton className="h-10 w-36" />
   339→        </div>
   340→        {Array.from({ length: 5 }).map((_, i) => (
   341→          <Skeleton key={i} className="h-14" />
   342→        ))}
   343→      </div>
   344→    )
   345→  }
   346→
   347→  if (error && sales.length === 0) {
   348→    return <ErrorState title="Failed to load sales" message={error} onRetry={fetchSales} />
   349→  }
   350→
   351→  return (
   352→    <div className="space-y-4">
   353→      {/* Filters */}
   354→      <div className="flex flex-col sm:flex-row gap-3">
   355→        <div className="relative flex-1">
   356→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   357→          <Input
   358→            placeholder="Search by invoice # or customer..."
   359→            className="pl-9"
   360→            value={search}
   361→            onChange={(e) => setSearch(e.target.value)}
   362→          />
   363→        </div>
   364→        <Select value={filterStatus} onValueChange={(v) => { setFilterStatus(v); setPage(1) }}>
   365→          <SelectTrigger className="w-full sm:w-40">
   366→            <SelectValue placeholder="Status" />
   367→          </SelectTrigger>
   368→          <SelectContent>
   369→            <SelectItem value="all">All Status</SelectItem>
   370→            <SelectItem value="completed">Completed</SelectItem>
   371→            <SelectItem value="pending">Pending</SelectItem>
   372→            <SelectItem value="cancelled">Cancelled</SelectItem>
   373→          </SelectContent>
   374→        </Select>
   375→        <Select value={filterPayment} onValueChange={setFilterPayment}>
   376→          <SelectTrigger className="w-full sm:w-40">
   377→            <SelectValue placeholder="Payment" />
   378→          </SelectTrigger>
   379→          <SelectContent>
   380→            <SelectItem value="all">All Payment</SelectItem>
   381→            <SelectItem value="cash">Cash</SelectItem>
   382→            <SelectItem value="card">Card</SelectItem>
   383→            <SelectItem value="mobile_money">Mobile Money</SelectItem>
   384→            <SelectItem value="credit">Credit</SelectItem>
   385→          </SelectContent>
   386→        </Select>
   387→      </div>
   388→
   389→      {/* Empty state or Table */}
   390→      {!loading && filteredSales.length === 0 ? (
   391→        <Card>
   392→          <CardContent className="py-12 text-center">
   393→            <ShoppingCart className="size-12 text-muted-foreground mx-auto mb-4" />
   394→            <h3 className="text-base font-semibold mb-1">
   395→              {search || filterStatus !== 'all' || filterPayment !== 'all' ? 'No sales match your filters' : 'No sales yet'}
   396→            </h3>
   397→            <p className="text-muted-foreground text-sm mb-4">
   398→              {search || filterStatus !== 'all' || filterPayment !== 'all'
   399→                ? 'Try adjusting your search or filters'
   400→                : 'Create your first sale to start tracking transactions'}
   401→            </p>
   402→            {!search && filterStatus === 'all' && filterPayment === 'all' && (
   403→              <Button onClick={onCreateSale} className="gap-2">
   404→                <Plus className="size-4 mr-2" />
   405→                Create Sale
   406→              </Button>
   407→            )}
   408→          </CardContent>
   409→        </Card>
   410→      ) : (
   411→      <div className="rounded-lg border overflow-hidden">
   412→        {/* Desktop table */}
   413→        <div className="hidden md:block overflow-x-auto">
   414→          <Table>
   415→            <TableHeader>
   416→              <TableRow>
   417→                <TableHead>Invoice #</TableHead>
   418→                <TableHead>Customer</TableHead>
   419→                <TableHead className="text-center">Items</TableHead>
   420→                <TableHead className="text-right">Subtotal</TableHead>
   421→                <TableHead className="text-right">Discount</TableHead>
   422→                <TableHead className="text-right">Total</TableHead>
   423→                <TableHead>Payment</TableHead>
   424→                <TableHead>Status</TableHead>
   425→                <TableHead>Date</TableHead>
   426→                <TableHead className="text-right">Actions</TableHead>
   427→              </TableRow>
   428→            </TableHeader>
   429→            <TableBody>
   430→              {(
   431→                filteredSales.map((sale) => (
   432→                  <TableRow key={sale.id} className="cursor-pointer hover:bg-muted/50" onClick={() => onViewSale(sale)}>
   433→                    <TableCell className="font-medium font-mono text-sm">{sale.invoiceNumber}</TableCell>
   434→                    <TableCell className="text-sm">{sale.customer?.name || 'Walk-in'}</TableCell>
   435→                    <TableCell className="text-center text-sm">{sale.items?.length ?? 0}</TableCell>
   436→                    <TableCell className="text-right text-sm">{formatETB(sale.subtotal)}</TableCell>
   437→                    <TableCell className="text-right text-sm">{sale.discount > 0 ? formatETB(sale.discount) : '—'}</TableCell>
   438→                    <TableCell className="text-right text-sm font-medium">{formatETB(sale.total)}</TableCell>
   439→                    <TableCell>
   440→                      <div className="flex items-center gap-1.5 text-sm">
   441→                        {paymentMethodIcon(sale.paymentMethod)}
   442→                        {paymentMethodLabel(sale.paymentMethod)}
   443→                      </div>
   444→                    </TableCell>
   445→                    <TableCell>{<StatusBadge status={sale.status} />}</TableCell>
   446→                    <TableCell className="text-sm whitespace-nowrap">{formatDateShort(sale.saleDate)}</TableCell>
   447→                    <TableCell className="text-right">
   448→                      <div className="flex items-center justify-end gap-1">

   450→                          <Eye className="size-4" />
   451→                        </Button>

   453→                          <Download className="size-4" />
   454→                        </Button>

   459→                    </TableCell>

   461→                ))
   462→              )}
   463→            </TableBody>
   464→          </Table>
   465→        </div>
   466→        {/* Mobile card view */}
   467→        <div className="md:hidden space-y-3 p-3">
   468→          {filteredSales.map((sale) => (
   469→            <Card key={sale.id} className="p-4 cursor-pointer hover:bg-muted/50 transition-colors" onClick={() => onViewSale(sale)}>
   470→              <div className="flex items-start justify-between mb-2">
   471→                <div className="min-w-0 flex-1">
   472→                  <p className="font-semibold text-sm">{sale.invoiceNumber}</p>

   474→                    {sale.customer?.name || 'Walk-in'} · {sale.items?.length ?? 0} item{(sale.items?.length ?? 0) !== 1 ? 's' : ''}

   476→                </div>
   477→                <div className="flex items-center gap-2">
   478→                  {<StatusBadge status={sale.status} />}
   479→                </div>
   480→              </div>
   481→              <div className="flex items-center justify-between mt-2">
   482→                <div className="flex items-center gap-2 text-xs text-muted-foreground">
   483→                  {paymentMethodIcon(sale.paymentMethod)}

   491→                  <Download className="size-3" /> Receipt
