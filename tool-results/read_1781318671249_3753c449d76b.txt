    35→// ============================================
    36→// Helpers

    38→function formatETB(amount: number): string {
    39→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    40→}
    41→
    42→function formatDate(dateStr: string | null | undefined): string {
    43→  if (!dateStr) return '—'
    44→  return new Date(dateStr).toLocaleDateString('en-US', {
    45→    year: 'numeric', month: 'short', day: 'numeric'
    46→  })
    47→}
    48→
    49→function getStatusBadge(status: string) {
    50→  switch (status) {
    51→    case 'pending':
    52→      return <Badge variant="outline" className="bg-yellow-100 text-yellow-800 border-yellow-300 dark:bg-yellow-900/30 dark:text-yellow-400 dark:border-yellow-700">Pending</Badge>
    53→    case 'partial':
    54→      return <Badge variant="outline" className="bg-blue-100 text-blue-800 border-blue-300 dark:bg-blue-900/30 dark:text-blue-400 dark:border-blue-700">Partial</Badge>
    55→    case 'paid':
    56→      return <Badge variant="outline" className="bg-emerald-100 text-emerald-800 border-emerald-300 dark:bg-emerald-900/30 dark:text-emerald-400 dark:border-emerald-700">Paid</Badge>
    57→    case 'overdue':
    58→      return <Badge variant="outline" className="bg-red-100 text-red-800 border-red-300 dark:bg-red-900/30 dark:text-red-400 dark:border-red-700">Overdue</Badge>
    59→    default:
    60→      return <Badge variant="outline">{status}</Badge>
    61→  }
    62→}
    63→
    64→function isOverdue(debt: Debt): boolean {
    65→  if (debt.status === 'paid') return false
    66→  if (!debt.dueDate) return false
    67→  return new Date(debt.dueDate) < new Date()
    68→}

   164→    if (!d.payments) return sum

   168→      d.payments
   169→        .filter((p) => {
   170→          const paidDate = new Date(p.paidAt)

   172→            paidDate.getMonth() === now.getMonth() &&
   173→            paidDate.getFullYear() === now.getFullYear()
   174→          )
   175→        })
   176→        .reduce((s, p) => s + p.amount, 0)
   177→    )
   178→  }, 0)

   239→            status: 'all',

   241→            limit: '100',
   242→            shopId,

   248→        }
   249→      }

   251→      toast.error(getNetworkErrorMessage(err))

   297→                <Users className="size-5 text-blue-600 dark:text-blue-400" />

   313→                <p className="text-sm text-muted-foreground">Supplier Debts</p>

   315→              </div>
   316→            </div>

   318→        </Card>

   320→          <CardContent className="p-4">
   321→            <div className="flex items-center gap-3">

   336→                <ArrowDownLeft className="size-5 text-primary" />
   337→              </div>
   338→              <div>
   339→                <p className="text-sm text-muted-foreground">Collected This Month</p>

   341→              </div>

   351→            <TabsTrigger value="customer" className="gap-1 sm:gap-2 text-xs sm:text-sm">

   353→              <span className="hidden sm:inline">Customer</span> Debts

   385→        <TabsContent value="customer">

   653→
   654→              {/* Add payment button */}
   655→              {selectedDebt.status !== 'paid' && (

   657→                  className="w-full gap-2"
   658→                  onClick={() => {
   659→                    setShowDetail(false)
   660→                    openRecordPayment(selectedDebt)

   662→                >
   663→                  <CreditCard className="size-4" />
   664→                  Record Payment
   665→                </Button>
   666→              )}
   667→            </div>
   668→          )}
   669→        </DialogContent>
   670→      </Dialog>
   671→    </div>
   672→  )
