    54→function formatETB(amount: number): string {
    55→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 0, maximumFractionDigits: 0 })}`

    58→function daysUntil(dateStr: string | null): number | null {
    59→  if (!dateStr) return null
    60→  const diff = new Date(dateStr).getTime() - Date.now()
    61→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
    62→}

   135→function getModuleStatus(m: ModuleData): ModuleStatus {
   136→  if (!m.orgStatus) return 'available'
   137→  if (m.orgStatus.status === 'active') return 'active'
   138→  if (m.orgStatus.status === 'trial') return 'trial'
   139→  if (m.orgStatus.status === 'requested') return 'requested'
   140→  if (m.orgStatus.status === 'expired' || m.orgStatus.status === 'cancelled') return 'expired'
   141→  // If orgStatus exists but status is unexpected, check isActive
   142→  if (m.orgStatus.isActive) return 'active'
   143→  return 'available'
   144→}

   193→              <Badge className={statusConfig.color} variant="outline">
   194→                <span className="flex items-center gap-1">

   220→            </span>
   221→          )}

   226→          <>
   227→            <Separator />
   228→            <div className="flex items-center gap-2 p-3 rounded-lg bg-violet-50 dark:bg-violet-900/20">

   233→              </div>
   234→            </div>
   235→          </>
   236→        )}
   237→
   238→        {/* Trial countdown with progress bar */}
   239→        {isTrial && (
   240→          <>
   241→            <Separator />
   242→            {days !== null && days > 0 ? (
   243→              <div className={`rounded-lg p-3 space-y-2 ${
   244→                isCritical
   245→                  ? 'bg-red-50 dark:bg-red-900/20 animate-pulse'
   246→                  : isWarning
   247→                    ? 'bg-amber-50 dark:bg-amber-900/20 animate-pulse'
   248→                    : 'bg-amber-50 dark:bg-amber-900/20'

   391→// ============================================
   392→// Request Confirmation Dialog
   393→// ============================================
   394→function RequestDialog({
   395→  open,
   396→  onOpenChange,
   397→  module: mod,
   398→  onSuccess,
   399→}: {
   400→  open: boolean
   401→  onOpenChange: (open: boolean) => void
   402→  module: EnrichedModule | null
   403→  onSuccess: () => void
   404→}) {
   405→  const { currentOrg } = useAuthStore()
   406→  const [loading, setLoading] = useState(false)
   407→
   408→  if (!mod || !currentOrg) return null
   409→
   410→  const handleSubmit = async () => {
   411→    setLoading(true)
   412→    try {
   413→      await api.requestModule(currentOrg.id, mod.key)
   414→      toast.success(`Request for ${mod.name} submitted! Our team will review it shortly.`)
   415→      onOpenChange(false)
   416→      onSuccess()
   417→    } catch (err) {
   418→      toast.error(getNetworkErrorMessage(err))
   419→    } finally {
   420→      setLoading(false)
   421→    }
   422→  }
   423→
   424→  return (
   425→    <Dialog open={open} onOpenChange={onOpenChange}>
   426→      <DialogContent className="sm:max-w-md">
   427→        <DialogHeader>
   428→          <DialogTitle className="flex items-center gap-2">
   429→            <Send className="size-5 text-primary" />
   430→            Request {mod.name}
   431→          </DialogTitle>
   432→          <DialogDescription>
   433→            {mod.description || `Request access to the ${mod.name} module for your organization.`}
   434→          </DialogDescription>
   435→        </DialogHeader>
   436→
   437→        <div className="space-y-3">
   438→          {/* Module info */}
   439→          <div className="rounded-lg border p-3 space-y-2">
   440→            <div className="flex items-center justify-between">
   441→              <span className="text-xs text-muted-foreground">Module</span>
   442→              <Badge className={CATEGORY_COLORS[mod.category] || CATEGORY_COLORS.core} variant="outline">
   443→                {mod.category}
   444→              </Badge>
   445→            </div>
   446→            <p className="text-sm font-medium">{mod.name}</p>
   447→            {!mod.isFree && (
   448→              <div className="flex items-center justify-between text-sm">
   449→                <span className="text-muted-foreground">Price</span>
   450→                <span className="font-semibold">{formatETB(mod.priceETB)}/mo</span>
   451→              </div>
   452→            )}
   453→            {mod.isFree && (
   454→              <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300" variant="outline">
   455→                Free — No payment required
   456→              </Badge>
   457→            )}
   458→          </div>
   459→
   460→          {/* Beta notice */}
   461→          <div className="rounded-lg border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-900/20 p-3">
   462→            <div className="flex items-start gap-2">
   463→              <Info className="size-4 text-emerald-600 shrink-0 mt-0.5" />
   464→              <div>
   465→                <p className="text-xs font-medium text-emerald-800 dark:text-emerald-300">Free during Beta</p>
   466→                <p className="text-[11px] text-emerald-700 dark:text-emerald-400">
   467→                  All modules are currently free while InvenSync is in beta. Request access and we&apos;ll activate it for you right away.
   468→                </p>
   469→              </div>
   470→            </div>
   471→          </div>
   472→        </div>
   473→
   474→        <DialogFooter>
   475→          <FormSubmitButton
   476→            isLoading={loading}
   477→            loadingText="Submitting..."
   478→            onClick={handleSubmit}
   479→            className="w-full gap-1.5"
   480→          >
   481→            <Send className="size-4" />
   482→            Submit Request
   483→          </FormSubmitButton>
   484→        </DialogFooter>
   485→      </DialogContent>
   486→    </Dialog>
   487→  )
   488→}

   517→      <p className={`text-lg font-semibold ${colorClass}`}>{count}</p>
   518→      <p className="text-xs text-muted-foreground">{label}</p>
   519→    </button>
   520→  )
   521→}

   524→// Modules Page
   525→// ============================================
   526→export function ModulesPage() {
   527→  const { currentOrg } = useAuthStore()
   528→  const [modules, setModules] = useState<EnrichedModule[]>([])
   529→  const [loading, setLoading] = useState(true)
   530→  const [fetchError, setFetchError] = useState<string | null>(null)
   531→  const [filter, setFilter] = useState<'all' | ModuleStatus>('all')
   532→  const [requestingKey, setRequestingKey] = useState<string | null>(null)
   533→  const [requestDialogOpen, setRequestDialogOpen] = useState(false)
   534→  const [selectedModule, setSelectedModule] = useState<EnrichedModule | null>(null)
   535→  const [activatingKey, setActivatingKey] = useState<string | null>(null)
   536→
   537→  const fetchModules = useCallback(async () => {
   538→    if (!currentOrg) return
   539→    try {
   540→      setFetchError(null)
   541→      const data = await api.getModules(currentOrg.id)
   542→      const enriched: EnrichedModule[] = data.modules.map((m: ModuleData) => ({
   543→        id: m.id,
   544→        key: m.key,
   545→        name: m.name,
   546→        description: m.description,
   547→        icon: m.icon,
   548→        category: m.category,
   549→        priceETB: m.priceETB,
   550→        isFree: m.isFree ?? m.priceETB === 0,
   551→        freeTrialDays: m.freeTrialDays || 30,
   552→        billingCycle: m.billingCycle || 'monthly',
   553→        order: m.order,
   554→        status: getModuleStatus(m),
   555→        orgStatus: m.orgStatus,
   556→        requested: m.orgStatus?.status === 'requested',
   557→      }))
   558→      setModules(enriched)
   559→    } catch (err) {
   560→      setFetchError(getNetworkErrorMessage(err))
   561→    } finally {
   562→      setLoading(false)
   563→    }
   564→  }, [currentOrg])
   565→
   566→  useEffect(() => { fetchModules() }, [fetchModules])
   567→
   568→  const handleRequestAccess = async (moduleKey: string) => {
   569→    if (!currentOrg) return
   570→    setRequestingKey(moduleKey)
   571→    try {
   572→      await api.requestModule(currentOrg.id, moduleKey)
   573→      toast.success('Request submitted! Our team will review it shortly.')
   574→      // Update local state to show "requested" status immediately
   575→      setModules(prev =>
   576→        prev.map(m =>
   577→          m.key === moduleKey ? { ...m, requested: true, status: 'requested' as ModuleStatus } : m
   578→        )
   579→      )
   580→    } catch (err) {
   581→      toast.error(getNetworkErrorMessage(err))
   582→    } finally {
   583→      setRequestingKey(null)
   584→    }
   585→  }
   586→
   587→  const handleActivate = async (moduleKey: string) => {
   588→    if (!currentOrg) return
   589→    setActivatingKey(moduleKey)
   590→    try {
   591→      await api.activateModule(currentOrg.id, moduleKey)
   592→      toast.success('Module activated successfully!')
   593→      fetchModules()
   594→    } catch (err) {
   595→      toast.error(getNetworkErrorMessage(err))
   596→    } finally {
   597→      setActivatingKey(null)
   598→    }
   599→  }
   600→
   601→  const handleOpenRequestDialog = (mod: EnrichedModule) => {
   602→    setSelectedModule(mod)
   603→    setRequestDialogOpen(true)
   604→  }
   605→
   606→  const filteredModules = modules.filter(m => {
   607→    if (filter === 'all') return true
   608→    return m.status === filter
   609→  })
   610→
   611→  const activeCount = modules.filter(m => m.status === 'active').length
   612→  const trialCount = modules.filter(m => m.status === 'trial').length
   613→  const requestedCount = modules.filter(m => m.status === 'requested').length
   614→  const availableCount = modules.filter(m => m.status === 'available').length
   615→  const expiredCount = modules.filter(m => m.status === 'expired').length
   616→
   617→  // Expiry warning data
   618→  const expiringModules = modules.filter(m => {
   619→    const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   620→    return days !== null && days <= 7
   621→  })
   622→  const criticalModules = expiringModules.filter(m => {
   623→    const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   624→    return days !== null && (days <= 1 || days <= 0)
   625→  })
   626→
   627→  if (loading) {
   628→    return (
   629→      <div className="space-y-6">
   630→        <Skeleton className="h-10 w-64" />
   631→        <Skeleton className="h-20" />
   632→        <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   633→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-20" />)}
   634→        </div>
   635→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   636→          {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-64" />)}
   637→        </div>
   638→      </div>
   639→    )
   640→  }
   641→
   642→  if (fetchError) {
   643→    return <ErrorState title="Failed to load modules" error={fetchError} onRetry={fetchModules} />
   644→  }
   645→
   646→  return (
   647→    <div className="space-y-6">
   648→      {/* Header */}
   649→      <div>
   650→        <h1 className="text-base font-semibold tracking-tight">My Modules</h1>
   651→        <p className="text-muted-foreground text-sm">Explore, request, and manage your InvenSync modules</p>
   652→      </div>
   653→
   654→      {/* Expiry Warning Banner */}
   655→      {expiringModules.length > 0 && (
   656→        <Alert className={
   657→          criticalModules.length > 0
   658→            ? 'border-red-300 bg-red-50 dark:border-red-800 dark:bg-red-900/20'
   659→            : 'border-amber-300 bg-amber-50 dark:border-amber-800 dark:bg-amber-900/20'
   660→        }>
   661→          <AlertTriangle className={
   662→            criticalModules.length > 0
   663→              ? 'size-4 text-red-600'
   664→              : 'size-4 text-amber-600'
   665→          } />
   666→          <AlertTitle className={
   667→            criticalModules.length > 0
   668→              ? 'text-sm font-medium text-red-800 dark:text-red-300'
   669→              : 'text-sm font-medium text-amber-800 dark:text-amber-300'
   670→          }>
   671→            {criticalModules.length > 0
   672→              ? 'Modules expiring or expired!'
   673→              : 'Modules expiring soon'}
   674→          </AlertTitle>
   675→          <AlertDescription className={
   676→            criticalModules.length > 0
   677→              ? 'text-xs text-red-700 dark:text-red-400'
   678→              : 'text-xs text-amber-700 dark:text-amber-400'
   679→          }>
   680→            {expiringModules.map(m => {
   681→              const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   682→              const label = days !== null && days <= 0
   683→                ? `${m.name} (expired)`
   684→                : days !== null
   685→                  ? `${m.name} (${days}d left)`
   686→                  : m.name

   691→              onClick={() => {

   693→                section?.scrollIntoView({ behavior: 'smooth' })
   694→              }}
   695→              className="underline font-medium"
   696→            >
   697→              View modules
   698→            </button>
   699→          </AlertDescription>
   700→        </Alert>
   701→      )}
   702→
   703→      {/* Beta Info Alert */}
   704→      <Alert className="border-primary/20 bg-primary/5">
   705→        <Info className="size-4 text-primary" />
   706→        <AlertTitle className="text-sm font-medium">Free During Beta</AlertTitle>
   707→        <AlertDescription className="text-xs">
   708→          All modules are currently free while InvenSync is in beta. Request access to any module and our team will activate it for you.
   709→        </AlertDescription>
   710→      </Alert>
   711→
   712→      {/* Summary Stats */}
   713→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   714→        <StatCard
   715→          count={activeCount}
   716→          label="Active"
   717→          colorClass="text-emerald-600"
   718→          bgColor="bg-emerald-50 dark:bg-emerald-900/20"
   719→          borderColor="border-emerald-200 dark:border-emerald-800"
   720→          active={filter === 'active'}
   721→          onClick={() => setFilter(filter === 'active' ? 'all' : 'active')}
