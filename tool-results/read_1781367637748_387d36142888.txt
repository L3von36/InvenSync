    85→// Main Component
    86→// ============================================
    87→export function ReportsPage() {
    88→  const { currentOrg, currentShop } = useAuthStore()
    89→  const orgId = currentOrg?.id || ''
    90→  const isMobile = useIsMobile()
    91→
    92→  const [reportData, setReportData] = useState<ReportData | null>(null)
    93→  const [loading, setLoading] = useState(true)
    94→  const [pageError, setPageError] = useState<string | null>(null)
    95→  const [dateError, setDateError] = useState<string | null>(null)
    96→  const [activeTab, setActiveTab] = useState('sales')
    97→
    98→  // Date range
    99→  const [periodType, setPeriodType] = useState('monthly')
   100→  const [customStart, setCustomStart] = useState('')
   101→  const [customEnd, setCustomEnd] = useState('')
   102→
   103→  const getDateRange = useCallback(() => {
   104→    const now = new Date()
   105→    let startDate: string
   106→    let endDate: string = now.toISOString().split('T')[0]
   107→
   108→    switch (periodType) {
   109→      case 'today':
   110→        startDate = endDate
   111→        break
   112→      case 'weekly': {
   113→        const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   114→        startDate = weekAgo.toISOString().split('T')[0]
   115→        break
   116→      }
   117→      case 'monthly':
   118→        startDate = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   119→        break
   120→      case 'quarterly': {
   121→        const quarter = Math.floor(now.getMonth() / 3)
   122→        startDate = new Date(now.getFullYear(), quarter * 3, 1).toISOString().split('T')[0]
   123→        break
   124→      }
   125→      case 'custom':
   126→        startDate = customStart || new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   127→        endDate = customEnd || now.toISOString().split('T')[0]
   128→        break
   129→      default:
   130→        startDate = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   131→    }
   132→    return { startDate, endDate }
   133→  }, [periodType, customStart, customEnd])
   134→
   135→  const fetchReport = useCallback(async () => {
   136→    if (!orgId) return
   137→
   138→    // Date range validation for custom period
   139→    if (periodType === 'custom' && customStart && customEnd) {
   140→      if (new Date(customStart) > new Date(customEnd)) {
   141→        setDateError('Start date must be before end date')
   142→        return
   143→      }
   144→    }
   145→    setDateError(null)
   146→    setPageError(null)
   147→    setLoading(true)
   148→    try {
   149→      const { startDate, endDate } = getDateRange()
   150→      const data = await api.getReports(orgId, {
   151→        startDate,
   152→        endDate,
   153→        period: periodType === 'custom' ? 'daily' : periodType,
   154→        shopId: currentShop?.id,
   155→      })
   156→      setReportData(data as unknown as ReportData)
   157→    } catch (err) {
   158→      const msg = getNetworkErrorMessage(err)
   159→      setPageError(msg)
   160→    } finally {
   161→      setLoading(false)
   162→    }
   163→  }, [orgId, getDateRange, periodType, currentShop])
   164→
   165→  useEffect(() => {
   166→    fetchReport()
   167→  }, [fetchReport])
   168→
   169→  const summary = reportData?.summary
   170→  const salesByDate = reportData?.salesByDate || []
   171→  const bestSelling = reportData?.bestSellingProducts || []
   172→  const inventoryVal = reportData?.inventoryValuation
   173→
   174→  const handleExportCSV = useCallback(() => {
   175→    if (!reportData) return
   176→
   177→    if (activeTab === 'sales') {
   178→      exportToCSV(
   179→        salesByDate.map(row => ({
   180→          date: row.date,
   181→          invoice: row.count,
   182→          customer: '',
   183→          total: row.revenue,
   184→          status: '',
   185→          paymentMethod: '',
   186→        })),
   187→        'sales-report',
   188→        [
   189→          { key: 'date', label: 'Date' },
   190→          { key: 'invoice', label: 'Transactions' },
   191→          { key: 'customer', label: 'Customer' },
   192→          { key: 'total', label: 'Total' },
   193→          { key: 'status', label: 'Status' },
   194→          { key: 'paymentMethod', label: 'Payment Method' },
   195→        ]
   196→      )
   197→    } else if (activeTab === 'profit') {
   198→      exportToCSV(
   199→        salesByDate.map(row => ({
   200→          date: row.date,
   201→          revenue: row.revenue,
   202→          cost: row.cost,
   203→          profit: row.profit,
   204→          margin: row.revenue > 0 ? ((row.profit / row.revenue) * 100).toFixed(1) + '%' : '0%',
   205→        })),
   206→        'profit-report',
   207→        [
   208→          { key: 'date', label: 'Date' },
   209→          { key: 'revenue', label: 'Revenue' },
   210→          { key: 'cost', label: 'Cost' },
   211→          { key: 'profit', label: 'Profit' },
   212→          { key: 'margin', label: 'Margin' },
   213→        ]
   214→      )
   215→    } else if (activeTab === 'inventory') {
   216→      const val = reportData.inventoryValuation
   217→      exportToCSV(
   218→        [{
   219→          productName: 'All Inventory',
   220→          type: 'Summary',
   221→          quantity: val?.totalItems || 0,
   222→          costPrice: val?.totalCostValue || 0,
   223→          sellingPrice: val?.totalRetailValue || 0,
   224→          value: val?.totalRetailValue || 0,
   225→        }],
   226→        'inventory-report',
   227→        [
   228→          { key: 'productName', label: 'Product Name' },
   229→          { key: 'type', label: 'Type' },
   230→          { key: 'quantity', label: 'Quantity' },
   231→          { key: 'costPrice', label: 'Cost Price' },
   232→          { key: 'sellingPrice', label: 'Selling Price' },
   233→          { key: 'value', label: 'Value' },
   234→        ]
   235→      )
   236→    } else if (activeTab === 'bestsellers') {
   237→      exportToCSV(
   238→        bestSelling.map(p => ({
   239→          productName: p.name,
   240→          sku: p.sku || '',
   241→          quantitySold: p.totalQuantity,
   242→          revenue: p.totalRevenue,
   243→          avgPrice: p.totalQuantity > 0 ? p.totalRevenue / p.totalQuantity : 0,
   244→        })),
   245→        'bestsellers-report',
   246→        [
   247→          { key: 'productName', label: 'Product Name' },
   248→          { key: 'sku', label: 'SKU' },
   249→          { key: 'quantitySold', label: 'Quantity Sold' },
   250→          { key: 'revenue', label: 'Revenue' },
   251→          { key: 'avgPrice', label: 'Avg Price' },
   252→        ]
   253→      )
   254→    }
   255→  }, [activeTab, reportData, salesByDate, bestSelling])
   256→
   257→  // Payment method distribution for pie chart (mock from sales data)
   258→  const paymentMethodData = [
   259→    { name: 'Cash', value: 45 },
   260→    { name: 'Card', value: 30 },
   261→    { name: 'Mobile Money', value: 25 },
   262→  ]
   263→
   264→  return (
   265→    <div className="space-y-6">
   266→      {/* Header */}
   267→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   268→        <div>
   269→          <h1 className="text-xl sm:text-2xl font-bold tracking-tight">Reports</h1>
   270→          <p className="text-muted-foreground text-sm">Analyze your sales, profit, and inventory performance</p>
   271→        </div>
   272→        <Button
   273→          variant="outline"
   274→          size="sm"
   275→          onClick={handleExportCSV}
   276→          disabled={!reportData || salesByDate.length === 0}
   277→          className="gap-2"
   278→        >
   279→          <Download className="size-4" />
   280→          Export CSV
   281→        </Button>
   282→      </div>
   283→
   284→      {/* Date Range Selector */}
   285→      <Card>
   286→        <CardContent className="p-4">
   287→          <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   288→            <div className="flex items-center gap-2">
   289→              <Calendar className="size-4 text-muted-foreground" />
   290→              <span className="text-sm font-medium">Period:</span>
   291→            </div>
   292→            <div className="flex flex-wrap items-center gap-2">
   293→              {[
   294→                { value: 'today', label: 'Today' },
   295→                { value: 'weekly', label: 'This Week' },
   296→                { value: 'monthly', label: 'This Month' },
   297→                { value: 'quarterly', label: 'This Quarter' },
   298→                { value: 'custom', label: 'Custom' },
   299→              ].map((p) => (
   300→                <Button
   301→                  key={p.value}
   302→                  variant={periodType === p.value ? 'default' : 'outline'}
   303→                  size="sm"
   304→                  onClick={() => setPeriodType(p.value)}
   305→                >
   306→                  {p.label}
   307→                </Button>
   308→              ))}
   309→            </div>
   310→            {periodType === 'custom' && (
   311→              <div className="flex items-center gap-4">
   312→                <div className="flex items-center gap-2">
   313→                  <Label className="text-xs whitespace-nowrap">From</Label>
   314→                  <Input
   315→                    type="date"
   316→                    value={customStart}
   317→                    onChange={(e) => { setCustomStart(e.target.value); setDateError(null) }}
   318→                    className="h-8 text-xs w-36"
   319→                  />
   320→                </div>
   321→                <div className="flex items-center gap-2">
   322→                  <Label className="text-xs whitespace-nowrap">To</Label>
   323→                  <Input
   324→                    type="date"
   325→                    value={customEnd}
   326→                    onChange={(e) => { setCustomEnd(e.target.value); setDateError(null) }}
   327→                    className="h-8 text-xs w-36"
   328→                  />
   329→                </div>
   330→              </div>
   331→            )}
   332→            {dateError && (
   333→              <p className="w-full text-xs text-destructive">{dateError}</p>
   334→            )}
   335→          </div>
   336→        </CardContent>
   337→      </Card>
   338→
   339→      {/* Tabs */}
   340→      <Tabs value={activeTab} onValueChange={setActiveTab} className="space-y-4">
   341→        <TabsList className="flex-wrap grid w-full grid-cols-2 sm:grid-cols-4">
   342→          <TabsTrigger value="sales" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   343→            <ShoppingCart className="size-3.5 sm:size-4" />
   344→            <span className="hidden sm:inline">Sales</span>
   345→          </TabsTrigger>
   346→          <TabsTrigger value="profit" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   347→            <TrendingUp className="size-3.5 sm:size-4" />
   348→            <span className="hidden sm:inline">Profit</span>
   349→          </TabsTrigger>
   350→          <TabsTrigger value="bestsellers" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   351→            <BarChart3 className="size-3.5 sm:size-4" />
   352→            <span className="hidden sm:inline">Best</span>
   353→          </TabsTrigger>
   354→          <TabsTrigger value="inventory" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   355→            <Package className="size-3.5 sm:size-4" />
   356→            <span className="hidden sm:inline">Inventory</span>
   357→          </TabsTrigger>
   358→        </TabsList>
   359→
   360→        {/* Sales Report */}
   361→        <TabsContent value="sales" className="space-y-4">
   362→          {loading ? (
   363→            <LoadingSkeleton cards={3} />
   364→          ) : pageError && !reportData ? (
   365→            <ErrorState title="Failed to load report data" message={pageError} onRetry={fetchReport} />
   366→          ) : !loading && reportData && salesByDate.length === 0 ? (
   367→            <EmptyState
   368→              title="No sales data for this period"
   369→              message="Try selecting a different date range, or make your first sale to see reports here."
   370→              icon={<BarChart3 className="size-7 text-muted-foreground" />}
   371→            />
   372→          ) : (
   373→            <>
   374→              {/* Summary Cards */}
   375→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 md:gap-4">
   376→                <Card>
   377→                  <CardContent className="p-4">
   378→                    <div className="flex items-center gap-3">
   379→                      <div className="size-10 rounded-lg bg-brand-50 dark:bg-brand-900/20 flex items-center justify-center">
   380→                        <DollarSign className="size-5 text-primary" />
   381→                      </div>
   382→                      <div>
   383→                        <p className="text-sm text-muted-foreground">Total Revenue</p>
   384→                        <p className="text-xl sm:text-2xl font-bold">{formatETB(summary?.totalRevenue || 0)}</p>
   385→                      </div>
   386→                    </div>
   387→                  </CardContent>
   388→                </Card>
   389→                <Card>
   390→                  <CardContent className="p-4">
   391→                    <div className="flex items-center gap-3">
   392→                      <div className="size-10 rounded-lg bg-blue-100 dark:bg-blue-900/30 flex items-center justify-center">
   393→                        <ShoppingCart className="size-5 text-blue-600 dark:text-blue-400" />
   394→                      </div>
   395→                      <div>
   396→                        <p className="text-sm text-muted-foreground">Total Transactions</p>
   397→                        <p className="text-xl sm:text-2xl font-bold">{summary?.totalSales || 0}</p>
   398→                      </div>
   399→                    </div>
   400→                  </CardContent>
   401→                </Card>
   402→                <Card>
   403→                  <CardContent className="p-4">
   404→                    <div className="flex items-center gap-3">
   405→                      <div className="size-10 rounded-lg bg-purple-100 dark:bg-purple-900/30 flex items-center justify-center">
   406→                        <TrendingUp className="size-5 text-purple-600 dark:text-purple-400" />
   407→                      </div>
   408→                      <div>
   409→                        <p className="text-sm text-muted-foreground">Avg Transaction</p>
   410→                        <p className="text-xl sm:text-2xl font-bold">{formatETB(summary?.averageSaleValue || 0)}</p>
   411→                      </div>
   412→                    </div>
   413→                  </CardContent>
   414→                </Card>
   415→              </div>
   416→
   417→              {/* Charts Row */}
   418→              <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 min-w-0">
   419→                {/* Daily Sales Bar Chart */}
   420→                <Card className="lg:col-span-2 overflow-hidden">
   421→                  <CardHeader className="pb-2">
   422→                    <CardTitle className="text-sm font-semibold">Daily Sales</CardTitle>
   423→                  </CardHeader>
   424→                  <CardContent>
   425→                    {salesByDate.length > 0 ? (
   426→                      <ResponsiveContainer width="100%" height={300}>
   427→                        <BarChart data={salesByDate}>
   428→                          <CartesianGrid strokeDasharray="3 3" className="opacity-30" />
   429→                          <XAxis
   430→                            dataKey="date"
   431→                            tick={{ fontSize: 11 }}
   432→                            tickFormatter={(val: string) => {
   433→                              const d = new Date(val)
   434→                              return `${d.getMonth() + 1}/${d.getDate()}`
   435→                            }}
   436→                          />
   437→                          <YAxis tick={{ fontSize: 11 }} tickFormatter={(val: number) => formatShortETB(val)} />
   438→                          <Tooltip
   439→                            formatter={(value: number) => formatETB(value)}
   440→                            labelFormatter={(label: string) => {
   441→                              const d = new Date(label)
   442→                              return d.toLocaleDateString('en-US', {
   443→                                weekday: 'short', year: 'numeric', month: 'short', day: 'numeric'
   444→                              })
   445→                            }}
   446→                          />
   447→                          <Legend />
   448→                          <Bar dataKey="revenue" name="Revenue" fill="oklch(0.637 0.237 25.331)" radius={[4, 4, 0, 0]} />
   449→                          <Bar dataKey="profit" name="Profit" fill="oklch(0.6 0.118 184.704)" radius={[4, 4, 0, 0]} />
   450→                        </BarChart>
   451→                      </ResponsiveContainer>
   452→                    ) : (
   453→                      <EmptyChart message="No sales data for this period" />
   454→                    )}
   455→                  </CardContent>
   456→                </Card>
   457→
   458→                {/* Payment Method Pie */}
   459→                <Card className="overflow-hidden">
   460→                  <CardHeader className="pb-2">
   461→                    <CardTitle className="text-sm font-semibold">By Payment Method</CardTitle>
   462→                  </CardHeader>
   463→                  <CardContent>
   464→                    <ResponsiveContainer width="100%" height={300}>
   465→                      <RechartsPie>
   466→                        <Pie
   467→                          data={paymentMethodData}
   468→                          cx="50%"
   469→                          cy="50%"
   470→                          innerRadius={60}
   471→                          outerRadius={90}
   472→                          paddingAngle={4}
   473→                          dataKey="value"
   474→                          label={({ name, percent }: { name: string; percent: number }) =>
   475→                            `${name} ${(percent * 100).toFixed(0)}%`
   476→                          }
   477→                        >
   478→                          {paymentMethodData.map((_, index) => (
   479→                            <Cell key={`cell-${index}`} fill={CHART_COLORS[index % CHART_COLORS.length]} />
   480→                          ))}
   481→                        </Pie>
   482→                        <Tooltip />
   483→                      </RechartsPie>
   484→                    </ResponsiveContainer>
   485→                  </CardContent>
   486→                </Card>
   487→              </div>
   488→
   489→              {/* Sales Table */}
   490→              <Card>
   491→                <CardHeader className="pb-2">
   492→                  <CardTitle className="text-sm font-semibold">Daily Sales Breakdown</CardTitle>
   493→                </CardHeader>
   494→                <CardContent className="p-0">
   495→                  {/* Desktop table */}
   496→                  <div className="hidden md:block overflow-x-auto max-h-96 overflow-y-auto custom-scrollbar">
   497→                    <Table>
   498→                      <TableHeader>
   499→                        <TableRow>
   500→                          <TableHead>Date</TableHead>
   501→                          <TableHead className="text-right">Revenue</TableHead>
   502→                          <TableHead className="text-right">Cost</TableHead>
   503→                          <TableHead className="text-right">Profit</TableHead>

   522→                              </TableCell>
   523→                              <TableCell className="text-right">{row.count}</TableCell>
   524→                            </TableRow>
   525→                          ))
   526→                        ) : (
   527→                          <TableRow>
   528→                            <TableCell colSpan={5} className="text-center py-8 text-muted-foreground">
   529→                              No sales data for this period
   530→                            </TableCell>
   531→                          </TableRow>
   532→                        )}
   533→                      </TableBody>
   534→                    </Table>
   535→                  </div>
   536→                  {/* Mobile card view */}
   537→                  <div className="md:hidden space-y-3 p-3 max-h-96 overflow-y-auto">
   538→                    {salesByDate.length > 0 ? (
   539→                      salesByDate.map((row) => (
