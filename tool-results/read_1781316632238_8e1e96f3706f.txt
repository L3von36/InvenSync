   105→   105→function formatMonthLabel(monthKey: string): string {
   106→   106→  const MONTH_LABELS: Record<string, string> = {
   107→   107→    '01': 'Jan', '02': 'Feb', '03': 'Mar', '04': 'Apr',
   108→   108→    '05': 'May', '06': 'Jun', '07': 'Jul', '08': 'Aug',
   109→   109→    '09': 'Sep', '10': 'Oct', '11': 'Nov', '12': 'Dec',
   110→   110→  }
   111→   111→  const [year, month] = monthKey.split('-')
   112→   112→  return `${MONTH_LABELS[month] || month} ${year.slice(2)}`
   113→   113→}
