   552→class ApiClient {
   553→  // Supabase Auth uses httpOnly cookies for session management.
   554→  // The token is no longer stored in localStorage for security.
   555→  // We keep a reference for backward compat but auth is primarily cookie-based.
   556→  private _token: string | null = null
   557→  // Guard flag: when true, 401 auto-logout is suppressed to prevent
   558→  // stale background requests from interfering with an active login attempt.
   559→  private _isLoggingIn: boolean = false
   560→
   561→  // Request deduplication: prevent concurrent identical GET requests
   562→  // If 3 components request the same endpoint simultaneously, only 1 fetch is made.
   563→  private _inflightRequests = new Map<string, Promise<unknown>>()
   564→
   565→  // Short-lived client-side response cache for GET requests (5s TTL)
   566→  // Prevents rapid re-fetches when multiple components mount simultaneously
   567→  private _responseCache = new Map<string, { data: unknown; expiresAt: number }>()
   568→  private static RESPONSE_CACHE_TTL = 5_000 // 5 seconds
   569→
   570→  /**
   571→   * Invalidate client-side response cache entries matching a prefix.
   572→   * Called after mutating operations to ensure stale data is not served.
   573→   */
   574→  invalidateCache(prefix: string): void {
   575→    // Invalidate our internal response cache
   576→    this._responseCache.forEach((_, key) => {
   577→      if (key.startsWith(prefix)) {
   578→        this._responseCache.delete(key)
   579→      }
   580→    })
   581→    // Also invalidate the shared client-cache (used by swrFetch)
   582→    try {
   583→      import('@/lib/client-cache').then(({ invalidateKeysByPrefix }) => {
   584→        invalidateKeysByPrefix(prefix)
   585→      }).catch(() => {
   586→        // client-cache may not be available during SSR
   587→      })
   588→    } catch {
   589→      // client-cache may not be available during SSR
   590→    }
   591→  }
   592→
   593→  private getToken(): string | null {
   594→    if (typeof window === 'undefined') return null
   595→    // Check memory first, then fallback to localStorage (for migration)
   596→    return this._token || localStorage.getItem('sb_token')
   597→  }
   598→
   599→  setToken(token: string): void {
   600→    if (typeof window === 'undefined') return
   601→    this._token = token
   602→    // Also keep in localStorage for migration period
   603→    localStorage.setItem('sb_token', token)
   604→  }
   605→
   606→  clearToken(): void {
   607→    if (typeof window === 'undefined') return
   608→    this._token = null
   609→    localStorage.removeItem('sb_token')
   610→  }
   611→
   612→  private async request<T>(
   613→    endpoint: string,
   614→    options: RequestInit = {}
   615→  ): Promise<T> {
   616→    const token = this.getToken()
   617→    const headers: Record<string, string> = {
   618→      'Content-Type': 'application/json',
   619→      ...(options.headers as Record<string, string>),
   620→    }
   621→
   622→    // Add JWT token to Authorization header if available
   623→    if (token) {
   624→      headers['Authorization'] = `Bearer ${token}`
   625→    }
   626→
   627→    // Offline interceptor: queue mutating requests, reject GET requests
   628→    const method = (options.method || 'GET').toUpperCase()
   629→    const isMutating = ['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)
   630→    const isGet = method === 'GET'
   631→
   632→    // ---- Request deduplication for GET requests ----
   633→    // If an identical GET request is already in-flight, return the same promise
   634→    // instead of firing a duplicate network request.
   635→    if (isGet) {
   636→      const cacheKey = `${method}:${endpoint}`
   637→
   638→      // Check short-lived response cache first
   639→      const cached = this._responseCache.get(cacheKey)
   640→      if (cached && Date.now() < cached.expiresAt) {
   641→        return cached.data as T
   642→      }
   643→
   644→      // Check for in-flight deduplication
   645→      const inflight = this._inflightRequests.get(cacheKey)
   646→      if (inflight) {
   647→        return inflight as Promise<T>
   648→      }
   649→    }
   650→
   651→    if (typeof window !== 'undefined' && !navigator.onLine) {
   652→      if (isMutating) {
   653→        // Queue the operation for later sync
   654→        try {
   655→          const { queueOperation } = await import('@/lib/offline-queue')
   656→          await queueOperation(
   657→            endpoint,
   658→            method,
   659→            headers,
   660→            options.body ? String(options.body) : null
   661→          )
   662→          // Return a mock "offline queued" response so callers don't break
   663→          return { offline: true, queued: true } as unknown as T
   664→        } catch {
   665→          // If queuing fails, fall through to normal error
   666→        }
   667→      }
   668→      throw new Error('You are offline. Please check your internet connection and try again.')
   669→    }
   670→
   671→    // Add timeout (30s) to prevent hanging requests
   672→    const controller = new AbortController()
   673→    const timeoutId = setTimeout(() => controller.abort(), 30000)
   674→    const signal = options.signal || controller.signal
   675→
   676→    // For GET requests, register the inflight promise for deduplication
   677→    const cacheKey = `${method}:${endpoint}`
   678→    let isInflightRegistered = false
   679→
   680→    if (isGet) {
   681→      isInflightRegistered = true
   682→    }
   683→
   684→    // Create the fetch promise (may be shared across callers)
   685→    const fetchPromise = (async () => {
   686→      let response: Response
   687→      try {
   688→        response = await fetch(endpoint, {
   689→          ...options,
   690→          headers,
   691→          signal,
   692→        })
   693→      } catch (err) {
   694→        clearTimeout(timeoutId)
   695→        // Clean up inflight registration
   696→        if (isInflightRegistered) {
   697→          this._inflightRequests.delete(cacheKey)
   698→        }
   699→        // If we got a network error during a mutating request, try to queue it
   700→        if (err instanceof TypeError && err.message === 'Failed to fetch' && isMutating) {
   701→          try {
   702→            const { queueOperation } = await import('@/lib/offline-queue')
   703→            await queueOperation(
   704→              endpoint,
   705→              method,
   706→              headers,
   707→              options.body ? String(options.body) : null
   708→            )
   709→            return { offline: true, queued: true } as unknown as T
   710→          } catch {
   711→            // Fall through to original error handling
   712→          }
   713→        }
   714→        if (err instanceof DOMException && err.name === 'AbortError') {
   715→          throw new Error('Request timed out — the server took too long to respond. Please try again.')
   716→        }
   717→        if (err instanceof TypeError && err.message === 'Failed to fetch') {
   718→          throw new Error('Network error — please check your internet connection and try again.')
   719→        }
   720→        throw new Error(err instanceof Error ? err.message : 'An unexpected network error occurred')
   721→      }
   722→
   723→      clearTimeout(timeoutId)
   724→
   725→      // Handle non-JSON responses gracefully
   726→      let data: Record<string, unknown>
   727→      try {
   728→        data = await response.json() as Record<string, unknown>
   729→      } catch {
   730→        // Clean up inflight registration on parse error
   731→        if (isInflightRegistered) {
   732→          this._inflightRequests.delete(cacheKey)
   733→        }
   734→        if (!response.ok) {
   735→          // Map common HTTP error codes to user-friendly messages — never expose raw status codes
   736→          if (response.status === 401) throw new Error('Your session has expired. Please log in again.')
   737→          if (response.status === 403) throw new Error('You do not have permission to perform this action')
   738→          if (response.status === 404) throw new Error('The requested resource was not found')
   739→          if (response.status === 429) throw new Error('Too many requests — please wait a moment and try again')
   740→          if (response.status >= 500) throw new Error('Server error — please try again later')
   741→          throw new Error('An unexpected error occurred. Please try again.')
   742→        }
   743→        throw new Error('Server returned an invalid response')
   744→      }
   745→
   746→      // Clean up inflight registration after response is parsed
   747→      if (isInflightRegistered) {
   748→        this._inflightRequests.delete(cacheKey)
   749→      }
   750→
   751→      if (!response.ok) {
   752→        const errorMessage = (data.error as string) || (data.message as string) || 'An unexpected error occurred. Please try again.'
   753→        const errorCode = data.code as string | undefined
   754→
   755→        // Handle database unreachable errors
   756→        if (response.status === 503 && errorCode === 'DB_UNREACHABLE') {
   757→          const dbError = new Error('Database is unreachable') as Error & { code: string }
   758→          dbError.code = 'DB_UNREACHABLE'
   759→          throw dbError
   760→        }
   761→
   762→        // Handle auth errors — clear token and trigger logout if 401 from a non-auth endpoint
   763→        // Auth endpoints (login, register, me) legitimately return 401 for wrong credentials
   764→        // Also suppress auto-logout during active login to prevent race conditions
   765→        const isAuthEndpoint = endpoint.startsWith('/api/auth/login') || endpoint.startsWith('/api/auth/register') || endpoint.startsWith('/api/auth/me')
   766→        if (response.status === 401 && token && !isAuthEndpoint && !this._isLoggingIn) {
   767→          // Only clear token and trigger logout if the token we sent is STILL the current token.
   768→          // This prevents a stale background request from clearing a freshly-set token
   769→          // after the user logged out and logged back in.
   770→          const currentToken = this.getToken()
   771→          if (token === currentToken) {
   772→            this.clearToken()
   773→            // Trigger logout via auth store to update UI state
   774→            try {
   775→              const { useAuthStore } = await import('@/lib/stores/auth-store')
   776→              const store = useAuthStore.getState()
   777→              if (store.isAuthenticated) {
   778→                // Use setTimeout to avoid race conditions with concurrent requests.
   779→                // If multiple 401s come in at the same time, only the first one will
   780→                // trigger logout (subsequent ones will see isAuthenticated = false).
   781→                setTimeout(() => {
   782→                  const currentStore = useAuthStore.getState()
   783→                  if (currentStore.isAuthenticated) {
   784→                    currentStore.logout()
   785→                  }
   786→                }, 100)
   787→              }
   788→            } catch {
   789→              // If import fails (e.g. during SSR), just clear the token
   790→            }
   791→          }
   792→          throw new Error('Your session has expired. Please log in again.')
   793→        }
   794→
   795→        throw new Error(errorMessage)
   796→      }
   797→
   798→      // Cache successful GET responses for short TTL
   799→      if (isGet) {
   800→        this._responseCache.set(cacheKey, {
   801→          data: data as T,
   802→          expiresAt: Date.now() + ApiClient.RESPONSE_CACHE_TTL,
   803→        })
   804→      }
   805→
   806→      return data as T
   807→    })()
   808→
   809→    // Register the inflight promise for GET deduplication
   810→    if (isGet) {
   811→      this._inflightRequests.set(cacheKey, fetchPromise)
   812→    }
   813→
   814→    return fetchPromise
   815→  }
