    35→    35→function getStockStatus(quantity: number, threshold: number): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    36→    36→  if (quantity === 0) return { label: 'Out of Stock', variant: 'destructive' }
    37→    37→  if (quantity <= threshold) return { label: 'Low Stock', variant: 'secondary' }
    38→    38→  return { label: 'In Stock', variant: 'default' }
    39→    39→}

    41→    41→function formatDate(dateStr: string): string {
    42→    42→  return new Date(dateStr).toLocaleDateString('en-US', {
    43→    43→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit'
    44→    44→  })
    45→    45→}

    82→    82→// ============================================
    83→    83→function StockAdjustmentDialog({
    84→    84→  open,
    85→    85→  onClose,
    86→    86→  product,
    87→    87→  orgId,
    88→    88→  shopId,
    89→    89→  onAdjusted,
    90→    90→}: {
    91→    91→  open: boolean
    92→    92→  onClose: () => void
    93→    93→  product: Product | null
    94→    94→  orgId: string
    95→    95→  shopId?: string
    96→    96→  onAdjusted: () => void
    97→    97→}) {
    98→    98→  const [loading, setLoading] = useState(false)
    99→    99→
   100→   100→  const form = useForm<StockAdjustmentFormData>({
   101→   101→    resolver: zodResolver(stockAdjustmentSchema),
   102→   102→    defaultValues: { type: 'in', quantity: '' as unknown as number, reason: '' },
   103→   103→  })
   104→   104→
   105→   105→  const watchedType = form.watch('type')
   106→   106→  const watchedQty = form.watch('quantity')
   107→   107→  const qty = typeof watchedQty === 'number' ? watchedQty : (parseInt(String(watchedQty)) || 0)
   108→   108→  const previewStock = watchedType === 'in'
   109→   109→    ? (product?.quantity ?? 0) + qty
   110→   110→    : watchedType === 'out'
   111→   111→    ? Math.max(0, (product?.quantity ?? 0) - qty)
   112→   112→    : qty
   113→   113→
   114→   114→  const handleSubmit = async (data: StockAdjustmentFormData) => {
   115→   115→    if (!product) return
   116→   116→
   117→   117→    setLoading(true)
   118→   118→    try {
   119→   119→      await api.addStockMovement(orgId, {
   120→   120→        productId: product.id,
   121→   121→        type: data.type,
   122→   122→        quantity: data.quantity,
   123→   123→        reason: data.reason.trim(),
   124→   124→        shopId,
   125→   125→      })
   126→   126→      toast.success('Stock adjusted successfully')
   127→   127→      onAdjusted()
   128→   128→      onClose()
   129→   129→    } catch (err: unknown) {
   130→   130→      toast.error(getNetworkErrorMessage(err))
   131→   131→    } finally {
   132→   132→      setLoading(false)
   133→   133→    }
   134→   134→  }
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    if (!open) {
   138→   138→      form.reset({ type: 'in', quantity: '' as unknown as number, reason: '' })
   139→   139→    }
   140→   140→  }, [open, form])
   141→   141→
   142→   142→  if (!product) return null
   143→   143→
   144→   144→  return (
   145→   145→    <Dialog open={open} onOpenChange={onClose}>
   146→   146→      <DialogContent className="sm:max-w-md">
   147→   147→        <DialogHeader>
   148→   148→          <DialogTitle>Adjust Stock</DialogTitle>
   149→   149→          <DialogDescription>
   150→   150→            Adjust stock for <span className="font-semibold">{product.name}</span> (Current: {product.quantity})
   151→   151→          </DialogDescription>
   152→   152→        </DialogHeader>
   153→   153→        <Form {...form}>
   154→   154→          <form onSubmit={form.handleSubmit(handleSubmit)} className="space-y-4 py-2">
   155→   155→            <FormSelectField
   156→   156→              name="type"
   157→   157→              label="Adjustment Type"
   158→   158→              options={[
   159→   159→                { value: 'in', label: 'Add Stock', icon: <Plus className="size-3" /> },
   160→   160→                { value: 'out', label: 'Remove Stock', icon: <Minus className="size-3" /> },
   161→   161→                { value: 'adjustment', label: 'Adjustment (Set)', icon: <RefreshCw className="size-3" /> },
   162→   162→              ]}
   163→   163→            />
   164→   164→            <FormInputField
   165→   165→              name="quantity"
   166→   166→              label="Quantity"
   167→   167→              placeholder="Enter quantity"
   168→   168→              type="number"
   169→   169→              required
   170→   170→              min={1}
   171→   171→            />
   172→   172→            <FormInputField
   173→   173→              name="reason"
   174→   174→              label="Reason"
   175→   175→              placeholder="Enter reason for adjustment"
   176→   176→              required
   177→   177→            />
   178→   178→            {qty > 0 && (
   179→   179→              <div className="rounded-lg border bg-muted/50 p-3">
   180→   180→                <p className="text-sm text-muted-foreground">Preview</p>
   181→   181→                <p className="text-sm font-bold">
   182→   182→                  {product.quantity} → {previewStock}
   183→   183→                </p>
   184→   184→              </div>
   185→   185→            )}
   186→   186→            <DialogFooter>
   187→   187→              <Button type="button" variant="outline" onClick={onClose}>Cancel</Button>
   188→   188→              <FormSubmitButton isLoading={loading} loadingText="Adjusting...">
   189→   189→                Confirm Adjustment
   190→   190→              </FormSubmitButton>
   191→   191→            </DialogFooter>
   192→   192→          </form>
   193→   193→        </Form>
   194→   194→      </DialogContent>
   195→   195→    </Dialog>
   196→   196→  )
   197→   197→}

   202→   202→function AllProductsTab({
   203→   203→  orgId,
   204→   204→  shopId,
   205→   205→  onAdjust,
   206→   206→}: {
   207→   207→  orgId: string
   208→   208→  shopId?: string
   209→   209→  onAdjust: (product: Product) => void
   210→   210→}) {
   211→   211→  const [products, setProducts] = useState<Product[]>([])
   212→   212→  const [productTypes, setProductTypes] = useState<ProductType[]>([])
   213→   213→  const [loading, setLoading] = useState(true)
   214→   214→  const [search, setSearch] = useState('')
   215→   215→  const [filterType, setFilterType] = useState<string>('all')
   216→   216→  const [sortCol, setSortCol] = useState<string>('name')
   217→   217→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   218→   218→  const [page, setPage] = useState(1)
   219→   219→  const [totalPages, setTotalPages] = useState(1)
   220→   220→
   221→   221→  const fetchData = useCallback(async () => {
   222→   222→    try {
   223→   223→      const [productsRes, typesRes] = await Promise.all([
   224→   224→        api.getProducts(orgId, {
   225→   225→          search: search || undefined,
   226→   226→          productTypeId: filterType !== 'all' ? filterType : undefined,
   227→   227→          page,
   228→   228→          limit: 10,
   229→   229→          shopId,
   230→   230→        }),
   231→   231→        api.getProductTypes(orgId, shopId),
   232→   232→      ])
   233→   233→      setProducts(productsRes.products)
   234→   234→      setTotalPages(productsRes.pagination.totalPages)
   235→   235→      setProductTypes(typesRes.productTypes)
   236→   236→    } catch (error) {
   237→   237→      toast.error(getNetworkErrorMessage(error))
   238→   238→    } finally {
   239→   239→      setLoading(false)
   240→   240→    }
   241→   241→  }, [orgId, shopId, search, filterType, page])
   242→   242→
   243→   243→  useEffect(() => {
   244→   244→    fetchData()
   245→   245→  }, [fetchData])
   246→   246→
   247→   247→  const handleSort = (col: string) => {
   248→   248→    if (sortCol === col) {
   249→   249→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   250→   250→    } else {
   251→   251→      setSortCol(col)
   252→   252→      setSortDir('asc')
   253→   253→    }
   254→   254→  }
   255→   255→
   256→   256→  const sortedProducts = [...products].sort((a, b) => {
   257→   257→    const dir = sortDir === 'asc' ? 1 : -1
   258→   258→    switch (sortCol) {
   259→   259→      case 'name': return a.name.localeCompare(b.name) * dir
   260→   260→      case 'sku': return (a.sku || '').localeCompare(b.sku || '') * dir
   261→   261→      case 'quantity': return (a.quantity - b.quantity) * dir
   262→   262→      case 'costPrice': return (a.costPrice - b.costPrice) * dir
   263→   263→      case 'sellingPrice': return (a.sellingPrice - b.sellingPrice) * dir
   264→   264→      default: return 0
   265→   265→    }
   266→   266→  })
   267→   267→
   268→   268→  const SortHeader = ({ col, children }: { col: string; children: React.ReactNode }) => (
   269→   269→    <TableHead className="cursor-pointer select-none hover:bg-muted/50" onClick={() => handleSort(col)}>
   270→   270→      <span className="flex items-center gap-1">
   271→   271→        {children}
   272→   272→        <ArrowUpDown className={`size-3 ${sortCol === col ? 'opacity-100' : 'opacity-30'}`} />
   273→   273→      </span>
   274→   274→    </TableHead>
   275→   275→  )
   276→   276→
   277→   277→  if (loading) {
   278→   278→    return (
   279→   279→      <div className="space-y-4">
   280→   280→        <div className="flex flex-col sm:flex-row gap-3">
   281→   281→          <Skeleton className="h-10 flex-1" />
   282→   282→          <Skeleton className="h-10 w-40" />
   283→   283→        </div>
   284→   284→        {Array.from({ length: 5 }).map((_, i) => (
   285→   285→          <Skeleton key={i} className="h-14 w-full" />
   286→   286→        ))}
   287→   287→      </div>
   288→   288→    )
   289→   289→  }
   290→   290→
   291→   291→  return (
   292→   292→    <div className="space-y-4">
   293→   293→      {/* Filters */}
   294→   294→      <div className="flex flex-col sm:flex-row gap-3">
   295→   295→        <div className="relative flex-1">
   296→   296→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   297→   297→          <Input
   298→   298→            placeholder="Search products..."
   299→   299→            className="pl-9"
   300→   300→            aria-label="Search inventory"
   301→   301→            value={search}
   302→   302→            onChange={(e) => { setSearch(e.target.value); setPage(1) }}
   303→   303→          />
   304→   304→        </div>
   305→   305→        <Select value={filterType} onValueChange={(v) => { setFilterType(v); setPage(1) }}>
   306→   306→          <SelectTrigger className="w-full sm:w-48">
   307→   307→            <Filter className="size-4 mr-2" />
   308→   308→            <SelectValue placeholder="Filter by type" />
   309→   309→          </SelectTrigger>
   310→   310→          <SelectContent>
   311→   311→            <SelectItem value="all">All Types</SelectItem>
   312→   312→            {productTypes.map((pt) => (
   313→   313→              <SelectItem key={pt.id} value={pt.id}>{pt.name}</SelectItem>
   314→   314→            ))}
   315→   315→          </SelectContent>
   316→   316→        </Select>
   317→   317→      </div>
   318→   318→
   319→   319→      {/* Table */}
   320→   320→      <div className="rounded-lg border overflow-hidden">
   321→   321→        {/* Desktop table */}
   322→   322→        <div className="hidden md:block overflow-x-auto -mx-4 md:mx-0">
   323→   323→        <Table>
   324→   324→          <TableHeader>
   325→   325→            <TableRow>
   326→   326→              <SortHeader col="name">Product Name</SortHeader>
   327→   327→              <SortHeader col="sku">SKU</SortHeader>
   328→   328→              <TableHead>Type</TableHead>
   329→   329→              <SortHeader col="quantity">Quantity</SortHeader>
   330→   330→              <SortHeader col="costPrice">Cost Price</SortHeader>
   331→   331→              <SortHeader col="sellingPrice">Selling Price</SortHeader>
   332→   332→              <TableHead>Value</TableHead>
   333→   333→              <TableHead>Status</TableHead>
   334→   334→              <TableHead className="text-right">Actions</TableHead>
   335→   335→            </TableRow>
   336→   336→          </TableHeader>
   337→   337→          <TableBody>
   338→   338→            {sortedProducts.length === 0 ? (
   339→   339→              <TableRow>
   340→   340→                <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   341→   341→                  No products found
   342→   342→                </TableCell>
   343→   343→              </TableRow>
   344→   344→            ) : (
   345→   345→              sortedProducts.map((product) => {
   346→   346→                const status = getStockStatus(product.quantity, product.lowStockThreshold)
   347→   347→                return (
   348→   348→                  <TableRow key={product.id}>
   349→   349→                    <TableCell className="font-medium max-w-[180px] truncate">{product.name}</TableCell>
   350→   350→                    <TableCell className="text-muted-foreground text-sm">{product.sku || '—'}</TableCell>
   351→   351→                    <TableCell>
   352→   352→                      <Badge variant="outline" className="text-xs">{product.productType?.name || '—'}</Badge>
   353→   353→                    </TableCell>
   354→   354→                    <TableCell className="font-mono">{product.quantity}</TableCell>
   355→   355→                    <TableCell className="text-sm">{formatETB(product.costPrice)}</TableCell>
   356→   356→                    <TableCell className="text-sm">{formatETB(product.sellingPrice)}</TableCell>
   357→   357→                    <TableCell className="text-sm font-medium">{formatETB(product.quantity * product.sellingPrice)}</TableCell>
   358→   358→                    <TableCell>
   359→   359→                      <Badge variant={status.variant} className="text-xs">{status.label}</Badge>
   360→   360→                    </TableCell>
   361→   361→                    <TableCell className="text-right">
   362→   362→                      <Button size="sm" variant="outline" onClick={() => onAdjust(product)}>
   363→   363→                        Adjust Stock
   364→   364→                      </Button>
   365→   365→                    </TableCell>
   366→   366→                  </TableRow>
   367→   367→                )
   368→   368→              })
   369→   369→            )}
   370→   370→          </TableBody>
   371→   371→        </Table>
   372→   372→        </div>
   373→   373→        {/* Mobile card view */}
   374→   374→        <div className="md:hidden space-y-3 p-3">
   375→   375→          {sortedProducts.length === 0 ? (
   376→   376→            <EmptyState
   377→   377→              title="No products found"
   378→   378→              message="No products match your current filters."
   379→   379→            />
   380→   380→          ) : (
   381→   381→            sortedProducts.map((product) => {
   382→   382→              const status = getStockStatus(product.quantity, product.lowStockThreshold)
   383→   383→              return (
   384→   384→                <Card key={product.id} className="p-4">
   385→   385→                  <div className="flex items-start justify-between mb-2">
   386→   386→                    <div className="min-w-0 flex-1">
   387→   387→                      <p className="font-semibold text-sm truncate">{product.name}</p>

   389→   389→                    </div>
   390→   390→                    <Badge variant={status.variant} className="text-xs shrink-0 ml-2">{status.label}</Badge>
   391→   391→                  </div>
   392→   392→                  <div className="flex items-center justify-between text-xs mt-2">
   393→   393→                    <div className="flex items-center gap-3 text-muted-foreground">
   394→   394→                      <span>Qty: <strong className="text-foreground">{product.quantity}</strong></span>

   396→   396→                    </div>
   397→   397→                    <Button size="sm" variant="outline" className="h-7 text-xs" onClick={() => onAdjust(product)}>Adjust</Button>
   398→   398→                  </div>
   399→   399→                </Card>

   401→   401→            })
   402→   402→          )}
   403→   403→        </div>
   404→   404→      </div>
   405→   405→
   406→   406→      {/* Pagination */}
   407→   407→      {totalPages > 1 && (
   408→   408→        <div className="flex items-center justify-between">
   409→   409→          <p className="text-sm text-muted-foreground">Page {page} of {totalPages}</p>
   410→   410→          <div className="flex gap-2">
   411→   411→            <Button size="sm" variant="outline" disabled={page <= 1} onClick={() => setPage(page - 1)}>
   412→   412→              <ChevronLeft className="size-4" />
   413→   413→            </Button>
   414→   414→            <Button size="sm" variant="outline" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>
   415→   415→              <ChevronRight className="size-4" />
   416→   416→            </Button>
   417→   417→          </div>
   418→   418→        </div>
   419→   419→      )}
   420→   420→    </div>
   421→   421→  )
   422→   422→}

   439→   439→  const [loading, setLoading] = useState(true)
   440→   440→
