   537→class ApiClient {
   538→  // Supabase Auth uses httpOnly cookies for session management.
   539→  // The token is no longer stored in localStorage for security.
   540→  // We keep a reference for backward compat but auth is primarily cookie-based.
   541→  private _token: string | null = null
   542→  // Guard flag: when true, 401 auto-logout is suppressed to prevent
   543→  // stale background requests from interfering with an active login attempt.
   544→  private _isLoggingIn: boolean = false
   545→
   546→  private getToken(): string | null {
   547→    if (typeof window === 'undefined') return null
   548→    // Check memory first, then fallback to localStorage (for migration)
   549→    return this._token || localStorage.getItem('sb_token')
   550→  }
   551→
   552→  setToken(token: string): void {
   553→    if (typeof window === 'undefined') return
   554→    this._token = token
   555→    // Also keep in localStorage for migration period
   556→    localStorage.setItem('sb_token', token)
   557→  }
   558→
   559→  clearToken(): void {
   560→    if (typeof window === 'undefined') return
   561→    this._token = null
   562→    localStorage.removeItem('sb_token')
   563→  }
   564→
   565→  private async request<T>(
   566→    endpoint: string,
   567→    options: RequestInit = {}
   568→  ): Promise<T> {
   569→    const token = this.getToken()
   570→    const headers: Record<string, string> = {
   571→      'Content-Type': 'application/json',
   572→      ...(options.headers as Record<string, string>),
   573→    }
   574→
   575→    // Add JWT token to Authorization header if available
   576→    if (token) {
   577→      headers['Authorization'] = `Bearer ${token}`
   578→    }
   579→
   580→    // Offline interceptor: queue mutating requests, reject GET requests
   581→    const method = (options.method || 'GET').toUpperCase()
   582→    const isMutating = ['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)
   583→
   584→    if (typeof window !== 'undefined' && !navigator.onLine) {
   585→      if (isMutating) {
   586→        // Queue the operation for later sync
   587→        try {
   588→          const { queueOperation } = await import('@/lib/offline-queue')
   589→          await queueOperation(
   590→            endpoint,
   591→            method,
   592→            headers,
   593→            options.body ? String(options.body) : null
   594→          )
   595→          // Return a mock "offline queued" response so callers don't break
   596→          return { offline: true, queued: true } as unknown as T
   597→        } catch {
   598→          // If queuing fails, fall through to normal error
   599→        }
   600→      }
   601→      throw new Error('You are offline. Please check your internet connection and try again.')
   602→    }
   603→
   604→    // Add timeout (30s) to prevent hanging requests
   605→    const controller = new AbortController()
   606→    const timeoutId = setTimeout(() => controller.abort(), 30000)
   607→    const signal = options.signal || controller.signal
   608→
   609→    let response: Response
   610→    try {
   611→      response = await fetch(endpoint, {
   612→        ...options,
   613→        headers,
   614→        signal,
   615→      })
   616→    } catch (err) {
   617→      clearTimeout(timeoutId)
   618→      // If we got a network error during a mutating request, try to queue it
   619→      if (err instanceof TypeError && err.message === 'Failed to fetch' && isMutating) {
   620→        try {
   621→          const { queueOperation } = await import('@/lib/offline-queue')
   622→          await queueOperation(
   623→            endpoint,
   624→            method,
   625→            headers,
   626→            options.body ? String(options.body) : null
   627→          )
   628→          return { offline: true, queued: true } as unknown as T
   629→        } catch {
   630→          // Fall through to original error handling
   631→        }
   632→      }
   633→      if (err instanceof DOMException && err.name === 'AbortError') {
   634→        throw new Error('Request timed out — the server took too long to respond. Please try again.')
   635→      }
   636→      if (err instanceof TypeError && err.message === 'Failed to fetch') {
   637→        throw new Error('Network error — please check your internet connection and try again.')
   638→      }
   639→      throw new Error(err instanceof Error ? err.message : 'An unexpected network error occurred')
   640→    }
   641→
   642→    clearTimeout(timeoutId)
   643→
   644→    // Handle non-JSON responses gracefully
   645→    let data: Record<string, unknown>
   646→    try {
   647→      data = await response.json() as Record<string, unknown>
   648→    } catch {
   649→      if (!response.ok) {
   650→        // Map common HTTP error codes to user-friendly messages — never expose raw status codes
   651→        if (response.status === 401) throw new Error('Your session has expired. Please log in again.')
   652→        if (response.status === 403) throw new Error('You do not have permission to perform this action')
   653→        if (response.status === 404) throw new Error('The requested resource was not found')
   654→        if (response.status === 429) throw new Error('Too many requests — please wait a moment and try again')
   655→        if (response.status >= 500) throw new Error('Server error — please try again later')
   656→        throw new Error('An unexpected error occurred. Please try again.')
   657→      }
   658→      throw new Error('Server returned an invalid response')
   659→    }
   660→
   661→    if (!response.ok) {
   662→      const errorMessage = (data.error as string) || (data.message as string) || 'An unexpected error occurred. Please try again.'
   663→      const errorCode = data.code as string | undefined
   664→
   665→      // Handle database unreachable errors
   666→      if (response.status === 503 && errorCode === 'DB_UNREACHABLE') {
   667→        const dbError = new Error('Database is unreachable') as Error & { code: string }
   668→        dbError.code = 'DB_UNREACHABLE'
   669→        throw dbError
   670→      }
   671→
   672→      // Handle auth errors — clear token and trigger logout if 401 from a non-auth endpoint
   673→      // Auth endpoints (login, register, me) legitimately return 401 for wrong credentials
   674→      // Also suppress auto-logout during active login to prevent race conditions
   675→      const isAuthEndpoint = endpoint.startsWith('/api/auth/login') || endpoint.startsWith('/api/auth/register') || endpoint.startsWith('/api/auth/me')
   676→      if (response.status === 401 && token && !isAuthEndpoint && !this._isLoggingIn) {
   677→        // Only clear token and trigger logout if the token we sent is STILL the current token.
   678→        // This prevents a stale background request from clearing a freshly-set token
   679→        // after the user logged out and logged back in.
   680→        const currentToken = this.getToken()
   681→        if (token === currentToken) {
   682→          this.clearToken()
   683→          // Trigger logout via auth store to update UI state
   684→          try {
   685→            const { useAuthStore } = await import('@/lib/stores/auth-store')
   686→            const store = useAuthStore.getState()
   687→            if (store.isAuthenticated) {
   688→              // Use setTimeout to avoid race conditions with concurrent requests.
   689→              // If multiple 401s come in at the same time, only the first one will
   690→              // trigger logout (subsequent ones will see isAuthenticated = false).
   691→              setTimeout(() => {
   692→                const currentStore = useAuthStore.getState()
   693→                if (currentStore.isAuthenticated) {
   694→                  currentStore.logout()
   695→                }
   696→              }, 100)
   697→            }
   698→          } catch {
   699→            // If import fails (e.g. during SSR), just clear the token
   700→          }
   701→        }
   702→        throw new Error('Your session has expired. Please log in again.')
   703→      }
   704→
   705→      throw new Error(errorMessage)
   706→    }
   707→
   708→    return data as T

   732→      }
   733→      return data
   734→    } finally {
   735→      // Always reset the guard flag, even if login fails
   736→      this._isLoggingIn = false
   737→    }
   738→  }
