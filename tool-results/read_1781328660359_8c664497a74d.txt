    32→// ============================================
    33→// Helpers
    34→// ============================================
    35→function formatETB(amount: number): string {
    36→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    37→}
    38→
    39→function formatShortETB(amount: number): string {
    40→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`
    41→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(1)}K`
    42→  return `ETB ${amount.toFixed(0)}`
    43→}

   118→        startDate = endDate
   119→        break

   121→        const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)

   131→        break

   135→        endDate = customEnd || now.toISOString().split('T')[0]

   138→        startDate = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   139→    }
