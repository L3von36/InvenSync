    55→function formatETB(amount: number): string {
    56→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 0, maximumFractionDigits: 0 })}`
    57→}
    58→
    59→function daysUntil(dateStr: string | null): number | null {
    60→  if (!dateStr) return null
    61→  const diff = new Date(dateStr).getTime() - Date.now()
    62→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
    63→}

   131→function getModuleStatus(m: ModuleData): ModuleStatus {
   132→  if (!m.orgStatus) return 'available'
   133→  if (m.orgStatus.status === 'active') return 'active'
   134→  if (m.orgStatus.status === 'trial') return 'trial'
   135→  if (m.orgStatus.status === 'expired' || m.orgStatus.status === 'cancelled') return 'expired'
   136→  // If orgStatus exists but status is unexpected, check isActive
   137→  if (m.orgStatus.isActive) return 'active'
   138→  return 'available'
   139→}

   144→function ModuleCard({
   145→  mod,
   146→  onRequestAccess,
   147→  requestingKey,
   148→  onSubscribe,
   149→  onRenew,
   150→  onActivate,
   151→  activatingKey,
   152→}: {
   153→  mod: EnrichedModule
   154→  onRequestAccess: (moduleKey: string) => void
   155→  requestingKey: string | null
   156→  onSubscribe: (mod: EnrichedModule) => void
   157→  onRenew: (mod: EnrichedModule) => void
   158→  onActivate: (moduleKey: string) => void
   159→  activatingKey: string | null
   160→}) {
   161→  const statusConfig = STATUS_CONFIG[mod.status]
   162→  const days = daysUntil(mod.orgStatus?.expiresAt ?? null)
   163→  const isExpired = mod.status === 'expired'
   164→  const isTrial = mod.status === 'trial'
   165→  const isAvailable = mod.status === 'available'
   166→  const isRequested = mod.requested
   167→

   169→  const opacityClass = isExpired ? 'opacity-75' : isAvailable ? 'opacity-90' : ''
   170→
   171→  // Trial progress calculation
   172→  const trialDays = mod.freeTrialDays || 30
   173→  const trialProgress = isTrial && days !== null && days > 0
   174→    ? Math.max(0, Math.min(100, ((trialDays - days) / trialDays) * 100))
   175→    : isTrial ? 100 : 0
   176→
   177→  // Urgency levels for trial
   178→  const isCritical = isTrial && days !== null && days <= 3 && days > 0
   179→  const isWarning = isTrial && days !== null && days <= 7 && days > 3
   180→

   182→    <Card className={`border-l-4 ${cardAccent} transition-shadow hover:shadow-md ${opacityClass}`}>
   183→      <CardContent className="p-4 space-y-3">
   184→        {/* Header */}
   185→        <div className="flex items-start gap-3">
   186→          <div className={`size-11 rounded-lg flex items-center justify-center shrink-0 ${ICON_BG[mod.category] || 'bg-gray-100 text-gray-600'}`}>
   187→            {getModuleIcon(mod.icon)}
   188→          </div>
   189→          <div className="flex-1 min-w-0">
   190→            <div className="flex items-center gap-2 flex-wrap">
   191→              <h3 className="font-semibold text-sm">{mod.name}</h3>
   192→              <Badge className={statusConfig.color} variant="outline">
   193→                <span className="flex items-center gap-1">
   194→                  {statusConfig.icon}
   195→                  {isTrial && days !== null && days > 0 ? `${days}d left` : statusConfig.label}
   196→                </span>
   197→              </Badge>
   198→            </div>
   199→            <p className="text-xs text-muted-foreground font-mono mt-0.5">{mod.key}</p>
   200→          </div>
   201→        </div>
   202→
   203→        {/* Description */}
   204→        <p className="text-xs text-muted-foreground line-clamp-2">{mod.description || 'No description available'}</p>
   205→
   206→        {/* Category & Price */}
   207→        <div className="flex items-center justify-between flex-wrap gap-1.5">
   208→          <Badge className={CATEGORY_COLORS[mod.category] || CATEGORY_COLORS.core} variant="outline">
   209→            {mod.category}
   210→          </Badge>
   211→          {mod.isFree ? (
   212→            <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300" variant="outline">
   213→              Free
   214→            </Badge>
   215→          ) : (
   216→            <span className="text-sm font-semibold">
   217→              {formatETB(mod.priceETB)}
   218→              <span className="text-xs font-normal text-muted-foreground">/mo</span>
   219→            </span>
   220→          )}
   221→        </div>
   222→
   223→        {/* Trial countdown with progress bar */}
   224→        {isTrial && (
   225→          <>
   226→            <Separator />
   227→            {days !== null && days > 0 ? (
   228→              <div className={`rounded-lg p-3 space-y-2 ${
   229→                isCritical
   230→                  ? 'bg-red-50 dark:bg-red-900/20 animate-pulse'
   231→                  : isWarning
   232→                    ? 'bg-amber-50 dark:bg-amber-900/20 animate-pulse'
   233→                    : 'bg-amber-50 dark:bg-amber-900/20'
   234→              }`}>
   235→                <div className="flex items-center justify-between">
   236→                  <div className="flex items-center gap-2">
   237→                    <Clock className={`size-4 shrink-0 ${
   238→                      isCritical ? 'text-red-600' : 'text-amber-600'
   239→                    }`} />
   240→                    <p className={`text-xs font-medium ${
   241→                      isCritical ? 'text-red-800 dark:text-red-300' : 'text-amber-800 dark:text-amber-300'
   242→                    }`}>
   243→                      Trial Period
   244→                    </p>
   245→                  </div>
   246→                  <span className={`text-xs font-bold ${
   247→                    isCritical ? 'text-red-700 dark:text-red-300' : 'text-amber-700 dark:text-amber-400'
   248→                  }`}>
   249→                    {days} day{days !== 1 ? 's' : ''} left
   250→                  </span>
   251→                </div>
   252→                <Progress
   253→                  value={100 - trialProgress}
   254→                  className={`h-2 ${
   255→                    isCritical
   256→                      ? '[&>[data-slot=progress-indicator]]:bg-red-500'
   257→                      : isWarning
   258→                        ? '[&>[data-slot=progress-indicator]]:bg-amber-500'
   259→                        : '[&>[data-slot=progress-indicator]]:bg-amber-400'

   274→                      ? '⚠ Trial ending very soon — contact us to continue!'

   276→                  </p>
   277→                )}
   278→              </div>

   280→              <div className="flex items-center gap-2 p-3 rounded-lg bg-red-50 dark:bg-red-900/20">
   281→                <AlertTriangle className="size-4 text-red-600 shrink-0" />
   282→                <div>
   283→                  <p className="text-xs font-medium text-red-800 dark:text-red-300">Trial Expired</p>
   284→                  <p className="text-[11px] text-red-700 dark:text-red-400">Contact support to renew your subscription</p>
   285→                </div>
   286→              </div>
   287→            )}
   288→          </>
   289→        )}
   290→
   291→        {/* Expired module info */}
   292→        {isExpired && (
   293→          <>
   294→            <Separator />
   295→            <div className="flex items-center gap-2 p-2 rounded-lg bg-red-50 dark:bg-red-900/20">
   296→              <XCircle className="size-4 text-red-600 shrink-0" />
   297→              <div>
   298→                <p className="text-xs font-medium text-red-800 dark:text-red-300">Module Expired</p>
   299→                <p className="text-[11px] text-red-700 dark:text-red-400">Contact support@invensync.com to renew</p>
   300→              </div>
   301→            </div>
   302→          </>
   303→        )}
   304→
   305→        {/* Action buttons */}
   306→        {isAvailable && (
   307→          <>
   308→            <Separator />
   309→            {mod.isFree ? (
   310→              <Button
   311→                size="sm"
   312→                className="w-full gap-1.5"
   313→                disabled={activatingKey === mod.key}
   314→                onClick={() => onActivate(mod.key)}

   337→          <>
   338→            <Separator />
   339→            <Button
   340→              size="sm"
   341→              className="w-full gap-1.5"
   342→              onClick={() => onSubscribe(mod)}
   343→            >
   344→              <Sparkles className="size-3.5" />

   346→            </Button>
   347→          </>
   348→        )}
   349→
   350→        {mod.status === 'active' && !mod.isFree && (
   351→          <>
   352→            <Separator />
   353→            <div className="space-y-2">
   354→              <div className="flex items-center justify-between text-xs text-muted-foreground">

   356→                  <CheckCircle2 className="size-3 text-emerald-600" /> Active

   359→                  <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300" variant="outline">
   360→                    Auto-renew
   361→                  </Badge>
   362→                )}
   363→              </div>
   364→              {!mod.orgStatus?.autoRenew && (
   365→                <Button
   366→                  size="sm"
   367→                  variant="outline"
   368→                  className="w-full gap-1.5"

   372→                  Enable Auto-Renew
   373→                </Button>
   374→              )}
   375→              {days !== null && days <= 7 && days > 0 && !mod.orgStatus?.autoRenew && (
   376→                <div className="flex items-center gap-2 p-2 rounded-lg bg-amber-50 dark:bg-amber-900/20">
   377→                  <AlertTriangle className="size-3.5 text-amber-600 shrink-0" />
   378→                  <div className="flex-1 min-w-0">
   379→                    <p className="text-[11px] font-medium text-amber-800 dark:text-amber-300">

   382→                  </div>

   399→            <Separator />

   417→// Subscribe Dialog
   418→// ============================================
   419→function SubscribeDialog({
   420→  open,
   421→  onOpenChange,
   422→  module: mod,
   423→  onSuccess,
   424→}: {
   425→  open: boolean
   426→  onOpenChange: (open: boolean) => void
   427→  module: EnrichedModule | null
   428→  onSuccess: () => void
   429→}) {
   430→  const { currentOrg } = useAuthStore()
   431→  const [billingCycle, setBillingCycle] = useState<'monthly' | 'yearly'>('monthly')
   432→  const [loading, setLoading] = useState(false)
   433→
   434→  if (!mod || !currentOrg) return null
   435→
   436→  const monthlyPrice = mod.priceETB
   437→  const yearlyPrice = mod.priceETB * 10
   438→  const currentPrice = billingCycle === 'monthly' ? monthlyPrice : yearlyPrice
   439→  const savingsPercent = Math.round((2 / 12) * 100)
   440→
   441→  const handleSubmit = async () => {
   442→    setLoading(true)
   443→    try {
   444→      await api.subscribeModule(currentOrg.id, mod.key, billingCycle)
   445→      toast.success(`Subscribed to ${mod.name} successfully!`)
   446→      onOpenChange(false)
   447→      onSuccess()
   448→    } catch (err) {
   449→      toast.error(getNetworkErrorMessage(err))
   450→    } finally {
   451→      setLoading(false)
   452→    }
   453→  }
   454→
   455→  return (
   456→    <Dialog open={open} onOpenChange={onOpenChange}>
   457→      <DialogContent className="sm:max-w-md">
   458→        <DialogHeader>
   459→          <DialogTitle className="flex items-center gap-2">
   460→            <CreditCard className="size-5 text-primary" />
   461→            Subscribe to {mod.name}
   462→          </DialogTitle>
   463→          <DialogDescription>
   464→            {mod.description || `Choose a billing cycle and subscribe to the ${mod.name} module.`}
   465→          </DialogDescription>
   466→        </DialogHeader>
   467→
   468→        <div className="space-y-4">
   469→          {/* Billing Cycle Toggle */}
   470→          <div className="space-y-2">
   471→            <p className="text-sm font-medium">Billing Cycle</p>
   472→            <ToggleGroup
   473→              type="single"
   474→              value={billingCycle}
   475→              onValueChange={(val) => {
   476→                if (val) setBillingCycle(val as 'monthly' | 'yearly')
   477→              }}
   478→              variant="outline"
   479→              className="w-full"
   480→            >
   481→              <ToggleGroupItem value="monthly" className="flex-1 text-xs">
   482→                Monthly
   483→              </ToggleGroupItem>
   484→              <ToggleGroupItem value="yearly" className="flex-1 text-xs">
   485→                Yearly
   486→              </ToggleGroupItem>
   487→            </ToggleGroup>
   488→          </div>
   489→
   490→          {/* Pricing Card */}
   491→          <div className="rounded-lg border p-4 space-y-3">
   492→            <div className="flex items-center justify-between">
   493→              <span className="text-sm text-muted-foreground">
   494→                {billingCycle === 'monthly' ? 'Monthly' : 'Yearly'} price
   495→              </span>
   496→              {billingCycle === 'yearly' && (
   497→                <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300" variant="outline">
   498→                  Save {savingsPercent}%
   499→                </Badge>
   500→              )}
   501→            </div>
   502→            <div className="flex items-baseline gap-1">
   503→              <span className="text-2xl font-bold">{formatETB(currentPrice)}</span>
   504→              <span className="text-sm text-muted-foreground">
   505→                /{billingCycle === 'monthly' ? 'mo' : 'yr'}
   506→              </span>
   507→            </div>
   508→            {billingCycle === 'yearly' && (
   509→              <p className="text-xs text-emerald-700 dark:text-emerald-400">
   510→                2 months free with yearly billing
   511→              </p>
   512→            )}
   513→            {billingCycle === 'monthly' && (
   514→              <p className="text-xs text-muted-foreground">
   515→                Billed monthly at {formatETB(monthlyPrice)}/month
   516→              </p>
   517→            )}
   518→          </div>
   519→        </div>
   520→
   521→        <DialogFooter>
   522→          <FormSubmitButton
   523→            isLoading={loading}
   524→            loadingText="Subscribing..."
   525→            onClick={handleSubmit}
   526→            className="w-full gap-1.5"
   527→          >
   528→            <CreditCard className="size-4" />

   540→function RenewDialog({
   541→  open,
   542→  onOpenChange,
   543→  module: mod,
   544→  onSuccess,
   545→}: {
   546→  open: boolean
   547→  onOpenChange: (open: boolean) => void

   549→  onSuccess: () => void
   550→}) {
   551→  const { currentOrg } = useAuthStore()

   553→
   554→  if (!mod || !currentOrg) return null
   555→
   556→  const currentExpiry = mod.orgStatus?.expiresAt
   557→  const newExpiry = currentExpiry
   558→    ? new Date(new Date(currentExpiry).getTime() + 30 * 24 * 60 * 60 * 1000)
   559→    : new Date(Date.now() + 30 * 24 * 60 * 60 * 1000)
   560→
   561→  const handleSubmit = async () => {
   562→    setLoading(true)
   563→    try {
   564→      await api.renewModule(currentOrg.id, mod.key)
   565→      toast.success(`Renewed ${mod.name} successfully!`)

   567→      onSuccess()
   568→    } catch (err) {
   569→      toast.error(getNetworkErrorMessage(err))

   576→    <Dialog open={open} onOpenChange={onOpenChange}>
   577→      <DialogContent className="sm:max-w-md">
   578→        <DialogHeader>
   579→          <DialogTitle className="flex items-center gap-2">
   580→            <RefreshCw className="size-5 text-primary" />
   581→            Renew {mod.name}
   582→          </DialogTitle>
   583→          <DialogDescription>
   584→            Renew your subscription to continue using the {mod.name} module.
   585→          </DialogDescription>
   586→        </DialogHeader>
   587→
   588→        <div className="space-y-3">
   589→          {/* Current expiry */}
   590→          {currentExpiry && (
   591→            <div className="rounded-lg border p-3 space-y-1">
   592→              <div className="flex items-center justify-between">
   593→                <span className="text-xs text-muted-foreground">Current expiry</span>
   594→                <CalendarDays className="size-3.5 text-muted-foreground" />
   595→              </div>
   596→              <p className="text-sm font-medium">
   597→                {new Date(currentExpiry).toLocaleDateString('en-US', {
   598→                  year: 'numeric', month: 'long', day: 'numeric',
   599→                })}
   600→              </p>
   601→            </div>
   602→          )}
   603→
   604→          {/* New expiry after renewal */}
   605→          <div className="rounded-lg border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-900/20 p-3 space-y-1">
   606→            <div className="flex items-center justify-between">
   607→              <span className="text-xs text-emerald-700 dark:text-emerald-400">New expiry after renewal</span>
   608→              <ArrowRight className="size-3.5 text-emerald-600" />
   609→            </div>
   610→            <p className="text-sm font-medium text-emerald-800 dark:text-emerald-300">
   611→              {newExpiry.toLocaleDateString('en-US', {
   612→                year: 'numeric', month: 'long', day: 'numeric',
   613→              })}
   614→            </p>
   615→          </div>
   616→
   617→          {/* Price info */}
   618→          {!mod.isFree && (
   619→            <div className="flex items-center justify-between text-sm px-1">
   620→              <span className="text-muted-foreground">Renewal cost</span>
   621→              <span className="font-semibold">{formatETB(mod.priceETB)}/mo</span>
   622→            </div>
   623→          )}
   624→        </div>
   625→
   626→        <DialogFooter>
   627→          <FormSubmitButton
   628→            isLoading={loading}
   629→            loadingText="Renewing..."
   630→            onClick={handleSubmit}
   631→            className="w-full gap-1.5"
   632→          >
   633→            <RefreshCw className="size-4" />
   634→            Renew Subscription
   635→          </FormSubmitButton>
   636→        </DialogFooter>
   637→      </DialogContent>
   638→    </Dialog>
   639→  )

   675→// ============================================
   676→// Modules Page
   677→// ============================================
   678→export function ModulesPage() {
   679→  const { currentOrg } = useAuthStore()
   680→  const [modules, setModules] = useState<EnrichedModule[]>([])
   681→  const [loading, setLoading] = useState(true)
   682→  const [fetchError, setFetchError] = useState<string | null>(null)
   683→  const [filter, setFilter] = useState<'all' | ModuleStatus>('all')
   684→  const [requestedKeys, setRequestedKeys] = useState<Set<string>>(new Set())
   685→  const [requestingKey, setRequestingKey] = useState<string | null>(null)
   686→  const [subscribeDialogOpen, setSubscribeDialogOpen] = useState(false)
   687→  const [renewDialogOpen, setRenewDialogOpen] = useState(false)
   688→  const [selectedModule, setSelectedModule] = useState<EnrichedModule | null>(null)
   689→  const [activatingKey, setActivatingKey] = useState<string | null>(null)
   690→
   691→  const fetchModules = useCallback(async () => {
   692→    if (!currentOrg) return
   693→    try {
   694→      setFetchError(null)
   695→      const data = await api.getModules(currentOrg.id)
   696→      const enriched: EnrichedModule[] = data.modules.map((m: ModuleData) => ({
   697→        id: m.id,
   698→        key: m.key,
   699→        name: m.name,
   700→        description: m.description,
   701→        icon: m.icon,
   702→        category: m.category,
   703→        priceETB: m.priceETB,
   704→        isFree: m.isFree ?? m.priceETB === 0,
   705→        freeTrialDays: m.freeTrialDays || 30,
   706→        billingCycle: m.billingCycle || 'monthly',
   707→        order: m.order,
   708→        status: getModuleStatus(m),
   709→        orgStatus: m.orgStatus,
   710→        requested: false,
   711→      }))
   712→      setModules(enriched)
   713→    } catch (err) {
   714→      setFetchError(getNetworkErrorMessage(err))
   715→    } finally {
   716→      setLoading(false)
   717→    }
   718→  }, [currentOrg])
   719→
   720→  useEffect(() => { fetchModules() }, [fetchModules])
   721→
   722→  const handleRequestAccess = async (moduleKey: string) => {
   723→    if (!currentOrg) return
   724→    setRequestingKey(moduleKey)
   725→    try {
   726→      await api.requestModule(currentOrg.id, moduleKey)
   727→      toast.success('Request sent! Our team will contact you shortly.')
   728→      setRequestedKeys(prev => new Set(prev).add(moduleKey))
   729→      setModules(prev =>
   730→        prev.map(m =>
   731→          m.key === moduleKey ? { ...m, requested: true } : m
   732→        )
   733→      )
   734→    } catch (err) {
   735→      toast.error(getNetworkErrorMessage(err))
   736→    } finally {
   737→      setRequestingKey(null)
   738→    }
   739→  }
   740→
   741→  const handleActivate = async (moduleKey: string) => {
   742→    if (!currentOrg) return
   743→    setActivatingKey(moduleKey)
   744→    try {
   745→      await api.activateModule(currentOrg.id, moduleKey)
   746→      toast.success('Module activated successfully!')
   747→      fetchModules()
   748→    } catch (err) {
   749→      toast.error(getNetworkErrorMessage(err))
   750→    } finally {
   751→      setActivatingKey(null)
   752→    }
   753→  }
   754→
   755→  const handleOpenSubscribe = (mod: EnrichedModule) => {
   756→    setSelectedModule(mod)
   757→    setSubscribeDialogOpen(true)
   758→  }
   759→
   760→  const handleOpenRenew = (mod: EnrichedModule) => {
   761→    setSelectedModule(mod)
   762→    setRenewDialogOpen(true)
   763→  }
   764→
   765→  const filteredModules = modules.filter(m => {
   766→    if (filter === 'all') return true
   767→    return m.status === filter
   768→  })
   769→

   771→  const trialCount = modules.filter(m => m.status === 'trial').length
   772→  const availableCount = modules.filter(m => m.status === 'available').length
   773→  const expiredCount = modules.filter(m => m.status === 'expired').length
   774→  const trialModulesCount = modules.filter(m => m.status === 'trial' || m.status === 'available').length
   775→
   776→  // Expiry warning data
   777→  const expiringModules = modules.filter(m => {
   778→    const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   779→    return days !== null && days <= 7
   780→  })
   781→  const criticalModules = expiringModules.filter(m => {
   782→    const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   783→    return days !== null && (days <= 1 || days <= 0)
   784→  })
   785→
   786→  if (loading) {
   787→    return (
   788→      <div className="space-y-6">
   789→        <Skeleton className="h-10 w-64" />
   790→        <Skeleton className="h-20" />
   791→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   792→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20" />)}

   795→          {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-64" />)}
   796→        </div>
   797→      </div>
   798→    )
   799→  }

   801→  if (fetchError) {

   803→  }
   804→
   805→  return (

   807→      {/* Header */}
   808→      <div>
   809→        <h1 className="text-base font-semibold tracking-tight">My Modules</h1>
   810→        <p className="text-muted-foreground text-sm">Explore, request, and manage your InvenSync modules</p>
   811→      </div>
   812→
   813→      {/* Expiry Warning Banner */}
   814→      {expiringModules.length > 0 && (
   815→        <Alert className={
   816→          criticalModules.length > 0
   817→            ? 'border-red-300 bg-red-50 dark:border-red-800 dark:bg-red-900/20'
   818→            : 'border-amber-300 bg-amber-50 dark:border-amber-800 dark:bg-amber-900/20'
   819→        }>
   820→          <AlertTriangle className={
   821→            criticalModules.length > 0
   822→              ? 'size-4 text-red-600'
   823→              : 'size-4 text-amber-600'
   824→          } />
   825→          <AlertTitle className={
   826→            criticalModules.length > 0
   827→              ? 'text-sm font-medium text-red-800 dark:text-red-300'
   828→              : 'text-sm font-medium text-amber-800 dark:text-amber-300'
   829→          }>
   830→            {criticalModules.length > 0
   831→              ? 'Modules expiring or expired!'
   832→              : 'Modules expiring soon'}
   833→          </AlertTitle>
   834→          <AlertDescription className={
   835→            criticalModules.length > 0
   836→              ? 'text-xs text-red-700 dark:text-red-400'
   837→              : 'text-xs text-amber-700 dark:text-amber-400'
   838→          }>
   839→            {expiringModules.map(m => {
   840→              const days = daysUntil(m.orgStatus?.expiresAt ?? null)
   841→              const label = days !== null && days <= 0
   842→                ? `${m.name} (expired)`
   843→                : days !== null
   844→                  ? `${m.name} (${days}d left)`
   845→                  : m.name
   846→              return label
   847→            }).join(', ')}
   848→            {' — '}
   849→            <button

   851→                const section = document.getElementById('modules-grid')
   852→                section?.scrollIntoView({ behavior: 'smooth' })
   853→              }}
   854→              className="underline font-medium"
   855→            >
   856→              View modules
   857→            </button>
   858→          </AlertDescription>
   859→        </Alert>
