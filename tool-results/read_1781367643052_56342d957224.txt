    86→    86→function getStockStatus(
    87→    87→  quantity: number,
    88→    88→  lowStockThreshold: number
    89→    89→): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    90→    90→  if (quantity <= 0) return { label: 'Out of Stock', variant: 'destructive' }
    91→    91→  if (quantity <= lowStockThreshold) return { label: 'Low Stock', variant: 'secondary' }
    92→    92→  return { label: 'In Stock', variant: 'default' }
    93→    93→}

   111→   111→/** Safely parse attribute options that may be double-encoded */
   112→   112→function parseAttributeOptions(raw: string | null | undefined): string[] | null {
   113→   113→  if (!raw) return null
   114→   114→  try {
   115→   115→    const parsed = JSON.parse(raw)
   116→   116→    if (Array.isArray(parsed)) {
   117→   117→      return parsed
   118→   118→    } else if (typeof parsed === 'string') {
   119→   119→      // Handle double-encoded: "Red,Blue,Green" instead of ["Red","Blue","Green"]
   120→   120→      return parsed.split(',').map((s: string) => s.trim()).filter(Boolean)
   121→   121→    }
   122→   122→    return null
   123→   123→  } catch {
   124→   124→    // If parsing fails, try splitting by comma
   125→   125→    return raw.split(',').map(s => s.trim()).filter(Boolean)
   126→   126→  }
   127→   127→}

   307→   307→          name: data.name.trim(),

   322→   322→      onSaved()
   323→   323→    } catch (error) {
