    46→interface Expense {
    47→  id: string
    48→  organizationId: string
    49→  shopId?: string | null
    50→  category: string
    51→  amount: number
    52→  description?: string | null
    53→  expenseDate: string
    54→  isRecurring: boolean
    55→  recurringPeriod?: string | null
    56→  createdAt: string
    57→  updatedAt: string
    58→  shop?: { id: string; name: string } | null
    59→}
    60→
    61→const EXPENSE_CATEGORIES = [
    62→  { value: 'rent', label: 'Rent' },
    63→  { value: 'salary', label: 'Salary' },
    64→  { value: 'utilities', label: 'Utilities' },
    65→  { value: 'marketing', label: 'Marketing' },
    66→  { value: 'supplies', label: 'Supplies' },
    67→  { value: 'transport', label: 'Transport' },
    68→  { value: 'other', label: 'Other' },
    69→]
    70→
    71→const expenseFormSchema = z.object({
    72→  category: z.string().min(1, 'Category is required'),
    73→  amount: z.number({ message: 'Amount is required' }).positive('Amount must be greater than 0'),
    74→  description: z.string().optional(),
    75→  expenseDate: z.string().min(1, 'Date is required'),
    76→  shopId: z.string().optional(),
    77→  isRecurring: z.boolean().optional(),
    78→  recurringPeriod: z.string().optional(),
    79→})
    80→
    81→type ExpenseFormData = z.infer<typeof expenseFormSchema>

    87→const etbFormatter = new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB' })
    88→const formatETB = (val: number) => etbFormatter.format(val)

    90→const CATEGORY_LABELS: Record<string, string> = {
    91→  rent: 'Rent', salary: 'Salary', utilities: 'Utilities',
    92→  marketing: 'Marketing', supplies: 'Supplies', transport: 'Transport', other: 'Other',
    93→}

    95→const CATEGORY_COLORS: Record<string, string> = {
    96→  rent: 'bg-blue-100 text-blue-700',
    97→  salary: 'bg-purple-100 text-purple-700',
    98→  utilities: 'bg-amber-100 text-amber-700',
    99→  marketing: 'bg-pink-100 text-pink-700',
   100→  supplies: 'bg-emerald-100 text-emerald-700',
   101→  transport: 'bg-cyan-100 text-cyan-700',
   102→  other: 'bg-gray-100 text-gray-700',
   103→}
   104→
   105→function formatMonthLabel(monthKey: string): string {
   106→  const MONTH_LABELS: Record<string, string> = {
   107→    '01': 'Jan', '02': 'Feb', '03': 'Mar', '04': 'Apr',
   108→    '05': 'May', '06': 'Jun', '07': 'Jul', '08': 'Aug',
   109→    '09': 'Sep', '10': 'Oct', '11': 'Nov', '12': 'Dec',
   110→  }
   111→  const [year, month] = monthKey.split('-')
   112→  return `${MONTH_LABELS[month] || month} ${year.slice(2)}`
   113→}

   119→export function ExpensesPage() {
   120→  const { currentOrg, shops } = useAuthStore()
   121→  const [expenses, setExpenses] = useState<Expense[]>([])
   122→  const [isLoading, setIsLoading] = useState(true)
   123→  const [error, setError] = useState<string | null>(null)
   124→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   125→  const [searchQuery, setSearchQuery] = useState('')
   126→  const [totalExpenses, setTotalExpenses] = useState(0)
   127→  const [monthlySummary, setMonthlySummary] = useState<Array<{ month: string; total: number }>>([])
   128→
   129→  // Dialog states
   130→  const [dialogOpen, setDialogOpen] = useState(false)
   131→  const [editingExpense, setEditingExpense] = useState<Expense | null>(null)
   132→  const [deleteTarget, setDeleteTarget] = useState<Expense | null>(null)
   133→  const [isSaving, setIsSaving] = useState(false)
   134→  const [isDeleting, setIsDeleting] = useState(false)
   135→
   136→  const form = useForm<ExpenseFormData>({
   137→    resolver: zodResolver(expenseFormSchema),
   138→    defaultValues: {
   139→      category: '',
   140→      amount: 0,
   141→      description: '',
   142→      expenseDate: new Date().toISOString().split('T')[0],
   143→      shopId: '',
   144→      isRecurring: false,
   145→      recurringPeriod: '',
   146→    },
   147→  })
   148→
   149→  const isRecurring = form.watch('isRecurring')
   150→

   152→    if (!currentOrg) return

   154→    setError(null)
   155→    try {
   156→      const params = new URLSearchParams({ organizationId: currentOrg.id })
   157→      if (categoryFilter && categoryFilter !== 'all') params.set('category', categoryFilter)
   158→
   159→      const response = await fetch(`/api/expenses?${params.toString()}`, {
   160→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   161→      })
   162→      if (!response.ok) {
   163→        const errData = await response.json().catch(() => ({}))
   164→        throw new Error(errData.error || 'Failed to fetch expenses')
   165→      }
   166→      const result = await response.json()
   167→      setExpenses(result.expenses || [])
   168→      setTotalExpenses(result.summary?.totalExpenses || 0)
   169→      setMonthlySummary(result.monthlySummary || [])
   170→    } catch (err) {
   171→      setError(getNetworkErrorMessage(err))
   172→    } finally {
   173→      setIsLoading(false)
   174→    }
   175→  }, [currentOrg, categoryFilter])
   176→
   177→  useEffect(() => {
   178→    fetchExpenses()
   179→  }, [fetchExpenses])

   181→  // Filter expenses client-side by search query
   182→  const filteredExpenses = searchQuery
   183→    ? expenses.filter(e =>
   184→        e.description?.toLowerCase().includes(searchQuery.toLowerCase()) ||
   185→        CATEGORY_LABELS[e.category]?.toLowerCase().includes(searchQuery.toLowerCase())
   186→      )
   187→    : expenses
   188→
   189→  function openCreateDialog() {

   191→    form.reset({
   192→      category: '',
   193→      amount: 0,
   194→      description: '',
   195→      expenseDate: new Date().toISOString().split('T')[0],
   196→      shopId: '',
   197→      isRecurring: false,
   198→      recurringPeriod: '',
   199→    })
   200→    setDialogOpen(true)
   201→  }
   202→
   203→  function openEditDialog(expense: Expense) {
   204→    setEditingExpense(expense)
   205→    form.reset({
   206→      category: expense.category,
   207→      amount: expense.amount,
   208→      description: expense.description || '',
   209→      expenseDate: new Date(expense.expenseDate).toISOString().split('T')[0],
   210→      shopId: expense.shopId || '',
   211→      isRecurring: expense.isRecurring,
   212→      recurringPeriod: expense.recurringPeriod || '',
   213→    })
   214→    setDialogOpen(true)
   215→  }
   216→
   217→  async function onSubmit(formData: ExpenseFormData) {
   218→    if (!currentOrg) return
   219→    setIsSaving(true)
   220→    try {
   221→      const payload = {
   222→        organizationId: currentOrg.id,
   223→        ...formData,
   224→        shopId: formData.shopId || null,
   225→        description: formData.description || null,
   226→        recurringPeriod: formData.isRecurring ? formData.recurringPeriod : null,
   227→      }
   228→
   229→      if (editingExpense) {
   230→        const res = await fetch(`/api/expenses/${editingExpense.id}?organizationId=${currentOrg.id}`, {
   231→          method: 'PUT',
   232→          headers: {
   233→            'Content-Type': 'application/json',
   234→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   235→          },
   236→          body: JSON.stringify(payload),
   237→        })
   238→        if (!res.ok) {
   239→          const errData = await res.json().catch(() => ({}))
   240→          throw new Error(errData.error || 'Failed to update expense')
   241→        }
   242→        toast.success('Expense updated successfully')
   243→      } else {
   244→        const res = await fetch('/api/expenses', {
   245→          method: 'POST',
   246→          headers: {
   247→            'Content-Type': 'application/json',
   248→            Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   249→          },
   250→          body: JSON.stringify(payload),
   251→        })
   252→        if (!res.ok) {
   253→          const errData = await res.json().catch(() => ({}))
   254→          throw new Error(errData.error || 'Failed to create expense')
   255→        }
   256→        toast.success('Expense created successfully')
   257→      }
   258→
   259→      setDialogOpen(false)
   260→      fetchExpenses()
   261→    } catch (err) {
   262→      toast.error(getNetworkErrorMessage(err))
   263→    } finally {
   264→      setIsSaving(false)
   265→    }
   266→  }
   267→
   268→  async function handleDelete() {

   270→    setIsDeleting(true)
   271→    try {
   272→      const res = await fetch(`/api/expenses/${deleteTarget.id}?organizationId=${currentOrg.id}`, {
   273→        method: 'DELETE',
   274→        headers: { Authorization: `Bearer ${localStorage.getItem('sb_token')}` },
   275→      })
   276→      if (!res.ok) {
   277→        const errData = await res.json().catch(() => ({}))
   278→        throw new Error(errData.error || 'Failed to delete expense')
   279→      }
   280→      toast.success('Expense deleted successfully')
   281→      setDeleteTarget(null)
   282→      fetchExpenses()
   283→    } catch (err) {
   284→      toast.error(getNetworkErrorMessage(err))
   285→    } finally {
   286→      setIsDeleting(false)
   287→    }
   288→  }
   289→
   290→  if (error) {
   291→    return <ErrorState title="Failed to load Expenses" message={error} onRetry={fetchExpenses} />
   292→  }
   293→

   296→      {/* Header */}
   297→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   298→        <div>
   299→          <h1 className="text-2xl font-bold tracking-tight">Expenses</h1>
   300→          <p className="text-muted-foreground text-sm mt-1">
   301→            Track and manage your business expenses

   307→        </Button>
   308→      </div>
   309→
   310→      {/* KPI Card */}
   311→      <Card>
   312→        <CardContent className="p-4 sm:p-6">
   313→          <div className="flex items-center justify-between">
   314→            <div>
   315→              <p className="text-sm text-muted-foreground font-medium">Total Expenses</p>
   316→              <p className="text-2xl sm:text-3xl font-bold tracking-tight">
   317→                {isLoading ? <Skeleton className="h-9 w-32 inline-block" /> : formatETB(totalExpenses)}
   318→              </p>
   319→            </div>
   320→            <div className="size-12 rounded-xl bg-red-100 text-red-600 flex items-center justify-center">
   321→              <TrendingDown className="size-6" />
   322→            </div>
   323→          </div>
   324→        </CardContent>
   325→      </Card>
   326→
   327→      {/* Monthly Chart */}
   328→      <Card>
   329→        <CardHeader>
   330→          <CardTitle className="text-base">Monthly Expense Summary</CardTitle>
   331→          <CardDescription>Last 6 months expense trend</CardDescription>
   332→        </CardHeader>
   333→        <CardContent>
   334→          {isLoading ? (
   335→            <Skeleton className="h-[250px] w-full" />
   336→          ) : monthlySummary.length === 0 ? (
   337→            <div className="flex items-center justify-center h-[250px] text-muted-foreground text-sm">
   338→              No expense data available yet
   339→            </div>
   340→          ) : (
   341→            <ResponsiveContainer width="100%" height={250}>
   342→              <BarChart data={monthlySummary.map(d => ({ ...d, month: formatMonthLabel(d.month) }))}>
   343→                <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
   344→                <XAxis dataKey="month" className="text-xs" />
   345→                <YAxis className="text-xs" tickFormatter={(v) => `${(v / 1000).toFixed(0)}k`} />
   346→                <Tooltip formatter={(value: number) => formatETB(value)} contentStyle={{ borderRadius: '8px', fontSize: '12px' }} />
   347→                <Bar dataKey="total" name="Expenses" fill="#ef4444" radius={[4, 4, 0, 0]} />
   348→              </BarChart>
   349→            </ResponsiveContainer>

   351→        </CardContent>
   352→      </Card>
   353→
   354→      {/* Filters */}
   355→      <div className="flex flex-col sm:flex-row gap-3">
   356→        <div className="relative flex-1">
   357→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   358→          <Input
   359→            placeholder="Search expenses..."
   360→            value={searchQuery}
   361→            onChange={(e) => setSearchQuery(e.target.value)}
   362→            className="pl-9"
   363→          />
   364→        </div>
   365→        <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   366→          <SelectTrigger className="w-full sm:w-[160px]">
   367→            <SelectValue placeholder="All Categories" />
   368→          </SelectTrigger>
   369→          <SelectContent>
   370→            <SelectItem value="all">All Categories</SelectItem>
   371→            {EXPENSE_CATEGORIES.map(cat => (
   372→              <SelectItem key={cat.value} value={cat.value}>{cat.label}</SelectItem>
   373→            ))}
   374→          </SelectContent>
   375→        </Select>
   376→        <Button variant="outline" size="icon" onClick={fetchExpenses} disabled={isLoading}>
   377→          <RefreshCw className={`size-4 ${isLoading ? 'animate-spin' : ''}`} />
   378→        </Button>
   379→      </div>
   380→
   381→      {/* Expense List */}
   382→      <Card>
   383→        <CardContent className="p-0">
   384→          {isLoading ? (
   385→            <div className="p-6 space-y-3">
   386→              {Array.from({ length: 5 }).map((_, i) => (
   387→                <Skeleton key={i} className="h-12 w-full" />
   388→              ))}
   389→            </div>
   390→          ) : filteredExpenses.length === 0 ? (
   391→            <EmptyState
   392→              title="No expenses found"
   393→              message={categoryFilter !== 'all' ? 'Try changing the filter or add a new expense' : 'Start by adding your first expense'}
   394→              action={{ label: 'Add Expense', onClick: openCreateDialog }}
   395→            />
   396→          ) : (
   397→            <>
   398→              {/* Desktop Table */}
   399→              <div className="hidden md:block">
   400→                <Table>
   401→                  <TableHeader>
   402→                    <TableRow>
   403→                      <TableHead>Date</TableHead>
   404→                      <TableHead>Category</TableHead>
   405→                      <TableHead>Description</TableHead>
   406→                      <TableHead>Shop</TableHead>
   407→                      <TableHead className="text-right">Amount</TableHead>
   408→                      <TableHead>Recurring</TableHead>
   409→                      <TableHead className="text-right">Actions</TableHead>
   410→                    </TableRow>
   411→                  </TableHeader>
   412→                  <TableBody>
   413→                    {filteredExpenses.map((expense) => (
   414→                      <TableRow key={expense.id}>
   415→                        <TableCell className="text-sm">
   416→                          {new Date(expense.expenseDate).toLocaleDateString('en-US', {

   426→                          {expense.description || '—'}
   427→                        </TableCell>
   428→                        <TableCell className="text-sm text-muted-foreground">

   430→                        </TableCell>

   445→                              variant="ghost"
   446→                              size="icon"
   447→                              className="size-8"
   448→                              onClick={() => openEditDialog(expense)}

   450→                              <Pencil className="size-3.5" />

   452→                            <Button
   453→                              variant="ghost"
   454→                              size="icon"
   455→                              className="size-8 text-destructive hover:text-destructive"
   456→                              onClick={() => setDeleteTarget(expense)}

   458→                              <Trash2 className="size-3.5" />
   459→                            </Button>

   461→                        </TableCell>
   462→                      </TableRow>
   463→                    ))}
   464→                  </TableBody>
   465→                </Table>
   466→              </div>
   467→
   468→              {/* Mobile Cards */}
   469→              <div className="md:hidden divide-y">
   470→                {filteredExpenses.map((expense) => (
   471→                  <div key={expense.id} className="p-4 space-y-2">
   472→                    <div className="flex items-center justify-between">
   473→                      <Badge variant="secondary" className={CATEGORY_COLORS[expense.category] || ''}>
   474→                        {CATEGORY_LABELS[expense.category] || expense.category}
   475→                      </Badge>

   477→                    </div>
   478→                    <div className="flex items-center justify-between text-sm text-muted-foreground">
   479→                      <span>
   480→                        {new Date(expense.expenseDate).toLocaleDateString('en-US', {
   481→                          month: 'short', day: 'numeric',

   483→                      </span>

   486→                          <Repeat className="size-3" />

   500→                      >
   501→                        <Pencil className="size-3" />
   502→                        Edit
   503→                      </Button>
   504→                      <Button
   505→                        variant="outline"
   506→                        size="sm"
   507→                        className="flex-1 gap-1 text-destructive hover:text-destructive"

   510→                        <Trash2 className="size-3" />
   511→                        Delete
   512→                      </Button>
   513→                    </div>
   514→                  </div>
   515→                ))}
   516→              </div>
   517→            </>
   518→          )}
   519→        </CardContent>
   520→      </Card>
   521→
   522→      {/* Add/Edit Dialog */}
   523→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   524→        <DialogContent className="sm:max-w-[480px]">
   525→          <DialogHeader>
   526→            <DialogTitle>{editingExpense ? 'Edit Expense' : 'Add Expense'}</DialogTitle>
   527→            <DialogDescription>
   528→              {editingExpense ? 'Update the expense details below' : 'Fill in the details for the new expense'}
   529→            </DialogDescription>
   530→          </DialogHeader>
   531→          <Form {...form}>
   532→            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   533→              {/* Category */}
   534→              <div className="space-y-2">
   535→                <label className="text-sm font-medium">
   536→                  Category <span className="text-destructive">*</span>
   537→                </label>
   538→                <Select
   539→                  value={form.watch('category')}
   540→                  onValueChange={(val) => form.setValue('category', val)}
   541→                >
   542→                  <SelectTrigger>
   543→                    <SelectValue placeholder="Select category" />
   544→                  </SelectTrigger>
   545→                  <SelectContent>
   546→                    {EXPENSE_CATEGORIES.map(cat => (
   547→                      <SelectItem key={cat.value} value={cat.value}>{cat.label}</SelectItem>

   549→                  </SelectContent>

   552→                  <p className="text-xs text-destructive">{form.formState.errors.category.message}</p>
   553→                )}
   554→              </div>
   555→
   556→              {/* Amount */}
   557→              <div className="space-y-2">
   558→                <label className="text-sm font-medium">
   559→                  Amount (ETB) <span className="text-destructive">*</span>
   560→                </label>
   561→                <Input
   562→                  type="number"
   563→                  step="0.01"
   564→                  min="0"
   565→                  placeholder="0.00"
   566→                  {...form.register('amount', { valueAsNumber: true })}
   567→                />
   568→                {form.formState.errors.amount && (
   569→                  <p className="text-xs text-destructive">{form.formState.errors.amount.message}</p>
   570→                )}
   571→              </div>
   572→
   573→              {/* Description */}
   574→              <div className="space-y-2">
   575→                <label className="text-sm font-medium">Description</label>
   576→                <Input
   577→                  placeholder="Optional description"
   578→                  {...form.register('description')}
   579→                />
   580→              </div>
   581→
   582→              {/* Date */}
   583→              <div className="space-y-2">
   584→                <label className="text-sm font-medium">
   585→                  Date <span className="text-destructive">*</span>
   586→                </label>
   587→                <Input
   588→                  type="date"
   589→                  {...form.register('expenseDate')}
   590→                />
   591→                {form.formState.errors.expenseDate && (
   592→                  <p className="text-xs text-destructive">{form.formState.errors.expenseDate.message}</p>
   593→                )}
   594→              </div>
   595→
   596→              {/* Shop */}
   597→              {shops.length > 1 && (
   598→                <div className="space-y-2">
   599→                  <label className="text-sm font-medium">Shop</label>
   600→                  <Select
   601→                    value={form.watch('shopId') || 'none'}
   602→                    onValueChange={(val) => form.setValue('shopId', val === 'none' ? '' : val)}
   603→                  >
   604→                    <SelectTrigger>
   605→                      <SelectValue placeholder="All shops" />
   606→                    </SelectTrigger>
   607→                    <SelectContent>
   608→                      <SelectItem value="none">All shops</SelectItem>
   609→                      {shops.map(shop => (
   610→                        <SelectItem key={shop.id} value={shop.id}>{shop.name}</SelectItem>
   611→                      ))}
   612→                    </SelectContent>
   613→                  </Select>
   614→                </div>
   615→              )}
   616→
   617→              {/* Recurring Toggle */}
   618→              <div className="flex items-center justify-between gap-3 rounded-lg border p-3">
   619→                <div className="space-y-0.5">
   620→                  <label className="text-sm font-medium">Recurring</label>
   621→                  <p className="text-xs text-muted-foreground">Set up a recurring expense</p>
   622→                </div>
   623→                <Switch
   624→                  checked={form.watch('isRecurring')}
   625→                  onCheckedChange={(val) => form.setValue('isRecurring', val)}
   626→                />
   627→              </div>
   628→
   629→              {/* Recurring Period */}
   630→              {isRecurring && (
   631→                <div className="space-y-2">
   632→                  <label className="text-sm font-medium">Recurring Period</label>
   633→                  <Select
   634→                    value={form.watch('recurringPeriod') || ''}
   635→                    onValueChange={(val) => form.setValue('recurringPeriod', val)}
   636→                  >
   637→                    <SelectTrigger>
   638→                      <SelectValue placeholder="Select frequency" />
   639→                    </SelectTrigger>
   640→                    <SelectContent>
   641→                      <SelectItem value="weekly">Weekly</SelectItem>
   642→                      <SelectItem value="monthly">Monthly</SelectItem>
   643→                      <SelectItem value="yearly">Yearly</SelectItem>
   644→                    </SelectContent>
   645→                  </Select>
   646→                </div>
   647→              )}
   648→
   649→              <DialogFooter>
   650→                <Button type="button" variant="outline" onClick={() => setDialogOpen(false)}>
   651→                  Cancel
   652→                </Button>
   653→                <Button type="submit" disabled={isSaving}>
   654→                  {isSaving ? (
   655→                    <>
   656→                      <Loader2 className="size-4 animate-spin" />
   657→                      <span>Saving...</span>
   658→                    </>
   659→                  ) : editingExpense ? 'Update' : 'Create'}
   660→                </Button>
   661→              </DialogFooter>
   662→            </form>
   663→          </Form>
   664→        </DialogContent>
   665→      </Dialog>
   666→
   667→      {/* Delete Confirmation */}
   668→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   669→        <AlertDialogContent>
   670→          <AlertDialogHeader>
   671→            <AlertDialogTitle>Delete Expense</AlertDialogTitle>
   672→            <AlertDialogDescription>
   673→              Are you sure you want to delete this expense of {deleteTarget ? formatETB(deleteTarget.amount) : ''}?
   674→              This action cannot be undone.
   675→            </AlertDialogDescription>
   676→          </AlertDialogHeader>
   677→          <AlertDialogFooter>
   678→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   679→            <AlertDialogAction
   680→              onClick={handleDelete}
   681→              disabled={isDeleting}
   682→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   683→            >
   684→              {isDeleting ? (
   685→                <>
   686→                  <Loader2 className="size-4 animate-spin" />
   687→                  Deleting...
   688→                </>
   689→              ) : 'Delete'}
   690→            </AlertDialogAction>
   691→          </AlertDialogFooter>
   692→        </AlertDialogContent>
   693→      </AlertDialog>
   694→    </div>
   695→  )
