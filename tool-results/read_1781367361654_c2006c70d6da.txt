   537→class ApiClient {
   538→  // Supabase Auth uses httpOnly cookies for session management.
   539→  // The token is no longer stored in localStorage for security.
   540→  // We keep a reference for backward compat but auth is primarily cookie-based.
   541→  private _token: string | null = null
   542→  // Guard flag: when true, 401 auto-logout is suppressed to prevent
   543→  // stale background requests from interfering with an active login attempt.
   544→  private _isLoggingIn: boolean = false
   545→
   546→  // Request deduplication: prevent concurrent identical GET requests
   547→  // If 3 components request the same endpoint simultaneously, only 1 fetch is made.
   548→  private _inflightRequests = new Map<string, Promise<unknown>>()
   549→
   550→  // Short-lived client-side response cache for GET requests (5s TTL)
   551→  // Prevents rapid re-fetches when multiple components mount simultaneously
   552→  private _responseCache = new Map<string, { data: unknown; expiresAt: number }>()
   553→  private static RESPONSE_CACHE_TTL = 5_000 // 5 seconds
   554→
   555→  /**
   556→   * Invalidate client-side response cache entries matching a prefix.
   557→   * Called after mutating operations to ensure stale data is not served.
   558→   */
   559→  invalidateCache(prefix: string): void {
   560→    // Invalidate our internal response cache
   561→    this._responseCache.forEach((_, key) => {
   562→      if (key.startsWith(prefix)) {
   563→        this._responseCache.delete(key)
   564→      }
   565→    })
   566→    // Also invalidate the shared client-cache (used by swrFetch)
   567→    try {
   568→      import('@/lib/client-cache').then(({ invalidateKeysByPrefix }) => {
   569→        invalidateKeysByPrefix(prefix)
   570→      }).catch(() => {
   571→        // client-cache may not be available during SSR
   572→      })
   573→    } catch {
   574→      // client-cache may not be available during SSR
   575→    }
   576→  }
   577→
   578→  private getToken(): string | null {
   579→    if (typeof window === 'undefined') return null
   580→    // Check memory first, then fallback to localStorage (for migration)
   581→    return this._token || localStorage.getItem('sb_token')
   582→  }
   583→
   584→  setToken(token: string): void {
   585→    if (typeof window === 'undefined') return
   586→    this._token = token
   587→    // Also keep in localStorage for migration period
   588→    localStorage.setItem('sb_token', token)
   589→  }
   590→
   591→  clearToken(): void {
   592→    if (typeof window === 'undefined') return
   593→    this._token = null
   594→    localStorage.removeItem('sb_token')
   595→  }
   596→
   597→  private async request<T>(
   598→    endpoint: string,
   599→    options: RequestInit = {}
   600→  ): Promise<T> {
   601→    const token = this.getToken()
   602→    const headers: Record<string, string> = {
   603→      'Content-Type': 'application/json',
   604→      ...(options.headers as Record<string, string>),
   605→    }
   606→
   607→    // Add JWT token to Authorization header if available
   608→    if (token) {
   609→      headers['Authorization'] = `Bearer ${token}`
   610→    }
   611→
   612→    // Offline interceptor: queue mutating requests, reject GET requests
   613→    const method = (options.method || 'GET').toUpperCase()
   614→    const isMutating = ['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)
   615→    const isGet = method === 'GET'
   616→
   617→    // ---- Request deduplication for GET requests ----
   618→    // If an identical GET request is already in-flight, return the same promise
   619→    // instead of firing a duplicate network request.
   620→    if (isGet) {
   621→      const cacheKey = `${method}:${endpoint}`
   622→
   623→      // Check short-lived response cache first
   624→      const cached = this._responseCache.get(cacheKey)
   625→      if (cached && Date.now() < cached.expiresAt) {
   626→        return cached.data as T
   627→      }
   628→
   629→      // Check for in-flight deduplication
   630→      const inflight = this._inflightRequests.get(cacheKey)
   631→      if (inflight) {
   632→        return inflight as Promise<T>
   633→      }
   634→    }
   635→
   636→    if (typeof window !== 'undefined' && !navigator.onLine) {
   637→      if (isMutating) {
   638→        // Queue the operation for later sync
   639→        try {
   640→          const { queueOperation } = await import('@/lib/offline-queue')
   641→          await queueOperation(
   642→            endpoint,
   643→            method,
   644→            headers,
   645→            options.body ? String(options.body) : null
   646→          )
   647→          // Return a mock "offline queued" response so callers don't break
   648→          return { offline: true, queued: true } as unknown as T
   649→        } catch {
   650→          // If queuing fails, fall through to normal error
   651→        }
   652→      }
   653→      throw new Error('You are offline. Please check your internet connection and try again.')
   654→    }
   655→
   656→    // Add timeout (30s) to prevent hanging requests
   657→    const controller = new AbortController()
   658→    const timeoutId = setTimeout(() => controller.abort(), 30000)
   659→    const signal = options.signal || controller.signal
   660→
   661→    // For GET requests, register the inflight promise for deduplication
   662→    const cacheKey = `${method}:${endpoint}`
   663→    let isInflightRegistered = false
   664→
   665→    if (isGet) {
   666→      isInflightRegistered = true
   667→    }
   668→
   669→    // Create the fetch promise (may be shared across callers)
   670→    const fetchPromise = (async () => {
   671→      let response: Response
   672→      try {
   673→        response = await fetch(endpoint, {
   674→          ...options,
   675→          headers,
   676→          signal,
   677→        })
   678→      } catch (err) {
   679→        clearTimeout(timeoutId)
   680→        // Clean up inflight registration
   681→        if (isInflightRegistered) {
   682→          this._inflightRequests.delete(cacheKey)
   683→        }
   684→        // If we got a network error during a mutating request, try to queue it
   685→        if (err instanceof TypeError && err.message === 'Failed to fetch' && isMutating) {
   686→          try {
   687→            const { queueOperation } = await import('@/lib/offline-queue')
   688→            await queueOperation(
   689→              endpoint,
   690→              method,
   691→              headers,
   692→              options.body ? String(options.body) : null
   693→            )
   694→            return { offline: true, queued: true } as unknown as T
   695→          } catch {
   696→            // Fall through to original error handling
   697→          }
   698→        }
   699→        if (err instanceof DOMException && err.name === 'AbortError') {
   700→          throw new Error('Request timed out — the server took too long to respond. Please try again.')
   701→        }
   702→        if (err instanceof TypeError && err.message === 'Failed to fetch') {
   703→          throw new Error('Network error — please check your internet connection and try again.')
   704→        }
   705→        throw new Error(err instanceof Error ? err.message : 'An unexpected network error occurred')
   706→      }
   707→
   708→      clearTimeout(timeoutId)
   709→
   710→      // Handle non-JSON responses gracefully
   711→      let data: Record<string, unknown>
   712→      try {
   713→        data = await response.json() as Record<string, unknown>
   714→      } catch {
   715→        // Clean up inflight registration on parse error
   716→        if (isInflightRegistered) {
   717→          this._inflightRequests.delete(cacheKey)
   718→        }
   719→        if (!response.ok) {
   720→          // Map common HTTP error codes to user-friendly messages — never expose raw status codes
   721→          if (response.status === 401) throw new Error('Your session has expired. Please log in again.')
   722→          if (response.status === 403) throw new Error('You do not have permission to perform this action')
   723→          if (response.status === 404) throw new Error('The requested resource was not found')
   724→          if (response.status === 429) throw new Error('Too many requests — please wait a moment and try again')
   725→          if (response.status >= 500) throw new Error('Server error — please try again later')
   726→          throw new Error('An unexpected error occurred. Please try again.')
   727→        }
   728→        throw new Error('Server returned an invalid response')
   729→      }
   730→
   731→      // Clean up inflight registration after response is parsed
   732→      if (isInflightRegistered) {
   733→        this._inflightRequests.delete(cacheKey)
   734→      }
   735→
   736→      if (!response.ok) {
   737→        const errorMessage = (data.error as string) || (data.message as string) || 'An unexpected error occurred. Please try again.'
   738→        const errorCode = data.code as string | undefined
   739→
   740→        // Handle database unreachable errors
   741→        if (response.status === 503 && errorCode === 'DB_UNREACHABLE') {
   742→          const dbError = new Error('Database is unreachable') as Error & { code: string }
   743→          dbError.code = 'DB_UNREACHABLE'
   744→          throw dbError
   745→        }
   746→
   747→        // Handle auth errors — clear token and trigger logout if 401 from a non-auth endpoint
   748→        // Auth endpoints (login, register, me) legitimately return 401 for wrong credentials
   749→        // Also suppress auto-logout during active login to prevent race conditions
   750→        const isAuthEndpoint = endpoint.startsWith('/api/auth/login') || endpoint.startsWith('/api/auth/register') || endpoint.startsWith('/api/auth/me')
   751→        if (response.status === 401 && token && !isAuthEndpoint && !this._isLoggingIn) {
   752→          // Only clear token and trigger logout if the token we sent is STILL the current token.
   753→          // This prevents a stale background request from clearing a freshly-set token
   754→          // after the user logged out and logged back in.
   755→          const currentToken = this.getToken()
   756→          if (token === currentToken) {
   757→            this.clearToken()
   758→            // Trigger logout via auth store to update UI state
   759→            try {
   760→              const { useAuthStore } = await import('@/lib/stores/auth-store')
   761→              const store = useAuthStore.getState()
   762→              if (store.isAuthenticated) {
   763→                // Use setTimeout to avoid race conditions with concurrent requests.
   764→                // If multiple 401s come in at the same time, only the first one will
   765→                // trigger logout (subsequent ones will see isAuthenticated = false).
   766→                setTimeout(() => {
   767→                  const currentStore = useAuthStore.getState()
   768→                  if (currentStore.isAuthenticated) {
   769→                    currentStore.logout()
   770→                  }
   771→                }, 100)
   772→              }
   773→            } catch {
   774→              // If import fails (e.g. during SSR), just clear the token
   775→            }
   776→          }
   777→          throw new Error('Your session has expired. Please log in again.')
   778→        }
   779→
   780→        throw new Error(errorMessage)
   781→      }
   782→
   783→      // Cache successful GET responses for short TTL
   784→      if (isGet) {
   785→        this._responseCache.set(cacheKey, {
   786→          data: data as T,
   787→          expiresAt: Date.now() + ApiClient.RESPONSE_CACHE_TTL,
   788→        })
   789→      }
   790→
   791→      return data as T
   792→    })()
   793→
   794→    // Register the inflight promise for GET deduplication
   795→    if (isGet) {
   796→      this._inflightRequests.set(cacheKey, fetchPromise)
   797→    }
   798→
   799→    return fetchPromise
   800→  }

   827→      this._isLoggingIn = false
   828→    }
