   140→   140→type DateRangePreset = 'today' | 'this_week' | 'this_month' | 'this_quarter' | 'custom'
   141→   141→
   142→   142→function getDateRange(preset: DateRangePreset): { from: string; to: string } {
   143→   143→  const now = new Date()
   144→   144→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   145→   145→
   146→   146→  switch (preset) {
   147→   147→    case 'today':
   148→   148→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   149→   149→    case 'this_week': {
   150→   150→      const dayOfWeek = now.getDay()
   151→   151→      const weekStart = new Date(today.getTime() - dayOfWeek * 24 * 60 * 60 * 1000)
   152→   152→      return { from: weekStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   153→   153→    }
   154→   154→    case 'this_month': {
   155→   155→      const monthStart = new Date(now.getFullYear(), now.getMonth(), 1)
   156→   156→      return { from: monthStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   157→   157→    }
   158→   158→    case 'this_quarter': {
   159→   159→      const quarter = Math.floor(now.getMonth() / 3)
   160→   160→      const quarterStart = new Date(now.getFullYear(), quarter * 3, 1)
   161→   161→      return { from: quarterStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   162→   162→    }
   163→   163→    default:
   164→   164→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   165→   165→  }
   166→   166→}
