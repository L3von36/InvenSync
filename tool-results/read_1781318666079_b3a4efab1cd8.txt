    34→function formatETB(amount: number): string {
    35→  return new Intl.NumberFormat('en-ET', {
    36→    style: 'decimal',
    37→    minimumFractionDigits: 2,
    38→    maximumFractionDigits: 2,
    39→  }).format(amount) + ' ETB'
    40→}
    41→
    42→function formatDate(dateStr: string): string {
    43→  return new Date(dateStr).toLocaleDateString('en-US', {
    44→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit'
    45→  })
    46→}
    47→
    48→function formatDateShort(dateStr: string): string {
    49→  return new Date(dateStr).toLocaleDateString('en-US', {
    50→    year: 'numeric', month: 'short', day: 'numeric'
    51→  })
    52→}

    54→function statusBadge(status: string) {
    55→  switch (status) {
    56→    case 'completed': return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300">Completed</Badge>
    57→    case 'pending': return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300">Pending</Badge>
    58→    case 'cancelled': return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300">Cancelled</Badge>
    59→    case 'refunded': return <Badge className="bg-blue-100 text-blue-800 dark:bg-blue-900/40 dark:text-blue-300">Refunded</Badge>
    60→    default: return <Badge variant="outline">{status}</Badge>
    61→  }
    62→}

    65→  switch (method) {

    86→// ============================================
    87→function SalesStatsCards({ sales }: { sales: Sale[] }) {
    88→  const today = new Date()
    89→  today.setHours(0, 0, 0, 0)
    90→
    91→  const startOfWeek = new Date(today)
    92→  startOfWeek.setDate(today.getDate() - today.getDay())
    93→
    94→  const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
    95→
    96→  const todaySales = sales.filter(s => new Date(s.saleDate) >= today)
    97→  const weekSales = sales.filter(s => new Date(s.saleDate) >= startOfWeek)
    98→  const monthSales = sales.filter(s => new Date(s.saleDate) >= startOfMonth)
    99→
   100→  const todayTotal = todaySales.reduce((acc, s) => acc + s.total, 0)
   101→  const weekTotal = weekSales.reduce((acc, s) => acc + s.total, 0)
   102→  const monthTotal = monthSales.reduce((acc, s) => acc + s.total, 0)
   103→  const avgSale = sales.length > 0 ? sales.reduce((acc, s) => acc + s.total, 0) / sales.length : 0
   104→
   105→  const cards = [
   106→    { title: "Today's Sales", value: formatETB(todayTotal), icon: ShoppingCart, color: 'text-primary', bg: 'bg-primary/10' },
   107→    { title: 'This Week', value: formatETB(weekTotal), icon: Calendar, color: 'text-primary', bg: 'bg-brand-50 dark:bg-brand-900/20' },
   108→    { title: 'This Month', value: formatETB(monthTotal), icon: TrendingUp, color: 'text-amber-600', bg: 'bg-amber-100 dark:bg-amber-900/30' },
   109→    { title: 'Avg Sale Value', value: formatETB(avgSale), icon: DollarSign, color: 'text-primary', bg: 'bg-primary/10' },
   110→  ]
   111→
   112→  return (
   113→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   114→      {cards.map((card) => (
   115→        <Card key={card.title}>
   116→          <CardContent className="p-4">
   117→            <div className="flex items-center gap-3">
   118→              <div className={`size-10 rounded-lg ${card.bg} flex items-center justify-center shrink-0`}>
   119→                <card.icon className={`size-5 ${card.color}`} />
   120→              </div>
   121→              <div className="min-w-0">
   122→                <p className="text-xs text-muted-foreground truncate">{card.title}</p>
   123→                <p className="text-base font-semibold truncate">{card.value}</p>
   124→              </div>
   125→            </div>
   126→          </CardContent>
   127→        </Card>
   128→      ))}
   129→    </div>
   130→  )
   131→}

   302→// All Sales Tab
   303→// ============================================
   304→function AllSalesTab({
   305→  orgId,
   306→  shopId,
   307→  onViewSale,
   308→  onCreateSale,
   309→}: {
   310→  orgId: string
   311→  shopId?: string
   312→  onViewSale: (sale: Sale) => void
   313→  onCreateSale: () => void
   314→}) {
   315→  const [sales, setSales] = useState<Sale[]>([])
   316→  const [loading, setLoading] = useState(true)
   317→  const [error, setError] = useState<string | null>(null)
   318→  const [search, setSearch] = useState('')
   319→  const [filterStatus, setFilterStatus] = useState<string>('all')
   320→  const [filterPayment, setFilterPayment] = useState<string>('all')
   321→  const [page, setPage] = useState(1)
   322→  const [totalPages, setTotalPages] = useState(1)
   323→
   324→  const fetchSales = useCallback(async () => {
   325→    setLoading(true)
   326→    setError(null)
   327→    try {
   328→      const res = await api.getSales(orgId, {
   329→        status: filterStatus !== 'all' ? filterStatus : undefined,
   330→        page,
   331→        limit: 10,
   332→        shopId,
   333→      })
   334→      setSales(res.sales)
   335→      setTotalPages(res.pagination.totalPages)
   336→    } catch (err) {
   337→      const msg = getNetworkErrorMessage(err)
   338→      setError(msg)
   339→      toast.error(msg)
   340→    } finally {
   341→      setLoading(false)
   342→    }
   343→  }, [orgId, filterStatus, page, shopId])
   344→
   345→  useEffect(() => {
   346→    fetchSales()
   347→  }, [fetchSales])
   348→
   349→  const filteredSales = sales.filter((s) => {
   350→    if (filterPayment !== 'all' && s.paymentMethod !== filterPayment) return false
   351→    if (search) {
   352→      const q = search.toLowerCase()
   353→      const matchInvoice = s.invoiceNumber.toLowerCase().includes(q)
   354→      const matchCustomer = s.customer?.name.toLowerCase().includes(q)
   355→      if (!matchInvoice && !matchCustomer) return false
   356→    }
   357→    return true
   358→  })
   359→
   360→  if (loading) {
   361→    return (
   362→      <div className="space-y-4">
   363→        <div className="flex flex-col sm:flex-row gap-3">
   364→          <Skeleton className="h-10 flex-1" />
   365→          <Skeleton className="h-10 w-36" />
   366→          <Skeleton className="h-10 w-36" />
   367→        </div>
   368→        {Array.from({ length: 5 }).map((_, i) => (
   369→          <Skeleton key={i} className="h-14" />
   370→        ))}
   371→      </div>
   372→    )
   373→  }
   374→
   375→  if (error && sales.length === 0) {
   376→    return <ErrorState title="Failed to load sales" message={error} onRetry={fetchSales} />
   377→  }
   378→
   379→  return (
   380→    <div className="space-y-4">
   381→      {/* Filters */}
   382→      <div className="flex flex-col sm:flex-row gap-3">
   383→        <div className="relative flex-1">
   384→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   385→          <Input
   386→            placeholder="Search by invoice # or customer..."
   387→            className="pl-9"
   388→            value={search}
   389→            onChange={(e) => setSearch(e.target.value)}
   390→          />
   391→        </div>
   392→        <Select value={filterStatus} onValueChange={(v) => { setFilterStatus(v); setPage(1) }}>
   393→          <SelectTrigger className="w-full sm:w-40">
   394→            <SelectValue placeholder="Status" />
   395→          </SelectTrigger>
   396→          <SelectContent>
   397→            <SelectItem value="all">All Status</SelectItem>
   398→            <SelectItem value="completed">Completed</SelectItem>
   399→            <SelectItem value="pending">Pending</SelectItem>
   400→            <SelectItem value="cancelled">Cancelled</SelectItem>
   401→          </SelectContent>
   402→        </Select>
   403→        <Select value={filterPayment} onValueChange={setFilterPayment}>
   404→          <SelectTrigger className="w-full sm:w-40">
   405→            <SelectValue placeholder="Payment" />
   406→          </SelectTrigger>
   407→          <SelectContent>
   408→            <SelectItem value="all">All Payment</SelectItem>
   409→            <SelectItem value="cash">Cash</SelectItem>
   410→            <SelectItem value="card">Card</SelectItem>
   411→            <SelectItem value="mobile_money">Mobile Money</SelectItem>
   412→            <SelectItem value="credit">Credit</SelectItem>
   413→          </SelectContent>
   414→        </Select>
   415→      </div>
   416→
   417→      {/* Empty state or Table */}
   418→      {!loading && filteredSales.length === 0 ? (
   419→        <Card>
   420→          <CardContent className="py-12 text-center">
   421→            <ShoppingCart className="size-12 text-muted-foreground mx-auto mb-4" />
   422→            <h3 className="text-lg font-semibold mb-1">
   423→              {search || filterStatus !== 'all' || filterPayment !== 'all' ? 'No sales match your filters' : 'No sales yet'}
   424→            </h3>
   425→            <p className="text-muted-foreground text-sm mb-4">
   426→              {search || filterStatus !== 'all' || filterPayment !== 'all'
   427→                ? 'Try adjusting your search or filters'
   428→                : 'Create your first sale to start tracking transactions'}
   429→            </p>
   430→            {!search && filterStatus === 'all' && filterPayment === 'all' && (
   431→              <Button onClick={onCreateSale} className="gap-2">
   432→                <Plus className="size-4 mr-2" />
   433→                Create Sale
   434→              </Button>
   435→            )}
   436→          </CardContent>
   437→        </Card>
   438→      ) : (
   439→      <div className="rounded-lg border overflow-hidden">
   440→        {/* Desktop table */}
   441→        <div className="hidden md:block overflow-x-auto">
   442→          <Table>
   443→            <TableHeader>
   444→              <TableRow>
   445→                <TableHead>Invoice #</TableHead>
   446→                <TableHead>Customer</TableHead>
   447→                <TableHead className="text-center">Items</TableHead>
   448→                <TableHead className="text-right">Subtotal</TableHead>
   449→                <TableHead className="text-right">Discount</TableHead>
   450→                <TableHead className="text-right">Total</TableHead>
   451→                <TableHead>Payment</TableHead>
   452→                <TableHead>Status</TableHead>
   453→                <TableHead>Date</TableHead>
   454→                <TableHead className="text-right">Actions</TableHead>
   455→              </TableRow>
   456→            </TableHeader>
   457→            <TableBody>
   458→              {(
   459→                filteredSales.map((sale) => (
   460→                  <TableRow key={sale.id} className="cursor-pointer hover:bg-muted/50" onClick={() => onViewSale(sale)}>
   461→                    <TableCell className="font-medium font-mono text-sm">{sale.invoiceNumber}</TableCell>
   462→                    <TableCell className="text-sm">{sale.customer?.name || 'Walk-in'}</TableCell>
   463→                    <TableCell className="text-center text-sm">{sale.items?.length ?? 0}</TableCell>
   464→                    <TableCell className="text-right text-sm">{formatETB(sale.subtotal)}</TableCell>
   465→                    <TableCell className="text-right text-sm">{sale.discount > 0 ? formatETB(sale.discount) : '—'}</TableCell>
   466→                    <TableCell className="text-right text-sm font-medium">{formatETB(sale.total)}</TableCell>
   467→                    <TableCell>
   468→                      <div className="flex items-center gap-1.5 text-sm">
   469→                        {paymentMethodIcon(sale.paymentMethod)}
   470→                        {paymentMethodLabel(sale.paymentMethod)}
   471→                      </div>
   472→                    </TableCell>
   473→                    <TableCell>{statusBadge(sale.status)}</TableCell>
   474→                    <TableCell className="text-sm whitespace-nowrap">{formatDateShort(sale.saleDate)}</TableCell>
   475→                    <TableCell className="text-right">
   476→                      <div className="flex items-center justify-end gap-1">
   477→                        <Button size="sm" variant="ghost" className="min-h-9 min-w-9" onClick={(e) => { e.stopPropagation(); onViewSale(sale) }} title="View Details">
   478→                          <Eye className="size-4" />
   479→                        </Button>
   480→                        <Button size="sm" variant="ghost" className="min-h-9 min-w-9" onClick={(e) => { e.stopPropagation(); window.open(`/api/sales/${sale.id}/receipt?orgId=${orgId}`, '_blank') }} title="Download Receipt">
   481→                          <Download className="size-4" />
   482→                        </Button>
   483→                        <Button size="sm" variant="ghost" className="min-h-9 min-w-9" onClick={(e) => { e.stopPropagation(); window.open(`/api/sales/${sale.id}/invoice?orgId=${orgId}`, '_blank') }} title="Print Invoice">
   484→                          <FileText className="size-4" />
   485→                        </Button>
   486→                      </div>
   487→                    </TableCell>
   488→                  </TableRow>
   489→                ))
   490→              )}
   491→            </TableBody>
   492→          </Table>
   493→        </div>
   494→        {/* Mobile card view */}
   495→        <div className="md:hidden space-y-3 p-3">
   496→          {filteredSales.map((sale) => (
   497→            <Card key={sale.id} className="p-4 cursor-pointer hover:bg-muted/50 transition-colors" onClick={() => onViewSale(sale)}>
   498→              <div className="flex items-start justify-between mb-2">
   499→                <div className="min-w-0 flex-1">
   500→                  <p className="font-semibold text-sm">{sale.invoiceNumber}</p>
