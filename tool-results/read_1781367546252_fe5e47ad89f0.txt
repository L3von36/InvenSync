   555→// ============================================
   556→// Mobile Bottom Navigation
   557→// ============================================
   558→export function MobileBottomNav() {
   559→  const { currentPage, setPage } = useAppStore()
   560→  const { user, currentOrg, currentOrgRole, activeModules, modulesLoaded } = useAuthStore()
   561→  const { theme, setTheme } = useTheme()
   562→  const [showMore, setShowMore] = useState(false)
   563→  const morePanelRef = useRef<HTMLDivElement>(null)
   564→  const moreButtonRef = useRef<HTMLButtonElement>(null)
   565→
   566→  const isService = currentOrg?.businessType === 'service'
   567→  const isMixed = currentOrg?.businessType === 'mixed'
   568→  const isManager = currentOrgRole === 'owner' || currentOrgRole === 'manager'
   569→  const isAdmin = user?.role === 'admin'
   570→
   571→  // Check if a module is accessible
   572→  const isModuleActive = (moduleKey?: string): boolean => {
   573→    if (!moduleKey) return true
   574→    if (!modulesLoaded) return true // Graceful degradation: modules not loaded yet, show all
   575→    return activeModules.includes(moduleKey)
   576→  }
   577→
   578→  const mainTabs: Array<{ page: Page; icon: React.ElementType; label: string }> = isAdmin
   579→    ? [
   580→        { page: 'admin-dashboard', icon: Shield, label: 'Dashboard' },
   581→        { page: 'admin-regions', icon: Globe2, label: 'Regions' },
   582→        { page: 'admin-organizations', icon: Building2, label: 'Orgs' },
   583→      ]
   584→    : [
   585→        { page: 'dashboard', icon: LayoutDashboard, label: 'Home' },
   586→        ...(isService && !isMixed && isModuleActive('services')
   587→          ? [{ page: 'services' as Page, icon: Wrench, label: 'Services' }]
   588→          : isModuleActive('inventory')
   589→            ? [{ page: 'products' as Page, icon: Package, label: 'Products' }]
   590→            : []),
   591→        ...(isModuleActive('sales') ? [{ page: 'sales' as Page, icon: ShoppingCart, label: 'Sales' }] : []),
   592→      ]
   593→
   594→  const moreItems: Array<{ page: Page; icon: React.ElementType; label: string }> = isAdmin
   595→    ? [
   596→        { page: 'admin-users' as Page, icon: UserCog, label: 'Users' },
   597→        { page: 'admin-modules' as Page, icon: Cog, label: 'Modules' },
   598→        { page: 'admin-sales-team' as Page, icon: Megaphone, label: 'Sales Team' },
   599→        { page: 'admin-notifications' as Page, icon: Bell, label: 'Notify' },
   600→      ]
   601→    : [
   602→        ...(isManager ? [{ page: 'branches' as Page, icon: GitBranch, label: 'Branches' }] : []),
   603→        ...(isMixed && isModuleActive('services') ? [{ page: 'services' as Page, icon: Wrench, label: 'Services' }] : []),
   604→        ...(isManager && isModuleActive('inventory') ? [{ page: 'inventory' as Page, icon: Warehouse, label: 'Inventory' }] : []),
   605→        ...(isModuleActive('customers') ? [{ page: 'customers' as Page, icon: Users, label: 'Customers' }] : []),
   606→        ...(isManager && isModuleActive('debts') ? [{ page: 'debts' as Page, icon: CreditCard, label: 'Debts' }] : []),
   607→        ...(isManager && isModuleActive('reports') ? [{ page: 'reports' as Page, icon: BarChart3, label: 'Reports' }] : []),
   608→        ...(isModuleActive('ai-assistant') ? [{ page: 'ai-assistant' as Page, icon: Bot, label: 'AI Chat' }] : []),
   609→        { page: 'modules' as Page, icon: Puzzle, label: 'Modules' },
   610→        { page: 'settings' as Page, icon: Settings, label: 'Settings' },
   611→      ]
   612→
   613→  const isMoreActive = !['dashboard', 'products', 'sales', 'services', 'admin-dashboard', 'admin-regions', 'admin-organizations'].includes(currentPage)
   614→
   615→  // Focus trap for "More" panel (accessibility)
   616→  const handleMoreKeyDown = useCallback((e: React.KeyboardEvent) => {
   617→    if (e.key === 'Escape') {
   618→      setShowMore(false)
   619→      moreButtonRef.current?.focus()
   620→    }
   621→  }, [])
   622→
   623→  // Focus the more panel when it opens
   624→  useEffect(() => {
   625→    if (showMore && morePanelRef.current) {
   626→      const firstButton = morePanelRef.current.querySelector<HTMLElement>('button')
   627→      firstButton?.focus()
   628→    }
   629→  }, [showMore])
   630→
   631→  return (
   632→    <>
   633→      {/* More Menu Overlay — ARIA dialog for accessibility */}
   634→      {showMore && (
   635→        <div
   636→          className="fixed inset-0 z-40 md:hidden"
   637→          onClick={() => { setShowMore(false); moreButtonRef.current?.focus() }}
   638→          onKeyDown={handleMoreKeyDown}
   639→          role="dialog"
   640→          aria-modal="true"
   641→          aria-label="More navigation options"
   642→        >
   643→          <div className="absolute inset-0 bg-black/40" aria-hidden="true" />
   644→          <div
   645→            ref={morePanelRef}
   646→            className="absolute bottom-14 left-0 right-0 bg-background border-t rounded-t-2xl shadow-2xl p-4 max-h-[60vh] overflow-y-auto"
   647→            onClick={(e) => e.stopPropagation()}
   648→          >
   649→            <div className="flex items-center justify-between mb-3">
   650→              <h3 id="more-panel-title" className="font-semibold text-sm">More Options</h3>
   651→              <button
   652→                onClick={() => { setShowMore(false); moreButtonRef.current?.focus() }}
   653→                className="text-muted-foreground p-2 min-w-11 min-h-11 flex items-center justify-center"
   654→                aria-label="Close more options"
   655→              >
   656→                <X className="size-5" aria-hidden="true" />
   657→              </button>
   658→            </div>
   659→            <div className="grid grid-cols-4 gap-3" role="list">
   660→              {moreItems.map(item => (
   661→                <button
   662→                  key={item.page}
   663→                  onClick={() => { setPage(item.page); setShowMore(false) }}
   664→                  className={`flex flex-col items-center gap-1.5 p-3 rounded-xl transition-colors min-h-11 ${
   665→                    currentPage === item.page
   666→                      ? 'bg-primary/10 text-primary'
   667→                      : 'hover:bg-muted text-muted-foreground'
   668→                  }`}
   669→                  role="listitem"
   670→                  aria-label={item.label}
   671→                  aria-current={currentPage === item.page ? 'page' : undefined}
   672→                >
   673→                  <item.icon className="size-5" aria-hidden="true" />
   674→                  <span className="text-xs font-medium">{item.label}</span>
   675→                </button>
   676→              ))}
   677→              {/* Theme toggle */}
   678→              <button
   679→                onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   680→                className="flex flex-col items-center gap-1.5 p-3 rounded-xl transition-colors hover:bg-muted text-muted-foreground min-h-11"
   681→                aria-label={`Switch to ${theme === 'dark' ? 'light' : 'dark'} theme`}
   682→              >
   683→                {theme === 'dark' ? <Sun className="size-5" aria-hidden="true" /> : <Moon className="size-5" aria-hidden="true" />}
   684→                <span className="text-xs font-medium">{theme === 'dark' ? 'Light' : 'Dark'}</span>
   685→              </button>
   686→            </div>
   687→          </div>
   688→        </div>
   689→      )}
   690→
   691→      {/* Bottom Nav Bar */}
   692→      <nav className="fixed bottom-0 left-0 right-0 z-50 md:hidden border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80" aria-label="Mobile navigation">
   693→        <div className="flex items-center justify-around h-14 pb-[env(safe-area-inset-bottom)]">
   694→          {mainTabs.map(tab => (
   695→            <button
   696→              key={tab.page}
   697→              onClick={() => setPage(tab.page)}
   698→              className={`flex flex-col items-center justify-center gap-0.5 min-w-0 flex-1 h-full transition-colors min-h-11 ${
   699→                currentPage === tab.page
   700→                  ? 'text-primary'
   701→                  : 'text-muted-foreground'
   702→              }`}
   703→              aria-label={tab.label}
   704→              aria-current={currentPage === tab.page ? 'page' : undefined}
   705→            >
   706→              <tab.icon className="size-5" aria-hidden="true" />
   707→              <span className="text-xs font-medium">{tab.label}</span>
   708→            </button>
   709→          ))}
   710→          <button
   711→            ref={moreButtonRef}
   712→            onClick={() => setShowMore(true)}
   713→            className={`flex flex-col items-center justify-center gap-0.5 min-w-0 flex-1 h-full transition-colors min-h-11 ${
   714→              isMoreActive
   715→                ? 'text-primary'
   716→                : 'text-muted-foreground'
   717→            }`}
   718→            aria-label="More options"
   719→            aria-expanded={showMore}
   720→            aria-haspopup="dialog"
   721→          >
   722→            <BarChart3 className="size-5" aria-hidden="true" />
   723→            <span className="text-xs font-medium">More</span>
   724→          </button>
   725→        </div>
   726→      </nav>
   727→    </>
   728→  )
   729→}
