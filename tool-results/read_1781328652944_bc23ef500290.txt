   128→function OrganizationTab({ orgId }: { orgId: string }) {
   129→  const [orgData, setOrgData] = useState<{
   130→    organization: Organization & {
   131→      members: Array<{
   132→        id: string
   133→        userId: string
   134→        role: string
   135→        user: { id: string; name: string; email: string; avatarUrl?: string | null }
   136→      }>
   137→    }
   138→  } | null>(null)
   139→  const [loading, setLoading] = useState(true)
   140→  const [saving, setSaving] = useState(false)
   141→  const [fetchError, setFetchError] = useState<string | null>(null)
   142→
   143→  // Org settings form
   144→  const orgForm = useForm<OrgSettingsFormData>({
   145→    resolver: zodResolver(orgSettingsSchema),
   146→    defaultValues: { orgName: '', orgLogoUrl: '', orgCurrency: 'ETB', orgCountry: 'Ethiopia' },
   147→  })
   148→
   149→  // Invite form
   150→  const inviteForm = useForm<InviteMemberFormData>({
   151→    resolver: zodResolver(inviteMemberSchema),
   152→    defaultValues: { email: '', role: 'member' },
   153→  })
   154→
   155→  // Invite dialog
   156→  const [showInvite, setShowInvite] = useState(false)
   157→  const [inviteSubmitting, setInviteSubmitting] = useState(false)
   158→
   159→  const fetchOrg = useCallback(async () => {
   160→    if (!orgId) return
   161→    setLoading(true)
   162→    try {
   163→      setFetchError(null)
   164→      const data = await api.getOrganization(orgId)
   165→      setOrgData(data)
   166→      orgForm.reset({
   167→        orgName: data.organization.name,
   168→        orgLogoUrl: data.organization.slug || '',
   169→        orgCurrency: data.organization.currency || 'ETB',
   170→        orgCountry: data.organization.country || 'Ethiopia',
   171→      })
   172→    } catch (err) {
   173→      setFetchError(getNetworkErrorMessage(err))
   174→    } finally {
   175→      setLoading(false)
   176→    }
   177→  }, [orgId])
   178→
   179→  useEffect(() => {
   180→    fetchOrg()
   181→  }, [fetchOrg])
   182→
   183→  const handleSave = async (data: OrgSettingsFormData) => {
   184→    if (!orgId) return
   185→    setSaving(true)
   186→    try {
   187→      const res = await fetch(`/api/organizations/${orgId}`, {
   188→        method: 'PATCH',
   189→        headers: {
   190→          'Content-Type': 'application/json',
   191→          Authorization: `Bearer ${localStorage.getItem('sb_token')}`,
   192→        },
   193→        body: JSON.stringify({
   194→          name: data.orgName,
   195→          currency: data.orgCurrency,
   196→          country: data.orgCountry,
   197→          logoUrl: data.orgLogoUrl || undefined,
   198→        }),
   199→      })
   200→      if (!res.ok) {
   201→        throw new Error(`Failed to update organization (${res.status})`)
   202→      }
   203→      toast.success('Organization updated successfully')
   204→      fetchOrg()
   205→    } catch (err) {
   206→      toast.error(getNetworkErrorMessage(err))
   207→    } finally {
   208→      setSaving(false)
   209→    }
   210→  }
   211→
   212→  const handleInvite = async (data: InviteMemberFormData) => {
   213→    setInviteSubmitting(true)
   214→    try {
   215→      // Simulated invite - no actual API endpoint
   216→      await new Promise((r) => setTimeout(r, 1000))
   217→      toast.success(`Invitation sent to ${data.email}`)
   218→      setShowInvite(false)
   219→      inviteForm.reset()
   220→    } catch (err) {
   221→      toast.error(getNetworkErrorMessage(err))
   222→    } finally {
   223→      setInviteSubmitting(false)
   224→    }
   225→  }
   226→
   227→  if (loading) {
   228→    return (
   229→      <div className="space-y-6">
   230→        <Card>
   231→          <CardContent className="p-6 space-y-4">
   232→            {Array.from({ length: 4 }).map((_, i) => (
   233→              <Skeleton key={i} className="h-10 w-full" />
   234→            ))}
   235→          </CardContent>
   236→        </Card>
   237→      </div>
   238→    )
   239→  }
   240→
   241→  if (fetchError) {
   242→    return <ErrorState title="Failed to load organization settings" error={fetchError} onRetry={fetchOrg} />
   243→  }
   244→
   245→  const members = orgData?.organization.members || []
   246→
   247→  return (
   248→    <div className="space-y-6">
   249→      {/* Org Info */}
   250→      <Card>
   251→        <CardHeader>
   252→          <CardTitle>Organization Details</CardTitle>
   253→          <CardDescription>Update your organization information</CardDescription>
   254→        </CardHeader>
   255→        <Form {...orgForm}>
   256→          <form onSubmit={orgForm.handleSubmit(handleSave)}>
   257→            <CardContent className="space-y-4">
   258→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   259→                <FormInputField<OrgSettingsFormData>
   260→                  name="orgName"
   261→                  label="Organization Name"
   262→                  required
   263→                />
   264→                <FormInputField<OrgSettingsFormData>
   265→                  name="orgLogoUrl"
   266→                  label="Logo URL"
   267→                  placeholder="https://..."
   268→                />
   269→                <FormSelectField<OrgSettingsFormData>
   270→                  name="orgCurrency"
   271→                  label="Currency"
   272→                  options={[
   273→                    { value: 'ETB', label: 'ETB - Ethiopian Birr' },
   274→                    { value: 'USD', label: 'USD - US Dollar' },
   275→                    { value: 'EUR', label: 'EUR - Euro' },
   276→                    { value: 'GBP', label: 'GBP - British Pound' },
   277→                    { value: 'KES', label: 'KES - Kenyan Shilling' },
   278→                  ]}
   279→                />
   280→                <FormSelectField<OrgSettingsFormData>
   281→                  name="orgCountry"
   282→                  label="Country"
   283→                  options={[
   284→                    { value: 'Ethiopia', label: 'Ethiopia' },
   285→                    { value: 'Kenya', label: 'Kenya' },
   286→                    { value: 'Nigeria', label: 'Nigeria' },
   287→                    { value: 'Ghana', label: 'Ghana' },
   288→                    { value: 'South Africa', label: 'South Africa' },
   289→                    { value: 'United States', label: 'United States' },
   290→                    { value: 'United Kingdom', label: 'United Kingdom' },
   291→                  ]}
   292→                />
   293→              </div>
   294→            </CardContent>
   295→            <CardFooter>
   296→              <FormSubmitButton isLoading={saving} loadingText="Saving..." className="gap-2">
   297→                <><Save className="size-4" /> Save Changes</>
   298→              </FormSubmitButton>
   299→            </CardFooter>
   300→          </form>
   301→        </Form>
   302→      </Card>
   303→
   304→      {/* Members */}
   305→      <Card>
   306→        <CardHeader>
   307→          <div className="flex items-center justify-between">
   308→            <div>
   309→              <CardTitle>Team Members</CardTitle>
   310→              <CardDescription>Manage who has access to your organization</CardDescription>
   311→            </div>
   312→            <Button onClick={() => setShowInvite(true)} className="gap-2">
   313→              <UserPlus className="size-4" />
   314→              Invite
   315→            </Button>
   316→          </div>
   317→        </CardHeader>
   318→        <CardContent className="p-0">
   319→          {/* Desktop table */}
   320→          <div className="hidden md:block">
   321→            <Table>
   322→              <TableHeader>
   323→                <TableRow>
   324→                  <TableHead>Name</TableHead>
   325→                  <TableHead>Email</TableHead>
   326→                  <TableHead>Role</TableHead>
   327→                </TableRow>
   328→              </TableHeader>
   329→              <TableBody>
   330→                {members.map((member) => (
   331→                  <TableRow key={member.id}>
   332→                    <TableCell className="font-medium">
   333→                      <div className="flex items-center gap-2">
   334→                        <div className="size-8 rounded-full bg-primary/10 flex items-center justify-center text-xs font-bold text-primary">
   335→                          {member.user.name.charAt(0).toUpperCase()}
   336→                        </div>
   337→                        {member.user.name}
   338→                      </div>
   339→                    </TableCell>
   340→                    <TableCell className="text-muted-foreground">{member.user.email}</TableCell>

   346→                            ? 'bg-amber-100 text-amber-800 border-amber-300 dark:bg-amber-900/30 dark:text-amber-400 dark:border-amber-700'

   348→                              ? 'bg-blue-100 text-blue-800 border-blue-300 dark:bg-blue-900/30 dark:text-blue-400 dark:border-blue-700'
   349→                              : 'bg-gray-100 text-gray-800 border-gray-300 dark:bg-gray-900/30 dark:text-gray-400 dark:border-gray-700'
   350→                        }

   426→                name="role"
   427→                label="Role"
   428→                options={[
   429→                  { value: 'member', label: 'Member' },

   432→              />
   433→            </form>

   435→          <DialogFooter>
   436→            <Button variant="outline" onClick={() => { setShowInvite(false); inviteForm.reset() }}>
   437→              Cancel
   438→            </Button>
   439→            <FormSubmitButton
   440→              isLoading={inviteSubmitting}

   443→            >
   444→              Send Invitation
   445→            </FormSubmitButton>
   446→          </DialogFooter>
   447→        </DialogContent>

  1168→function ModulesTab({ orgId }: { orgId: string }) {
  1169→  const [modules, setModules] = useState<Array<{
  1170→    id: string; key: string; name: string; description: string; icon: string;
  1171→    category: string; priceETB: number; isActive: boolean; order: number;
  1172→  }>>([])
  1173→  const [loading, setLoading] = useState(true)
  1174→  const [fetchError, setFetchError] = useState<string | null>(null)
  1175→  const [search, setSearch] = useState('')
  1176→  const [categoryFilter, setCategoryFilter] = useState('all')
  1177→  const [togglingKey, setTogglingKey] = useState<string | null>(null)
  1178→  const [deactivateTarget, setDeactivateTarget] = useState<typeof modules[0] | null>(null)
  1179→  const [deactivating, setDeactivating] = useState(false)
  1180→
  1181→  const fetchModules = useCallback(async () => {
  1182→    if (!orgId) return
  1183→    setLoading(true)
  1184→    try {
  1185→      setFetchError(null)
  1186→      const data = await api.getModules(orgId)
  1187→      setModules(data.modules)
  1188→    } catch (err) {
  1189→      setFetchError(getNetworkErrorMessage(err))
  1190→    } finally { setLoading(false) }
  1191→  }, [orgId])
  1192→
  1193→  useEffect(() => { fetchModules() }, [fetchModules])
  1194→
  1195→  const handleToggleModule = async (mod: typeof modules[0]) => {
  1196→    const activating = !mod.isActive
  1197→    // Require confirmation before deactivating
  1198→    if (!activating) {
  1199→      setDeactivateTarget(mod)
  1200→      return
  1201→    }
  1202→    setTogglingKey(mod.key)
  1203→    try {
  1204→      await api.activateModule(orgId, mod.key)

  1207→    } catch (err) {
  1208→      toast.error(getNetworkErrorMessage(err))

  1210→    finally { setTogglingKey(null) }

  1212→
  1213→  const handleConfirmDeactivate = async () => {

  1218→      await api.deactivateModule(orgId, deactivateTarget.key)
  1219→      toast.success(`${deactivateTarget.name} deactivated`)
  1220→      setDeactivateTarget(null)
  1221→      fetchModules()
  1222→    } catch (err) {
  1223→      toast.error(getNetworkErrorMessage(err))
  1224→    }
  1225→    finally { setTogglingKey(null); setDeactivating(false) }
  1226→  }
  1227→
  1228→  const filteredModules = modules.filter((m) => {
  1229→    const matchesSearch = search === '' || m.name.toLowerCase().includes(search.toLowerCase()) || m.description.toLowerCase().includes(search.toLowerCase())
  1230→    const matchesCategory = categoryFilter === 'all' || m.category === categoryFilter
  1231→    return matchesSearch && matchesCategory
  1232→  })
  1233→
  1234→  const activeModules = modules.filter(m => m.isActive)
  1235→  const totalMonthlyCost = activeModules.reduce((sum, m) => sum + m.priceETB, 0)
  1236→
  1237→  const categories = [

  1240→    { key: 'analytics', label: 'Analytics' },

  1243→  ]
  1244→

  1247→      case 'core': return 'bg-brand-50 dark:bg-brand-900/20 text-primary'
  1248→      case 'analytics': return 'bg-sky-50 dark:bg-sky-900/30 text-sky-600 dark:text-sky-400'
  1249→      case 'ai': return 'bg-violet-50 dark:bg-violet-900/30 text-violet-600 dark:text-violet-400'
  1250→      case 'integration': return 'bg-emerald-50 dark:bg-emerald-900/30 text-emerald-600 dark:text-emerald-400'
  1251→      default: return 'bg-gray-50 dark:bg-gray-900/30 text-gray-600 dark:text-gray-400'
  1252→    }
  1253→  }
  1254→
  1255→  if (loading) {
  1256→    return (
  1257→      <div className="space-y-6">
  1258→        <Skeleton className="h-24 w-full rounded-lg" />
  1259→        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
  1260→          {Array.from({ length: 6 }).map((_, i) => (
  1261→            <Card key={i}><CardContent className="p-4 space-y-3">
  1262→              <Skeleton className="h-10 w-10 rounded-lg" /><Skeleton className="h-5 w-32" />
  1263→              <Skeleton className="h-4 w-full" /><Skeleton className="h-4 w-20" />

  1270→
  1271→  if (fetchError) {
  1272→    return <ErrorState title="Failed to load modules" error={fetchError} onRetry={fetchModules} />

  1274→
  1275→  return (
  1276→    <div className="space-y-6">
  1277→      {/* Active Module Summary */}
  1278→      <Card>

  1300→          </div>
  1301→          {activeModules.length > 0 && (
