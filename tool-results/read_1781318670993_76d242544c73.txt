    28→// ============================================
    29→// Types

    31→
    32→interface CreditLimitData {
    33→  id: string
    34→  customerId: string
    35→  creditLimit: number
    36→  usedCredit: number
    37→  availableCredit: number
    38→  alertThreshold: number
    39→  isBlocked: boolean
    40→  createdAt: string
    41→  updatedAt: string
    42→  customer: {
    43→    id: string
    44→    name: string
    45→    email: string | null
    46→    phone: string | null
    47→  }
    48→}
    49→
    50→interface CreditSummary {
    51→  totalCreditExtended: number
    52→  totalUsed: number
    53→  totalAvailable: number
    54→  blockedAccounts: number
    55→}
    56→
    57→// ============================================
    58→// Helpers
    59→// ============================================
    60→
    61→function formatETB(amount: number): string {
    62→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    63→}
    64→
    65→function getUsagePercentage(used: number, limit: number): number {
    66→  if (limit <= 0) return 0
    67→  return Math.min(Math.round((used / limit) * 100), 100)
    68→}
    69→
    70→function getUsageColor(percentage: number): string {
    71→  if (percentage > 80) return 'text-red-600'
    72→  if (percentage >= 60) return 'text-yellow-600'
    73→  return 'text-emerald-600'
    74→}
    75→
    76→function getUsageBarClass(percentage: number): string {
    77→  if (percentage > 80) return '[&_[data-slot=progress-indicator]]:bg-red-500'
    78→  if (percentage >= 60) return '[&_[data-slot=progress-indicator]]:bg-yellow-500'
    79→  return '[&_[data-slot=progress-indicator]]:bg-emerald-500'
    80→}
    81→
    82→function isAlertTriggered(used: number, limit: number, threshold: number): boolean {
    83→  if (limit <= 0) return false
    84→  return (used / limit) > threshold
    85→}

   171→  }

   177→      toast.error('Please enter a valid credit limit amount')

   180→    const threshold = parseFloat(alertThreshold)

   182→      toast.error('Alert threshold must be between 0 and 100')
   183→      return

   186→    setFormSubmitting(true)

   190→        await api.request(`/api/credit-limits/${editLimit.id}`, {
   191→          method: 'PUT',
   192→          body: JSON.stringify({

   194→            creditLimit: amount,
   195→            alertThreshold: threshold / 100,
   196→          }),

   198→        toast.success('Credit limit updated successfully')
   199→      } else {
   200→        // Need to select a customer — this path is for the "add new" flow
   201→        // For simplicity, the set dialog will always edit from the list

   205→      fetchCreditLimits()

   209→      setFormSubmitting(false)
   210→    }

   227→      toast.error(err instanceof Error ? err.message : 'Failed to toggle block')

   250→          <CardContent className="p-4">

   255→              <div>

   259→            </div>
   260→          </CardContent>
   261→        </Card>
   262→        <Card>
   263→          <CardContent className="p-4">
   264→            <div className="flex items-center gap-3">
   265→              <div className="size-10 rounded-lg bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   266→                <DollarSign className="size-5 text-red-600" />
   267→              </div>
   268→              <div>
   269→                <p className="text-sm text-muted-foreground">Total Used</p>
   270→                <p className="text-base font-semibold">{formatETB(summary.totalUsed)}</p>
   271→              </div>
   272→            </div>
   273→          </CardContent>
   274→        </Card>
   275→        <Card>
   276→          <CardContent className="p-4">

   278→              <div className="size-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   279→                <DollarSign className="size-5 text-emerald-600" />
   280→              </div>
   281→              <div>
   282→                <p className="text-sm text-muted-foreground">Total Available</p>
   283→                <p className="text-base font-semibold">{formatETB(summary.totalAvailable)}</p>
   284→              </div>
   285→            </div>

   287→        </Card>
   288→        <Card>
   289→          <CardContent className="p-4">
   290→            <div className="flex items-center gap-3">
   291→              <div className="size-10 rounded-lg bg-orange-100 dark:bg-orange-900/30 flex items-center justify-center">
   292→                <ShieldAlert className="size-5 text-orange-600" />

   305→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   306→        <Input
   307→          placeholder="Search by customer name..."
   308→          value={searchQuery}
   309→          onChange={(e) => setSearchQuery(e.target.value)}
   310→          className="pl-9"
   311→        />
   312→      </div>
   313→
   314→      {/* Content */}
   315→      {pageError ? (
   316→        <ErrorState title="Failed to load credit limits" message={pageError} onRetry={fetchCreditLimits} />
   317→      ) : loading ? (

   319→          <CardContent className="p-6">
   320→            <div className="space-y-4">

   322→                <div key={i} className="flex items-center gap-4">
   323→                  <Skeleton className="size-10 rounded-full" />
   324→                  <div className="flex-1 space-y-2">
   325→                    <Skeleton className="h-4 w-40" />
   326→                    <Skeleton className="h-3 w-24" />
   327→                  </div>
   328→                  <Skeleton className="h-4 w-24" />
   329→                  <Skeleton className="h-2 w-32" />
   330→                </div>
   331→              ))}
   332→            </div>

   334→        </Card>
   335→      ) : creditLimits.length === 0 ? (
   336→        <EmptyState
   337→          title="No credit limits found"
   338→          message={searchQuery ? 'Try adjusting your search query' : 'Credit limits will appear here when set for customers'}
   339→          icon={<CreditCard className="size-7 text-muted-foreground" />}

   341→      ) : (
   342→        <>
   343→          {/* Desktop Table */}
   344→          <div className="hidden lg:block">
   345→            <Card>
   346→              <Table>
   347→                <TableHeader>
   348→                  <TableRow>
   349→                    <TableHead>Customer</TableHead>
   350→                    <TableHead className="text-right">Credit Limit</TableHead>
   351→                    <TableHead className="text-right">Used</TableHead>
   352→                    <TableHead className="text-right">Available</TableHead>
   353→                    <TableHead>Usage</TableHead>
   354→                    <TableHead>Status</TableHead>
   355→                    <TableHead className="text-right">Actions</TableHead>
   356→                  </TableRow>
   357→                </TableHeader>
   358→                <TableBody>

   360→                    const usagePct = getUsagePercentage(cl.usedCredit, cl.creditLimit)
   361→                    const alertTriggered = isAlertTriggered(cl.usedCredit, cl.creditLimit, cl.alertThreshold)
   362→                    return (
   363→                      <TableRow key={cl.id}>
   364→                        <TableCell>
   365→                          <div>
   366→                            <p className="font-medium">{cl.customer.name}</p>
   367→                            {cl.customer.phone && (
   368→                              <p className="text-xs text-muted-foreground">{cl.customer.phone}</p>
   369→                            )}
   370→                          </div>
   371→                        </TableCell>
   372→                        <TableCell className="text-right font-medium">
   373→                          {formatETB(cl.creditLimit)}
   374→                        </TableCell>

   379→                        </TableCell>

   381→                          {formatETB(cl.availableCredit)}

   384→                          <div className="flex items-center gap-2 min-w-[120px]">

   390→                              {usagePct}%
   391→                            </span>
   392→                          </div>
   393→                        </TableCell>

   398→                                <Lock className="size-3" />

   401→                            )}

   403→                              <Badge variant="outline" className="text-xs gap-1 border-yellow-300 text-yellow-600 bg-yellow-50">
   404→                                <AlertTriangle className="size-3" />
   405→                                Alert
   406→                              </Badge>
   407→                            )}
   408→                            {!cl.isBlocked && !alertTriggered && (
   409→                              <Badge variant="outline" className="text-xs border-emerald-300 text-emerald-600 bg-emerald-50">
   410→                                Active
   411→                              </Badge>
   412→                            )}
   413→                          </div>
   414→                        </TableCell>
   415→                        <TableCell className="text-right">
   416→                          <div className="flex items-center justify-end gap-1">
   417→                            <Button
   418→                              variant="ghost"
   419→                              size="icon"
   420→                              className="size-8"
   421→                              onClick={() => openSetLimit(cl)}
   422→                              title="Edit credit limit"
   423→                            >
   424→                              <Pencil className="size-4" />
   425→                            </Button>
   426→                            <Button
   427→                              variant="ghost"
   428→                              size="icon"
   429→                              className="size-8"
   430→                              disabled={togglingBlock === cl.id}
   431→                              onClick={() => handleToggleBlock(cl)}
   432→                              title={cl.isBlocked ? 'Unblock credit' : 'Block credit'}

   436→                              ) : cl.isBlocked ? (

   439→                                <Lock className="size-4 text-red-600" />

   441→                            </Button>

   443→                        </TableCell>
   444→                      </TableRow>
   445→                    )

   448→              </Table>

   479→                        </Badge>

   481→                    </div>
   482→
   483→                    {/* Usage Bar */}
   484→                    <div className="mb-3">
   485→                      <div className="flex items-center justify-between mb-1">
   486→                        <span className="text-xs text-muted-foreground">
   487→                          {formatETB(cl.usedCredit)} of {formatETB(cl.creditLimit)}
   488→                        </span>
   489→                        <span className={`text-xs font-medium ${getUsageColor(usagePct)}`}>{usagePct}%</span>
   490→                      </div>
   491→                      <Progress
   492→                        value={usagePct}
   493→                        className={`h-2 ${getUsageBarClass(usagePct)}`}
   494→                      />
   495→                    </div>
   496→
   497→                    {/* Amounts Grid */}
   498→                    <div className="grid grid-cols-3 gap-2 mb-3 text-center">
   499→                      <div>
   500→                        <p className="text-xs text-muted-foreground">Limit</p>
   501→                        <p className="text-sm font-medium">{formatETB(cl.creditLimit)}</p>
   502→                      </div>
   503→                      <div>
   504→                        <p className="text-xs text-muted-foreground">Used</p>

   507→                      <div>
   508→                        <p className="text-xs text-muted-foreground">Available</p>

   510→                      </div>
   511→                    </div>
   512→
   513→                    {/* Actions */}
   514→                    <div className="flex items-center gap-2">
   515→                      <Button
   516→                        variant="outline"
   517→                        size="sm"
   518→                        className="h-8 text-xs gap-1 flex-1"
   519→                        onClick={() => openSetLimit(cl)}
   520→                      >
   521→                        <Pencil className="size-3.5" />
   522→                        Edit Limit

   525→                        variant="outline"
   526→                        size="sm"
   527→                        className={`h-8 text-xs gap-1 ${cl.isBlocked ? 'text-emerald-600' : 'text-red-600'}`}
   528→                        disabled={togglingBlock === cl.id}
   529→                        onClick={() => handleToggleBlock(cl)}

   540→                    </div>
   541→                  </CardContent>
   542→                </Card>
   543→              )
   544→            })}
   545→          </div>
   546→
   547→          {/* Pagination */}
   548→          {totalPages > 1 && (
   549→            <div className="flex items-center justify-between">
   550→              <p className="text-sm text-muted-foreground">
