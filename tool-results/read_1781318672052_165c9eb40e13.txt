     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import { useForm } from 'react-hook-form'
     5→import { zodResolver } from '@hookform/resolvers/zod'
     6→import {
     7→  Truck, Plus, Search, Eye, Pencil, Trash2,
     8→  DollarSign, MapPin, Mail, Phone,
     9→  Building2, ChevronLeft, ChevronRight, CreditCard, Loader2,
    10→} from 'lucide-react'
    11→import { api, type Supplier, type Debt } from '@/lib/api-client'
    12→import { useAuthStore } from '@/lib/stores/auth-store'
    13→import { supplierSchema, type SupplierFormData } from '@/lib/validations'
    14→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Badge } from '@/components/ui/badge'
    18→import {
    19→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
    20→} from '@/components/ui/table'
    21→import {
    22→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription
    23→} from '@/components/ui/dialog'
    24→import {
    25→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    26→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    27→} from '@/components/ui/alert-dialog'
    28→import { Form } from '@/components/ui/form'
    29→import { FormInputField, FormTextareaField, FormSubmitButton } from '@/components/shared/form-fields'
    30→import { Separator } from '@/components/ui/separator'
    31→import { Skeleton } from '@/components/ui/skeleton'
    32→import { toast } from 'sonner'
    33→import { getNetworkErrorMessage } from '@/lib/validation'
    34→import { ErrorState } from '@/components/shared/error-states'
    35→
    36→// ============================================
    37→// Helpers
    38→// ============================================
    39→function formatETB(amount: number): string {
    40→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    41→}
    42→
    43→function formatDate(dateStr: string | null | undefined): string {
    44→  if (!dateStr) return '—'
    45→  return new Date(dateStr).toLocaleDateString('en-US', {
    46→    year: 'numeric', month: 'short', day: 'numeric'
    47→  })
    48→}
    49→
    50→// ============================================
    51→// Main Component
    52→// ============================================
    53→export function SuppliersPage() {
    54→  const { currentOrg, currentShop } = useAuthStore()
    55→  const orgId = currentOrg?.id || ''
    56→  const shopId = currentShop?.id
    57→
    58→  const [suppliers, setSuppliers] = useState<Supplier[]>([])
    59→  const [loading, setLoading] = useState(true)
    60→  const [pageError, setPageError] = useState<string | null>(null)
    61→  const [searchQuery, setSearchQuery] = useState('')
    62→  const [page, setPage] = useState(1)
    63→  const [totalPages, setTotalPages] = useState(1)
    64→  const [total, setTotal] = useState(0)
    65→  const limit = 10
    66→
    67→  // Dialogs
    68→  const [showAddEdit, setShowAddEdit] = useState(false)
    69→  const [showDetail, setShowDetail] = useState(false)
    70→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false)
    71→  const [selectedSupplier, setSelectedSupplier] = useState<Supplier | null>(null)
    72→  const [isEditing, setIsEditing] = useState(false)
    73→
    74→  // Form state
    75→  const [formSubmitting, setFormSubmitting] = useState(false)
    76→  const [deleting, setDeleting] = useState(false)
    77→
    78→  const form = useForm<SupplierFormData>({
    79→    resolver: zodResolver(supplierSchema),
    80→    defaultValues: {
    81→      name: '',
    82→      email: '',
    83→      phone: '',
    84→      address: '',
    85→    },
    86→  })
    87→
    88→  // Supplier detail data
    89→  const [supplierDebts, setSupplierDebts] = useState<Debt[]>([])
    90→  const [detailLoading, setDetailLoading] = useState(false)
    91→  const [totalOwedToSuppliers, setTotalOwedToSuppliers] = useState(0)
    92→
    93→  const fetchSuppliers = useCallback(async () => {
    94→    if (!orgId) return
    95→    setLoading(true)
    96→    setPageError(null)
    97→    try {
    98→      const data = await api.getSuppliers(orgId, {
    99→        search: searchQuery || undefined,
   100→        page,
   101→        limit,
   102→        shopId,
   103→      })
   104→      setSuppliers(data.suppliers)
   105→      setTotalPages(data.pagination.totalPages)
   106→      setTotal(data.pagination.total)
   107→
   108→      // Get total owed to suppliers from debts API
   109→      if (page === 1 && !searchQuery) {
   110→        try {
   111→          const debtsData = await api.getDebts(orgId, { type: 'supplier_debt', status: 'all', limit: '100', shopId })
   112→          const outstanding = debtsData.debts
   113→            .filter((d) => d.status !== 'paid')
   114→            .reduce((sum, d) => sum + (d.amount - d.paidAmount), 0)
   115→          setTotalOwedToSuppliers(outstanding)
   116→        } catch {
   117→          // Partial data load — debt stats may be incomplete but suppliers still show
   118→        }
   119→      }
   120→    } catch (err) {
   121→      const msg = getNetworkErrorMessage(err)
   122→      setPageError(msg)
   123→      toast.error(msg)
   124→    } finally {
   125→      setLoading(false)
   126→    }
   127→  }, [orgId, searchQuery, page, shopId])
   128→
   129→  useEffect(() => {
   130→    fetchSuppliers()
   131→  }, [fetchSuppliers])
   132→
   133→  // Reset page when search changes
   134→  useEffect(() => {
   135→    setPage(1)
   136→  }, [searchQuery])
   137→
   138→  // Active suppliers = those with non-zero debts (recently active)
   139→  const activeSuppliers = suppliers.length
   140→
   141→  // ============================================
   142→  // Handlers
   143→  // ============================================
   144→  const openAddDialog = () => {
   145→    setIsEditing(false)
   146→    setSelectedSupplier(null)
   147→    form.reset({ name: '', email: '', phone: '', address: '' })
   148→    setShowAddEdit(true)
   149→  }
   150→
   151→  const openEditDialog = (supplier: Supplier) => {
   152→    setIsEditing(true)
   153→    setSelectedSupplier(supplier)
   154→    form.reset({
   155→      name: supplier.name,
   156→      email: supplier.email || '',
   157→      phone: supplier.phone || '',
   158→      address: supplier.address || '',
   159→    })
   160→    setShowAddEdit(true)
   161→  }
   162→
   163→  const handleSave = async (data: SupplierFormData) => {
   164→    if (!orgId) return
   165→
   166→    setFormSubmitting(true)
   167→    try {
   168→      if (isEditing && selectedSupplier) {
   169→        await api.updateSupplier(selectedSupplier.id, orgId, {
   170→          name: data.name.trim(),
   171→          email: data.email?.trim() || undefined,
   172→          phone: data.phone?.trim() || undefined,
   173→          address: data.address?.trim() || undefined,
   174→        })
   175→        toast.success('Supplier updated successfully')
   176→      } else {
   177→        await api.createSupplier(orgId, {
   178→          name: data.name.trim(),
   179→          email: data.email?.trim() || undefined,
   180→          phone: data.phone?.trim() || undefined,
   181→          address: data.address?.trim() || undefined,
   182→          shopId,
   183→        })
   184→        toast.success('Supplier created successfully')
   185→      }
   186→      setShowAddEdit(false)
   187→      fetchSuppliers()
   188→    } catch (err) {
   189→      toast.error(getNetworkErrorMessage(err))
   190→    } finally {
   191→      setFormSubmitting(false)
   192→    }
   193→  }
   194→
   195→  const openDeleteConfirm = (supplier: Supplier) => {
   196→    setSelectedSupplier(supplier)
   197→    setShowDeleteConfirm(true)
   198→  }
   199→
   200→  const handleDelete = async () => {
   201→    if (!selectedSupplier || !orgId) return
   202→    setDeleting(true)
   203→    try {
   204→      await api.deleteSupplier(selectedSupplier.id, orgId)
   205→      toast.success('Supplier deleted successfully')
   206→      setShowDeleteConfirm(false)
   207→      setSelectedSupplier(null)
   208→      fetchSuppliers()
   209→    } catch (err) {
   210→      toast.error(getNetworkErrorMessage(err))
   211→    } finally {
   212→      setDeleting(false)
   213→    }
   214→  }
   215→
   216→  const openDetail = async (supplier: Supplier) => {
   217→    setSelectedSupplier(supplier)
   218→    setShowDetail(true)
   219→    setDetailLoading(true)
   220→    try {
   221→      const debtsData = await api.getDebts(orgId, { type: 'supplier_debt', limit: '100', shopId })
   222→      const filteredDebts = debtsData.debts.filter((d) => d.supplierId === supplier.id)
   223→      setSupplierDebts(filteredDebts)
   224→    } catch (err) {
   225→      toast.error(getNetworkErrorMessage(err))
   226→    } finally {
   227→      setDetailLoading(false)
   228→    }
   229→  }
   230→
   231→  // ============================================
   232→  // Render
   233→  // ============================================
   234→  return (
   235→    <div className="space-y-6">
   236→      {/* Header */}
   237→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   238→        <div>
   239→          <h1 className="text-base font-semibold tracking-tight">Suppliers</h1>
   240→          <p className="text-muted-foreground text-sm">Manage supplier information and track supplier debts</p>
   241→        </div>
   242→        <Button onClick={openAddDialog} className="gap-2">
   243→          <Plus className="size-4" />
   244→          Add Supplier
   245→        </Button>
   246→      </div>
   247→
   248→      {/* Stats Cards */}
   249→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 md:gap-4">
   250→        <Card>
   251→          <CardContent className="p-4">
   252→            <div className="flex items-center gap-3">
   253→              <div className="size-10 rounded-lg bg-primary/10 flex items-center justify-center">
   254→                <Truck className="size-5 text-primary" />
   255→              </div>
   256→              <div className="min-w-0">
   257→                <p className="text-xs sm:text-sm text-muted-foreground">Total Suppliers</p>
   258→                <p className="text-base font-semibold">{total}</p>
   259→              </div>
   260→            </div>
   261→          </CardContent>
   262→        </Card>
   263→        <Card>
   264→          <CardContent className="p-4">
   265→            <div className="flex items-center gap-3">
   266→              <div className="size-10 rounded-lg bg-brand-50 dark:bg-brand-900/20 flex items-center justify-center">
   267→                <Building2 className="size-5 text-primary" />
   268→              </div>
   269→              <div className="min-w-0">
   270→                <p className="text-xs sm:text-sm text-muted-foreground">Active Suppliers</p>
   271→                <p className="text-base font-semibold">{activeSuppliers}</p>
   272→              </div>
   273→            </div>
   274→          </CardContent>
   275→        </Card>
   276→        <Card>
   277→          <CardContent className="p-4">
   278→            <div className="flex items-center gap-3">
   279→              <div className="size-10 rounded-lg bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   280→                <DollarSign className="size-5 text-red-600 dark:text-red-400" />
   281→              </div>
   282→              <div className="min-w-0">
   283→                <p className="text-xs sm:text-sm text-muted-foreground">Total Owed to Suppliers</p>
   284→                <p className="text-base font-semibold">{formatETB(totalOwedToSuppliers)}</p>
   285→              </div>
   286→            </div>
   287→          </CardContent>
   288→        </Card>
   289→      </div>
   290→
   291→      {/* Search */}
   292→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   293→        <div className="relative w-full sm:w-72">
   294→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   295→          <Input
   296→            placeholder="Search suppliers..."
   297→            value={searchQuery}
   298→            onChange={(e) => setSearchQuery(e.target.value)}
   299→            className="pl-9"
   300→          />
   301→        </div>
   302→      </div>
   303→
   304→      {/* Table */}
   305→      {loading ? (
   306→        <Card>
   307→          <CardContent className="p-6 space-y-4">
   308→            {Array.from({ length: 6 }).map((_, i) => (
   309→              <Skeleton key={i} className="h-12 w-full" />
   310→            ))}
   311→          </CardContent>
   312→        </Card>
   313→      ) : pageError && suppliers.length === 0 ? (
   314→        <ErrorState title="Failed to load suppliers" message={pageError} onRetry={fetchSuppliers} />
   315→      ) : suppliers.length === 0 ? (
   316→        <Card>
   317→          <CardContent className="py-12 text-center">
   318→            <Truck className="size-12 text-muted-foreground mx-auto mb-4" />
   319→            <p className="text-muted-foreground font-medium">
   320→              {searchQuery ? 'No suppliers match your search' : 'No suppliers yet'}
   321→            </p>
   322→            <p className="text-xs text-muted-foreground mt-1">
   323→              {searchQuery ? 'Try a different search term' : 'Add your first supplier to get started'}
   324→            </p>
   325→            {!searchQuery && (
   326→              <Button onClick={openAddDialog} className="mt-4 gap-2" variant="outline">
   327→                <Plus className="size-4" />
   328→                Add Supplier
   329→              </Button>
   330→            )}
   331→          </CardContent>
   332→        </Card>
   333→      ) : (
   334→        <Card>
   335→          <CardContent className="p-0">
   336→            {/* Desktop table */}
   337→            <div className="hidden md:block overflow-x-auto">
   338→              <Table>
   339→                <TableHeader>
   340→                  <TableRow>
   341→                    <TableHead>Name</TableHead>
   342→                    <TableHead className="hidden lg:table-cell">Email</TableHead>
   343→                    <TableHead>Phone</TableHead>
   344→                    <TableHead className="hidden lg:table-cell">Address</TableHead>
   345→                    <TableHead className="text-right">Debt Owed</TableHead>
   346→                    <TableHead className="hidden lg:table-cell">Created</TableHead>
   347→                    <TableHead className="text-right">Actions</TableHead>
   348→                  </TableRow>
   349→                </TableHeader>
   350→                <TableBody>
   351→                  {suppliers.map((supplier) => (
   352→                    <TableRow
   353→                      key={supplier.id}
   354→                      className="cursor-pointer hover:bg-muted/50"
   355→                      onClick={() => openDetail(supplier)}
   356→                    >
   357→                      <TableCell>
   358→                        <div className="flex items-center gap-3">
   359→                          <div className="size-8 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   360→                            <Truck className="size-4 text-primary" />
   361→                          </div>
   362→                          <div>
   363→                            <p className="font-medium">{supplier.name}</p>
   364→                            <p className="text-xs text-muted-foreground md:hidden">
   365→                              {supplier.email || supplier.phone || ''}
   366→                            </p>
   367→                          </div>
   368→                        </div>
   369→                      </TableCell>
   370→                      <TableCell className="hidden md:table-cell text-muted-foreground">
   371→                        {supplier.email || '—'}
   372→                      </TableCell>
   373→                      <TableCell className="hidden sm:table-cell text-muted-foreground">
   374→                        {supplier.phone || '—'}
   375→                      </TableCell>
   376→                      <TableCell className="hidden lg:table-cell text-muted-foreground max-w-[200px] truncate">
   377→                        {supplier.address || '—'}
   378→                      </TableCell>
   379→                      <TableCell className="text-right">
   380→                        <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800">
   381→                          {formatETB(0)}
   382→                        </Badge>
   383→                      </TableCell>
   384→                      <TableCell className="hidden md:table-cell text-muted-foreground text-sm">
   385→                        {formatDate(supplier.createdAt)}
   386→                      </TableCell>
   387→                      <TableCell className="text-right">
   388→                        <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   389→                          <Button
   390→                            variant="ghost"
   391→                            size="sm"
   392→                            onClick={() => openDetail(supplier)}
   393→                            title="View Details"
   394→                          >
   395→                            <Eye className="size-4" />
   396→                          </Button>
   397→                          <Button
   398→                            variant="ghost"
   399→                            size="sm"
   400→                            onClick={() => openEditDialog(supplier)}
   401→                            title="Edit"
   402→                          >
   403→                            <Pencil className="size-4" />
   404→                          </Button>
   405→                          <Button
   406→                            variant="ghost"
   407→                            size="sm"
   408→                            onClick={() => openDeleteConfirm(supplier)}
   409→                            title="Delete"
   410→                            className="text-destructive hover:text-destructive"
   411→                          >
   412→                            <Trash2 className="size-4" />
   413→                          </Button>
   414→                        </div>
   415→                      </TableCell>
   416→                    </TableRow>
   417→                  ))}
   418→                </TableBody>
   419→              </Table>
   420→            </div>
   421→            {/* Mobile card view */}
   422→            <div className="md:hidden space-y-3 p-3">
   423→              {suppliers.map((supplier) => (
   424→                <Card
   425→                  key={supplier.id}
   426→                  className="p-4 cursor-pointer hover:bg-muted/50 transition-colors"
   427→                  onClick={() => openDetail(supplier)}
   428→                >
   429→                  <div className="flex items-start justify-between mb-2">
   430→                    <div className="flex items-center gap-3">
   431→                      <div className="size-9 rounded-full bg-primary/10 flex items-center justify-center shrink-0">
   432→                        <Truck className="size-5 text-primary" />
   433→                      </div>
   434→                      <div className="min-w-0">
   435→                        <p className="font-semibold text-sm truncate">{supplier.name}</p>
   436→                        <p className="text-xs text-muted-foreground truncate">{supplier.email || supplier.phone || 'No contact'}</p>
   437→                      </div>
   438→                    </div>
   439→                    <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800 shrink-0 text-xs">
   440→                      {formatETB(0)}
   441→                    </Badge>
   442→                  </div>
   443→                  <div className="flex items-center justify-between mt-2">
   444→                    <span className="text-xs text-muted-foreground">{supplier.address ? supplier.address.substring(0, 30) : 'No address'}</span>
   445→                    <div className="flex gap-1" onClick={(e) => e.stopPropagation()}>
   446→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0" onClick={() => openEditDialog(supplier)}><Pencil className="size-3.5" /></Button>
   447→                      <Button variant="ghost" size="sm" className="h-9 w-9 p-0 text-destructive" onClick={() => openDeleteConfirm(supplier)}><Trash2 className="size-3.5" /></Button>
   448→                    </div>
   449→                  </div>
   450→                </Card>
   451→              ))}
   452→            </div>
   453→          </CardContent>
   454→        </Card>
   455→      )}
   456→
   457→      {/* Pagination */}
   458→      {totalPages > 1 && (
   459→        <div className="flex items-center justify-between">
   460→          <p className="text-xs sm:text-sm text-muted-foreground">
   461→            Page {page} of {totalPages}
   462→          </p>
   463→          <div className="flex items-center gap-2">
   464→            <Button
   465→              variant="outline"
   466→              size="sm"
   467→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   468→              disabled={page === 1}
   469→              className="h-9 min-w-9 sm:min-w-auto"
   470→            >
   471→              <ChevronLeft className="size-4" />
   472→              <span className="hidden sm:inline ml-1">Previous</span>
   473→            </Button>
   474→            <Button
   475→              variant="outline"
   476→              size="sm"
   477→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   478→              disabled={page === totalPages}
   479→              className="h-9 min-w-9 sm:min-w-auto"
   480→            >
   481→              <span className="hidden sm:inline mr-1">Next</span>
   482→              <ChevronRight className="size-4" />
   483→            </Button>
   484→          </div>
   485→        </div>
   486→      )}
   487→
   488→      {/* Add/Edit Supplier Dialog */}
   489→      <Dialog open={showAddEdit} onOpenChange={setShowAddEdit}>
   490→        <DialogContent className="sm:max-w-md">
   491→          <DialogHeader>
   492→            <DialogTitle>{isEditing ? 'Edit Supplier' : 'Add New Supplier'}</DialogTitle>
   493→            <DialogDescription>
   494→              {isEditing ? 'Update supplier information' : 'Enter details for the new supplier'}
   495→            </DialogDescription>
   496→          </DialogHeader>
   497→          <Form {...form}>
   498→            <form onSubmit={form.handleSubmit(handleSave)} className="space-y-4">
   499→              <FormInputField
   500→                name="name"
   501→                label="Name"
   502→                placeholder="Supplier name"
   503→                required
   504→              />
   505→              <FormInputField
   506→                name="email"
   507→                label="Email"
   508→                placeholder="supplier@email.com"
   509→                type="email"
   510→              />
   511→              <FormInputField
   512→                name="phone"
   513→                label="Phone"
   514→                placeholder="+251 9XX XXX XXXX"
   515→                type="tel"
   516→              />
   517→              <FormTextareaField
   518→                name="address"
   519→                label="Address"
   520→                placeholder="Supplier address..."
   521→                rows={3}
   522→              />
   523→              <DialogFooter>
   524→                <Button variant="outline" type="button" onClick={() => setShowAddEdit(false)}>
   525→                  Cancel
   526→                </Button>
   527→                <FormSubmitButton isLoading={formSubmitting} loadingText={isEditing ? 'Updating...' : 'Saving...'}>
   528→                  {isEditing ? 'Update Supplier' : 'Add Supplier'}
   529→                </FormSubmitButton>
   530→              </DialogFooter>
   531→            </form>
   532→          </Form>
   533→        </DialogContent>
   534→      </Dialog>
   535→
   536→      {/* Delete Confirmation */}
   537→      <AlertDialog open={showDeleteConfirm} onOpenChange={setShowDeleteConfirm}>
   538→        <AlertDialogContent>
   539→          <AlertDialogHeader>
   540→            <AlertDialogTitle>Delete Supplier</AlertDialogTitle>
   541→            <AlertDialogDescription>
   542→              Are you sure you want to delete <strong>{selectedSupplier?.name}</strong>?
   543→              This action cannot be undone. The supplier and all their data will be permanently deleted.
   544→              Any associated debt records will be preserved.
   545→            </AlertDialogDescription>
   546→          </AlertDialogHeader>
   547→          <AlertDialogFooter>
   548→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   549→            <AlertDialogAction
   550→              onClick={handleDelete}
   551→              disabled={deleting}
   552→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   553→            >
   554→              {deleting && <Loader2 className="size-4 animate-spin" />}
