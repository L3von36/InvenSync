    88→function getStockStatus(
    89→  quantity: number,
    90→  lowStockThreshold: number
    91→): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    92→  if (quantity <= 0) return { label: 'Out of Stock', variant: 'destructive' }
    93→  if (quantity <= lowStockThreshold) return { label: 'Low Stock', variant: 'secondary' }
    94→  return { label: 'In Stock', variant: 'default' }
    95→}
    96→
    97→/** Safely parse attribute options that may be double-encoded */
    98→function parseAttributeOptions(raw: string | null | undefined): string[] | null {
    99→  if (!raw) return null
   100→  try {
   101→    const parsed = JSON.parse(raw)
   102→    if (Array.isArray(parsed)) {
   103→      return parsed
   104→    } else if (typeof parsed === 'string') {
   105→      // Handle double-encoded: "Red,Blue,Green" instead of ["Red","Blue","Green"]
   106→      return parsed.split(',').map((s: string) => s.trim()).filter(Boolean)
   107→    }
   108→    return null
   109→  } catch {
   110→    // If parsing fails, try splitting by comma
   111→    return raw.split(',').map(s => s.trim()).filter(Boolean)
   112→  }
   113→}

   116→// Create/Edit Product Dialog

   118→function ProductDialog({
   119→  open,
   120→  onOpenChange,
   121→  product,
   122→  productTypes,
   123→  onSaved,
   124→}: {
   125→  open: boolean
   126→  onOpenChange: (open: boolean) => void
   127→  product: Product | null
   128→  productTypes: ProductType[]
   129→  onSaved: () => void
   130→}) {
   131→  const { currentOrg, currentShop } = useAuthStore()
   132→  const [saving, setSaving] = useState(false)
   133→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   134→  const [attributeValues, setAttributeValues] = useState<Record<string, string>>({})
   135→  const [attrErrors, setAttrErrors] = useState<Record<string, string>>({})
   136→  const isEditing = !!product
   137→
   138→  // Separate state to track selected product type ID reliably
   139→  const [selectedTypeId, setSelectedTypeId] = useState<string>('')
   140→
   141→  const form = useForm<ProductFormData>({
   142→    resolver: zodResolver(productSchema),
   143→    defaultValues: {
   144→      productTypeId: '',
   145→      name: '',
   146→      sku: '',
   147→      description: '',
   148→      costPrice: undefined as unknown as number,
   149→      sellingPrice: undefined as unknown as number,
   150→      lowStockThreshold: 10,
   151→      quantity: 0,
   152→    },
   153→  })
   154→
   155→  // Watch productTypeId for reactive UI
   156→  const watchedProductTypeId = form.watch('productTypeId')
   157→
   158→  // Use selectedTypeId as the source of truth (falls back to form value)
   159→  const activeTypeId = selectedTypeId || watchedProductTypeId
   160→  const selectedType = productTypes.find((pt) => pt.id === activeTypeId)
   161→  const typeAttributes = selectedType?.attributes || []
   162→
   163→  // Reset form when dialog opens
   164→  useEffect(() => {
   165→    if (open) {
   166→      setAttrErrors({})
   167→      if (product) {
   168→        const attrVals: Record<string, string> = {}
   169→        product.attributeValues?.forEach((av) => {
   170→          attrVals[av.attributeDefinitionId] = av.value
   171→        })
   172→        setAttributeValues(attrVals)
   173→        setSelectedTypeId(product.productTypeId)
   174→        form.reset({
   175→          productTypeId: product.productTypeId,
   176→          name: product.name,
   177→          sku: product.sku || '',
   178→          description: product.description || '',
   179→          costPrice: product.costPrice,
   180→          sellingPrice: product.sellingPrice,
   181→          lowStockThreshold: product.lowStockThreshold,
   182→          quantity: product.quantity,
   183→        })
   184→        setImagePreview(product.imageUrl || null)
   185→      } else {
   186→        setAttributeValues({})
   187→        setSelectedTypeId('')
   188→        form.reset({
   189→          productTypeId: '',
   190→          name: '',
   191→          sku: '',
   192→          description: '',
   193→          costPrice: undefined as unknown as number,
   194→          sellingPrice: undefined as unknown as number,
   195→          lowStockThreshold: 10,
   196→          quantity: 0,
   197→        })
   198→        setImagePreview(null)
   199→      }
   200→    } else {
   201→      // Clean up state when dialog closes
   202→      setSelectedTypeId('')
   203→      setAttributeValues({})
   204→      setAttrErrors({})
   205→      setImagePreview(null)
   206→    }
   207→  }, [open, product, form])
   208→
   209→  // Handle product type selection
   210→  const handleSelectType = (typeId: string) => {
   211→    setSelectedTypeId(typeId)
   212→    form.setValue('productTypeId', typeId, { shouldValidate: true, shouldDirty: true, shouldTouch: true })
   213→    // Clear attribute values when switching types
   214→    setAttributeValues({})
   215→    setAttrErrors({})
   216→  }
   217→
   218→  const updateAttributeValue = (attrId: string, value: string) => {
   219→    setAttributeValues((prev) => ({ ...prev, [attrId]: value }))
   220→    setAttrErrors((prev) => {
   221→      const next = { ...prev }
   222→      delete next[`attr_${attrId}`]
   223→      return next
   224→    })
   225→  }
   226→
   227→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   228→    const file = e.target.files?.[0]
   229→    if (!file) return
   230→    const reader = new FileReader()
   231→    reader.onloadend = () => {
   232→      const base64 = reader.result as string
   233→      setImagePreview(base64)
   234→      typeAttributes
   235→        .filter((a) => a.fieldType === 'image')
   236→        .forEach((a) => {
   237→          updateAttributeValue(a.id, base64)
   238→        })
   239→    }
   240→    reader.readAsDataURL(file)
   241→  }
   242→
   243→  const onSubmit = async (data: ProductFormData) => {
   244→    if (!currentOrg) return
   245→
   246→    // Use selectedTypeId as the authoritative source for productTypeId
   247→    const productTypeId = selectedTypeId || data.productTypeId || activeTypeId
   248→    if (!productTypeId) {
   249→      toast.error('Please select a product type')
   250→      return
   251→    }
   252→
   253→    // Safely convert form values to numbers (form may store them as strings from input onChange)
   254→    const costPrice = Number(data.costPrice)
   255→    const sellingPrice = Number(data.sellingPrice)
   256→    const lowStockThreshold = Number(data.lowStockThreshold) || 10
   257→    const quantity = Number(data.quantity) || 0
   258→
   259→    // Validate converted numbers
   260→    if (!costPrice || Number.isNaN(costPrice) || costPrice <= 0) {
   261→      toast.error('Cost price must be a valid number greater than 0')
   262→      return
   263→    }
   264→    if (!sellingPrice || Number.isNaN(sellingPrice) || sellingPrice <= 0) {
   265→      toast.error('Selling price must be a valid number greater than 0')
   266→      return
   267→    }
   268→
   269→    setSaving(true)
   270→    try {
   271→      const attrVals = Object.entries(attributeValues)
   272→        .filter(([, value]) => value.trim() !== '')
   273→        .map(([attributeDefinitionId, value]) => ({
   274→          attributeDefinitionId,
   275→          value,
   276→        }))
   277→
   278→      if (isEditing && product) {
   279→        await api.updateProduct(product.id, currentOrg.id, {
   280→          name: data.name.trim(),
   281→          sku: data.sku?.trim() || undefined,
   282→          description: data.description?.trim() || undefined,
   283→          imageUrl: imagePreview || product?.imageUrl || undefined,
   284→          costPrice,
   285→          sellingPrice,
   286→          lowStockThreshold,
   287→          attributeValues: attrVals,
   288→        })
   289→        toast.success('Product updated successfully')
   290→      } else {
   291→        await api.createProduct(currentOrg.id, {
   292→          productTypeId,
   293→          name: data.name.trim(),
   294→          sku: data.sku?.trim() || undefined,
   295→          description: data.description?.trim() || undefined,
   296→          imageUrl: imagePreview || product?.imageUrl || undefined,
   297→          quantity,
   298→          costPrice,
   299→          sellingPrice,
   300→          lowStockThreshold,
   301→          attributeValues: attrVals,
   302→          shopId: currentShop?.id,
   303→        })
   304→        toast.success('Product created successfully')
   305→      }
   306→
   307→      onOpenChange(false)
   308→      onSaved()
   309→    } catch (error) {
   310→      toast.error(getNetworkErrorMessage(error))
   311→    } finally {
   312→      setSaving(false)
   313→    }
   314→  }
   315→
   316→  const handleSaveClick = async () => {
   317→    // First validate attributes manually
   318→    const missingRequired = typeAttributes
   319→      .filter((a) => a.required)
   320→      .filter((a) => !attributeValues[a.id] || attributeValues[a.id].trim() === '')
   321→    if (missingRequired.length > 0) {
   322→      const errors: Record<string, string> = {}
   323→      missingRequired.forEach((a) => {
   324→        errors[`attr_${a.id}`] = `${a.name} is required`
   325→      })
   326→      setAttrErrors(errors)
   327→      return
   328→    }
   329→
   330→    // Ensure productTypeId is set in form before validation
   331→    if (selectedTypeId) {
   332→      form.setValue('productTypeId', selectedTypeId, { shouldValidate: true, shouldDirty: true })
   333→    }
   334→
   335→    // Validate all form fields including productTypeId
   336→    const isValid = await form.trigger(['productTypeId', 'name', 'costPrice', 'sellingPrice', 'lowStockThreshold', 'quantity'])
   337→    if (!isValid) return
   338→
   339→    // Final check: ensure we have a productTypeId
   340→    if (!selectedTypeId && !form.getValues('productTypeId')) {
   341→      toast.error('Please select a product type')
   342→      return
   343→    }
   344→
   345→    const data = form.getValues()
   346→    onSubmit(data)
   347→  }
   348→
   349→  return (
   350→    <Dialog open={open} onOpenChange={onOpenChange}>
   351→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto w-[calc(100%-2rem)] sm:w-full">
   352→        <DialogHeader>
   353→          <DialogTitle>{isEditing ? 'Edit Product' : 'Add Product'}</DialogTitle>
   354→          <DialogDescription>Fill in the product details below.</DialogDescription>
   355→        </DialogHeader>
   356→
   357→        <Form {...form}>
   358→          <div className="space-y-6">
   359→            {/* Section 1: Product Type Selection */}
   360→            <div className="space-y-2">
   361→              <Label className="text-sm font-medium">Product Type *</Label>
   362→              {productTypes.length === 0 ? (
   363→                <div className="text-center py-8 text-muted-foreground border border-dashed rounded-lg">
   364→                  <Package className="size-8 mx-auto mb-2 opacity-50" />
   365→                  <p className="text-sm">No product types available.</p>
   366→                  <p className="text-xs mt-1">Create a product type first.</p>
   367→                </div>
   368→              ) : (
   369→                <>
   370→                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   371→                    {productTypes.map((pt) => (
   372→                      <button
   373→                        key={pt.id}
   374→                        type="button"
   375→                        onClick={() => handleSelectType(pt.id)}
   376→                        className={`p-3 sm:p-4 rounded-lg border-2 text-center transition-all cursor-pointer ${
   377→                          activeTypeId === pt.id
   378→                            ? 'border-primary bg-primary/5 ring-2 ring-primary/20'
   379→                            : 'border-border hover:border-primary/50'
   380→                        }`}
   381→                      >
   382→                        <div className="text-2xl mb-1">{pt.icon || '📦'}</div>
   383→                        <div className="text-sm font-medium">{pt.name}</div>
   384→                        <div className="text-xs text-muted-foreground mt-0.5">
   385→                          {pt.attributes?.length || 0} attributes
   386→                        </div>
   387→                      </button>
   388→                    ))}
   389→                  </div>
   390→                  {form.formState.errors.productTypeId && (
   391→                    <p className="text-sm text-destructive">{form.formState.errors.productTypeId.message}</p>
   392→                  )}
   393→                </>
   394→              )}
   395→            </div>
   396→
   397→            {/* Section 2: Basic Information — shown only after a type is selected */}
   398→            {activeTypeId && (
   399→              <div className="space-y-2">
   400→                <Label className="text-sm font-medium">Basic Information</Label>
   401→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   402→                  <FormInputField
   403→                    name="name"
   404→                    label="Product Name"
   405→                    placeholder='e.g., Samsung 55" 4K Smart TV'
   406→                    required
   407→                    className="sm:col-span-2"
   408→                  />
   409→
   410→                  <FormInputField
   411→                    name="sku"
   412→                    label="SKU"
   413→                    placeholder="e.g., TV-SAM-55"
   414→                  />
   415→
   416→                  <FormTextareaField
   417→                    name="description"
   418→                    label="Description"
   419→                    placeholder="Product description..."
   420→                    rows={2}
   421→                  />
   422→
   423→                  <FormInputField
   424→                    name="costPrice"
   425→                    label="Cost Price (ETB)"
   426→                    type="number"
   427→                    step="0.01"
   428→                    min="0.01"
   429→                    placeholder="e.g. 5000"
   430→                    required
   431→                  />
   432→
   433→                  <FormInputField
   434→                    name="sellingPrice"
   435→                    label="Selling Price (ETB)"
   436→                    type="number"
   437→                    step="0.01"
   438→                    min="0.01"
   439→                    placeholder="e.g. 8000"
   440→                    required
   441→                  />
   442→
   443→                  <FormInputField
   444→                    name="lowStockThreshold"
   445→                    label="Low Stock Threshold"
   446→                    type="number"
   447→                    min="0"
   448→                  />
   449→
   450→                  {!isEditing && (
   451→                    <FormInputField
   452→                      name="quantity"
   453→                      label="Initial Quantity"
   454→                      type="number"
   455→                      min="0"
   456→                    />
   457→                  )}
   458→                </div>
   459→              </div>
   460→            )}
   461→
   462→            {/* Section 3: Custom Attributes — shown only when type has attributes */}
   463→            {activeTypeId && typeAttributes.length > 0 && (
   464→              <div className="space-y-2">
   465→                <Label className="text-sm font-medium">Custom Attributes</Label>
   466→                <div className="space-y-4">
   467→                  {typeAttributes.map((attr) => {
   468→                    const options = parseAttributeOptions(attr.options)
   469→                    return (
   470→                      <div key={attr.id} className="space-y-2">
   471→                        <Label className="flex items-center gap-1">
   472→                          {attr.name}
   473→                          {attr.required && <span className="text-destructive">*</span>}
   474→                        </Label>
   475→
   476→                        {(attr.fieldType === 'text' || attr.fieldType === 'textarea') && (
   477→                          <Input
   478→                            placeholder={`Enter ${attr.name.toLowerCase()}`}
   479→                            value={attributeValues[attr.id] || ''}
   480→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   481→                          />
   482→                        )}
   483→
   484→                        {attr.fieldType === 'number' && (
   485→                          <Input
   486→                            type="number"
   487→                            step="any"
   488→                            placeholder={`Enter ${attr.name.toLowerCase()}`}
   489→                            value={attributeValues[attr.id] || ''}
   490→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   491→                          />
   492→                        )}
   493→
   494→                        {(attr.fieldType === 'boolean' || attr.fieldType === 'switch') && (
   495→                          <div className="flex items-center gap-2">
   496→                            <Switch
   497→                              checked={attributeValues[attr.id] === 'true'}
   498→                              onCheckedChange={(checked) =>
   499→                                updateAttributeValue(attr.id, checked ? 'true' : 'false')
   500→                              }
   501→                            />
   502→                            <span className="text-sm text-muted-foreground">
   503→                              {attributeValues[attr.id] === 'true' ? 'Yes' : 'No'}
   504→                            </span>
   505→                          </div>
   506→                        )}
   507→
   508→                        {attr.fieldType === 'select' && options && (
   509→                          <Select
   510→                            value={attributeValues[attr.id] || ''}
   511→                            onValueChange={(v) => updateAttributeValue(attr.id, v)}
   512→                          >
   513→                            <SelectTrigger>
   514→                              <SelectValue placeholder={`Select ${attr.name.toLowerCase()}`} />
   515→                            </SelectTrigger>
   516→                            <SelectContent>
   517→                              {options.map((opt: string) => (
   518→                                <SelectItem key={opt} value={opt}>
   519→                                  {opt}
   520→                                </SelectItem>
   521→                              ))}
   522→                            </SelectContent>
   523→                          </Select>
   524→                        )}
   525→
   526→                        {attr.fieldType === 'image' && (
   527→                          <div className="space-y-2">
   528→                            <div className="flex items-center gap-2">
   529→                              <Button
   530→                                type="button"
   531→                                variant="outline"
   532→                                size="sm"
   533→                                className="gap-1"
   534→                                onClick={() =>
   535→                                  document.getElementById(`img-upload-${attr.id}`)?.click()
   536→                                }
   537→                              >
   538→                                <Upload className="size-3.5" />
   539→                                Upload Image
   540→                              </Button>
   541→                              {attributeValues[attr.id] && (
   542→                                <Button
   543→                                  type="button"
   544→                                  variant="ghost"
   545→                                  size="sm"
   546→                                  onClick={() => {
   547→                                    updateAttributeValue(attr.id, '')
   548→                                    setImagePreview(null)
   549→                                  }}
   550→                                >
   551→                                  <X className="size-3.5" />
   552→                                </Button>
   553→                              )}
   554→                            </div>
   555→                            <input
   556→                              id={`img-upload-${attr.id}`}
   557→                              type="file"
   558→                              accept="image/*"
   559→                              className="hidden"
   560→                              onChange={handleImageUpload}
   561→                            />
   562→                            {(attributeValues[attr.id] || imagePreview) && (
   563→                              <div className="size-20 rounded-lg border overflow-hidden">
   564→                                <img
   565→                                  src={attributeValues[attr.id] || imagePreview || ''}
   566→                                  alt="Preview"
   567→                                  className="size-full object-cover"
   568→                                  loading="lazy"
   569→                                />
   570→                              </div>
   571→                            )}
   572→                          </div>
   573→                        )}
   574→
   575→                        {attr.fieldType === 'date' && (
   576→                          <Input
   577→                            type="date"
   578→                            value={attributeValues[attr.id] || ''}
   579→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   580→                          />
   581→                        )}
   582→
   583→                        {attrErrors[`attr_${attr.id}`] && (
   584→                          <p className="text-sm text-destructive">{attrErrors[`attr_${attr.id}`]}</p>
   585→                        )}
   586→                      </div>
   587→                    )
   588→                  })}
   589→                </div>
   590→              </div>
   591→            )}
   592→          </div>
   593→        </Form>
   594→
   595→        <DialogFooter>
   596→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving} type="button">
   597→            Cancel
   598→          </Button>
   599→          <Button onClick={handleSaveClick} disabled={saving || !activeTypeId} type="button">
   600→            {saving && <Loader2 className="size-4 animate-spin" />}
   601→            {saving ? (isEditing ? 'Saving...' : 'Creating...') : (isEditing ? 'Save Changes' : 'Create Product')}
   602→          </Button>
   603→        </DialogFooter>
   604→      </DialogContent>
   605→    </Dialog>
   606→  )
   607→}

   612→function ProductDetailView({
   613→  product,
   614→  onClose,

   616→}: {
   617→  product: Product & {
   618→    stockMovements?: StockMovement[]
   619→    productType?: {

   621→      name: string
   622→      icon?: string | null
