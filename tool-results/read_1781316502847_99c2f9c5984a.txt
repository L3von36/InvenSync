   175→        return true

   555→// Mobile Bottom Navigation
   556→// ============================================
   557→export function MobileBottomNav() {
   558→  const { currentPage, setPage } = useAppStore()
   559→  const { user, currentOrg, currentOrgRole, activeModules } = useAuthStore()
   560→  const { theme, setTheme } = useTheme()
   561→  const [showMore, setShowMore] = useState(false)
   562→  const morePanelRef = useRef<HTMLDivElement>(null)
   563→  const moreButtonRef = useRef<HTMLButtonElement>(null)
   564→
   565→  const isService = currentOrg?.businessType === 'service'
   566→  const isMixed = currentOrg?.businessType === 'mixed'
   567→  const isManager = currentOrgRole === 'owner' || currentOrgRole === 'manager'
   568→  const isAdmin = user?.role === 'admin'
   569→
   570→  // Check if a module is accessible
   571→  const isModuleActive = (moduleKey?: string): boolean => {
   572→    if (!moduleKey) return true
   573→    if (activeModules.length === 0) return true // Graceful degradation
   574→    return activeModules.includes(moduleKey)
   575→  }
   576→
   577→  const mainTabs: Array<{ page: Page; icon: React.ElementType; label: string }> = isAdmin
   578→    ? [
   579→        { page: 'admin-dashboard', icon: Shield, label: 'Dashboard' },
   580→        { page: 'admin-regions', icon: Globe2, label: 'Regions' },
   581→        { page: 'admin-organizations', icon: Building2, label: 'Orgs' },
   582→      ]
   583→    : [
   584→        { page: 'dashboard', icon: LayoutDashboard, label: 'Home' },
   585→        ...(isService && !isMixed && isModuleActive('services')
   586→          ? [{ page: 'services' as Page, icon: Wrench, label: 'Services' }]
   587→          : isModuleActive('inventory')
   588→            ? [{ page: 'products' as Page, icon: Package, label: 'Products' }]
   589→            : []),
   590→        ...(isModuleActive('sales') ? [{ page: 'sales' as Page, icon: ShoppingCart, label: 'Sales' }] : []),
   591→      ]
   592→
   593→  const moreItems: Array<{ page: Page; icon: React.ElementType; label: string }> = isAdmin
   594→    ? [
   595→        { page: 'admin-users' as Page, icon: UserCog, label: 'Users' },
   596→        { page: 'admin-modules' as Page, icon: Cog, label: 'Modules' },
   597→        { page: 'admin-sales-team' as Page, icon: Megaphone, label: 'Sales Team' },
   598→        { page: 'admin-notifications' as Page, icon: Bell, label: 'Notify' },
   599→      ]
   600→    : [
   601→        ...(isManager ? [{ page: 'branches' as Page, icon: GitBranch, label: 'Branches' }] : []),
   602→        ...(isMixed && isModuleActive('services') ? [{ page: 'services' as Page, icon: Wrench, label: 'Services' }] : []),
   603→        ...(isManager && isModuleActive('inventory') ? [{ page: 'inventory' as Page, icon: Warehouse, label: 'Inventory' }] : []),
   604→        ...(isModuleActive('customers') ? [{ page: 'customers' as Page, icon: Users, label: 'Customers' }] : []),
   605→        ...(isManager && isModuleActive('debts') ? [{ page: 'debts' as Page, icon: CreditCard, label: 'Debts' }] : []),
   606→        ...(isManager && isModuleActive('reports') ? [{ page: 'reports' as Page, icon: BarChart3, label: 'Reports' }] : []),
   607→        ...(isModuleActive('ai-assistant') ? [{ page: 'ai-assistant' as Page, icon: Bot, label: 'AI Chat' }] : []),
   608→        { page: 'modules' as Page, icon: Puzzle, label: 'Modules' },
   609→        { page: 'settings' as Page, icon: Settings, label: 'Settings' },
   610→      ]
   611→
   612→  const isMoreActive = !['dashboard', 'products', 'sales', 'services', 'admin-dashboard', 'admin-regions', 'admin-organizations'].includes(currentPage)
   613→
   614→  // Focus trap for "More" panel (accessibility)
   615→  const handleMoreKeyDown = useCallback((e: React.KeyboardEvent) => {
   616→    if (e.key === 'Escape') {
   617→      setShowMore(false)
   618→      moreButtonRef.current?.focus()
   619→    }
   620→  }, [])
   621→
   622→  // Focus the more panel when it opens
   623→  useEffect(() => {
   624→    if (showMore && morePanelRef.current) {
   625→      const firstButton = morePanelRef.current.querySelector<HTMLElement>('button')
   626→      firstButton?.focus()
   627→    }
   628→  }, [showMore])
   629→
   630→  return (
   631→    <>
   632→      {/* More Menu Overlay — ARIA dialog for accessibility */}
   633→      {showMore && (
   634→        <div
   635→          className="fixed inset-0 z-40 md:hidden"
   636→          onClick={() => { setShowMore(false); moreButtonRef.current?.focus() }}
   637→          onKeyDown={handleMoreKeyDown}
   638→          role="dialog"
   639→          aria-modal="true"
   640→          aria-label="More navigation options"
   641→        >
   642→          <div className="absolute inset-0 bg-black/40" aria-hidden="true" />
   643→          <div
   644→            ref={morePanelRef}
   645→            className="absolute bottom-14 left-0 right-0 bg-background border-t rounded-t-2xl shadow-2xl p-4 max-h-[60vh] overflow-y-auto"
   646→            onClick={(e) => e.stopPropagation()}
   647→          >
   648→            <div className="flex items-center justify-between mb-3">
   649→              <h3 id="more-panel-title" className="font-semibold text-sm">More Options</h3>
   650→              <button
   651→                onClick={() => { setShowMore(false); moreButtonRef.current?.focus() }}
   652→                className="text-muted-foreground p-2 min-w-11 min-h-11 flex items-center justify-center"
   653→                aria-label="Close more options"
   654→              >
   655→                <X className="size-5" aria-hidden="true" />
   656→              </button>
   657→            </div>
   658→            <div className="grid grid-cols-4 gap-3" role="list">
   659→              {moreItems.map(item => (
   660→                <button
   661→                  key={item.page}
   662→                  onClick={() => { setPage(item.page); setShowMore(false) }}
   663→                  className={`flex flex-col items-center gap-1.5 p-3 rounded-xl transition-colors min-h-11 ${
   664→                    currentPage === item.page
   665→                      ? 'bg-primary/10 text-primary'
   666→                      : 'hover:bg-muted text-muted-foreground'
   667→                  }`}
   668→                  role="listitem"
   669→                  aria-label={item.label}
   670→                  aria-current={currentPage === item.page ? 'page' : undefined}
   671→                >
   672→                  <item.icon className="size-5" aria-hidden="true" />
   673→                  <span className="text-xs font-medium">{item.label}</span>
   674→                </button>
   675→              ))}
   676→              {/* Theme toggle */}
   677→              <button
   678→                onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   679→                className="flex flex-col items-center gap-1.5 p-3 rounded-xl transition-colors hover:bg-muted text-muted-foreground min-h-11"
   680→                aria-label={`Switch to ${theme === 'dark' ? 'light' : 'dark'} theme`}
   681→              >
   682→                {theme === 'dark' ? <Sun className="size-5" aria-hidden="true" /> : <Moon className="size-5" aria-hidden="true" />}
   683→                <span className="text-xs font-medium">{theme === 'dark' ? 'Light' : 'Dark'}</span>
   684→              </button>
   685→            </div>
   686→          </div>
   687→        </div>
   688→      )}
   689→
   690→      {/* Bottom Nav Bar */}
   691→      <nav className="fixed bottom-0 left-0 right-0 z-50 md:hidden border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80" aria-label="Mobile navigation">
   692→        <div className="flex items-center justify-around h-14 pb-[env(safe-area-inset-bottom)]">
   693→          {mainTabs.map(tab => (
   694→            <button
   695→              key={tab.page}
   696→              onClick={() => setPage(tab.page)}
   697→              className={`flex flex-col items-center justify-center gap-0.5 min-w-0 flex-1 h-full transition-colors min-h-11 ${
   698→                currentPage === tab.page
   699→                  ? 'text-primary'
   700→                  : 'text-muted-foreground'
   701→              }`}
   702→              aria-label={tab.label}
   703→              aria-current={currentPage === tab.page ? 'page' : undefined}
   704→            >
   705→              <tab.icon className="size-5" aria-hidden="true" />
   706→              <span className="text-xs font-medium">{tab.label}</span>
   707→            </button>
   708→          ))}
   709→          <button
   710→            ref={moreButtonRef}
   711→            onClick={() => setShowMore(true)}
   712→            className={`flex flex-col items-center justify-center gap-0.5 min-w-0 flex-1 h-full transition-colors min-h-11 ${
   713→              isMoreActive
   714→                ? 'text-primary'
   715→                : 'text-muted-foreground'
   716→            }`}
   717→            aria-label="More options"
   718→            aria-expanded={showMore}
   719→            aria-haspopup="dialog"
   720→          >
   721→            <BarChart3 className="size-5" aria-hidden="true" />
   722→            <span className="text-xs font-medium">More</span>
   723→          </button>
   724→        </div>
   725→      </nav>
   726→    </>
   727→  )
   728→}
   729→
