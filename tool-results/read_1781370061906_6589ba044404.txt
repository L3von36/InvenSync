    86→// Main Component
    87→// ============================================
    88→export function ReportsPage() {
    89→  const { currentOrg, currentShop } = useAuthStore()
    90→  const orgId = currentOrg?.id || ''
    91→  const isMobile = useIsMobile()
    92→
    93→  const [reportData, setReportData] = useState<ReportData | null>(null)
    94→  const [loading, setLoading] = useState(true)
    95→  const [pageError, setPageError] = useState<string | null>(null)
    96→  const [dateError, setDateError] = useState<string | null>(null)
    97→  const [activeTab, setActiveTab] = useState('sales')
    98→
    99→  // Date range
   100→  const [periodType, setPeriodType] = useState('monthly')
   101→  const [customStart, setCustomStart] = useState('')
   102→  const [customEnd, setCustomEnd] = useState('')
   103→
   104→  const getDateRange = useCallback(() => {
   105→    const now = new Date()
   106→    let startDate: string
   107→    let endDate: string = now.toISOString().split('T')[0]
   108→
   109→    switch (periodType) {
   110→      case 'today':
   111→        startDate = endDate
   112→        break
   113→      case 'weekly': {
   114→        const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   115→        startDate = weekAgo.toISOString().split('T')[0]
   116→        break
   117→      }
   118→      case 'monthly':
   119→        startDate = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   120→        break
   121→      case 'quarterly': {
   122→        const quarter = Math.floor(now.getMonth() / 3)
   123→        startDate = new Date(now.getFullYear(), quarter * 3, 1).toISOString().split('T')[0]
   124→        break
   125→      }
   126→      case 'custom':
   127→        startDate = customStart || new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   128→        endDate = customEnd || now.toISOString().split('T')[0]
   129→        break
   130→      default:
   131→        startDate = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split('T')[0]
   132→    }
   133→    return { startDate, endDate }
   134→  }, [periodType, customStart, customEnd])
   135→
   136→  const fetchReport = useCallback(async () => {
   137→    if (!orgId) return
   138→
   139→    // Date range validation for custom period
   140→    if (periodType === 'custom' && customStart && customEnd) {
   141→      if (new Date(customStart) > new Date(customEnd)) {
   142→        setDateError('Start date must be before end date')
   143→        return
   144→      }
   145→    }
   146→    setDateError(null)
   147→    setPageError(null)
   148→    setLoading(true)
   149→    try {
   150→      const { startDate, endDate } = getDateRange()
   151→      const data = await api.getReports(orgId, {
   152→        startDate,
   153→        endDate,
   154→        period: periodType === 'custom' ? 'daily' : periodType,
   155→        shopId: currentShop?.id,
   156→      })
   157→      setReportData(data as unknown as ReportData)
   158→    } catch (err) {
   159→      const msg = getNetworkErrorMessage(err)
   160→      setPageError(msg)
   161→    } finally {
   162→      setLoading(false)
   163→    }
   164→  }, [orgId, getDateRange, periodType, currentShop])
   165→
   166→  useEffect(() => {
   167→    fetchReport()
   168→  }, [fetchReport])
   169→
   170→  const summary = reportData?.summary
   171→  const salesByDate = reportData?.salesByDate || []
   172→  const bestSelling = reportData?.bestSellingProducts || []
   173→  const inventoryVal = reportData?.inventoryValuation
   174→
   175→  const handleExportCSV = useCallback(() => {
   176→    if (!reportData) return
   177→
   178→    if (activeTab === 'sales') {
   179→      exportToCSV(
   180→        salesByDate.map(row => ({
   181→          date: row.date,
   182→          invoice: row.count,
   183→          customer: '',
   184→          total: row.revenue,
   185→          status: '',
   186→          paymentMethod: '',
   187→        })),
   188→        'sales-report',
   189→        [
   190→          { key: 'date', label: 'Date' },
   191→          { key: 'invoice', label: 'Transactions' },
   192→          { key: 'customer', label: 'Customer' },
   193→          { key: 'total', label: 'Total' },
   194→          { key: 'status', label: 'Status' },
   195→          { key: 'paymentMethod', label: 'Payment Method' },
   196→        ]
   197→      )
   198→    } else if (activeTab === 'profit') {
   199→      exportToCSV(
   200→        salesByDate.map(row => ({
   201→          date: row.date,
   202→          revenue: row.revenue,
   203→          cost: row.cost,
   204→          profit: row.profit,
   205→          margin: row.revenue > 0 ? ((row.profit / row.revenue) * 100).toFixed(1) + '%' : '0%',
   206→        })),
   207→        'profit-report',
   208→        [
   209→          { key: 'date', label: 'Date' },
   210→          { key: 'revenue', label: 'Revenue' },
   211→          { key: 'cost', label: 'Cost' },
   212→          { key: 'profit', label: 'Profit' },
   213→          { key: 'margin', label: 'Margin' },
   214→        ]
   215→      )
   216→    } else if (activeTab === 'inventory') {
   217→      const val = reportData.inventoryValuation
   218→      exportToCSV(
   219→        [{
   220→          productName: 'All Inventory',
   221→          type: 'Summary',
   222→          quantity: val?.totalItems || 0,
   223→          costPrice: val?.totalCostValue || 0,
   224→          sellingPrice: val?.totalRetailValue || 0,
   225→          value: val?.totalRetailValue || 0,
   226→        }],
   227→        'inventory-report',
   228→        [
   229→          { key: 'productName', label: 'Product Name' },
   230→          { key: 'type', label: 'Type' },
   231→          { key: 'quantity', label: 'Quantity' },
   232→          { key: 'costPrice', label: 'Cost Price' },
   233→          { key: 'sellingPrice', label: 'Selling Price' },
   234→          { key: 'value', label: 'Value' },
   235→        ]
   236→      )
   237→    } else if (activeTab === 'bestsellers') {
   238→      exportToCSV(
   239→        bestSelling.map(p => ({
   240→          productName: p.name,
   241→          sku: p.sku || '',
   242→          quantitySold: p.totalQuantity,
   243→          revenue: p.totalRevenue,
   244→          avgPrice: p.totalQuantity > 0 ? p.totalRevenue / p.totalQuantity : 0,
   245→        })),
   246→        'bestsellers-report',
   247→        [
   248→          { key: 'productName', label: 'Product Name' },
   249→          { key: 'sku', label: 'SKU' },
   250→          { key: 'quantitySold', label: 'Quantity Sold' },
   251→          { key: 'revenue', label: 'Revenue' },
   252→          { key: 'avgPrice', label: 'Avg Price' },
   253→        ]
   254→      )
   255→    }
   256→  }, [activeTab, reportData, salesByDate, bestSelling])
   257→
   258→  // Payment method distribution for pie chart (mock from sales data)
   259→  const paymentMethodData = [
   260→    { name: 'Cash', value: 45 },
   261→    { name: 'Card', value: 30 },
   262→    { name: 'Mobile Money', value: 25 },
   263→  ]
   264→
   265→  return (
   266→    <div className="space-y-6">
   267→      {/* Header */}
   268→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   269→        <div>
   270→          <h1 className="text-xl sm:text-2xl font-bold tracking-tight">Reports</h1>
   271→          <p className="text-muted-foreground text-sm">Analyze your sales, profit, and inventory performance</p>
   272→        </div>
   273→        <Button
   274→          variant="outline"
   275→          size="sm"
   276→          onClick={handleExportCSV}
   277→          disabled={!reportData || salesByDate.length === 0}
   278→          className="gap-2"
   279→        >
   280→          <Download className="size-4" />
   281→          Export CSV
   282→        </Button>
   283→      </div>
   284→
   285→      {/* Date Range Selector */}
   286→      <Card>
   287→        <CardContent className="p-4">
   288→          <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   289→            <div className="flex items-center gap-2">
   290→              <Calendar className="size-4 text-muted-foreground" />
   291→              <span className="text-sm font-medium">Period:</span>
   292→            </div>
   293→            <div className="flex flex-wrap items-center gap-2">
   294→              {[
   295→                { value: 'today', label: 'Today' },
   296→                { value: 'weekly', label: 'This Week' },
   297→                { value: 'monthly', label: 'This Month' },
   298→                { value: 'quarterly', label: 'This Quarter' },
   299→                { value: 'custom', label: 'Custom' },
   300→              ].map((p) => (
   301→                <Button
   302→                  key={p.value}
   303→                  variant={periodType === p.value ? 'default' : 'outline'}
   304→                  size="sm"
   305→                  onClick={() => setPeriodType(p.value)}
   306→                >
   307→                  {p.label}
   308→                </Button>
   309→              ))}
   310→            </div>
   311→            {periodType === 'custom' && (
   312→              <div className="flex items-center gap-4">
   313→                <div className="flex items-center gap-2">
   314→                  <Label className="text-xs whitespace-nowrap">From</Label>
   315→                  <Input
   316→                    type="date"
   317→                    value={customStart}
   318→                    onChange={(e) => { setCustomStart(e.target.value); setDateError(null) }}
   319→                    className="h-8 text-xs w-36"
   320→                  />
   321→                </div>
   322→                <div className="flex items-center gap-2">
   323→                  <Label className="text-xs whitespace-nowrap">To</Label>
   324→                  <Input
   325→                    type="date"
   326→                    value={customEnd}
   327→                    onChange={(e) => { setCustomEnd(e.target.value); setDateError(null) }}
   328→                    className="h-8 text-xs w-36"
   329→                  />
   330→                </div>
   331→              </div>
   332→            )}
   333→            {dateError && (
   334→              <p className="w-full text-xs text-destructive">{dateError}</p>
   335→            )}
   336→          </div>
   337→        </CardContent>
   338→      </Card>
   339→
   340→      {/* Tabs */}
   341→      <Tabs value={activeTab} onValueChange={setActiveTab} className="space-y-4">
   342→        <TabsList className="flex-wrap grid w-full grid-cols-2 sm:grid-cols-4">
   343→          <TabsTrigger value="sales" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   344→            <ShoppingCart className="size-3.5 sm:size-4" />
   345→            <span className="hidden sm:inline">Sales</span>
   346→          </TabsTrigger>
   347→          <TabsTrigger value="profit" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   348→            <TrendingUp className="size-3.5 sm:size-4" />
   349→            <span className="hidden sm:inline">Profit</span>
   350→          </TabsTrigger>
   351→          <TabsTrigger value="bestsellers" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   352→            <BarChart3 className="size-3.5 sm:size-4" />
   353→            <span className="hidden sm:inline">Best</span>
   354→          </TabsTrigger>
   355→          <TabsTrigger value="inventory" className="gap-1 text-xs sm:text-sm px-1 sm:px-3">
   356→            <Package className="size-3.5 sm:size-4" />
   357→            <span className="hidden sm:inline">Inventory</span>
   358→          </TabsTrigger>
   359→        </TabsList>
   360→
   361→        {/* Sales Report */}
   362→        <TabsContent value="sales" className="space-y-4">
   363→          {loading ? (
   364→            <LoadingSkeleton cards={3} />
   365→          ) : pageError && !reportData ? (
   366→            <ErrorState title="Failed to load report data" message={pageError} onRetry={fetchReport} />
   367→          ) : !loading && reportData && salesByDate.length === 0 ? (
   368→            <EmptyState
   369→              title="No sales data for this period"
   370→              message="Try selecting a different date range, or make your first sale to see reports here."
   371→              icon={<BarChart3 className="size-7 text-muted-foreground" />}
   372→            />
   373→          ) : (
   374→            <>
   375→              {/* Summary Cards */}
   376→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 md:gap-4">
   377→                <Card>
   378→                  <CardContent className="p-4">
   379→                    <div className="flex items-center gap-3">
   380→                      <div className="size-10 rounded-lg bg-brand-50 dark:bg-brand-900/20 flex items-center justify-center">
   381→                        <DollarSign className="size-5 text-primary" />
   382→                      </div>
   383→                      <div>
   384→                        <p className="text-sm text-muted-foreground">Total Revenue</p>
   385→                        <p className="text-xl sm:text-2xl font-bold">{formatETB(summary?.totalRevenue || 0)}</p>
   386→                      </div>
   387→                    </div>
   388→                  </CardContent>
   389→                </Card>
   390→                <Card>
   391→                  <CardContent className="p-4">
   392→                    <div className="flex items-center gap-3">
   393→                      <div className="size-10 rounded-lg bg-blue-100 dark:bg-blue-900/30 flex items-center justify-center">
   394→                        <ShoppingCart className="size-5 text-blue-600 dark:text-blue-400" />
   395→                      </div>
   396→                      <div>
   397→                        <p className="text-sm text-muted-foreground">Total Transactions</p>
   398→                        <p className="text-xl sm:text-2xl font-bold">{summary?.totalSales || 0}</p>
   399→                      </div>
   400→                    </div>
   401→                  </CardContent>
   402→                </Card>
   403→                <Card>
   404→                  <CardContent className="p-4">
   405→                    <div className="flex items-center gap-3">
   406→                      <div className="size-10 rounded-lg bg-purple-100 dark:bg-purple-900/30 flex items-center justify-center">
   407→                        <TrendingUp className="size-5 text-purple-600 dark:text-purple-400" />
   408→                      </div>
   409→                      <div>
   410→                        <p className="text-sm text-muted-foreground">Avg Transaction</p>
   411→                        <p className="text-xl sm:text-2xl font-bold">{formatETB(summary?.averageSaleValue || 0)}</p>
   412→                      </div>
   413→                    </div>
   414→                  </CardContent>
   415→                </Card>
   416→              </div>
   417→
   418→              {/* Charts Row */}
   419→              <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 min-w-0">
   420→                {/* Daily Sales Bar Chart */}
   421→                <Card className="lg:col-span-2 overflow-hidden">
   422→                  <CardHeader className="pb-2">
   423→                    <CardTitle className="text-sm font-semibold">Daily Sales</CardTitle>
   424→                  </CardHeader>
   425→                  <CardContent>
   426→                    {salesByDate.length > 0 ? (
   427→                      <ResponsiveContainer width="100%" height={300}>
   428→                        <BarChart data={salesByDate}>
   429→                          <CartesianGrid strokeDasharray="3 3" className="opacity-30" />
   430→                          <XAxis
   431→                            dataKey="date"
   432→                            tick={{ fontSize: 11 }}
   433→                            tickFormatter={(val: string) => {
   434→                              const d = new Date(val)
   435→                              return `${d.getMonth() + 1}/${d.getDate()}`
   436→                            }}
   437→                          />
   438→                          <YAxis tick={{ fontSize: 11 }} tickFormatter={(val: number) => formatShortETB(val)} />
   439→                          <Tooltip
   440→                            formatter={(value: number) => formatETB(value)}
   441→                            labelFormatter={(label: string) => {
   442→                              const d = new Date(label)
   443→                              return d.toLocaleDateString('en-US', {
   444→                                weekday: 'short', year: 'numeric', month: 'short', day: 'numeric'
   445→                              })
   446→                            }}
   447→                          />
   448→                          <Legend />
   449→                          <Bar dataKey="revenue" name="Revenue" fill="oklch(0.637 0.237 25.331)" radius={[4, 4, 0, 0]} />
   450→                          <Bar dataKey="profit" name="Profit" fill="oklch(0.6 0.118 184.704)" radius={[4, 4, 0, 0]} />
   451→                        </BarChart>
   452→                      </ResponsiveContainer>
   453→                    ) : (
   454→                      <EmptyChart message="No sales data for this period" />
   455→                    )}
   456→                  </CardContent>
   457→                </Card>
   458→
   459→                {/* Payment Method Pie */}
   460→                <Card className="overflow-hidden">
   461→                  <CardHeader className="pb-2">
   462→                    <CardTitle className="text-sm font-semibold">By Payment Method</CardTitle>
   463→                  </CardHeader>
   464→                  <CardContent>
   465→                    <ResponsiveContainer width="100%" height={300}>
   466→                      <RechartsPie>
   467→                        <Pie
   468→                          data={paymentMethodData}
   469→                          cx="50%"
   470→                          cy="50%"
   471→                          innerRadius={60}
   472→                          outerRadius={90}
   473→                          paddingAngle={4}
   474→                          dataKey="value"
   475→                          label={({ name, percent }: { name: string; percent: number }) =>
   476→                            `${name} ${(percent * 100).toFixed(0)}%`
   477→                          }
   478→                        >
   479→                          {paymentMethodData.map((_, index) => (
   480→                            <Cell key={`cell-${index}`} fill={CHART_COLORS[index % CHART_COLORS.length]} />
   481→                          ))}
   482→                        </Pie>
   483→                        <Tooltip />
   484→                      </RechartsPie>
   485→                    </ResponsiveContainer>
   486→                  </CardContent>
   487→                </Card>
   488→              </div>
   489→
   490→              {/* Sales Table */}
   491→              <Card>
   492→                <CardHeader className="pb-2">
   493→                  <CardTitle className="text-sm font-semibold">Daily Sales Breakdown</CardTitle>
   494→                </CardHeader>
   495→                <CardContent className="p-0">
   496→                  {/* Desktop table */}
   497→                  <div className="hidden md:block overflow-x-auto max-h-96 overflow-y-auto custom-scrollbar">
   498→                    <Table>
   499→                      <TableHeader>
   500→                        <TableRow>
   501→                          <TableHead>Date</TableHead>
   502→                          <TableHead className="text-right">Revenue</TableHead>
   503→                          <TableHead className="text-right">Cost</TableHead>
   504→                          <TableHead className="text-right">Profit</TableHead>
   505→                          <TableHead className="text-right">Transactions</TableHead>
   506→                        </TableRow>

   522→                                </span>
   523→                              </TableCell>
   524→                              <TableCell className="text-right">{row.count}</TableCell>

   526→                          ))
   527→                        ) : (
   528→                          <TableRow>
   529→                            <TableCell colSpan={5} className="text-center py-8 text-muted-foreground">
   530→                              No sales data for this period
   531→                            </TableCell>
   532→                          </TableRow>
   533→                        )}
   534→                      </TableBody>
   535→                    </Table>
   536→                  </div>
   537→                  {/* Mobile card view */}
   538→                  <div className="md:hidden space-y-3 p-3 max-h-96 overflow-y-auto">
   539→                    {salesByDate.length > 0 ? (
   540→                      salesByDate.map((row) => (
