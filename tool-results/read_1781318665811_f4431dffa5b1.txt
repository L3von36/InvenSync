    86→function getStockStatus(
    87→  quantity: number,
    88→  lowStockThreshold: number
    89→): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    90→  if (quantity <= 0) return { label: 'Out of Stock', variant: 'destructive' }
    91→  if (quantity <= lowStockThreshold) return { label: 'Low Stock', variant: 'secondary' }
    92→  return { label: 'In Stock', variant: 'default' }
    93→}
    94→
    95→function formatCurrency(value: number): string {
    96→  return new Intl.NumberFormat('en-ET', {
    97→    style: 'currency',
    98→    currency: 'ETB',
    99→    maximumFractionDigits: 0,
   100→  }).format(value)
   101→}
   102→
   103→function formatDate(dateStr: string): string {
   104→  return new Date(dateStr).toLocaleDateString('en-US', {
   105→    year: 'numeric',
   106→    month: 'short',
   107→    day: 'numeric',
   108→  })
   109→}
   110→
   111→/** Safely parse attribute options that may be double-encoded */
   112→function parseAttributeOptions(raw: string | null | undefined): string[] | null {
   113→  if (!raw) return null
   114→  try {
   115→    const parsed = JSON.parse(raw)
   116→    if (Array.isArray(parsed)) {
   117→      return parsed
   118→    } else if (typeof parsed === 'string') {
   119→      // Handle double-encoded: "Red,Blue,Green" instead of ["Red","Blue","Green"]
   120→      return parsed.split(',').map((s: string) => s.trim()).filter(Boolean)
   121→    }
   122→    return null
   123→  } catch {
   124→    // If parsing fails, try splitting by comma
   125→    return raw.split(',').map(s => s.trim()).filter(Boolean)
   126→  }
   127→}

   132→function ProductDialog({
   133→  open,
   134→  onOpenChange,
   135→  product,
   136→  productTypes,
   137→  onSaved,
   138→}: {
   139→  open: boolean
   140→  onOpenChange: (open: boolean) => void
   141→  product: Product | null
   142→  productTypes: ProductType[]
   143→  onSaved: () => void
   144→}) {
   145→  const { currentOrg, currentShop } = useAuthStore()
   146→  const [saving, setSaving] = useState(false)
   147→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   148→  const [attributeValues, setAttributeValues] = useState<Record<string, string>>({})
   149→  const [attrErrors, setAttrErrors] = useState<Record<string, string>>({})
   150→  const isEditing = !!product
   151→
   152→  // Separate state to track selected product type ID reliably
   153→  const [selectedTypeId, setSelectedTypeId] = useState<string>('')
   154→
   155→  const form = useForm<ProductFormData>({
   156→    resolver: zodResolver(productSchema),
   157→    defaultValues: {
   158→      productTypeId: '',
   159→      name: '',
   160→      sku: '',
   161→      description: '',
   162→      costPrice: undefined as unknown as number,
   163→      sellingPrice: undefined as unknown as number,
   164→      lowStockThreshold: 10,
   165→      quantity: 0,
   166→    },
   167→  })
   168→
   169→  // Watch productTypeId for reactive UI
   170→  const watchedProductTypeId = form.watch('productTypeId')
   171→
   172→  // Use selectedTypeId as the source of truth (falls back to form value)
   173→  const activeTypeId = selectedTypeId || watchedProductTypeId
   174→  const selectedType = productTypes.find((pt) => pt.id === activeTypeId)
   175→  const typeAttributes = selectedType?.attributes || []
   176→
   177→  // Reset form when dialog opens
   178→  useEffect(() => {
   179→    if (open) {
   180→      setAttrErrors({})
   181→      if (product) {
   182→        const attrVals: Record<string, string> = {}
   183→        product.attributeValues?.forEach((av) => {
   184→          attrVals[av.attributeDefinitionId] = av.value
   185→        })
   186→        setAttributeValues(attrVals)
   187→        setSelectedTypeId(product.productTypeId)
   188→        form.reset({
   189→          productTypeId: product.productTypeId,
   190→          name: product.name,
   191→          sku: product.sku || '',
   192→          description: product.description || '',
   193→          costPrice: product.costPrice,
   194→          sellingPrice: product.sellingPrice,
   195→          lowStockThreshold: product.lowStockThreshold,
   196→          quantity: product.quantity,
   197→        })
   198→        setImagePreview(product.imageUrl || null)
   199→      } else {
   200→        setAttributeValues({})
   201→        setSelectedTypeId('')
   202→        form.reset({
   203→          productTypeId: '',
   204→          name: '',
   205→          sku: '',
   206→          description: '',
   207→          costPrice: undefined as unknown as number,
   208→          sellingPrice: undefined as unknown as number,
   209→          lowStockThreshold: 10,
   210→          quantity: 0,
   211→        })
   212→        setImagePreview(null)
   213→      }
   214→    } else {
   215→      // Clean up state when dialog closes
   216→      setSelectedTypeId('')
   217→      setAttributeValues({})
   218→      setAttrErrors({})
   219→      setImagePreview(null)
   220→    }
   221→  }, [open, product, form])
   222→
   223→  // Handle product type selection
   224→  const handleSelectType = (typeId: string) => {
   225→    setSelectedTypeId(typeId)
   226→    form.setValue('productTypeId', typeId, { shouldValidate: true, shouldDirty: true, shouldTouch: true })
   227→    // Clear attribute values when switching types
   228→    setAttributeValues({})
   229→    setAttrErrors({})
   230→  }
   231→
   232→  const updateAttributeValue = (attrId: string, value: string) => {
   233→    setAttributeValues((prev) => ({ ...prev, [attrId]: value }))
   234→    setAttrErrors((prev) => {
   235→      const next = { ...prev }
   236→      delete next[`attr_${attrId}`]
   237→      return next
   238→    })
   239→  }
   240→
   241→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   242→    const file = e.target.files?.[0]
   243→    if (!file) return
   244→    const reader = new FileReader()
   245→    reader.onloadend = () => {
   246→      const base64 = reader.result as string
   247→      setImagePreview(base64)
   248→      typeAttributes
   249→        .filter((a) => a.fieldType === 'image')
   250→        .forEach((a) => {
   251→          updateAttributeValue(a.id, base64)
   252→        })
   253→    }
   254→    reader.readAsDataURL(file)
   255→  }
   256→
   257→  const onSubmit = async (data: ProductFormData) => {
   258→    if (!currentOrg) return
   259→
   260→    // Use selectedTypeId as the authoritative source for productTypeId
   261→    const productTypeId = selectedTypeId || data.productTypeId || activeTypeId
   262→    if (!productTypeId) {
   263→      toast.error('Please select a product type')
   264→      return
   265→    }
   266→
   267→    // Safely convert form values to numbers (form may store them as strings from input onChange)
   268→    const costPrice = Number(data.costPrice)
   269→    const sellingPrice = Number(data.sellingPrice)
   270→    const lowStockThreshold = Number(data.lowStockThreshold) || 10
   271→    const quantity = Number(data.quantity) || 0
   272→
   273→    // Validate converted numbers
   274→    if (!costPrice || Number.isNaN(costPrice) || costPrice <= 0) {
   275→      toast.error('Cost price must be a valid number greater than 0')
   276→      return
   277→    }
   278→    if (!sellingPrice || Number.isNaN(sellingPrice) || sellingPrice <= 0) {
   279→      toast.error('Selling price must be a valid number greater than 0')
   280→      return
   281→    }
   282→
   283→    setSaving(true)
   284→    try {
   285→      const attrVals = Object.entries(attributeValues)
   286→        .filter(([, value]) => value.trim() !== '')
   287→        .map(([attributeDefinitionId, value]) => ({
   288→          attributeDefinitionId,
   289→          value,
   290→        }))
   291→
   292→      if (isEditing && product) {
   293→        await api.updateProduct(product.id, currentOrg.id, {
   294→          name: data.name.trim(),
   295→          sku: data.sku?.trim() || undefined,
   296→          description: data.description?.trim() || undefined,
   297→          imageUrl: imagePreview || product?.imageUrl || undefined,
   298→          costPrice,
   299→          sellingPrice,
   300→          lowStockThreshold,
   301→          attributeValues: attrVals,
   302→        })
   303→        toast.success('Product updated successfully')
   304→      } else {
   305→        await api.createProduct(currentOrg.id, {
   306→          productTypeId,
   307→          name: data.name.trim(),
   308→          sku: data.sku?.trim() || undefined,
   309→          description: data.description?.trim() || undefined,
   310→          imageUrl: imagePreview || product?.imageUrl || undefined,
   311→          quantity,
   312→          costPrice,
   313→          sellingPrice,
   314→          lowStockThreshold,
   315→          attributeValues: attrVals,
   316→          shopId: currentShop?.id,
   317→        })
   318→        toast.success('Product created successfully')
   319→      }
   320→
   321→      onOpenChange(false)
   322→      onSaved()
   323→    } catch (error) {
   324→      toast.error(getNetworkErrorMessage(error))
   325→    } finally {
   326→      setSaving(false)
   327→    }
   328→  }
   329→
   330→  const handleSaveClick = async () => {
   331→    // First validate attributes manually
   332→    const missingRequired = typeAttributes
   333→      .filter((a) => a.required)
   334→      .filter((a) => !attributeValues[a.id] || attributeValues[a.id].trim() === '')
   335→    if (missingRequired.length > 0) {
   336→      const errors: Record<string, string> = {}
   337→      missingRequired.forEach((a) => {
   338→        errors[`attr_${a.id}`] = `${a.name} is required`
   339→      })
   340→      setAttrErrors(errors)
   341→      return
   342→    }
   343→
   344→    // Ensure productTypeId is set in form before validation
   345→    if (selectedTypeId) {
   346→      form.setValue('productTypeId', selectedTypeId, { shouldValidate: true, shouldDirty: true })
   347→    }
   348→
   349→    // Validate all form fields including productTypeId
   350→    const isValid = await form.trigger(['productTypeId', 'name', 'costPrice', 'sellingPrice', 'lowStockThreshold', 'quantity'])
   351→    if (!isValid) return
   352→
   353→    // Final check: ensure we have a productTypeId
   354→    if (!selectedTypeId && !form.getValues('productTypeId')) {
   355→      toast.error('Please select a product type')
   356→      return
   357→    }
   358→
   359→    const data = form.getValues()
   360→    onSubmit(data)
   361→  }
   362→
   363→  return (
   364→    <Dialog open={open} onOpenChange={onOpenChange}>
   365→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto w-[calc(100%-2rem)] sm:w-full">
   366→        <DialogHeader>
   367→          <DialogTitle>{isEditing ? 'Edit Product' : 'Add Product'}</DialogTitle>
   368→          <DialogDescription>Fill in the product details below.</DialogDescription>
   369→        </DialogHeader>
   370→
   371→        <Form {...form}>
   372→          <div className="space-y-6">
   373→            {/* Section 1: Product Type Selection */}
   374→            <div className="space-y-2">
   375→              <Label className="text-sm font-medium">Product Type *</Label>
   376→              {productTypes.length === 0 ? (
   377→                <div className="text-center py-8 text-muted-foreground border border-dashed rounded-lg">
   378→                  <Package className="size-8 mx-auto mb-2 opacity-50" />
   379→                  <p className="text-sm">No product types available.</p>
   380→                  <p className="text-xs mt-1">Create a product type first.</p>
   381→                </div>
   382→              ) : (
   383→                <>
   384→                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   385→                    {productTypes.map((pt) => (
   386→                      <button
   387→                        key={pt.id}
   388→                        type="button"
   389→                        onClick={() => handleSelectType(pt.id)}
   390→                        className={`p-3 sm:p-4 rounded-lg border-2 text-center transition-all cursor-pointer ${
   391→                          activeTypeId === pt.id
   392→                            ? 'border-primary bg-primary/5 ring-2 ring-primary/20'
   393→                            : 'border-border hover:border-primary/50'
   394→                        }`}
   395→                      >
   396→                        <div className="text-2xl mb-1">{pt.icon || '📦'}</div>
   397→                        <div className="text-sm font-medium">{pt.name}</div>
   398→                        <div className="text-xs text-muted-foreground mt-0.5">
   399→                          {pt.attributes?.length || 0} attributes
   400→                        </div>
   401→                      </button>
   402→                    ))}
   403→                  </div>
   404→                  {form.formState.errors.productTypeId && (
   405→                    <p className="text-sm text-destructive">{form.formState.errors.productTypeId.message}</p>
   406→                  )}
   407→                </>
   408→              )}
   409→            </div>
   410→
   411→            {/* Section 2: Basic Information — shown only after a type is selected */}
   412→            {activeTypeId && (
   413→              <div className="space-y-2">
   414→                <Label className="text-sm font-medium">Basic Information</Label>
   415→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   416→                  <FormInputField
   417→                    name="name"
   418→                    label="Product Name"
   419→                    placeholder='e.g., Samsung 55" 4K Smart TV'
   420→                    required
   421→                    className="sm:col-span-2"
   422→                  />
   423→
   424→                  <FormInputField
   425→                    name="sku"
   426→                    label="SKU"
   427→                    placeholder="e.g., TV-SAM-55"
   428→                  />
   429→
   430→                  <FormTextareaField
   431→                    name="description"
   432→                    label="Description"
   433→                    placeholder="Product description..."
   434→                    rows={2}
   435→                  />
   436→
   437→                  <FormInputField
   438→                    name="costPrice"
   439→                    label="Cost Price (ETB)"
   440→                    type="number"
   441→                    step="0.01"
   442→                    min="0.01"
   443→                    placeholder="e.g. 5000"
   444→                    required
   445→                  />
   446→
   447→                  <FormInputField
   448→                    name="sellingPrice"
   449→                    label="Selling Price (ETB)"
   450→                    type="number"
   451→                    step="0.01"
   452→                    min="0.01"
   453→                    placeholder="e.g. 8000"
   454→                    required
   455→                  />
   456→
   457→                  <FormInputField
   458→                    name="lowStockThreshold"
   459→                    label="Low Stock Threshold"
   460→                    type="number"
   461→                    min="0"
   462→                  />
   463→
   464→                  {!isEditing && (
   465→                    <FormInputField
   466→                      name="quantity"
   467→                      label="Initial Quantity"
   468→                      type="number"
   469→                      min="0"
   470→                    />
   471→                  )}
   472→                </div>
   473→              </div>
   474→            )}
   475→
   476→            {/* Section 3: Custom Attributes — shown only when type has attributes */}
   477→            {activeTypeId && typeAttributes.length > 0 && (
   478→              <div className="space-y-2">
   479→                <Label className="text-sm font-medium">Custom Attributes</Label>
   480→                <div className="space-y-4">
   481→                  {typeAttributes.map((attr) => {
   482→                    const options = parseAttributeOptions(attr.options)
   483→                    return (
   484→                      <div key={attr.id} className="space-y-2">
   485→                        <Label className="flex items-center gap-1">
   486→                          {attr.name}
   487→                          {attr.required && <span className="text-destructive">*</span>}
   488→                        </Label>
   489→
   490→                        {(attr.fieldType === 'text' || attr.fieldType === 'textarea') && (
   491→                          <Input
   492→                            placeholder={`Enter ${attr.name.toLowerCase()}`}
   493→                            value={attributeValues[attr.id] || ''}
   494→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   495→                          />
   496→                        )}
   497→
   498→                        {attr.fieldType === 'number' && (
   499→                          <Input
   500→                            type="number"
   501→                            step="any"
   502→                            placeholder={`Enter ${attr.name.toLowerCase()}`}
   503→                            value={attributeValues[attr.id] || ''}
   504→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   505→                          />
   506→                        )}
   507→
   508→                        {(attr.fieldType === 'boolean' || attr.fieldType === 'switch') && (
   509→                          <div className="flex items-center gap-2">
   510→                            <Switch
   511→                              checked={attributeValues[attr.id] === 'true'}
   512→                              onCheckedChange={(checked) =>
   513→                                updateAttributeValue(attr.id, checked ? 'true' : 'false')
   514→                              }
   515→                            />
   516→                            <span className="text-sm text-muted-foreground">
   517→                              {attributeValues[attr.id] === 'true' ? 'Yes' : 'No'}
   518→                            </span>
   519→                          </div>
   520→                        )}
   521→
   522→                        {attr.fieldType === 'select' && options && (
   523→                          <Select
   524→                            value={attributeValues[attr.id] || ''}
   525→                            onValueChange={(v) => updateAttributeValue(attr.id, v)}
   526→                          >
   527→                            <SelectTrigger>
   528→                              <SelectValue placeholder={`Select ${attr.name.toLowerCase()}`} />
   529→                            </SelectTrigger>
   530→                            <SelectContent>
   531→                              {options.map((opt: string) => (
   532→                                <SelectItem key={opt} value={opt}>
   533→                                  {opt}
   534→                                </SelectItem>
   535→                              ))}
   536→                            </SelectContent>
   537→                          </Select>
   538→                        )}
   539→
   540→                        {attr.fieldType === 'image' && (
   541→                          <div className="space-y-2">
   542→                            <div className="flex items-center gap-2">
   543→                              <Button
   544→                                type="button"
   545→                                variant="outline"
   546→                                size="sm"
   547→                                className="gap-1"
   548→                                onClick={() =>
   549→                                  document.getElementById(`img-upload-${attr.id}`)?.click()
   550→                                }
   551→                              >
   552→                                <Upload className="size-3.5" />
   553→                                Upload Image
   554→                              </Button>
   555→                              {attributeValues[attr.id] && (
   556→                                <Button
   557→                                  type="button"
   558→                                  variant="ghost"
   559→                                  size="sm"
   560→                                  onClick={() => {
   561→                                    updateAttributeValue(attr.id, '')
   562→                                    setImagePreview(null)
   563→                                  }}
   564→                                >
   565→                                  <X className="size-3.5" />
   566→                                </Button>
   567→                              )}
   568→                            </div>
   569→                            <input
   570→                              id={`img-upload-${attr.id}`}
   571→                              type="file"
   572→                              accept="image/*"
   573→                              className="hidden"
   574→                              onChange={handleImageUpload}
   575→                            />
   576→                            {(attributeValues[attr.id] || imagePreview) && (
   577→                              <div className="size-20 rounded-lg border overflow-hidden">
   578→                                <img
   579→                                  src={attributeValues[attr.id] || imagePreview || ''}
   580→                                  alt="Preview"
   581→                                  className="size-full object-cover"
   582→                                />
   583→                              </div>
   584→                            )}
   585→                          </div>
   586→                        )}
   587→
   588→                        {attr.fieldType === 'date' && (
   589→                          <Input
   590→                            type="date"
   591→                            value={attributeValues[attr.id] || ''}
   592→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   593→                          />
   594→                        )}
   595→
   596→                        {attrErrors[`attr_${attr.id}`] && (
   597→                          <p className="text-sm text-destructive">{attrErrors[`attr_${attr.id}`]}</p>
   598→                        )}
   599→                      </div>
   600→                    )
   601→                  })}
   602→                </div>
   603→              </div>
   604→            )}
   605→          </div>
   606→        </Form>
   607→
   608→        <DialogFooter>
   609→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving} type="button">
   610→            Cancel
   611→          </Button>
   612→          <Button onClick={handleSaveClick} disabled={saving || !activeTypeId} type="button">
   613→            {saving && <Loader2 className="size-4 animate-spin" />}
   614→            {saving ? (isEditing ? 'Saving...' : 'Creating...') : (isEditing ? 'Save Changes' : 'Create Product')}
   615→          </Button>
   616→        </DialogFooter>
   617→      </DialogContent>
   618→    </Dialog>
   619→  )
   620→}

   632→    productType?: {

   635→      icon?: string | null

   646→  const [currentProduct, setCurrentProduct] = useState(product)
   647→
   648→  const stockStatus = getStockStatus(currentProduct.quantity, currentProduct.lowStockThreshold)
   649→
   650→  const handleStockAdjust = async () => {
   651→    if (!currentOrg || !adjustQty) return

   654→      toast.error('Please enter a valid quantity')

   656→    }

   659→      const result = await api.addStockMovement(currentOrg.id, {
   660→        productId: currentProduct.id,
   661→        type: adjustType,
   662→        quantity: qty,
   663→        reason: `${adjustType === 'in' ? 'Stock in' : 'Stock out'} (manual adjustment)`,
   664→        shopId: currentShop?.id,
   665→      })
   666→      setCurrentProduct((prev) => ({
   667→        ...prev,
   668→        quantity: result.movement.newStock,
   669→      }))
   670→      setAdjustQty('')
   671→      toast.success(`Stock ${adjustType === 'in' ? 'added' : 'removed'} successfully`)
   672→    } catch (error) {
   673→      toast.error(getNetworkErrorMessage(error))
   674→    } finally {
   675→      setAdjusting(false)
   676→    }
   677→  }

   701→          <div className="flex items-start justify-between gap-4">
   702→            <div>

   704→              <div className="flex items-center gap-2 mt-1 text-sm text-muted-foreground">
   705→                <span>{currentProduct.productType?.name}</span>
   706→                {currentProduct.sku && (
   707→                  <>
   708→                    <span>•</span>
   709→                    <span>SKU: {currentProduct.sku}</span>
   710→                  </>
   711→                )}
   712→              </div>
   713→            </div>
   714→            <div className="flex gap-2">
   715→              <Button
   716→                variant="outline"
   717→                size="sm"
   718→                className="gap-1"
   719→                onClick={() => onEdit(currentProduct)}
   720→              >
   721→                <Pencil className="size-3.5" />
   722→                Edit
   723→              </Button>
   724→            </div>
   725→          </div>
   726→          {currentProduct.description && (
   727→            <p className="text-muted-foreground mt-2 text-sm">{currentProduct.description}</p>
   728→          )}
   729→        </div>
   730→      </div>
   731→
   732→      {/* Stats Cards */}
   733→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 md:gap-4">
   734→        <Card>
   735→          <CardContent className="p-4">

   739→              {stockStatus.label}
   740→            </Badge>
   741→          </CardContent>
   742→        </Card>
   743→        <Card>

   745→            <div className="text-sm text-muted-foreground">Cost Price</div>

   868→                      className="text-xs"
   869→                    >
   870→                      {movement.type.toUpperCase()}
   871→                    </Badge>
   872→                    <span className="text-muted-foreground">
   873→                      {movement.reason || 'No reason'}
   874→                    </span>
   875→                  </div>
