    59→function getModuleIcon(iconName: string) {
    60→  const iconMap: Record<string, React.ReactNode> = {
    61→    'package': <ShoppingCart className="size-5" />,
    62→    'bar-chart': <BarChart3 className="size-5" />,
    63→    'brain': <Zap className="size-5" />,
    64→    'puzzle': <Puzzle className="size-5" />,
    65→    'palette': <Palette className="size-5" />,
    66→    'link': <Link2 className="size-5" />,
    67→    'shield': <ShieldCheck className="size-5" />,
    68→    'eye': <Eye className="size-5" />,
    69→    'tag': <Tag className="size-5" />,
    70→    'clock': <Clock className="size-5" />,
    71→  }
    72→  return iconMap[iconName] || <Puzzle className="size-5" />
    73→}

    76→// Tab 1: Module Catalog
    77→// ============================================
    78→function ModuleCatalogTab() {
    79→  const [modules, setModules] = useState<AdminModule[]>([])
    80→  const [loading, setLoading] = useState(true)
    81→  const [search, setSearch] = useState('')
    82→  const [categoryFilter, setCategoryFilter] = useState('all')
    83→  const [createOpen, setCreateOpen] = useState(false)
    84→  const [editingPrice, setEditingPrice] = useState<string | null>(null)
    85→  const [editPriceValue, setEditPriceValue] = useState('')
    86→  const [editingTrial, setEditingTrial] = useState<string | null>(null)
    87→  const [editTrialValue, setEditTrialValue] = useState('')
    88→
    89→  // Quick Assign state
    90→  const [quickAssignModule, setQuickAssignModule] = useState<string | null>(null) // module key
    91→  const [quickAssignOrgs, setQuickAssignOrgs] = useState<AdminOrg[]>([])
    92→  const [quickAssignOrgSearch, setQuickAssignOrgSearch] = useState('')
    93→  const [quickAssignSelectedOrg, setQuickAssignSelectedOrg] = useState('')
    94→  const [quickAssignLoading, setQuickAssignLoading] = useState(false)
    95→  const [quickAssignOrgsLoading, setQuickAssignOrgsLoading] = useState(false)
    96→
    97→  // Create form
    98→  const createForm = useForm<AdminModuleFormData>({
    99→    resolver: zodResolver(adminModuleSchema),
   100→    defaultValues: {
   101→      key: '',
   102→      name: '',
   103→      description: '',
   104→      icon: 'puzzle',
   105→      category: 'core',
   106→      priceETB: 0,
   107→      billingCycle: 'monthly',
   108→      isFree: true,
   109→      freeTrialDays: 30,
   110→      order: 0,
   111→    },
   112→  })
   113→
   114→  const [fetchError, setFetchError] = useState<string | null>(null)
   115→  const [creating, setCreating] = useState(false)
   116→  const [deletingId, setDeletingId] = useState<string | null>(null)
   117→  const [deleteTarget, setDeleteTarget] = useState<AdminModule | null>(null)
   118→  const [deleting, setDeleting] = useState(false)
   119→
   120→  const fetchModules = useCallback(async () => {
   121→    setFetchError(null)
   122→    try {
   123→      const data = await api.getAdminModules()
   124→      setModules(data.modules || [])
   125→    } catch (err) {
   126→      const msg = getNetworkErrorMessage(err)
   127→      setFetchError(msg)
   128→      toast.error(msg)
   129→    } finally {
   130→      setLoading(false)
   131→    }
   132→  }, [])
   133→
   134→  useEffect(() => { fetchModules() }, [fetchModules])
   135→
   136→  const handleCreateModule = async (data: AdminModuleFormData) => {
   137→    setCreating(true)
   138→    try {
   139→      await api.createAdminModule(data)
   140→      toast.success('Module created successfully')
   141→      setCreateOpen(false)
   142→      createForm.reset()
   143→      fetchModules()
   144→    } catch (err) {
   145→      toast.error(getNetworkErrorMessage(err))
   146→    } finally {
   147→      setCreating(false)
   148→    }
   149→  }
   150→
   151→  const handleToggleFree = async (mod: AdminModule) => {
   152→    try {
   153→      await api.updateAdminModule(mod.id, { isFree: !mod.isFree, priceETB: !mod.isFree ? 0 : mod.priceETB || 500 })
   154→      toast.success(`${mod.name} is now ${!mod.isFree ? 'free' : 'paid'}`)
   155→      fetchModules()
   156→    } catch (err) {
   157→      toast.error(getNetworkErrorMessage(err))
   158→    }
   159→  }
   160→
   161→  const handlePriceSave = async (mod: AdminModule) => {
   162→    const price = parseFloat(editPriceValue)
   163→    if (isNaN(price) || price < 0) {
   164→      toast.error('Price must be a valid non-negative number')
   165→      return
   166→    }
   167→    try {
   168→      await api.updateAdminModule(mod.id, { priceETB: price })
   169→      toast.success('Price updated')
   170→      setEditingPrice(null)
   171→      fetchModules()
   172→    } catch (err) {
   173→      toast.error(getNetworkErrorMessage(err))
   174→    }
   175→  }
   176→
   177→  const handleTrialSave = async (mod: AdminModule) => {
   178→    const days = parseInt(editTrialValue)
   179→    if (isNaN(days) || days < 0) {
   180→      toast.error('Trial days must be a valid non-negative number')
   181→      return
   182→    }
   183→    try {
   184→      await api.updateAdminModule(mod.id, { freeTrialDays: days })
   185→      toast.success('Trial days updated')
   186→      setEditingTrial(null)
   187→      fetchModules()
   188→    } catch (err) {
   189→      toast.error(getNetworkErrorMessage(err))
   190→    }
   191→  }
   192→
   193→  const handleDelete = async () => {
   194→    if (!deleteTarget) return
   195→    setDeleting(true)
   196→    setDeletingId(deleteTarget.id)
   197→    try {
   198→      await api.deleteAdminModule(deleteTarget.id)
   199→      toast.success('Module deactivated')
   200→      setDeleteTarget(null)
   201→      fetchModules()
   202→    } catch (err) {
   203→      toast.error(getNetworkErrorMessage(err))
   204→    } finally {
   205→      setDeletingId(null)
   206→      setDeleting(false)
   207→    }
   208→  }
   209→
   210→  const handleOpenQuickAssign = async (moduleKey: string) => {
   211→    setQuickAssignModule(moduleKey)
   212→    setQuickAssignSelectedOrg('')
   213→    setQuickAssignOrgSearch('')
   214→    setQuickAssignOrgsLoading(true)
   215→    try {
   216→      const data = await api.getAdminOrganizations({ limit: 100 })
   217→      setQuickAssignOrgs(data.organizations || [])
   218→    } catch (err) {
   219→      toast.error(getNetworkErrorMessage(err))
   220→    } finally {
   221→      setQuickAssignOrgsLoading(false)
   222→    }
   223→  }
   224→
   225→  const handleQuickAssign = async () => {
   226→    if (!quickAssignSelectedOrg || !quickAssignModule) {
   227→      toast.error('Select an organization')
   228→      return
   229→    }
   230→    setQuickAssignLoading(true)
   231→    try {
   232→      await api.assignModuleToOrg(quickAssignSelectedOrg, { moduleKey: quickAssignModule, status: 'trial', durationDays: 30 })
   233→      toast.success('Module assigned with 30-day trial')
   234→      setQuickAssignModule(null)
   235→      setQuickAssignSelectedOrg('')
   236→      fetchModules()
   237→    } catch (err) {
   238→      toast.error(getNetworkErrorMessage(err))
   239→    } finally {
   240→      setQuickAssignLoading(false)
   241→    }
   242→  }
   243→
   244→  const filteredModules = modules.filter(m => {
   245→    if (categoryFilter !== 'all' && m.category !== categoryFilter) return false
   246→    if (search && !m.name.toLowerCase().includes(search.toLowerCase()) && !m.key.toLowerCase().includes(search.toLowerCase())) return false
   247→    return true
   248→  })
   249→
   250→  if (loading) {
   251→    return (
   252→      <div className="space-y-4">
   253→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   254→          {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-56" />)}
   255→        </div>
   256→      </div>
   257→    )
   258→  }
   259→
   260→  if (fetchError && modules.length === 0) {
   261→    return (
   262→      <ErrorState
   263→        title="Failed to load modules"
   264→        message={fetchError}
   265→        onRetry={() => { setLoading(true); fetchModules() }}
   266→      />
   267→    )
   268→  }
   269→
   270→  return (
   271→    <div className="space-y-4">
   272→      {/* Toolbar */}
   273→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   274→        <div className="flex flex-col sm:flex-row gap-3 flex-1 w-full">
   275→          <div className="relative flex-1">
   276→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   277→            <Input placeholder="Search modules..." value={search} onChange={e => setSearch(e.target.value)} className="pl-9" />
   278→          </div>
   279→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   280→            <SelectTrigger className="w-full sm:w-40">
   281→              <SelectValue placeholder="Category" />
   282→            </SelectTrigger>
   283→            <SelectContent>
   284→              <SelectItem value="all">All Categories</SelectItem>
   285→              <SelectItem value="core">Core</SelectItem>
   286→              <SelectItem value="ai">AI</SelectItem>
   287→              <SelectItem value="integration">Integration</SelectItem>
   288→              <SelectItem value="analytics">Analytics</SelectItem>
   289→              <SelectItem value="communication">Communication</SelectItem>
   290→              <SelectItem value="reporting">Reporting</SelectItem>
   291→            </SelectContent>
   292→          </Select>
   293→        </div>
   294→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   295→          <DialogTrigger asChild>
   296→            <Button className="gap-1.5 w-full sm:w-auto">
   297→              <Plus className="size-4" /> Create Module
   298→            </Button>
   299→          </DialogTrigger>
   300→          <DialogContent className="max-w-lg">
   301→            <DialogHeader>
   302→              <DialogTitle>Create New Module</DialogTitle>
   303→              <DialogDescription>Add a new module to the platform catalog.</DialogDescription>
   304→            </DialogHeader>
   305→            <Form {...createForm}>
   306→              <form onSubmit={createForm.handleSubmit(handleCreateModule)} className="space-y-4">
   307→                <div className="grid grid-cols-2 gap-3">
   308→                  <FormInputField name="key" label="Module Key" placeholder="e.g. ai-analytics" required />
   309→                  <FormInputField name="name" label="Display Name" placeholder="e.g. AI Analytics" required />
   310→                </div>
   311→                <FormTextareaField name="description" label="Description" placeholder="What does this module do?" rows={2} />
   312→                <div className="grid grid-cols-2 gap-3">
   313→                  <FormSelectField name="category" label="Category" options={[
   314→                    { value: 'core', label: 'Core' },
   315→                    { value: 'ai', label: 'AI' },
   316→                    { value: 'integration', label: 'Integration' },
   317→                    { value: 'analytics', label: 'Analytics' },
   318→                    { value: 'communication', label: 'Communication' },
   319→                    { value: 'reporting', label: 'Reporting' },
   320→                  ]} />
   321→                  <FormSelectField name="icon" label="Icon" options={[
   322→                    { value: 'puzzle', label: 'Puzzle' },
   323→                    { value: 'package', label: 'Package' },
   324→                    { value: 'bar-chart', label: 'Bar Chart' },
   325→                    { value: 'brain', label: 'Brain/AI' },
   326→                    { value: 'palette', label: 'Palette' },
   327→                    { value: 'link', label: 'Link' },
   328→                    { value: 'shield', label: 'Shield' },
   329→                    { value: 'eye', label: 'Eye' },
   330→                    { value: 'tag', label: 'Tag' },
   331→                    { value: 'clock', label: 'Clock' },
   332→                  ]} />
   333→                </div>
   334→                <div className="grid grid-cols-2 gap-3">
   335→                  <FormInputField name="priceETB" label="Price (ETB/month)" type="number" min={0} />
   336→                  <FormSelectField name="billingCycle" label="Billing Cycle" options={[
   337→                    { value: 'monthly', label: 'Monthly' },
   338→                    { value: 'yearly', label: 'Yearly' },
   339→                    { value: 'one-time', label: 'One-time' },
   340→                  ]} />
   341→                </div>
   342→                <div className="flex items-center justify-between gap-3">
   343→                  <FormSwitchField name="isFree" label="Free module" className="flex-1" />
   344→                  <FormInputField name="freeTrialDays" label="Free Trial Days" type="number" min={0} max={365} inputClassName="w-24" />
   345→                </div>
   346→                <DialogFooter>
   347→                  <Button variant="outline" type="button" onClick={() => setCreateOpen(false)}>Cancel</Button>
   348→                  <FormSubmitButton isLoading={creating} loadingText="Creating...">Create Module</FormSubmitButton>
   349→                </DialogFooter>
   350→              </form>
   351→            </Form>
   352→          </DialogContent>
   353→        </Dialog>
   354→      </div>
   355→
   356→      {/* Module Grid */}
   357→      {filteredModules.length === 0 ? (
   358→        <div className="text-center py-12">
   359→          <Puzzle className="size-12 text-muted-foreground mx-auto mb-3" />
   360→          <p className="text-muted-foreground">No modules found</p>
   361→        </div>
   362→      ) : (
   363→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   364→          {filteredModules.map(mod => {
   365→            const priceType = mod.isFree ? 'free' : 'paid'
   366→            return (
   367→              <Card
   368→                key={mod.id}
   369→                className={`border-l-4 ${CARD_ACCENT[mod.category] || 'border-l-gray-400'} transition-shadow hover:shadow-md ${!mod.isActive ? 'opacity-60' : ''}`}
   370→              >
   371→                <CardContent className="p-4 space-y-3">
   372→                  {/* Header */}
   373→                  <div className="flex items-start justify-between">
   374→                    <div className="flex items-center gap-3">
   375→                      <div className={`size-10 rounded-lg flex items-center justify-center shrink-0 ${
   376→                        mod.category === 'ai' ? 'bg-purple-100 dark:bg-purple-900/30 text-purple-600' :
   377→                        mod.category === 'integration' ? 'bg-cyan-100 dark:bg-cyan-900/30 text-cyan-600' :
   378→                        mod.category === 'core' ? 'bg-emerald-100 dark:bg-emerald-900/30 text-emerald-600' :
   379→                        'bg-amber-100 dark:bg-amber-900/30 text-amber-600'
   380→                      }`}>
   381→                        {getModuleIcon(mod.icon)}
   382→                      </div>
   383→                      <div className="min-w-0">
   384→                        <h3 className="font-semibold text-sm truncate">{mod.name}</h3>
   385→                        <p className="text-xs text-muted-foreground font-mono">{mod.key}</p>
   386→                      </div>
   387→                    </div>
   388→                    <div className="flex gap-1 shrink-0">
   389→                      <Button variant="ghost" size="icon" className="size-7 text-muted-foreground hover:text-sky-600" title="Quick Assign" onClick={() => handleOpenQuickAssign(mod.key)}>
   390→                        <PackageCheck className="size-3.5" />
   391→                      </Button>
   392→                      <Button variant="ghost" size="icon" className="size-7 text-muted-foreground hover:text-destructive" onClick={() => setDeleteTarget(mod)} disabled={deletingId === mod.id}>
   393→                        {deletingId === mod.id ? <Loader2 className="size-3.5 animate-spin" /> : <Trash2 className="size-3.5" />}
   394→                      </Button>
   395→                    </div>
   396→                  </div>
   397→
   398→                  {/* Description */}
   399→                  <p className="text-xs text-muted-foreground line-clamp-2">{mod.description || 'No description'}</p>
   400→
   401→                  {/* Badges */}
   402→                  <div className="flex flex-wrap gap-1.5">
   403→                    <Badge className={categoryBadge(mod.category)} variant="outline">{mod.category}</Badge>
   404→                    {mod.isFree && <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300" variant="outline">Free</Badge>}
   405→                    {!mod.isActive && <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300" variant="outline">Inactive</Badge>}
   406→                    <Badge variant="outline" className="text-[10px]">{mod.billingCycle}</Badge>
   407→                  </div>
   408→
   409→                  <Separator />
   410→
   411→                  {/* Price - Inline Editable */}
   412→                  <div className="flex items-center justify-between">
   413→                    <div className="flex items-center gap-2">
   414→                      {editingPrice === mod.id ? (
   415→                        <div className="flex items-center gap-1">
   416→                          <span className="text-xs text-muted-foreground">ETB</span>
   417→                          <Input
   418→                            type="number"
   419→                            min={0}
   420→                            value={editPriceValue}
   421→                            onChange={e => setEditPriceValue(e.target.value)}
   422→                            onKeyDown={e => { if (e.key === 'Enter') handlePriceSave(mod); if (e.key === 'Escape') setEditingPrice(null) }}
   423→                            className="w-24 h-7 text-xs"
   424→                            autoFocus
   425→                          />
   426→                          <Button variant="ghost" size="icon" className="size-6" onClick={() => handlePriceSave(mod)}>
   427→                            <CheckCircle2 className="size-3.5 text-emerald-600" />
   428→                          </Button>
   429→                          <Button variant="ghost" size="icon" className="size-6" onClick={() => setEditingPrice(null)}>
   430→                            <XCircle className="size-3.5 text-red-500" />
   431→                          </Button>
   432→                        </div>
   433→                      ) : (
   434→                        <button
   435→                          className="flex items-center gap-1 group cursor-pointer"
   436→                          onClick={() => { setEditingPrice(mod.id); setEditPriceValue(mod.priceETB.toString()) }}
   437→                        >
   438→                          <span className={`text-sm font-bold ${PRICE_COLORS[priceType].text}`}>
   439→                            {mod.isFree ? 'Free' : formatETB(mod.priceETB)}
   440→                          </span>
   441→                          <Edit3 className="size-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity" />
   442→                        </button>
   443→                      )}
   444→                      <span className="text-[10px] text-muted-foreground">/mo</span>
   445→                    </div>
   446→                    <span className="text-xs text-muted-foreground">{mod.orgCount} orgs</span>
   447→                  </div>
   448→
   449→                  {/* Trial Days - Inline Editable */}
   450→                  <div className="flex items-center justify-between">
   451→                    <div className="flex items-center gap-2 text-xs">
   452→                      <Clock className="size-3 text-muted-foreground" />
   453→                      {editingTrial === mod.id ? (
   454→                        <div className="flex items-center gap-1">
   455→                          <Input
   456→                            type="number"
   457→                            min={0}
   458→                            max={365}
   459→                            value={editTrialValue}
   460→                            onChange={e => setEditTrialValue(e.target.value)}
   461→                            onKeyDown={e => { if (e.key === 'Enter') handleTrialSave(mod); if (e.key === 'Escape') setEditingTrial(null) }}
   462→                            className="w-16 h-6 text-xs"
   463→                            autoFocus
   464→                          />
   465→                          <span className="text-[10px]">days trial</span>
   466→                          <Button variant="ghost" size="icon" className="size-5" onClick={() => handleTrialSave(mod)}>
   467→                            <CheckCircle2 className="size-3 text-emerald-600" />
   468→                          </Button>
   469→                          <Button variant="ghost" size="icon" className="size-5" onClick={() => setEditingTrial(null)}>
   470→                            <XCircle className="size-3 text-red-500" />
   471→                          </Button>
   472→                        </div>
   473→                      ) : (
   474→                        <button
   475→                          className="flex items-center gap-1 group cursor-pointer"
   476→                          onClick={() => { setEditingTrial(mod.id); setEditTrialValue(mod.freeTrialDays.toString()) }}
   477→                        >
   478→                          <span>{mod.freeTrialDays}-day trial</span>
   479→                          <Edit3 className="size-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity" />
   480→                        </button>
   481→                      )}
   482→                    </div>

  1029→function RevenueSummaryTab() {
  1030→  const [modules, setModules] = useState<AdminModule[]>([])
  1031→  const [orgs, setOrgs] = useState<AdminOrg[]>([])
  1032→  const [loading, setLoading] = useState(true)
  1033→  const [fetchError, setFetchError] = useState<string | null>(null)
  1034→
  1035→  const fetchData = useCallback(async () => {
  1036→    setFetchError(null)
  1037→    try {
  1038→      const [modData, orgData] = await Promise.all([
  1039→        api.getAdminModules(),
  1040→        api.getAdminOrganizations({ limit: 100 }),
  1041→      ])
  1042→      setModules(modData.modules || [])
  1043→      setOrgs(orgData.organizations || [])
  1044→    } catch (err) {
  1045→      const msg = getNetworkErrorMessage(err)
  1046→      setFetchError(msg)
  1047→      toast.error(msg)
  1048→    } finally {
  1049→      setLoading(false)
  1050→    }
  1051→  }, [])
  1052→
  1053→  useEffect(() => { fetchData() }, [fetchData])
  1054→
  1055→  if (loading) {
  1056→    return (
  1057→      <div className="space-y-4">
  1058→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-4">
  1059→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24" />)}
  1060→        </div>
  1061→        <Skeleton className="h-64" />
  1062→      </div>
  1063→    )
  1064→  }
  1065→
  1066→  if (fetchError && modules.length === 0 && orgs.length === 0) {
  1067→    return (
  1068→      <ErrorState
  1069→        title="Failed to load revenue data"
  1070→        message={fetchError}
  1071→        onRetry={() => { setLoading(true); fetchData() }}
  1072→      />
  1073→    )
  1074→  }
  1075→
  1076→  // Calculate revenue metrics
  1077→  const paidModules = modules.filter(m => !m.isFree && m.isActive)
  1078→
  1079→  const monthlyRecurringRevenue = orgs.reduce((sum, org) => {
  1080→    return sum + (org.modules?.filter(m => m.status === 'active' && m.priceAtActivation > 0)
  1081→      .reduce((s, m) => s + m.priceAtActivation, 0) || 0)
  1082→  }, 0)
  1083→
  1084→  const totalTrialSubs = orgs.reduce((sum, org) => {
  1085→    return sum + (org.modules?.filter(m => m.status === 'trial').length || 0)
  1086→  }, 0)
  1087→
  1088→  const totalActiveSubs = orgs.reduce((sum, org) => {
  1089→    return sum + (org.modules?.filter(m => m.status === 'active').length || 0)
  1090→  }, 0)
  1091→
  1092→  const conversionRate = totalTrialSubs + totalActiveSubs > 0
  1093→    ? Math.round((totalActiveSubs / (totalTrialSubs + totalActiveSubs)) * 100)
  1094→    : 0
  1095→
  1096→  // Revenue by module
  1097→  const revenueByModule = paidModules.map(mod => {
  1098→    const revenue = orgs.reduce((sum, org) => {
  1099→      const sub = org.modules?.find(m => m.key === mod.key && (m.status === 'active' || m.status === 'trial'))
  1100→      return sum + (sub?.priceAtActivation || 0)
  1101→    }, 0)
  1102→    const subs = orgs.filter(org => org.modules?.some(m => m.key === mod.key && (m.status === 'active' || m.status === 'trial'))).length
  1103→    return { name: mod.name, revenue, subs, category: mod.category }
  1104→  }).filter(m => m.subs > 0).sort((a, b) => b.revenue - a.revenue)

  1110→      return sum + (org.modules?.filter(m => m.status === 'active' && m.priceAtActivation > 0)
  1111→        .reduce((s, m) => s + m.priceAtActivation, 0) || 0)
  1112→    }, 0)
  1113→    return { type: type.charAt(0).toUpperCase() + type.slice(1), revenue, count: typeOrgs.length }
  1114→  }).filter(t => t.count > 0)
  1115→
  1116→  // Adoption data for chart

  1118→    .filter(m => m.isActive)
  1119→    .sort((a, b) => b.orgCount - a.orgCount)

  1122→
  1123→  const stats = [
  1124→    { title: 'Monthly Recurring Revenue', value: formatETB(monthlyRecurringRevenue), icon: DollarSign, color: 'text-emerald-600', bg: 'bg-emerald-100 dark:bg-emerald-900/30' },
  1125→    { title: 'Active Subscriptions', value: totalActiveSubs, icon: CheckCircle2, color: 'text-primary', bg: 'bg-brand-50 dark:bg-brand-900/20' },
  1126→    { title: 'Trial Subscriptions', value: totalTrialSubs, icon: Clock, color: 'text-amber-600', bg: 'bg-amber-100 dark:bg-amber-900/30' },
  1127→    { title: 'Trial→Active Rate', value: `${conversionRate}%`, icon: TrendingUp, color: 'text-purple-600', bg: 'bg-purple-100 dark:bg-purple-900/30' },
  1128→  ]
  1129→
  1130→  return (
  1131→    <div className="space-y-4">
  1132→      {/* Stats */}
  1133→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-4">
  1134→        {stats.map(s => (
  1135→          <Card key={s.title}>
  1136→            <CardContent className="p-4">
  1137→              <div className="flex items-center gap-3">

  1142→                  <p className="text-xs text-muted-foreground truncate">{s.title}</p>
  1143→                  <p className="text-base font-semibold truncate">{s.value}</p>
  1144→                </div>
  1145→              </div>
  1146→            </CardContent>
  1147→          </Card>
