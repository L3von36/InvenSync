    39→    39→function formatETB(amount: number): string {
    40→    40→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    41→    41→}

    43→    43→function formatDate(dateStr: string | null | undefined): string {
    44→    44→  if (!dateStr) return '—'
    45→    45→  return new Date(dateStr).toLocaleDateString('en-US', {
    46→    46→    year: 'numeric', month: 'short', day: 'numeric'
    47→    47→  })
    48→    48→}
