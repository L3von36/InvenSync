   407→// ============================================
   408→// Shops Map Tab
   409→// ============================================
   410→function ShopsMapTab({ shops: initialShops }: { shops: ShopData[] }) {
   411→  const [shops, setShops] = useState<ShopData[]>(initialShops)
   412→  const [shopSearch, setShopSearch] = useState('')
   413→  const [shopTypeFilter, setShopTypeFilter] = useState('all')
   414→  const [loading, setLoading] = useState(false)
   415→
   416→  // Fetch shops from API for filtering (server-side handles the filters)
   417→  const fetchShops = useCallback(async () => {
   418→    setLoading(true)
   419→    try {
   420→      const data = await api.getAdminShops({
   421→        search: shopSearch || undefined,
   422→        businessType: shopTypeFilter !== 'all' ? shopTypeFilter : undefined,
   423→        limit: 100,
   424→      })
   425→      setShops(data.shops || [])
   426→    } catch (err) {
   427→      toast.error(getNetworkErrorMessage(err))
   428→      setShops(initialShops)
   429→    } finally {
   430→      setLoading(false)
   431→    }
   432→  }, [shopSearch, shopTypeFilter, initialShops])
   433→
   434→  useEffect(() => {
   435→    // Debounce search
   436→    const timer = setTimeout(() => fetchShops(), 300)
   437→    return () => clearTimeout(timer)
   438→  }, [fetchShops])
   439→
   440→  // Server already filters — just use the results directly
   441→  const filteredShops = shops
   442→
   443→  return (
   444→    <div className="space-y-4">
   445→      <Card>
   446→        <CardHeader>
   447→          <CardTitle className="text-base flex items-center gap-2">
   448→            <MapPin className="size-4 text-primary" /> Shops Map
   449→          </CardTitle>
   450→          <CardDescription>Geographic distribution of all registered shops</CardDescription>
   451→        </CardHeader>
   452→        <CardContent>
   453→          {/* Lazy loaded map component (YouTube pattern) */}
   454→          <Suspense fallback={<MapSkeleton />}>
   455→            <LazyShopsMap shops={filteredShops} />
   456→          </Suspense>
   457→        </CardContent>
   458→      </Card>
   459→
   460→      {/* Search and filter */}
   461→      <div className="flex flex-col sm:flex-row gap-3">
   462→        <div className="relative flex-1">
   463→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   464→          <Input placeholder="Search shops by name or city..." value={shopSearch} onChange={(e) => setShopSearch(e.target.value)} className="pl-9" />
   465→        </div>
   466→        <Select value={shopTypeFilter} onValueChange={setShopTypeFilter}>
   467→          <SelectTrigger className="w-full sm:w-40">
   468→            <SelectValue placeholder="Type" />
   469→          </SelectTrigger>
   470→          <SelectContent>
   471→            <SelectItem value="all">All Types</SelectItem>
   472→            <SelectItem value="retail">Retail</SelectItem>
   473→            <SelectItem value="service">Service</SelectItem>
   474→            <SelectItem value="mixed">Mixed</SelectItem>
   475→          </SelectContent>
   476→        </Select>
   477→      </div>
   478→
   479→      {/* Shops list */}
   480→      <div className="space-y-2 max-h-96 overflow-y-auto">
   481→        {loading ? (
   482→          <div className="space-y-2">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-16" />)}</div>
   483→        ) : filteredShops.length === 0 ? (
   484→          <p className="text-sm text-muted-foreground text-center py-8">No shops found</p>
   485→        ) : (
   486→          filteredShops.map(shop => (
   487→            <Card key={shop.id} className="p-3 hover:bg-muted/30 transition-colors">
   488→              <div className="flex items-center gap-3">
   489→                <div className={`size-8 rounded-lg flex items-center justify-center shrink-0 ${
   490→                  shop.businessType === 'retail' ? 'bg-emerald-100 dark:bg-emerald-900/30' :
   491→                  shop.businessType === 'service' ? 'bg-blue-100 dark:bg-blue-900/30' :
   492→                  'bg-amber-100 dark:bg-amber-900/30'
