    87→    87→function SalesStatsCards({ sales }: { sales: Sale[] }) {
    88→    88→  const today = new Date()
    89→    89→  today.setHours(0, 0, 0, 0)
    90→    90→
    91→    91→  const startOfWeek = new Date(today)
    92→    92→  startOfWeek.setDate(today.getDate() - today.getDay())
    93→    93→
    94→    94→  const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1)
    95→    95→
    96→    96→  const todaySales = sales.filter(s => new Date(s.saleDate) >= today)
    97→    97→  const weekSales = sales.filter(s => new Date(s.saleDate) >= startOfWeek)
    98→    98→  const monthSales = sales.filter(s => new Date(s.saleDate) >= startOfMonth)
    99→    99→
   100→   100→  const todayTotal = todaySales.reduce((acc, s) => acc + s.total, 0)
   101→   101→  const weekTotal = weekSales.reduce((acc, s) => acc + s.total, 0)
   102→   102→  const monthTotal = monthSales.reduce((acc, s) => acc + s.total, 0)
   103→   103→  const avgSale = sales.length > 0 ? sales.reduce((acc, s) => acc + s.total, 0) / sales.length : 0
   104→   104→
   105→   105→  const cards = [
   106→   106→    { title: "Today's Sales", value: formatETB(todayTotal), icon: ShoppingCart, color: 'text-primary', bg: 'bg-primary/10' },
   107→   107→    { title: 'This Week', value: formatETB(weekTotal), icon: Calendar, color: 'text-primary', bg: 'bg-brand-50 dark:bg-brand-900/20' },
   108→   108→    { title: 'This Month', value: formatETB(monthTotal), icon: TrendingUp, color: 'text-amber-600', bg: 'bg-amber-100 dark:bg-amber-900/30' },
   109→   109→    { title: 'Avg Sale Value', value: formatETB(avgSale), icon: DollarSign, color: 'text-primary', bg: 'bg-primary/10' },
   110→   110→  ]
   111→   111→
   112→   112→  return (
   113→   113→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   114→   114→      {cards.map((card) => (
   115→   115→        <Card key={card.title}>
   116→   116→          <CardContent className="p-4">
   117→   117→            <div className="flex items-center gap-3">
   118→   118→              <div className={`size-10 rounded-lg ${card.bg} flex items-center justify-center shrink-0`}>
   119→   119→                <card.icon className={`size-5 ${card.color}`} />
   120→   120→              </div>
   121→   121→              <div className="min-w-0">
   122→   122→                <p className="text-xs text-muted-foreground truncate">{card.title}</p>
   123→   123→                <p className="text-xl sm:text-2xl font-bold truncate">{card.value}</p>
   124→   124→              </div>
   125→   125→            </div>
   126→   126→          </CardContent>
   127→   127→        </Card>
   128→   128→      ))}
   129→   129→    </div>
   130→   130→  )
   131→   131→}

   304→   304→function AllSalesTab({
   305→   305→  orgId,
   306→   306→  shopId,
   307→   307→  onViewSale,
   308→   308→  onCreateSale,
   309→   309→}: {
   310→   310→  orgId: string
   311→   311→  shopId?: string
   312→   312→  onViewSale: (sale: Sale) => void
   313→   313→  onCreateSale: () => void
   314→   314→}) {
   315→   315→  const [sales, setSales] = useState<Sale[]>([])
   316→   316→  const [loading, setLoading] = useState(true)
   317→   317→  const [error, setError] = useState<string | null>(null)
   318→   318→  const [search, setSearch] = useState('')
   319→   319→  const [filterStatus, setFilterStatus] = useState<string>('all')
   320→   320→  const [filterPayment, setFilterPayment] = useState<string>('all')
   321→   321→  const [page, setPage] = useState(1)
   322→   322→  const [totalPages, setTotalPages] = useState(1)
   323→   323→
   324→   324→  const fetchSales = useCallback(async () => {
   325→   325→    setLoading(true)
   326→   326→    setError(null)
   327→   327→    try {
   328→   328→      const res = await api.getSales(orgId, {
   329→   329→        status: filterStatus !== 'all' ? filterStatus : undefined,
   330→   330→        page,
   331→   331→        limit: 10,
   332→   332→        shopId,
   333→   333→      })
   334→   334→      setSales(res.sales)
   335→   335→      setTotalPages(res.pagination.totalPages)
   336→   336→    } catch (err) {
   337→   337→      const msg = getNetworkErrorMessage(err)
   338→   338→      setError(msg)
   339→   339→      toast.error(msg)
   340→   340→    } finally {
   341→   341→      setLoading(false)
   342→   342→    }
   343→   343→  }, [orgId, filterStatus, page, shopId])
   344→   344→
   345→   345→  useEffect(() => {
   346→   346→    fetchSales()
   347→   347→  }, [fetchSales])
   348→   348→
   349→   349→  const filteredSales = sales.filter((s) => {
   350→   350→    if (filterPayment !== 'all' && s.paymentMethod !== filterPayment) return false
   351→   351→    if (search) {
   352→   352→      const q = search.toLowerCase()
   353→   353→      const matchInvoice = s.invoiceNumber.toLowerCase().includes(q)
   354→   354→      const matchCustomer = s.customer?.name.toLowerCase().includes(q)
   355→   355→      if (!matchInvoice && !matchCustomer) return false
   356→   356→    }
   357→   357→    return true
   358→   358→  })
   359→   359→
   360→   360→  if (loading) {
   361→   361→    return (
   362→   362→      <div className="space-y-4">
   363→   363→        <div className="flex flex-col sm:flex-row gap-3">
   364→   364→          <Skeleton className="h-10 flex-1" />
   365→   365→          <Skeleton className="h-10 w-36" />
   366→   366→          <Skeleton className="h-10 w-36" />
   367→   367→        </div>
   368→   368→        {Array.from({ length: 5 }).map((_, i) => (
   369→   369→          <Skeleton key={i} className="h-14" />
   370→   370→        ))}
   371→   371→      </div>
   372→   372→    )
   373→   373→  }
   374→   374→
   375→   375→  if (error && sales.length === 0) {
   376→   376→    return <ErrorState title="Failed to load sales" message={error} onRetry={fetchSales} />
   377→   377→  }
   378→   378→
   379→   379→  return (
   380→   380→    <div className="space-y-4">
   381→   381→      {/* Filters */}
   382→   382→      <div className="flex flex-col sm:flex-row gap-3">
   383→   383→        <div className="relative flex-1">
   384→   384→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   385→   385→          <Input
   386→   386→            placeholder="Search by invoice # or customer..."
   387→   387→            className="pl-9"
   388→   388→            value={search}
   389→   389→            onChange={(e) => setSearch(e.target.value)}
   390→   390→          />
   391→   391→        </div>
   392→   392→        <Select value={filterStatus} onValueChange={(v) => { setFilterStatus(v); setPage(1) }}>
   393→   393→          <SelectTrigger className="w-full sm:w-40">
   394→   394→            <SelectValue placeholder="Status" />
   395→   395→          </SelectTrigger>
   396→   396→          <SelectContent>
   397→   397→            <SelectItem value="all">All Status</SelectItem>
   398→   398→            <SelectItem value="completed">Completed</SelectItem>
   399→   399→            <SelectItem value="pending">Pending</SelectItem>
   400→   400→            <SelectItem value="cancelled">Cancelled</SelectItem>
   401→   401→          </SelectContent>
   402→   402→        </Select>
   403→   403→        <Select value={filterPayment} onValueChange={setFilterPayment}>
   404→   404→          <SelectTrigger className="w-full sm:w-40">
   405→   405→            <SelectValue placeholder="Payment" />
   406→   406→          </SelectTrigger>
   407→   407→          <SelectContent>
   408→   408→            <SelectItem value="all">All Payment</SelectItem>
   409→   409→            <SelectItem value="cash">Cash</SelectItem>
   410→   410→            <SelectItem value="card">Card</SelectItem>
   411→   411→            <SelectItem value="mobile_money">Mobile Money</SelectItem>
   412→   412→            <SelectItem value="credit">Credit</SelectItem>
   413→   413→          </SelectContent>
   414→   414→        </Select>
   415→   415→      </div>
   416→   416→
   417→   417→      {/* Empty state or Table */}
   418→   418→      {!loading && filteredSales.length === 0 ? (
   419→   419→        <Card>
   420→   420→          <CardContent className="py-12 text-center">
   421→   421→            <ShoppingCart className="size-12 text-muted-foreground mx-auto mb-4" />
   422→   422→            <h3 className="text-base font-semibold mb-1">
   423→   423→              {search || filterStatus !== 'all' || filterPayment !== 'all' ? 'No sales match your filters' : 'No sales yet'}
   424→   424→            </h3>
   425→   425→            <p className="text-muted-foreground text-sm mb-4">
   426→   426→              {search || filterStatus !== 'all' || filterPayment !== 'all'
   427→   427→                ? 'Try adjusting your search or filters'
   428→   428→                : 'Create your first sale to start tracking transactions'}
   429→   429→            </p>
   430→   430→            {!search && filterStatus === 'all' && filterPayment === 'all' && (
   431→   431→              <Button onClick={onCreateSale} className="gap-2">
   432→   432→                <Plus className="size-4 mr-2" />
   433→   433→                Create Sale
   434→   434→              </Button>
   435→   435→            )}
   436→   436→          </CardContent>
