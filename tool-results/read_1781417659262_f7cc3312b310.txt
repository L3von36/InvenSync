    18→
    19→export interface AttributeTemplate {
    20→  name: string
    21→  fieldType: 'text' | 'number' | 'select' | 'image'
    22→  options?: string[] // For select fields
    23→  required?: boolean
    24→}
    25→
    26→export interface ProductTypeTemplate {
    27→  name: string
    28→  icon: string
    29→  attributes: AttributeTemplate[]
    30→}
    31→
    32→export interface FormFieldConfig {
    33→  field: string
    34→  label: string
    35→  required: boolean
    36→  order: number
    37→}
    38→
    39→export interface BusinessTemplate {
    40→  key: BusinessTypeKey
    41→  label: string
    42→  description: string
    43→  icon: string
    44→  productTypes: ProductTypeTemplate[]
    45→  addProductFormFields: FormFieldConfig[]
    46→}
