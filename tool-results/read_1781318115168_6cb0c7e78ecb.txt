   557→   557→export function MobileBottomNav() {
   558→   558→  const { currentPage, setPage } = useAppStore()
   559→   559→  const { user, currentOrg, currentOrgRole, activeModules } = useAuthStore()
   560→   560→  const { theme, setTheme } = useTheme()
   561→   561→  const [showMore, setShowMore] = useState(false)
   562→   562→  const morePanelRef = useRef<HTMLDivElement>(null)
   563→   563→  const moreButtonRef = useRef<HTMLButtonElement>(null)
   564→   564→
   565→   565→  const isService = currentOrg?.businessType === 'service'
   566→   566→  const isMixed = currentOrg?.businessType === 'mixed'

   569→   569→

   575→   575→  }
   576→   576→
   577→   577→  const mainTabs: Array<{ page: Page; icon: React.ElementType; label: string }> = isAdmin
   578→   578→    ? [
