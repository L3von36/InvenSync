    44→function bookingStatusBadge(status: string) {
    45→  switch (status) {
    46→    case 'scheduled': return <Badge className="bg-blue-100 text-blue-800 dark:bg-blue-900/40 dark:text-blue-300">Scheduled</Badge>
    47→    case 'completed': return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300">Completed</Badge>
    48→    case 'cancelled': return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300">Cancelled</Badge>
    49→    case 'no_show': return <Badge className="bg-gray-100 text-gray-800 dark:bg-gray-900/40 dark:text-gray-300">No Show</Badge>
    50→    default: return <Badge variant="outline">{status}</Badge>

    54→function formatBookingDate(dateStr: string): string {
    55→  try {
    56→    const d = new Date(dateStr)
    57→    return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric', year: 'numeric' })
    58→  } catch {
    59→    return dateStr
    60→  }
    61→}

    63→function addMinutesToTime(time: string | undefined | null, minutes: number): string {
    64→  if (!time) return '0:00'
    65→  const [h, m] = time.split(':').map(Number)
    66→  const totalMinutes = (h * 60 + m + minutes) % (24 * 60)
    67→  const newH = Math.floor(totalMinutes / 60)
    68→  const newM = totalMinutes % 60
    69→  return `${String(newH).padStart(2, '0')}:${String(newM).padStart(2, '0')}`
    70→}

    75→function ServiceTypeDialog({
    76→  open,
    77→  onOpenChange,
    78→  serviceType,
    79→  onSaved,
    80→}: {
    81→  open: boolean
    82→  onOpenChange: (open: boolean) => void
    83→  serviceType: ServiceType | null
    84→  onSaved: () => void
    85→}) {
    86→  const { currentOrg } = useAuthStore()
    87→  const [saving, setSaving] = useState(false)
    88→  const isEditing = !!serviceType
    89→
    90→  const form = useForm<ServiceTypeFormData>({
    91→    resolver: zodResolver(serviceTypeSchema),
    92→    defaultValues: {
    93→      name: '',
    94→      description: '',
    95→      duration: 30,
    96→      price: 0,
    97→      imageUrl: '',
    98→    },
    99→  })
   100→
   101→  useEffect(() => {
   102→    if (open) {
   103→      if (serviceType) {
   104→        form.reset({
   105→          name: serviceType.name,
   106→          description: serviceType.description || '',
   107→          duration: serviceType.duration,
   108→          price: serviceType.price,
   109→          imageUrl: serviceType.imageUrl || '',
   110→        })
   111→      } else {
   112→        form.reset({
   113→          name: '',
   114→          description: '',
   115→          duration: 30,
   116→          price: 0,
   117→          imageUrl: '',
   118→        })
   119→      }
   120→    }
   121→  }, [open, serviceType, form])
   122→
   123→  const handleSave = async (data: ServiceTypeFormData) => {
   124→    if (!currentOrg) return
   125→    setSaving(true)
   126→    try {
   127→      if (isEditing && serviceType) {
   128→        await api.updateServiceType(serviceType.id, currentOrg.id, {
   129→          name: data.name.trim(),
   130→          description: data.description?.trim() || undefined,
   131→          duration: data.duration ?? 30,
   132→          price: data.price ?? 0,
   133→          imageUrl: data.imageUrl?.trim() || undefined,
   134→        })
   135→        toast.success('Service type updated')
   136→      } else {
   137→        await api.createServiceType(currentOrg.id, {
   138→          name: data.name.trim(),
   139→          description: data.description?.trim() || undefined,
   140→          duration: data.duration ?? 30,
   141→          price: data.price ?? 0,
   142→          imageUrl: data.imageUrl?.trim() || undefined,
   143→        })
   144→        toast.success('Service type created')
   145→      }
   146→      onOpenChange(false)
   147→      onSaved()
   148→    } catch (err) {
   149→      toast.error(getNetworkErrorMessage(err))
   150→    } finally {
   151→      setSaving(false)
   152→    }
   153→  }
   154→
   155→  return (
   156→    <Dialog open={open} onOpenChange={onOpenChange}>
   157→      <DialogContent className="sm:max-w-md">
   158→        <DialogHeader>
   159→          <DialogTitle>{isEditing ? 'Edit Service Type' : 'Add Service Type'}</DialogTitle>
   160→          <DialogDescription>
   161→            {isEditing ? 'Update the service details' : 'Define a new service you offer'}
   162→          </DialogDescription>
   163→        </DialogHeader>
   164→        <Form {...form}>
   165→          <form onSubmit={form.handleSubmit(handleSave)} className="space-y-4">
   166→            <FormInputField name="name" label="Service Name" placeholder="e.g., Haircut, Consultation" required />
   167→            <FormTextareaField name="description" label="Description" placeholder="Brief description of this service..." rows={2} />

   169→              <FormInputField name="duration" label="Duration (min)" type="number" min={5} />
   170→              <FormInputField name="price" label="Price (ETB)" type="number" min={0} step="0.01" placeholder="0.00" />

   202→

   204→    resolver: zodResolver(bookingSchema),

   206→      serviceTypeId: '',

   209→      bookingDate: new Date().toISOString().split('T')[0],
   210→      startTime: '09:00',

   216→  const watchedServiceTypeId = form.watch('serviceTypeId')
   217→  const watchedStartTime = form.watch('startTime')
   218→  const selectedService = serviceTypes.find(st => st.id === watchedServiceTypeId)
   219→

   221→  useEffect(() => {
   222→    if (selectedService && watchedStartTime) {
   223→      form.setValue('endTime', addMinutesToTime(watchedStartTime, selectedService.duration))
   224→    }
   225→  }, [selectedService, watchedStartTime, form])
   226→
   227→  useEffect(() => {
   228→    if (open) {
   229→      form.reset({
   230→        serviceTypeId: '',
   231→        customerName: '',
   232→        customerPhone: '',
   233→        bookingDate: new Date().toISOString().split('T')[0],
   234→        startTime: '09:00',
   235→        endTime: '09:30',
   236→        notes: '',
   237→      })
   238→    }
   239→  }, [open, form])
   240→
   241→  const handleSave = async (data: BookingFormData) => {
   242→    if (!currentOrg) return
   243→    setSaving(true)

   261→      setSaving(false)
   262→    }
   263→  }

   335→function ServiceTypesTab({ orgId, onRefresh }: { orgId: string; onRefresh: () => void }) {
   336→  const [serviceTypes, setServiceTypes] = useState<ServiceType[]>([])
   337→  const [loading, setLoading] = useState(true)
   338→  const [fetchError, setFetchError] = useState<string | null>(null)
   339→  const [dialogOpen, setDialogOpen] = useState(false)
   340→  const [editingType, setEditingType] = useState<ServiceType | null>(null)
   341→  const [deleteId, setDeleteId] = useState<string | null>(null)
   342→  const [deleting, setDeleting] = useState(false)
   343→
   344→  const fetchData = useCallback(async () => {
   345→    try {
   346→      setFetchError(null)
   347→      const data = await api.getServiceTypes(orgId)
   348→      setServiceTypes(data.serviceTypes || [])
   349→    } catch (err) {
   350→      setFetchError(getNetworkErrorMessage(err))
   351→    } finally {
   352→      setLoading(false)
   353→    }
   354→  }, [orgId])
   355→
   356→  useEffect(() => { fetchData() }, [fetchData])
   357→
   358→  const handleToggle = async (st: ServiceType) => {
   359→    try {
   360→      await api.updateServiceType(st.id, orgId, { isActive: !st.isActive })
   361→      toast.success(st.isActive ? 'Service deactivated' : 'Service activated')
   362→      fetchData()
   363→    } catch (err) {
   364→      toast.error(getNetworkErrorMessage(err))
   365→    }
   366→  }
   367→
   368→  const handleDelete = async () => {
   369→    if (!deleteId) return
   370→    setDeleting(true)
   371→    try {
   372→      await api.deleteServiceType(deleteId, orgId)
   373→      toast.success('Service type deleted')
   374→      setDeleteId(null)
   375→      fetchData()
   376→      onRefresh()
   377→    } catch (err) {
   378→      toast.error(getNetworkErrorMessage(err))
   379→    } finally {
   380→      setDeleting(false)
   381→    }
   382→  }
   383→
   384→  if (loading) {
   385→    return (
   386→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   387→        {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-44" />)}
   388→      </div>
   389→    )
   390→  }
   391→
   392→  if (fetchError) {
   393→    return <ErrorState title="Failed to load service types" error={fetchError} onRetry={fetchData} />
   394→  }
   395→
   396→  return (
   397→    <div className="space-y-4">
   398→      <div className="flex justify-end">
   399→        <Button onClick={() => { setEditingType(null); setDialogOpen(true) }} className="gap-2">
   400→          <Plus className="size-4" /> Add Service Type
   401→        </Button>
   402→      </div>
   403→
   404→      {serviceTypes.length === 0 ? (
   405→        <Card>
   406→          <CardContent className="py-12 text-center">
   407→            <Wrench className="size-12 text-muted-foreground mx-auto mb-4" />
   408→            <p className="font-medium text-muted-foreground">No service types yet</p>
   409→            <p className="text-sm text-muted-foreground mt-1">Create your first service type to get started</p>
   410→            <Button className="mt-4 gap-2" onClick={() => { setEditingType(null); setDialogOpen(true) }}>
   411→              <Plus className="size-4" /> Add Service Type
   412→            </Button>
   413→          </CardContent>
   414→        </Card>
   415→      ) : (
   416→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   417→          {serviceTypes.map(st => (
   418→            <Card key={st.id} className={`transition-opacity ${!st.isActive ? 'opacity-60' : ''}`}>
   419→              <CardContent className="p-4 space-y-3">
   420→                <div className="flex items-start justify-between">
   421→                  <div className="min-w-0 flex-1">
   422→                    <h4 className="font-semibold truncate">{st.name}</h4>
   423→                    {st.description && (
   424→                      <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">{st.description}</p>
   425→                    )}
   426→                    <div className="flex items-center gap-3 mt-1.5 text-sm text-muted-foreground">
   427→                      <span className="flex items-center gap-1"><Clock className="size-3.5" /> {st.duration} min</span>
   428→                      <span className="flex items-center gap-1"><DollarSign className="size-3.5" /> {formatETB(st.price)}</span>
   429→                    </div>
   430→                  </div>
   431→                  <Switch checked={st.isActive} onCheckedChange={() => handleToggle(st)} />
   432→                </div>
   433→                <div className="flex items-center gap-2">
   434→                  <Badge variant={st.isActive ? 'default' : 'secondary'} className="text-xs">
   435→                    {st.isActive ? 'Active' : 'Inactive'}
   436→                  </Badge>
   437→                </div>
   438→                <div className="flex gap-2 pt-1">
   439→                  <Button size="sm" variant="outline" className="gap-1 flex-1" onClick={() => { setEditingType(st); setDialogOpen(true) }}>
   440→                    <Pencil className="size-3" /> Edit
   441→                  </Button>
   442→                  <Button size="sm" variant="outline" className="text-destructive hover:text-destructive gap-1" onClick={() => setDeleteId(st.id)}>
   443→                    <Trash2 className="size-3" />
   444→                  </Button>
   445→                </div>
   446→              </CardContent>
   447→            </Card>
   448→          ))}
   449→        </div>
   450→      )}
   451→
   452→      <ServiceTypeDialog
   453→        open={dialogOpen}
   454→        onOpenChange={setDialogOpen}
   455→        serviceType={editingType}
   456→        onSaved={() => { fetchData(); onRefresh() }}
   457→      />
   458→
   459→      {/* Delete confirmation */}
   460→      <AlertDialog open={!!deleteId} onOpenChange={(open) => { if (!open) setDeleteId(null) }}>
   461→        <AlertDialogContent>
   462→          <AlertDialogHeader>
   463→            <AlertDialogTitle>Delete Service Type</AlertDialogTitle>
   464→            <AlertDialogDescription>
   465→              Are you sure you want to delete this service type? This action cannot be undone.
   466→              The service type and all its configuration will be permanently deleted.
   467→              Any existing bookings using this service type will not be affected.
   468→            </AlertDialogDescription>
   469→          </AlertDialogHeader>
   470→          <AlertDialogFooter>
   471→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   472→            <AlertDialogAction onClick={handleDelete} disabled={deleting} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">
   473→              {deleting && <Loader2 className="size-4 animate-spin" />}
   474→              {deleting ? 'Deleting...' : 'Delete'}
   475→            </AlertDialogAction>
   476→          </AlertDialogFooter>
   477→        </AlertDialogContent>
   478→      </AlertDialog>
   479→    </div>
   480→  )
   481→}

   632→                              {updatingBookingId === b.id ? <Loader2 className="size-3 animate-spin" /> : <CheckCircle2 className="size-3" />} Done
   633→                            </Button>
   634→                            <Button size="sm" variant="outline" className="text-red-600 gap-1 h-7 text-xs" onClick={() => handleStatusUpdate(b.id, 'cancelled')} disabled={updatingBookingId === b.id}>
   635→                              <XCircle className="size-3" /> Cancel
   636→                            </Button>
   637→                            <Button size="sm" variant="outline" className="text-gray-600 gap-1 h-7 text-xs" onClick={() => handleStatusUpdate(b.id, 'no_show')} disabled={updatingBookingId === b.id}>
   638→                              <EyeOff className="size-3" /> No Show
   639→                            </Button>
   640→                          </div>

   648→          </div>
   649→          <div className="md:hidden space-y-3">
   650→            {bookings.map(b => (

   662→                </div>
   663→                {b.status === 'scheduled' && (
   664→                  <div className="flex gap-2">

   666→                      {updatingBookingId === b.id ? <Loader2 className="size-3 animate-spin" /> : <CheckCircle2 className="size-3" />} Complete
   667→                    </Button>
   668→                    <Button size="sm" variant="outline" className="text-red-600 gap-1 flex-1 h-8 text-xs" onClick={() => handleStatusUpdate(b.id, 'cancelled')} disabled={updatingBookingId === b.id}>
   669→                      <XCircle className="size-3" /> Cancel
   670→                    </Button>
   671→                    <Button size="sm" variant="outline" className="text-gray-600 gap-1 h-8 text-xs" onClick={() => handleStatusUpdate(b.id, 'no_show')} disabled={updatingBookingId === b.id}>
   672→                      <EyeOff className="size-3" />
   673→                    </Button>
   674→                  </div>
   675→                )}
   676→              </Card>
   677→            ))}
   678→          </div>
   679→        </>
   680→      )}
   681→
   682→      <BookingDialog
   683→        open={dialogOpen}
   684→        onOpenChange={setDialogOpen}
   685→        serviceTypes={serviceTypes}
   686→        onSaved={fetchBookings}
   687→      />
   688→    </div>
   689→  )
   690→}
