    89→function getStockStatus(
    90→  quantity: number,
    91→  lowStockThreshold: number
    92→): { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' } {
    93→  if (quantity <= 0) return { label: 'Out of Stock', variant: 'destructive' }
    94→  if (quantity <= lowStockThreshold) return { label: 'Low Stock', variant: 'secondary' }
    95→  return { label: 'In Stock', variant: 'default' }
    96→}
    97→
    98→/** Safely parse attribute options that may be double-encoded */
    99→function parseAttributeOptions(raw: string | null | undefined): string[] | null {
   100→  if (!raw) return null
   101→  try {
   102→    const parsed = JSON.parse(raw)
   103→    if (Array.isArray(parsed)) {
   104→      return parsed
   105→    } else if (typeof parsed === 'string') {
   106→      // Handle double-encoded: "Red,Blue,Green" instead of ["Red","Blue","Green"]
   107→      return parsed.split(',').map((s: string) => s.trim()).filter(Boolean)
   108→    }
   109→    return null
   110→  } catch {
   111→    // If parsing fails, try splitting by comma
   112→    return raw.split(',').map(s => s.trim()).filter(Boolean)
   113→  }
   114→}

   119→function ProductDialog({

   124→  onSaved,

   126→  open: boolean

   128→  product: Product | null
   129→  productTypes: ProductType[]
   130→  onSaved: () => void
   131→}) {
   132→  const { currentOrg, currentShop } = useAuthStore()
   133→  const [saving, setSaving] = useState(false)
   134→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   135→  const [attributeValues, setAttributeValues] = useState<Record<string, string>>({})
   136→  const [attrErrors, setAttrErrors] = useState<Record<string, string>>({})
   137→  const isEditing = !!product
   138→
   139→  // Separate state to track selected product type ID reliably
   140→  const [selectedTypeId, setSelectedTypeId] = useState<string>('')
   141→
   142→  const form = useForm<ProductFormData>({
   143→    resolver: zodResolver(productSchema),
   144→    defaultValues: {
   145→      productTypeId: '',
   146→      name: '',
   147→      sku: '',
   148→      description: '',
   149→      costPrice: undefined as unknown as number,
   150→      sellingPrice: undefined as unknown as number,
   151→      lowStockThreshold: 10,
   152→      quantity: 0,
   153→    },
   154→  })
   155→
   156→  // Watch productTypeId for reactive UI
   157→  const watchedProductTypeId = form.watch('productTypeId')
   158→
   159→  // Use selectedTypeId as the source of truth (falls back to form value)
   160→  const activeTypeId = selectedTypeId || watchedProductTypeId
   161→  const selectedType = productTypes.find((pt) => pt.id === activeTypeId)
   162→  const typeAttributes = selectedType?.attributes || []
   163→
   164→  // Reset form when dialog opens
   165→  useEffect(() => {
   166→    if (open) {
   167→      setAttrErrors({})
   168→      if (product) {
   169→        const attrVals: Record<string, string> = {}
   170→        product.attributeValues?.forEach((av) => {
   171→          attrVals[av.attributeDefinitionId] = av.value
   172→        })
   173→        setAttributeValues(attrVals)
   174→        setSelectedTypeId(product.productTypeId)
   175→        form.reset({
   176→          productTypeId: product.productTypeId,
   177→          name: product.name,
   178→          sku: product.sku || '',
   179→          description: product.description || '',
   180→          costPrice: product.costPrice,
   181→          sellingPrice: product.sellingPrice,
   182→          lowStockThreshold: product.lowStockThreshold,
   183→          quantity: product.quantity,
   184→        })
   185→        setImagePreview(product.imageUrl || null)
   186→      } else {
   187→        setAttributeValues({})
   188→        setSelectedTypeId('')
   189→        form.reset({
   190→          productTypeId: '',
   191→          name: '',
   192→          sku: '',
   193→          description: '',
   194→          costPrice: undefined as unknown as number,
   195→          sellingPrice: undefined as unknown as number,
   196→          lowStockThreshold: 10,
   197→          quantity: 0,
   198→        })
   199→        setImagePreview(null)
   200→      }
   201→    } else {
   202→      // Clean up state when dialog closes
   203→      setSelectedTypeId('')
   204→      setAttributeValues({})
   205→      setAttrErrors({})
   206→      setImagePreview(null)
   207→    }
   208→  }, [open, product, form])
   209→
   210→  // Handle product type selection
   211→  const handleSelectType = (typeId: string) => {
   212→    setSelectedTypeId(typeId)
   213→    form.setValue('productTypeId', typeId, { shouldValidate: true, shouldDirty: true, shouldTouch: true })
   214→    // Clear attribute values when switching types
   215→    setAttributeValues({})
   216→    setAttrErrors({})
   217→  }
   218→
   219→  const updateAttributeValue = (attrId: string, value: string) => {
   220→    setAttributeValues((prev) => ({ ...prev, [attrId]: value }))
   221→    setAttrErrors((prev) => {
   222→      const next = { ...prev }
   223→      delete next[`attr_${attrId}`]
   224→      return next
   225→    })
   226→  }
   227→
   228→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   229→    const file = e.target.files?.[0]
   230→    if (!file) return
   231→    const reader = new FileReader()
   232→    reader.onloadend = () => {
   233→      const base64 = reader.result as string
   234→      setImagePreview(base64)
   235→      typeAttributes
   236→        .filter((a) => a.fieldType === 'image')
   237→        .forEach((a) => {
   238→          updateAttributeValue(a.id, base64)
   239→        })
   240→    }
   241→    reader.readAsDataURL(file)
   242→  }
   243→
   244→  const onSubmit = async (data: ProductFormData) => {
   245→    if (!currentOrg) return
   246→
   247→    // Use selectedTypeId as the authoritative source for productTypeId
   248→    const productTypeId = selectedTypeId || data.productTypeId || activeTypeId
   249→    if (!productTypeId) {
   250→      toast.error('Please select a product type')
   251→      return
   252→    }
   253→
   254→    // Safely convert form values to numbers (form may store them as strings from input onChange)
   255→    const costPrice = Number(data.costPrice)
   256→    const sellingPrice = Number(data.sellingPrice)
   257→    const lowStockThreshold = Number(data.lowStockThreshold) || 10
   258→    const quantity = Number(data.quantity) || 0
   259→
   260→    // Validate converted numbers
   261→    if (!costPrice || Number.isNaN(costPrice) || costPrice <= 0) {
   262→      toast.error('Cost price must be a valid number greater than 0')
   263→      return
   264→    }
   265→    if (!sellingPrice || Number.isNaN(sellingPrice) || sellingPrice <= 0) {
   266→      toast.error('Selling price must be a valid number greater than 0')
   267→      return
   268→    }
   269→
   270→    setSaving(true)
   271→    try {
   272→      const attrVals = Object.entries(attributeValues)
   273→        .filter(([, value]) => value.trim() !== '')
   274→        .map(([attributeDefinitionId, value]) => ({
   275→          attributeDefinitionId,
   276→          value,
   277→        }))
   278→
   279→      if (isEditing && product) {
   280→        await api.updateProduct(product.id, currentOrg.id, {
   281→          name: data.name.trim(),
   282→          sku: data.sku?.trim() || undefined,
   283→          description: data.description?.trim() || undefined,
   284→          imageUrl: imagePreview || product?.imageUrl || undefined,
   285→          costPrice,
   286→          sellingPrice,
   287→          lowStockThreshold,
   288→          attributeValues: attrVals,
   289→        })
   290→        toast.success('Product updated successfully')
   291→      } else {
   292→        await api.createProduct(currentOrg.id, {
   293→          productTypeId,
   294→          name: data.name.trim(),
   295→          sku: data.sku?.trim() || undefined,
   296→          description: data.description?.trim() || undefined,
   297→          imageUrl: imagePreview || product?.imageUrl || undefined,
   298→          quantity,
   299→          costPrice,
   300→          sellingPrice,
   301→          lowStockThreshold,
   302→          attributeValues: attrVals,
   303→          shopId: currentShop?.id,
   304→        })
   305→        toast.success('Product created successfully')
   306→      }
   307→
   308→      onOpenChange(false)
   309→      onSaved()
   310→    } catch (error) {
   311→      toast.error(getNetworkErrorMessage(error))
   312→    } finally {
   313→      setSaving(false)
   314→    }
   315→  }
   316→
   317→  const handleSaveClick = async () => {
   318→    // First validate attributes manually
   319→    const missingRequired = typeAttributes
   320→      .filter((a) => a.required)
   321→      .filter((a) => !attributeValues[a.id] || attributeValues[a.id].trim() === '')
   322→    if (missingRequired.length > 0) {
   323→      const errors: Record<string, string> = {}
   324→      missingRequired.forEach((a) => {
   325→        errors[`attr_${a.id}`] = `${a.name} is required`
   326→      })
   327→      setAttrErrors(errors)
   328→      return
   329→    }
   330→
   331→    // Ensure productTypeId is set in form before validation
   332→    if (selectedTypeId) {
   333→      form.setValue('productTypeId', selectedTypeId, { shouldValidate: true, shouldDirty: true })
   334→    }
   335→
   336→    // Validate all form fields including productTypeId
   337→    const isValid = await form.trigger(['productTypeId', 'name', 'costPrice', 'sellingPrice', 'lowStockThreshold', 'quantity'])
   338→    if (!isValid) return
   339→
   340→    // Final check: ensure we have a productTypeId
   341→    if (!selectedTypeId && !form.getValues('productTypeId')) {
   342→      toast.error('Please select a product type')
   343→      return
   344→    }
   345→
   346→    const data = form.getValues()
   347→    onSubmit(data)
   348→  }
   349→
   350→  return (
   351→    <Dialog open={open} onOpenChange={onOpenChange}>
   352→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto w-[calc(100%-2rem)] sm:w-full">
   353→        <DialogHeader>
   354→          <DialogTitle>{isEditing ? 'Edit Product' : 'Add Product'}</DialogTitle>
   355→          <DialogDescription>Fill in the product details below.</DialogDescription>
   356→        </DialogHeader>
   357→
   358→        <Form {...form}>
   359→          <div className="space-y-6">
   360→            {/* Section 1: Product Type Selection */}
   361→            <div className="space-y-2">
   362→              <Label className="text-sm font-medium">Product Type *</Label>
   363→              {productTypes.length === 0 ? (
   364→                <div className="text-center py-8 text-muted-foreground border border-dashed rounded-lg">
   365→                  <Package className="size-8 mx-auto mb-2 opacity-50" />
   366→                  <p className="text-sm">No product types available.</p>
   367→                  <p className="text-xs mt-1">Create a product type first.</p>
   368→                </div>
   369→              ) : (
   370→                <>
   371→                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   372→                    {productTypes.map((pt) => (
   373→                      <button
   374→                        key={pt.id}
   375→                        type="button"
   376→                        onClick={() => handleSelectType(pt.id)}
   377→                        className={`p-3 sm:p-4 rounded-lg border-2 text-center transition-all cursor-pointer ${
   378→                          activeTypeId === pt.id
   379→                            ? 'border-primary bg-primary/5 ring-2 ring-primary/20'
   380→                            : 'border-border hover:border-primary/50'
   381→                        }`}
   382→                      >
   383→                        <div className="text-2xl mb-1">{pt.icon || '📦'}</div>
   384→                        <div className="text-sm font-medium">{pt.name}</div>
   385→                        <div className="text-xs text-muted-foreground mt-0.5">
   386→                          {pt.attributes?.length || 0} attributes
   387→                        </div>
   388→                      </button>
   389→                    ))}
   390→                  </div>
   391→                  {form.formState.errors.productTypeId && (
   392→                    <p className="text-sm text-destructive">{form.formState.errors.productTypeId.message}</p>
   393→                  )}
   394→                </>
   395→              )}
   396→            </div>
   397→
   398→            {/* Section 2: Basic Information — shown only after a type is selected */}
   399→            {activeTypeId && (
   400→              <div className="space-y-2">
   401→                <Label className="text-sm font-medium">Basic Information</Label>
   402→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   403→                  <FormInputField
   404→                    name="name"
   405→                    label="Product Name"
   406→                    placeholder='e.g., Samsung 55" 4K Smart TV'
   407→                    required
   408→                    className="sm:col-span-2"
   409→                  />
   410→
   411→                  <FormInputField
   412→                    name="sku"
   413→                    label="SKU"
   414→                    placeholder="e.g., TV-SAM-55"
   415→                  />
   416→
   417→                  <FormTextareaField
   418→                    name="description"
   419→                    label="Description"
   420→                    placeholder="Product description..."
   421→                    rows={2}
   422→                  />
   423→
   424→                  <FormInputField
   425→                    name="costPrice"
   426→                    label="Cost Price (ETB)"
   427→                    type="number"
   428→                    step="0.01"
   429→                    min="0.01"
   430→                    placeholder="e.g. 5000"
   431→                    required
   432→                  />
   433→
   434→                  <FormInputField
   435→                    name="sellingPrice"
   436→                    label="Selling Price (ETB)"
   437→                    type="number"
   438→                    step="0.01"
   439→                    min="0.01"
   440→                    placeholder="e.g. 8000"
   441→                    required
   442→                  />
   443→
   444→                  <FormInputField
   445→                    name="lowStockThreshold"
   446→                    label="Low Stock Threshold"
   447→                    type="number"
   448→                    min="0"
   449→                  />
   450→
   451→                  {!isEditing && (
   452→                    <FormInputField
   453→                      name="quantity"
   454→                      label="Initial Quantity"
   455→                      type="number"
   456→                      min="0"
   457→                    />

   459→                </div>

   463→            {/* Section 3: Custom Attributes — shown only when type has attributes */}

   466→                <Label className="text-sm font-medium">Custom Attributes</Label>

   470→                    return (
   471→                      <div key={attr.id} className="space-y-2">
   472→                        <Label className="flex items-center gap-1">
   473→                          {attr.name}
   474→                          {attr.required && <span className="text-destructive">*</span>}
   475→                        </Label>
   476→
   477→                        {(attr.fieldType === 'text' || attr.fieldType === 'textarea') && (
   478→                          <Input
   479→                            placeholder={`Enter ${attr.name.toLowerCase()}`}
   480→                            value={attributeValues[attr.id] || ''}
   481→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   482→                          />
   483→                        )}
   484→
   485→                        {attr.fieldType === 'number' && (
   486→                          <Input
   487→                            type="number"
   488→                            step="any"
   489→                            placeholder={`Enter ${attr.name.toLowerCase()}`}

   491→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}

   500→                                updateAttributeValue(attr.id, checked ? 'true' : 'false')

   502→                            />

   504→                              {attributeValues[attr.id] === 'true' ? 'Yes' : 'No'}
   505→                            </span>
   506→                          </div>
   507→                        )}
   508→
   509→                        {attr.fieldType === 'select' && options && (

   515→                              <SelectValue placeholder={`Select ${attr.name.toLowerCase()}`} />
   516→                            </SelectTrigger>
   517→                            <SelectContent>
   518→                              {options.map((opt: string) => (
   519→                                <SelectItem key={opt} value={opt}>
   520→                                  {opt}
   521→                                </SelectItem>
   522→                              ))}
   523→                            </SelectContent>
   524→                          </Select>
   525→                        )}
   526→
   527→                        {attr.fieldType === 'image' && (
   528→                          <div className="space-y-2">
   529→                            <div className="flex items-center gap-2">
   530→                              <Button
   531→                                type="button"
   532→                                variant="outline"
   533→                                size="sm"
   534→                                className="gap-1"
   535→                                onClick={() =>
   536→                                  document.getElementById(`img-upload-${attr.id}`)?.click()
   537→                                }
   538→                              >
   539→                                <Upload className="size-3.5" />
   540→                                Upload Image
   541→                              </Button>
   542→                              {attributeValues[attr.id] && (
   543→                                <Button
   544→                                  type="button"
   545→                                  variant="ghost"
   546→                                  size="sm"
   547→                                  onClick={() => {
   548→                                    updateAttributeValue(attr.id, '')
   549→                                    setImagePreview(null)
   550→                                  }}
   551→                                >
   552→                                  <X className="size-3.5" />
   553→                                </Button>
   554→                              )}
   555→                            </div>
   556→                            <input
   557→                              id={`img-upload-${attr.id}`}
   558→                              type="file"
   559→                              accept="image/*"
   560→                              className="hidden"
   561→                              onChange={handleImageUpload}
   562→                            />
   563→                            {(attributeValues[attr.id] || imagePreview) && (
   564→                              <div className="size-20 rounded-lg border overflow-hidden">
   565→                                <img
   566→                                  src={attributeValues[attr.id] || imagePreview || ''}
   567→                                  alt="Preview"
   568→                                  className="size-full object-cover"
   569→                                  loading="lazy"
   570→                                />
   571→                              </div>
   572→                            )}
   573→                          </div>
   574→                        )}
   575→
   576→                        {attr.fieldType === 'date' && (
   577→                          <Input
   578→                            type="date"
   579→                            value={attributeValues[attr.id] || ''}
   580→                            onChange={(e) => updateAttributeValue(attr.id, e.target.value)}
   581→                          />
   582→                        )}
   583→
   584→                        {attrErrors[`attr_${attr.id}`] && (
   585→                          <p className="text-sm text-destructive">{attrErrors[`attr_${attr.id}`]}</p>
   586→                        )}
   587→                      </div>
   588→                    )
   589→                  })}
   590→                </div>
   591→              </div>
   592→            )}
   593→          </div>
   594→        </Form>
   595→
   596→        <DialogFooter>
   597→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving} type="button">
   598→            Cancel
   599→          </Button>
   600→          <Button onClick={handleSaveClick} disabled={saving || !activeTypeId} type="button">
   601→            {saving && <Loader2 className="size-4 animate-spin" />}
   602→            {saving ? (isEditing ? 'Saving...' : 'Creating...') : (isEditing ? 'Save Changes' : 'Create Product')}
   603→          </Button>
   604→        </DialogFooter>
   605→      </DialogContent>
   606→    </Dialog>
   607→  )
   608→}

   617→}: {

   620→    productType?: {
