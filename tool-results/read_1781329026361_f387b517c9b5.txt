   140→type DateRangePreset = 'today' | 'this_week' | 'this_month' | 'this_quarter' | 'custom'
   141→
   142→function getDateRange(preset: DateRangePreset): { from: string; to: string } {
   143→  const now = new Date()
   144→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   145→
   146→  switch (preset) {
   147→    case 'today':
   148→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   149→    case 'this_week': {
   150→      const dayOfWeek = now.getDay()
   151→      const weekStart = new Date(today.getTime() - dayOfWeek * 24 * 60 * 60 * 1000)
   152→      return { from: weekStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   153→    }
   154→    case 'this_month': {
   155→      const monthStart = new Date(now.getFullYear(), now.getMonth(), 1)
   156→      return { from: monthStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   157→    }
   158→    case 'this_quarter': {
   159→      const quarter = Math.floor(now.getMonth() / 3)
   160→      const quarterStart = new Date(now.getFullYear(), quarter * 3, 1)
   161→      return { from: quarterStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   162→    }
   163→    default:
   164→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   165→  }
   166→}

   403→  selectedPreset,
   404→  onPresetChange,

   408→  onCustomToChange,

   478→            </div>
