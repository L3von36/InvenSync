    35→    35→function formatETB(amount: number): string {
    36→    36→  return `ETB ${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    37→    37→}
    38→    38→
    39→    39→function formatShortETB(amount: number): string {
    40→    40→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`
    41→    41→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(1)}K`
    42→    42→  return `ETB ${amount.toFixed(0)}`
    43→    43→}
