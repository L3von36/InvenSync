   119→type DateRangePreset = 'today' | 'this_week' | 'this_month' | 'this_quarter' | 'custom'
   120→
   121→function getDateRange(preset: DateRangePreset): { from: string; to: string } {
   122→  const now = new Date()
   123→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   124→
   125→  switch (preset) {
   126→    case 'today':
   127→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   128→    case 'this_week': {
   129→      const dayOfWeek = now.getDay()
   130→      const weekStart = new Date(today.getTime() - dayOfWeek * 24 * 60 * 60 * 1000)
   131→      return { from: weekStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   132→    }
   133→    case 'this_month': {
   134→      const monthStart = new Date(now.getFullYear(), now.getMonth(), 1)
   135→      return { from: monthStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   136→    }
   137→    case 'this_quarter': {
   138→      const quarter = Math.floor(now.getMonth() / 3)
   139→      const quarterStart = new Date(now.getFullYear(), quarter * 3, 1)
   140→      return { from: quarterStart.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   141→    }
   142→    default:
   143→      return { from: today.toISOString().split('T')[0], to: today.toISOString().split('T')[0] }
   144→  }
   145→}

   384→  selectedPreset,

   388→  onCustomFromChange,
   389→  onCustomToChange,
   390→}: {

   410→          <p className="text-sm font-medium">Date Range</p>
   411→          <div className="grid grid-cols-2 gap-1.5">
   412→            {(Object.keys(PRESET_LABELS) as DateRangePreset[]).map((preset) => (

   414→                key={preset}
   415→                variant={selectedPreset === preset ? 'default' : 'outline'}

   417→                className="text-xs justify-start"
   418→                onClick={() => {
   419→                  if (preset !== 'custom') {
   420→                    onPresetChange(preset)

   422→                  } else {

   434→                <label className="text-xs text-muted-foreground">From</label>

   436→                  type="date"

   459→            </div>
